#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_data_mx.h"

#define DIM 16
#define BF16_PER_WORD 4
#define BANK_NUM 4
#define BANK_ROWS 2048

// Use BLK=64 to saturate accumulator and amortize per-invocation overhead
#define BLK 64

typedef uint8_t elem_t;
typedef uint8_t out_t;

// tiles within one BLK-sized block
#define TILES   (BLK / DIM)           // = 4
#define TILES_K (MATMUL_K / DIM)      // = 20

// Scale-factor byte counts for one BLK-wide block
#define GK       (MATMUL_K / 32)
#define A_S_BYTES (GK * BLK)
#define B_S_BYTES (GK * BLK)
#define BLK_OUTC  (BLK / BF16_PER_WORD)

#define BLOCKS_M (MATMUL_M / BLK)     // = 2
#define BLOCKS_N (MATMUL_N / BLK)     // = 2

// Staging buffers for scale factors (one block's worth)
static uint8_t A_s[GK][BLK];
static uint8_t B_s[GK][BLK];

// Output scale factors required by API
static uint8_t scale_factors[MATMUL_M * (MATMUL_N / 32)];

void solution(void) {
  // Scratchpad layout for BLK=64, TILES=4, TILES_K=20:
  //   A: a_base=0, rows = TILES*TILES_K*DIM = 4*20*16 = 1280
  //   B: b_base = 8192 - TILES*TILES_K*DIM = 8192 - 1280 = 6912
  //   No overlap: A ends at 1280, B starts at 6912 ✓
  //   Accumulator: TILES*TILES*DIM = 4*4*16 = 256 rows (exactly full) ✓

  uint32_t a_base   = 0;
  uint32_t b_base   = BANK_NUM * BANK_ROWS - TILES * TILES_K * DIM;
  int      SPAD_DEST = 128;

  // One-time hardware initialization — hoisted outside all loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(
    WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
    /*C_stride*/1, /*A_stride*/1,
    /*A_transpose*/0, /*B_transpose*/0,
    /*set_only_strides*/false,
    /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

  // Output stride configured once
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {

    // Stage B scale factors for this j0 block (independent of i0)
    for (int g = 0; g < GK; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

    // MVIN B tiles: B_in[k*DIM][j0*BLK+j*DIM]
    // B row stride = MATMUL_N bytes (fp8: 1 byte/elem)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < TILES; j++)
      for (int k = 0; k < TILES_K; k++)
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * TILES + j) * DIM,
          DIM, DIM);

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {

      // Stage A scale factors for this (i0, j0) block
      for (int g = 0; g < GK; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

      // Programming model sequence must be followed strictly:
      // 1. config_ex (already done once above, persistent)
      // 2. load A scales (sel=0), then B scales (sel=1)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // 3. MVIN A tiles: A_in[i0*BLK+i*DIM][k*DIM]
      // A row stride = MATMUL_K bytes (fp8: 1 byte/elem)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < TILES; i++)
        for (int k = 0; k < TILES_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * TILES_K + k) * DIM,
            DIM, DIM);

      // 4. Configure requantizer mvout bounds matching loop dimensions exactly
      gemmini_mxquant_config_mvout(
        (uint64_t)scale_factors,
        /*i_bound*/TILES, /*j_bound*/TILES, /*k_bound*/TILES_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

      // 5. Execute fused tiled matmul: 4x4x20 tiles of 16x16 MACs
      // mx_loop_spad_marker is one-shot; gemmini_loop_ws_spad re-issues
      // funct 24 (LOOP_WS_CONFIG_SPAD_AB) before every invocation
      gemmini_loop_ws_spad(
        TILES, TILES, TILES_K,
        0, 0, 0,
        a_base, BANK_NUM * BANK_ROWS,
        0, SPAD_DEST,
        false, false,
        false, false, false,
        NO_ACTIVATION, 0, 0, false,
        0x38);

      // 6. Drain BF16 results: single bulk DMA for the entire BLK x BLK region
      // smem layout: SPAD_DEST*16 is word offset; BLK*BLK BF16 values total
      gemmini_mx_read_smem(
        &C_hw[i0 * BLK][(j0 * BLK) / BF16_PER_WORD],
        SPAD_DEST * 16,
        BLK * BLK);

      // 7. Fence must complete before host reads output
      gemmini_fence();
    }
  }
}
