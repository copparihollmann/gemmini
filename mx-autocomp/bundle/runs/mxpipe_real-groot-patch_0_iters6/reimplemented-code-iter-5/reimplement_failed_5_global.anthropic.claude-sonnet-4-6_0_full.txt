Looking at the error, the issue is that `K_CHUNK` is defined with `#define K_CHUNK 8` but the compiler says it's undeclared. This means the `#define` is not being included in the compilation unit - it's only in the solution output but not in the header/preamble that gets compiled.

The fix is to move the constant inline into the function itself as a local constant, removing the `#define` from outside the function.

```c
void solution(void) {
  const int K_CHUNK = 8;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;  // = 32
      int pad_i   = BLK - mrows;
      int pad_j   = BLK - ncols;
      int SPAD_DEST = 128;
      uint32_t a_base = 0;

      gemmini_flush(0);

      bool first_chunk = true;
      for (int kc = 0; kc < tiles_K; kc += K_CHUNK) {
        int cur_tiles_K = tiles_K - kc;
        if (cur_tiles_K > K_CHUNK) cur_tiles_K = K_CHUNK;

        // Stage scale factors for this K-chunk only
        for (int g = 0; g < cur_tiles_K; g++) {
          memcpy(&A_s[g][0], &A_scales_row[kc + g][i0 * BLK], BLK);
          memcpy(&B_s[g][0], &B_scales_col[kc + g][j0 * BLK], BLK);
        }

        uint32_t b_base = BANK_NUM * BANK_ROWS - (uint32_t)(tiles * cur_tiles_K * DIM);

        // Step 1: Configure execution (must precede scale loads)
        gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                    1, 1, 0, 0, false, 0, 0, 3, 0);

        // Step 2: Load scale factors for this K-chunk
        // sel=0 => A (activation) scales, sel=1 => B (weight) scales
        gemmini_mx_load_scales((uint64_t)A_s, (uint64_t)(cur_tiles_K * BLK), 0);
        gemmini_mx_load_scales((uint64_t)B_s, (uint64_t)(cur_tiles_K * BLK), 1);

        // Step 3: Mvin A tiles for this K-chunk (stride = full K width)
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i = 0; i < tiles; i++)
          for (int k = 0; k < cur_tiles_K; k++)
            gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + (kc + k) * DIM),
              a_base + (uint32_t)((i * cur_tiles_K + k) * DIM),
              DIM, DIM);

        // Step 4: Mvin B tiles for this K-chunk (stride = full N width)
        gemmini_config_ld(MATMUL_N * sizeof(elem_t));
        for (int j = 0; j < tiles; j++)
          for (int k = 0; k < cur_tiles_K; k++)
            gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + ((kc + k) * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (uint32_t)((k * tiles + j) * DIM),
              DIM, DIM);

        // Step 5: Configure requantized mvout with cur_tiles_K as k_bound
        gemmini_config_st(BLK_OUTC * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                     tiles, tiles, cur_tiles_K,
                                     0, 0, 1);

        // Step 6: Compute — accumulate into smem after first chunk
        gemmini_loop_ws_spad(
          tiles, tiles, cur_tiles_K,
          pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false,
          false, false,
          /*ex_accumulate=*/(!first_chunk),
          NO_ACTIVATION, 0, 0, false, 0x38);

        first_chunk = false;
      }

      // Step 7: Drain fully-accumulated BF16 result from smem to DRAM
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
```
