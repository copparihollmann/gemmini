2. **BLK vs tiles calculation**: With MATMUL_M=41, MATMUL_N=512, MATMUL_K=512, DIM=16: BLK must be a multiple of DIM. tiles = BLK/DIM, tiles_K = K/DIM = 32.

The main issue is likely the scratchpad overflow. With BLK=32 (tiles=2, tiles_K=32): (2*32 + 2*32)*16 = 2048 ≤ 8192. ✓ Accumulator: 2*2*16=64 ≤ 256. ✓

Also, `gemmini_mx_read_smem` for BF16 output: ncols elements per row, offset needs to account for row stride in smem.

void solution(void) {
  const int tiles_K = MATMUL_K / DIM;  // 32

  // Use BLK=32 (tiles=2) to fit in scratchpad:
  // A: tiles_I*tiles_K*DIM = 2*32*16=1024 rows
  // B: tiles_J*tiles_K*DIM = 2*32*16=1024 rows
  // Total: 2048 <= 8192 rows. OK.
  const int BLK_TILE = 2;  // tiles per block (BLK = BLK_TILE*DIM = 32)
  const int BLK_SZ   = BLK_TILE * DIM;  // 32 elements per block

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - BLK_TILE * tiles_K * DIM;
  int SPAD_DEST = 128;

  int BLOCKS_M_loc = (REAL_M + BLK_SZ - 1) / BLK_SZ;
  int BLOCKS_N_loc = (MATMUL_N + BLK_SZ - 1) / BLK_SZ;

  for (int j0 = 0; j0 < BLOCKS_N_loc; j0++) {
    int ncols = MATMUL_N - j0 * BLK_SZ;
    if (ncols > BLK_SZ) ncols = BLK_SZ;
    if (ncols <= 0) continue;
    int jtiles = (ncols + DIM - 1) / DIM;
    if (jtiles > BLK_TILE) jtiles = BLK_TILE;
    int pad_j = BLK_TILE * DIM - ncols;

    for (int i0 = 0; i0 < BLOCKS_M_loc; i0++) {
      int mrows = REAL_M - i0 * BLK_SZ;
      if (mrows > BLK_SZ) mrows = BLK_SZ;
      if (mrows <= 0) continue;
      int itiles = (mrows + DIM - 1) / DIM;
      if (itiles > BLK_TILE) itiles = BLK_TILE;
      int pad_i = BLK_TILE * DIM - mrows;

      // Stage A scales for this i0 block: A_scales_row[gk][row], gk in [0, tiles_K)
      // A_s is [tiles_K][BLK_SZ] of e8m0 scale bytes
      for (int g = 0; g < tiles_K; g++)
        for (int ii = 0; ii < itiles * DIM && ii < mrows; ii++)
          A_s[g][ii] = A_scales_row[g][i0 * BLK_SZ + ii];

      // Stage B scales for this j0 block: B_scales_col[gk][col]
      for (int g = 0; g < tiles_K; g++)
        for (int jj = 0; jj < jtiles * DIM && jj < ncols; jj++)
          B_s[g][jj] = B_scales_col[g][j0 * BLK_SZ + jj];

      // Step 1: Configure execution format
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load A scales (sel=0) and B scales (sel=1)
      gemmini_mx_load_scales((uint64_t)A_s, (size_t)(tiles_K * BLK_TILE * DIM), 0);
      gemmini_mx_load_scales((uint64_t)B_s, (size_t)(tiles_K * BLK_TILE * DIM), 1);

      // Step 3: MVIN A tiles into scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < BLK_TILE; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK_SZ + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Step 3b: MVIN B tiles into scratchpad
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < BLK_TILE; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK_SZ + j * DIM)),
            b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      // Step 4: Configure output stride and requantizer
      gemmini_config_st(MATMUL_N * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   BLK_TILE, BLK_TILE, tiles_K,
                                   0, 0, 1);

      // Step 5: Execute fused weight-stationary matmul
      gemmini_loop_ws_spad(BLK_TILE, BLK_TILE, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          (void*)(&C_hw[i0 * BLK_SZ + r][(j0 * BLK_SZ)]),
          SPAD_DEST * 16 + r * BLK_TILE * DIM,
          ncols);

      // Step 7: Full memory barrier
      gemmini_fence();
    }
  }
}
