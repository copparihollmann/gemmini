Looking at the error, the issue is that `scale_factors` is being defined as a macro somewhere in the included headers (the error shows `scale_factors` expanding to something with `*` and a numeric constant). I need to rename the local variable to avoid the macro conflict.

Additionally, the `gemmini_mx_read_smem` call has an issue with the B matrix mvin stride - B is stored row-major in DRAM with stride `MATMUL_N`, but the `b_base` layout needs `(k * tiles + j)` ordering to match how `gemmini_loop_ws_spad` expects B tiles laid out.

The fix is to rename `scale_factors` to something that won't conflict with any macro, and ensure all other logic is correct.

void solution(void) {
  const int GK = MATMUL_K / 32;

  static uint8_t pre_A_s[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t pre_B_s[BLOCKS_N][MATMUL_K / 32][BLK];

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_A_s[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);

  for (int j0 = 0; j0 < BLOCKS_N; j0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_B_s[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);

  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  const int tiles   = BLK / DIM;
  const int tiles_K = MATMUL_K / DIM;
  const uint32_t a_base    = 0;
  const uint32_t b_base    = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  const int      SPAD_DEST = 128;

  static uint32_t mx_sf_buf[512] __attribute__((aligned(32)));

  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      gemmini_mx_load_scales((uint64_t)pre_A_s[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)pre_B_s[j0], B_S_BYTES, 1);

      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      gemmini_mxquant_config_mvout((uint64_t)mx_sf_buf,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);
      gemmini_fence();
    }
  }
}
