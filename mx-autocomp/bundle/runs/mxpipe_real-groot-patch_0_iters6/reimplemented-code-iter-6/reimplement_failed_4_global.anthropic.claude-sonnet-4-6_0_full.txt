Looking at the error, the issue is clear: B tiles are being loaded with `gemmini_extended_mvin2` (port 1), but `gemmini_loop_ws_spad` requires all inputs to use port 0 (`gemmini_extended_mvin`). The fix is to load both A and B tiles using `gemmini_extended_mvin` (port 0), but they need to be in non-overlapping scratchpad regions.

The scratchpad layout must keep A and B tiles separate:
- A tiles at `a_base` (starting from 0)
- B tiles at `b_base` (top of scratchpad, growing downward)

Both loaded via `gemmini_extended_mvin` (port 0), just to different spad addresses.

```c
void solution(void) {
  gemmini_flush(0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST   = 128;

  // j0 outer: B block is fixed for all i0 — load B tiles once per j0
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // Stage B scales for this j0 block (once, amortized over all i0)
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

    // Load B tiles into scratchpad once for all i0 iterations
    // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);

    // i0 inner: A changes each iteration, B stays resident in scratchpad
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Stage A scales for this i0 block
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

      // Programming model sequence:
      // 1. config_ex
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // 2. load_scales: A (sel=0) then B (sel=1) — both required before compute
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // 3. mvin A tiles using port 0 (gemmini_extended_mvin)
      //    B tiles already resident from outer j0 loop, also loaded via port 0
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // 4. config mvout + requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // 5. compute (funct 24 marker is one-shot, re-issued by macro on every call)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // 6. drain results from mx_smem to DRAM, then fence
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      gemmini_fence();
    }
  }
}
```
