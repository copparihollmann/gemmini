Looking at the code, I need to identify the bugs:

1. **B tile layout is wrong**: The B mvin uses `b_base + (k * tiles + j) * DIM` but it should be `b_base + (j * tiles_K + k) * DIM` to match the documented layout `b_base + (j*tiles_K + k)*DIM`.

2. **B mvin uses wrong stride**: B is stored column-major in DRAM (stride = `MATMUL_N`), but the access pattern for B should be `B_in[k*DIM][j0*BLK + j*DIM]` with stride `MATMUL_N`.

3. **`gemmini_mx_read_smem` offset calculation**: The smem offset uses `r * BLK` but BLK is in elements, and the smem is addressed in 16-bit word units. Since ncols is in BF16 elements (each 2 bytes = 1 u16 word), we need `r * tiles * DIM` words per row. Actually `SPAD_DEST*16 + r * BLK` — BLK here represents number of BF16 values per row of the output block, which should be correct if BLK=tiles*DIM. But let me check: smem_off_bf16 is in 16-bit word units, and each row has `BLK` BF16 values, so `r * BLK` is correct.

4. **A scales sel=0, B scales sel=1**: The code has `sel=0` for A and `sel=1` for B which is correct per rule 11.

5. **The B mvin layout bug** is the main issue: `b_base + (k * tiles + j) * DIM` should be `b_base + (j * tiles_K + k) * DIM`.

void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST = 128;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows_max = REAL_M - i0 * BLK;
    if (mrows_max <= 0) continue;
    if (mrows_max > BLK) mrows_max = BLK;

    // Pack and load A scales (sel=0 for A/activation scales).
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
    gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = mrows_max;
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Pack and load B scales (sel=1 for B/weight scales).
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Mvin A tiles: layout a_base + (i*tiles_K + k)*DIM, stride = MATMUL_K.
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // Mvin B tiles: layout b_base + (j*tiles_K + k)*DIM, stride = MATMUL_N.
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);

      // Configure requantized mvout; bounds must match tiles used in loop_ws_spad.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Compute tiled weight-stationary matmul over scratchpad.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM, one row at a time.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);
      gemmini_fence();
    }
  }
}
