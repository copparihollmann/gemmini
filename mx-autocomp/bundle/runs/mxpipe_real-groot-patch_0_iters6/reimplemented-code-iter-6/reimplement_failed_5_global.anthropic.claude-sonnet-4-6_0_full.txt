void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles_I = BLK / DIM;  // tiles per block in I dimension
  int tiles_J = BLK / DIM;  // tiles per block in J dimension
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base = 0;
  // B placed at top of scratchpad, growing downward
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
  int SPAD_DEST = 128;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    int pad_i = BLK - mrows;

    // Load A scales for this i0 row-block (invariant over j0)
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
    gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  // sel=0 -> A scales

    // Mvin A tiles into scratchpad — A layout: a_base + (i*tiles_K + k)*DIM

    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_j = BLK - ncols;

      // Load B scales for this j0 column-block
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  // sel=1 -> B scales

      // Mvin B tiles into scratchpad — B layout: b_base + (j*tiles_K + k)*DIM

              b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K, 0, 0, 1);

      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0,

                           false, false, false, false, false,

  }
}
