Looking at the code, I need to identify the issues:

1. **B tile layout in scratchpad is wrong**: B tiles should be at `b_base + (j*tiles_K + k)*DIM` but the code uses `(k * tiles + j) * DIM` - the j and k indices are swapped in the stride calculation.

2. **`gemmini_mx_read_smem` offset calculation is wrong**: The smem offset should be in BF16 word units. `SPAD_DEST * 16` gives the base, then per-row offset should be `r * BLK` BF16 elements (since BLK = tiles*DIM elements per row).

3. **Scale loading order**: The programming model requires A scales loaded before B scales before each compute. The code re-issues A scales inside j0 loop which is correct, but the initial A scale load at i0 level before j0 loop is redundant and could cause issues.

4. **`A_S_BYTES` and `B_S_BYTES`**: The scale buffer sizes need to match `(K/32) * BLK` bytes.

5. **B mvin stride**: B matrix is stored as `[K][N]` so the stride should be `MATMUL_N * sizeof(elem_t)` which matches B row stride.

6. **The `gemmini_mx_read_smem` num_bf16 should be `ncols` BF16 values per row, offset advances by `tiles*DIM` (=BLK) per row**.

7. **B base offset formula**: `b_base + (j*tiles_K + k)*DIM` not `(k*tiles + j)*DIM`.

```c
void solution(void) {
  static uint8_t A_scales_blk[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_blk[BLOCKS_N][MATMUL_K / 32][BLK];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows <= 0) continue;
    if (mrows > BLK) mrows = BLK;
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_scales_blk[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols <= 0) continue;
    if (ncols > BLK) ncols = BLK;
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_scales_blk[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
  }

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base    = 0;
  uint32_t b_base    = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 0;

  // A_S_BYTES = number of scale bytes for one i0 block: (K/32) * BLK
  // B_S_BYTES = number of scale bytes for one j0 block: (K/32) * BLK
  size_t A_S_BYTES = (MATMUL_K / 32) * BLK;
  size_t B_S_BYTES = (MATMUL_K / 32) * BLK;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Step 1: Configure execution
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false,
                                  /*act_mx_fmt*/0, /*wgt_mx_fmt*/0,
                                  /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: Load scale factors - A first (sel=0), then B (sel=1)
      gemmini_mx_load_scales((uint64_t)A_scales_blk[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_blk[j0], B_S_BYTES, 1);

      // Step 4: Mvin A tiles into scratchpad
      // A layout: a_base + (i*tiles_K + k)*DIM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 4: Mvin B tiles into scratchpad
      // B layout: b_base + (j*tiles_K + k)*DIM
      // B is stored as [K][N], stride = MATMUL_N bytes per row
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 5: Configure requantized mvout
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Step 6: Compute
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Read results from mx_smem to DRAM
      // Results are packed BF16: SPAD_DEST*16 is base in BF16 word units
      // Each row of output has BLK BF16 values; rows are contiguous in smem
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(
          (void*)&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);
      }
      gemmini_fence();
    }
  }
}
```
