Looking at the plan, I need to implement double-buffering for A tiles: pre-load A tiles for i0=0 before the loop, then after each `gemmini_loop_ws_spad`, prefetch A tiles for i0+1 into the alternate buffer.

Let me verify scratchpad layout:
- `tiles = BLK/DIM = 4` (assuming BLK=64), `tiles_K = MATMUL_K/DIM = 32`
- Each A buffer: `tiles * tiles_K * DIM = 4 * 32 * 16 = 2048` rows
- Two A buffers: `2 * 2048 = 4096` rows
- B tiles: `tiles * tiles_K * DIM = 2048` rows at top
- Total: `4096 + 2048 = 6144 ≤ 8192` ✓
- `b_base = BANK_NUM*BANK_ROWS - tiles*tiles_K*DIM = 8192 - 2048 = 6144`, no overlap with A buffers ending at 4096 ✓

Key ordering rules:
- `config_ex` once (format doesn't change)
- A scales loaded once per `i0` (outside `j0` loop)
- B scales + B mvin + config + compute per `(i0,j0)`
- Prefetch next A tiles after `loop_ws_spad` (DMA overlaps with compute)
- `read_smem` + `fence` after each block

```c
void solution(void) {
  const int GK = MATMUL_K / 32;

  static uint8_t pre_A_s[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t pre_B_s[BLOCKS_N][MATMUL_K / 32][BLK];

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_A_s[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);

  for (int j0 = 0; j0 < BLOCKS_N; j0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_B_s[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);

  const int tiles   = BLK / DIM;
  const int tiles_K = MATMUL_K / DIM;

  // Double-buffered A tile bases: buffer 0 starts at row 0, buffer 1 starts
  // after buffer 0.  B tiles are placed at the top of the scratchpad.
  const uint32_t a_buf_rows = (uint32_t)(tiles * tiles_K * DIM);
  const uint32_t a_base_buf[2] = { 0, a_buf_rows };
  const uint32_t b_base = (uint32_t)(BANK_NUM * BANK_ROWS) - (uint32_t)(tiles * tiles_K * DIM);
  const int SPAD_DEST = 128;

  // One-time hardware configuration: format never changes across blocks.
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  // Helper lambda (inlined): issue mvin instructions for one A block (i0) into
  // the specified scratchpad buffer base.  Declared as a local inline function
  // via a nested loop to keep the structure readable.
#define LOAD_A_TILES(i0_val, a_base_val)                                              \
  do {                                                                                 \
    int _mrows = REAL_M - (i0_val) * BLK;                                             \
    if (_mrows > BLK) _mrows = BLK;                                                   \
    if (_mrows > 0) {                                                                  \
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));                                   \
      for (int _i = 0; _i < tiles; _i++)                                              \
        for (int _k = 0; _k < tiles_K; _k++)                                          \
          gemmini_extended_mvin(                                                        \
            (void*)(((elem_t*)A_in) +                                                  \
                    (size_t)((i0_val) * BLK + _i * DIM) * MATMUL_K + _k * DIM),      \
            (a_base_val) + (_i * tiles_K + _k) * DIM,                                 \
            DIM, DIM);                                                                 \
    }                                                                                  \
  } while (0)

  // Prologue: load A tiles for i0=0 into buffer 0 before the main loop.
  LOAD_A_TILES(0, a_base_buf[0]);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    int pad_i = BLK - mrows;
    uint32_t cur_a_base = a_base_buf[i0 & 1];
    uint32_t nxt_a_base = a_base_buf[(i0 + 1) & 1];

    // Load A scales once per i0, before any j0 compute that uses them.
    gemmini_mx_load_scales((uint64_t)pre_A_s[i0], A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_j = BLK - ncols;

      // Load B scales for this (i0, j0) block.
      gemmini_mx_load_scales((uint64_t)pre_B_s[j0], B_S_BYTES, 1);

      // Load B tiles into scratchpad.  B layout: slot (k*tiles + j) for B[k][j].
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (size_t)(k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM,
            DIM, DIM);

      // Configure output stride and requantizer bounds.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                    tiles, tiles, tiles_K,
                                    0, 0, 1);

      // Issue the fused tiled matmul.  mx_loop_spad_marker is one-shot so
      // gemmini_loop_ws_spad internally re-issues funct 24 before funct 8.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                            pad_i, pad_j, 0,
                            cur_a_base, BANK_NUM * BANK_ROWS,
                            0, SPAD_DEST,
                            false, false,
                            false, false, false,
                            NO_ACTIVATION, 0, 0, false, 0x38);

      // Prefetch A tiles for i0+1 into the alternate buffer immediately after
      // issuing loop_ws_spad.  The DMA engine can overlap this load with the
      // systolic array compute, hiding the A-load latency.
      // Only prefetch at the first j0 to avoid redundant reloads; the A tiles
      // for i0+1 are identical for all j0, so one prefetch suffices.
      if (j0 == 0 && (i0 + 1) < BLOCKS_M) {
        LOAD_A_TILES(i0 + 1, nxt_a_base);
      }

      // Drain results from mx_smem to DRAM, one row at a time.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }

#undef LOAD_A_TILES
}
```
