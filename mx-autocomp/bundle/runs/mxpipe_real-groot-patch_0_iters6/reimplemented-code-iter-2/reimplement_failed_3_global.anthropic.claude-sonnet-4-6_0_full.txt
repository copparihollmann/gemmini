        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      /* 1. Load A and B scales */
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  /* sel=0 → A */
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  /* sel=1 → B */

  }
