Looking at the failing code, I need to identify the bugs:

1. **B tile layout is wrong**: The code uses `b_base + (k * tiles + j) * DIM` but the spec says B tile layout is `b_base + (j*tiles_K + k)*DIM`. The j and k indices are swapped in the stride calculation.

2. **B mvin stride is wrong**: B is stored in DRAM as `[K][N]` (row = k*DIM row of B, column = j0*BLK + j*DIM). The stride should be `MATMUL_N * sizeof(elem_t)` which is correct, but the DRAM address calculation needs to index `B_in[k*DIM][j0*BLK + j*DIM]`.

3. **`gemmini_mx_read_smem` smem offset**: The offset `SPAD_DEST * 16 + r * BLK` — since BF16 results are packed 4 per uint64 (2 bytes each), one row of `ncols` BF16 values spans `ncols` 16-bit words. The smem word offset for row `r` should be `SPAD_DEST*16 + r * (BLK)` where BLK is in BF16 units (tiles * DIM = BLK). This seems correct if BLK = tiles*DIM.

4. **The B mvin DRAM pointer**: B is `[K][N]`, so element `[k*DIM][j0*BLK + j*DIM]` = `B_in + (k*DIM)*MATMUL_N + j0*BLK + j*DIM`. This looks correct.

5. **B scratchpad address**: Should be `b_base + (j*tiles_K + k)*DIM` not `(k*tiles + j)*DIM`. This is the primary bug.

6. **`gemmini_config_ld` for B**: stride should be `MATMUL_N * sizeof(elem_t)` — correct.

7. **mvin2 should be used for B** (not mvin) to allow overlapping with A loads.

void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false,
                               0, 0, 3, 0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // Stage per-block scale factors (A row-scales and B col-scales).
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles_I = BLK / DIM;
      int tiles_J = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Load per-block scale factors into accelerator scale memory.
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles: stride = MATMUL_K (full packed row width).
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // MVIN B tiles via mvin2: stride = MATMUL_N (full packed row width).
      // B tile layout: b_base + (j*tiles_K + k)*DIM  (spec-mandated layout)
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      // Configure requantizer: output scale-factor destination and tile bounds.
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // Fused weight-stationary tiled matmul over scratchpad tiles.
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from shared memory to DRAM output buffer.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);

      gemmini_fence();
    }
}
