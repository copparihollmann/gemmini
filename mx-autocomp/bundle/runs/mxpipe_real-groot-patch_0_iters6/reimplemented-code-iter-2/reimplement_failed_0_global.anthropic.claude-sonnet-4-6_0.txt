#include "include/gemmini.h"
#include "include/gemmini_params.h"
#include "matmul_params.h"
#include <string.h>

#ifndef BLK
#define BLK 64
#endif

#define BLOCKS_M ((REAL_M + BLK - 1) / BLK)
#define BLOCKS_N ((REAL_N + BLK - 1) / BLK)

#define BLK_TILES (BLK / DIM)
#define TILES_K   (MATMUL_K / DIM)
#define GK        (MATMUL_K / 32)

#define A_S_BYTES (GK * BLK)
#define B_S_BYTES (GK * BLK)

#define BF16_PER_WORD 4

static uint8_t A_s[GK][BLK];
static uint8_t B_s[GK][BLK];

void solution(void) {
  gemmini_flush(0);

  // Configure execution once: FP8 inputs, BF16 output, weight-stationary
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int tiles_I = BLK / DIM;
      int tiles_J = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;

      // Stage scale factors contiguously for this block
      for (int g = 0; g < GK; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Step 1 already done (config_ex issued once at top)

      // Step 2: Load scale factors for this block
      // sel=0 → A (activation) scale memory, sel=1 → B (weight) scale memory
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN A tiles into scratchpad
      // A layout: a_base + (i*tiles_K + k)*DIM
      uint32_t a_base = 0;
      // B placed at TOP of scratchpad growing downward
      // b_base = BANK_NUM*BANK_ROWS - tiles_J*tiles_K*DIM
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;

      // Verify no overlap: a_base + tiles_I*tiles_K*DIM <= b_base
      // tiles_I*tiles_K*DIM = 4*4*16=256, b_base=8192-256=7936 → OK

      // MVIN A via port mvin (k_MVIN), stride = MATMUL_K * sizeof(elem_t)
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *a_ptr = ((const elem_t *)A_in)
              + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM;
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
          gemmini_extended_mvin(a_ptr, spad_addr, DIM, DIM);
        }
      }

      // MVIN B via port mvin2 (k_MVIN2), stride = MATMUL_N * sizeof(elem_t)
      // B layout: b_base + (j*tiles_K + k)*DIM
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *b_ptr = ((const elem_t *)B_in)
              + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
          gemmini_extended_mvin2(b_ptr, spad_addr, DIM, DIM);
        }
      }

      // Step 5: Configure requantized mvout
      int SPAD_DEST = 128;
      int pad_i = (BLK - mrows) / DIM;  // in tiles
      int pad_j = (BLK - ncols) / DIM;  // in tiles

      gemmini_config_st(REAL_N * sizeof(out_t));
      gemmini_mxquant_config_mvout(
          (uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K,
          0, 0, 1);

      // Step 6: Execute fused tiled matmul
      // gemmini_loop_ws_spad issues funct 24 then funct 8 internally
      gemmini_loop_ws_spad(
          tiles_I, tiles_J, tiles_K,
          pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false,
          false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain BF16 results from mx_smem to output DRAM
      // mx_smem is addressed in 16-bit words; C spad word offset = SPAD_DEST*16
      // Results are packed 4 BF16 per uint64; output row stride = REAL_N elements
      // smem layout: row r has BLK bf16 values starting at SPAD_DEST*16 + r*BLK
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(
            (void *)(&C_hw[i0 * BLK + r][j0 * BLK / BF16_PER_WORD]),
            SPAD_DEST * 16 + r * BLK,
            ncols);
      }
      gemmini_fence();
    }
  }
}
