void solution(void) {
  // Hoist loop-invariant flush and config outside all loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base   = 0;
  uint32_t b_base   = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  // Outer loop over J (weight columns) — B tiles are weight-stationary across I
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    int pad_j = BLK - ncols;

    // Stage B scales for this j0 block (reused across all i0)
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

    // Load B tiles into fixed high scratchpad region once for this j0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);

    // Inner loop over I (activation rows) — A tiles change, B tiles reused
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows;

      // Stage A scales for this i0 block
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

      // Programming model sequence: load A scales (sel=0) then B scales (sel=1)
      // before mvin and compute. B scales are re-loaded each i0 iteration to
      // satisfy the required ordering: both scale buffers must be loaded after
      // the most recent gemmini_extended3_config_ex and before loop_ws_spad.
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Load A tiles for this i0 block
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Configure output stride and requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Compute fused tiled matmul (mx_loop_spad_marker is one-shot;
      // gemmini_loop_ws_spad macro re-issues funct 24 before funct 8)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results from mx_smem to DRAM
      if (mrows == BLK && ncols == BLK) {
        // Full block: single smem read covers all rows contiguously
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + 0][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16, BLK * BLK);
      } else {
        // Partial block: per-row reads to handle mrows/ncols boundary
        for (int r = 0; r < mrows; r++)
          gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
  }
}
