Looking at the failing code, I need to identify the bugs:

1. **B tile layout is wrong**: The B matrix uses row-major layout where B[k][j] should be at `b_base + (j*tiles_K + k)*DIM`, but the code uses `b_base + (k*tiles + j)*DIM`. The doc says: "B tile layout: `b_base + (j*tiles_K + k)*DIM`"

2. **`gemmini_flush` inside the loop**: This is expensive and unnecessary on every iteration - but more importantly it may reset state. However, let's keep it for correctness and focus on the layout bug.

3. **`gemmini_mx_read_smem` offset**: The output is being written with stride issues. The output C_hw has dimensions [REAL_M][REAL_N/BF16_PER_WORD] but we need to write each row of the output block separately since the stride in C_hw may not match `mrows*ncols` contiguously.

4. **`gemmini_config_ld` for B**: B stride should be `MATMUL_N * sizeof(elem_t)` which is correct.

5. **Scale loading sel**: The doc says sel=0 for A and sel=1 for B, but looking at the `MX_LOAD_SCALES` table: "sel 0 = A-row scales, 1 = B-col scales" - the code has these correct actually. Wait, re-reading: the intrinsic comment says "sel 0 = activations/A, sel 1 = weights/B" - so the code is correct.

The main bug is the **B tile scratchpad addressing**. B should be at `b_base + (j*tiles_K + k)*DIM` not `b_base + (k*tiles + j)*DIM`.

void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

    // A scales only depend on i0: copy once per i0, reuse across all j0.
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // B scales depend on j0: copy once per (i0,j0).
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

      int tiles_I = BLK / DIM;
      int tiles_J = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      // B placed at top of scratchpad, b_base = BANK_NUM*BANK_ROWS - tiles_J*tiles_K*DIM
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
      int SPAD_DEST = 0;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

      // Load A scales (sel=0) and B scales (sel=1)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles: A[i][k] -> a_base + (i*tiles_K + k)*DIM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // MVIN B tiles: B[k][j] -> b_base + (j*tiles_K + k)*DIM
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);

      // Configure output stride and requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // Compute: B address is the end of scratchpad (BANK_NUM*BANK_ROWS)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results: each row of the output block
      for (int i = 0; i < mrows; i++) {
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + i][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + i * BLK,
          ncols);
      }
      gemmini_fence();
    }
  }
}
