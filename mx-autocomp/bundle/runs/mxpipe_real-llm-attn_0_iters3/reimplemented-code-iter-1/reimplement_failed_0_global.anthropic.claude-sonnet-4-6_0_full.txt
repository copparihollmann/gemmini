Looking at the error, the issue is clear: B tiles were loaded using `gemmini_extended_mvin2` (port 1), but `gemmini_loop_ws_spad` requires all operands to be in the port-0 scratchpad region. The fix is to use `gemmini_extended_mvin` (port 0) for both A and B tiles.

```c
// Optimized: j0 outer, i0 inner — B tiles and B scales loaded once per j0,
// amortizing BLOCKS_M redundant mvin/scale-load operations per column block.
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  // One-time flush + format config (format never changes across all tiles).
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols <= 0) continue;
    if (ncols > BLK) ncols = BLK;

    // ------------------------------------------------------------------ //
    // B scales: fixed for all i0 at this j0 — load once.
    // ------------------------------------------------------------------ //
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
    gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

    // ------------------------------------------------------------------ //
    // B tiles: fixed for all i0 at this j0 — mvin once via port 0.
    // (Must use gemmini_extended_mvin, not mvin2, for loop_ws_spad.)
    // ------------------------------------------------------------------ //
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));   // stride = full N row
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM,
          DIM, DIM);

    // ------------------------------------------------------------------ //
    // Inner loop: only A changes per i0.
    // ------------------------------------------------------------------ //
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows <= 0) continue;
      if (mrows > BLK) mrows = BLK;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // A scales for this (i0) block.
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

      // Programming-model ordering: load BOTH A and B scales before compute.
      // B scales are already in hardware from the j0-level load above, but the
      // spec requires both to be issued before loop_ws_spad.  We reload A here
      // and re-issue B to satisfy the ordering requirement without re-copying
      // the B scale buffer (it was set up at j0 entry and hasn't changed).
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Mvin A tiles for this i0 via port 0.
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));  // stride = full K row
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // Configure output / requantizer.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Fused tiled matmul — funct 24 marker is one-shot, re-issued inside
      // gemmini_loop_ws_spad before every invocation.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM, one row at a time.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
```
