Looking at the original code, I need to understand what it does:
- Computes a 256×256×512 matrix multiplication tiled over BLOCKS_M × BLOCKS_N output blocks
- Each block is BLK×BLK output, using FP8 inputs with BF16 output
- The main inefficiency is the sequential nature: for each (i0,j0) block, it does full config, scale load, mvin, compute, readout

The plan mentions "max retries reached" which suggests previous optimization attempts failed. I'll focus on correctness while applying safe optimizations:

1. **Avoid redundant `gemmini_extended3_config_ex`** - config persists, only issue once outside the loop
2. **Avoid redundant `gemmini_flush`** - only needed once at start
3. **Fix B tile layout** - the original B mvin loop has `(k*tiles + j)` but should be `(j*tiles_K + k)` per the spec
4. **Keep the programming model sequence correct** per the rules

Let me trace the B layout: spec says `b_base + (j*tiles_K + k)*DIM`. Original uses `(k*tiles + j)*DIM`. These differ — `k*tiles + j` vs `j*tiles_K + k`. With `tiles = tiles_K` only if BLK==MATMUL_K, but here tiles=BLK/DIM and tiles_K=MATMUL_K/DIM which differ. I'll fix this.

Also note `gemmini_config_ld` for B: B is stored as K×N (row=k*DIM, col=j0*BLK+j*DIM), stride should be `MATMUL_N * sizeof(elem_t)` which the original has correctly.

void solution(void) {
  // One-time hardware configuration — config fields persist until changed.
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  const int tiles   = BLK / DIM;          // tiles along I and J within a block
  const int tiles_K = MATMUL_K / DIM;     // tiles along K
  const uint32_t a_base   = 0;
  const uint32_t b_base   = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  const int      SPAD_DEST = 128;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      // --- Stage per-block scale factors (cpu) ---
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // --- Load scale factors (must precede mvin and compute) ---
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  // sel=0 → A scales
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  // sel=1 → B scales

      // --- MVIN A tiles: layout a_base + (i*tiles_K + k)*DIM ---
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // --- MVIN B tiles: layout b_base + (j*tiles_K + k)*DIM ---
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // --- Configure requantized mvout ---
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // --- Fused hardware matmul (funct 24 one-shot marker + funct 8) ---
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           0, 0, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // --- Drain BF16 results from mx_smem to DRAM ---
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      // --- Unpack BF16 results to output array ---
      for (int r = 0; r < BLK; r++) {
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
      }
    }
  }
}
