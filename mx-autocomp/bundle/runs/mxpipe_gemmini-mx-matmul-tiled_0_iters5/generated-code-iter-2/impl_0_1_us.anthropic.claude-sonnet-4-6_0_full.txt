Looking at the original code, I need to understand what it does and why it might be failing:

1. The original processes a 256×256×512 matmul in BLK×BLK blocks
2. It uses `memcpy` to stage scales, then loads tiles into scratchpad
3. The B matrix mvin loop has a bug: `b_base + (k * tiles + j) * DIM` - the layout should be `(j * tiles_K + k) * DIM` per the spec

Key issues to fix:
- B tile layout should be `b_base + (j * tiles_K + k) * DIM` not `(k * tiles + j) * DIM`
- The scratchpad/accumulator sizes in the problem statement say 256KB/64KB which means BANK_ROWS=4096 (not 2048), but rule 7 explicitly states BANK_ROWS=2048, BANK_NUM=4, so I must use those
- Verify scratchpad fits: `(tiles_I*tiles_K + tiles_J*tiles_K)*DIM ≤ 8192`
  - tiles=BLK/DIM, tiles_K=MATMUL_K/DIM. For BLK=32, DIM=16: tiles=2, tiles_K=32. `(2*32 + 2*32)*16 = 2048 ≤ 8192` ✓
- The `gemmini_extended_mvin2` should be used for B (mvin2 port) to overlap with A loads
- Use `gemmini_extended_mvin` for A and `gemmini_extended_mvin2` for B

```c
void solution(void) {
  int tiles    = BLK / DIM;
  int tiles_K  = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST   = 128;

  gemmini_flush(0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      /* ---- stage per-block scales into contiguous scratch buffers ---- */
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      /* 1. configure execution (fp8 in, bf16 out, weight-stationary) */
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false,
                                  /*act_mx_fmt*/0, /*wgt_mx_fmt*/0,
                                  /*out_mx_fmt*/3, /*uselut*/0);

      /* 2. load scale factors: sel=0 → A (activation), sel=1 → B (weight) */
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      /* 4a. configure and mvin A tiles (stride = K columns) */
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      /* 4b. configure and mvin B tiles via port 2 (stride = N columns) */
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      /* 5. configure requantized mvout */
      gemmini_extended_config_st(BLK_OUTC * sizeof(out_t), 0, ACC_SCALE_IDENTITY);
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      /* 6. fused weight-stationary compute (one-shot marker + loop) */
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           0, 0, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      /* 7. drain BF16 results from mx_smem to DRAM */
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      /* cpu_store: unpack 4×BF16 per uint64 word into C_acc */
      for (int r = 0; r < BLK; r++) {
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
      }
    }
  }
}
```
