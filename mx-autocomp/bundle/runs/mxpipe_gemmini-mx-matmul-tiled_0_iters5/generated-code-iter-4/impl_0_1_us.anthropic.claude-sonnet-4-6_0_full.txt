Looking at the original code, I need to understand what it does:
- It performs a 256×256×512 (MATMUL_M×MATMUL_N×MATMUL_K) matmul using FP8 MX format
- It tiles over BLOCKS_M × BLOCKS_N output blocks, each of size BLK×BLK
- The main bottleneck areas are the per-block memcpy for scales and the sequential mvin loops

Key optimizations I can make:
1. **Load scales directly from original arrays** without memcpy (the scales are already in DRAM)
2. **Batch mvin commands** using `gemmini_extended3_config_ld` with `block_mvin_stride` and larger block lengths (up to MAX_BLOCK_LEN=4)
3. **Use mvin2 for B tiles** to overlap A and B loads (separate ports)
4. **Eliminate the gemmini_flush** per iteration (only needed once at start)
5. **Avoid redundant config_ex re-issue** across iterations since format doesn't change
6. **Increase lut_update_granularity** since we're doing BF16 output (no LUT needed, but G controls requant refresh)

Let me check scratchpad constraints:
- tiles = BLK/DIM, tiles_K = MATMUL_K/DIM
- For BLK=32: tiles=2, tiles_K=32, A needs 2*32*16=1024 rows, B needs 2*32*16=1024 rows, total 2048 ≤ 8192 ✓

The main win is removing the per-block memcpy for scales by pointing directly to the DRAM scale arrays with the correct offset.

```c
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base   = 0;
  uint32_t b_base   = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  // Issue config_ex once — format fields are constant across all blocks.
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  // Pre-configure load strides once each (A stride = K cols, B stride = N cols).
  // We will re-issue them only when switching between A and B loads.
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      // ----------------------------------------------------------------
      // 1. Load scales directly from DRAM — no memcpy needed.
      //    A_scales_row[g][m]  layout: row-major, dim [MATMUL_K/32][MATMUL_M]
      //    B_scales_col[g][n]  layout: row-major, dim [MATMUL_K/32][MATMUL_N]
      //    We need the BLK-wide slice for this (i0,j0) block.
      //    Since the hardware streams 'len' bytes starting at the given
      //    address, and the scales for this block are NOT contiguous in
      //    the original 2-D arrays, we still need a staging buffer for
      //    the scale slice.  However we fuse the copy into a single pass
      //    and avoid the separate A_s/B_s globals by using stack arrays.
      // ----------------------------------------------------------------
      uint8_t a_scales_blk[MATMUL_K / 32][BLK];
      uint8_t b_scales_blk[MATMUL_K / 32][BLK];

      int GK = MATMUL_K / 32;
      for (int g = 0; g < GK; g++) {
        memcpy(&a_scales_blk[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&b_scales_blk[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      gemmini_mx_load_scales((uint64_t)a_scales_blk,
                             (uint64_t)(GK * BLK * sizeof(uint8_t)), 0);
      gemmini_mx_load_scales((uint64_t)b_scales_blk,
                             (uint64_t)(GK * BLK * sizeof(uint8_t)), 1);

      // ----------------------------------------------------------------
      // 2. MVIN A tiles — stride = MATMUL_K bytes (one FP8 row of A).
      //    Use mvin port 0 (gemmini_extended_mvin).
      //    Batch up to MAX_BLOCK_LEN=4 consecutive k-tiles per command.
      // ----------------------------------------------------------------
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t),
                                  MVIN_SCALE_IDENTITY, false, 0);

      for (int i = 0; i < tiles; i++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;

          // Each batch issues one mvin covering 'batch' consecutive k-tiles
          // in the same i-row.  The DRAM pointer advances by batch*DIM cols,
          // and the spad address advances by batch*DIM rows.
          // gemmini_extended_mvin transfers (cols=batch*DIM, rows=DIM).
          const elem_t *a_ptr =
              ((const elem_t *)A_in) +
              (i0 * BLK + i * DIM) * MATMUL_K + k * DIM;
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;

          gemmini_extended_mvin((void *)a_ptr, spad_addr,
                                batch * DIM, DIM);
          k += batch;
        }
      }

      // ----------------------------------------------------------------
      // 3. MVIN B tiles — stride = MATMUL_N bytes (one FP8 row of B).
      //    Use mvin2 port (gemmini_extended_mvin2) so A and B loads can
      //    be pipelined by the DMA scheduler.
      //    B layout in DRAM: B_in[k*DIM .. k*DIM+DIM-1][j0*BLK+j*DIM .. +DIM]
      //    B layout in spad: b_base + (k*tiles + j)*DIM
      //    We batch consecutive j-tiles for fixed k (j varies faster).
      // ----------------------------------------------------------------
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t),
                                  MVIN_SCALE_IDENTITY, false, 1);

      for (int k = 0; k < tiles_K; k++) {
        int j = 0;
        while (j < tiles) {
          int batch = tiles - j;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;

          const elem_t *b_ptr =
              ((const elem_t *)B_in) +
              (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (k * tiles + j) * DIM;

          gemmini_extended_mvin2((void *)b_ptr, spad_addr,
                                 batch * DIM, DIM);
          j += batch;
        }
      }

      // ----------------------------------------------------------------
      // 4. Configure requantized mvout and execute the fused loop.
      // ----------------------------------------------------------------
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, tiles_K);

      gemmini_loop_ws_spad(tiles, tiles, tiles_K, 0, 0, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // ----------------------------------------------------------------
      // 5. Drain BF16 results from mx_smem to C_blk, then to C_acc.
      // ----------------------------------------------------------------
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      for (int r = 0; r < BLK; r++)
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
    }
  }
}
```
