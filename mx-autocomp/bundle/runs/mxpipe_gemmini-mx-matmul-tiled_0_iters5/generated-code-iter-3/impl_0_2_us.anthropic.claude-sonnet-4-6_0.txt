void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  // Scratchpad layout (fixed across all outer tiles)
  uint32_t a_base = 0;
  uint32_t b_base = (uint32_t)(BANK_NUM * BANK_ROWS) - (uint32_t)(tiles * tiles_K * DIM);
  int SPAD_DEST   = 128;

  // One-time hardware configuration — format fields are constant
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      // --- Stage per-block e8m0 scales into host-side buffers ---
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // --- Load scale factors (A=sel 0, B=sel 1) ---
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // --- MVIN A tiles: A[i0*BLK + i*DIM .. +DIM][k*DIM .. +DIM] ---
      // A is stored row-major with row stride = MATMUL_K elements
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *a_ptr = ((const elem_t *)A_in)
                                + (size_t)(i0 * BLK + i * DIM) * MATMUL_K
                                + (size_t)(k * DIM);
          uint32_t spad_row = a_base + (uint32_t)((i * tiles_K + k) * DIM);
          gemmini_extended_mvin((void *)a_ptr, spad_row, DIM, DIM);
        }
      }

      // --- MVIN B tiles via mvin2: B[k*DIM .. +DIM][j0*BLK + j*DIM .. +DIM] ---
      // B is stored row-major with row stride = MATMUL_N elements
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *b_ptr = ((const elem_t *)B_in)
                                + (size_t)(k * DIM) * MATMUL_N
                                + (size_t)(j0 * BLK + j * DIM);
          uint32_t spad_row = b_base + (uint32_t)((j * tiles_K + k) * DIM);
          gemmini_extended_mvin2((void *)b_ptr, spad_row, DIM, DIM);
        }
      }

      // --- Configure output stride and requantizer ---
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles, tiles, tiles_K,
          0, 0, 1);

      // --- Fused weight-stationary tiled matmul (one-shot marker re-issued inside macro) ---
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
          0, 0, 0,
          a_base, (uint32_t)(BANK_NUM * BANK_ROWS), 0, SPAD_DEST,
          false, false,
          false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // --- Drain BF16 results from mx_smem to DRAM ---
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      // --- Unpack BF16 words into float output matrix ---
      for (int r = 0; r < BLK; r++) {
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
      }
    }
  }
}
