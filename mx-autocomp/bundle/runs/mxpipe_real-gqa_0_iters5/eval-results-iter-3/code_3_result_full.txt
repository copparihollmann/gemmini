                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);

      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J_v, tiles_K, 0, 0, 1);

      gemmini_fence();
