An elegant and highly effective optimization strategy for this CPU-bound bottleneck is **Strategy 2: Loop Unrolling**.

### Optimization Plan

The current MHA kernel has a high latency of **786,953 cycles**, heavily bounded by the host CPU executing the sequential float-free integer softmax. We can target this inefficiency by applying aggressive loop unrolling:

1. **Outer Loop Unrolling (Attention Heads)**:
   Since the number of attention heads `ATT_H` is exactly 4 and `ATT_HKV` is 2, the head loop runs exactly 4 iterations. We will **fully unroll the head loop `h` (from 0 to 3)**. This achieves the following:
   * Eliminates the dynamic `kv != last_kv` state tracking and conditional branch checks completely.
   * Statically schedules the $K^T$ and $V$ matrix scratchpad loading. We hardcode $K^T$ and $V$ loading only on $h=0$ (for $kv=0$) and $h=2$ (for $kv=1$). On $h=1$ and $h=3$, we skip loading entirely, reusing the cached matrices already in the scratchpad.

2. **Inner Softmax Loop Unrolling**:
   The float-free integer softmax on the CPU is highly sequential and has multiple nested loop levels. We will unroll these inner loops:
   * **Unroll the `l` loop fully** (from 0 to 3) to process 4 packed `BF16` elements in parallel, eliminating division/modulo loop indexing.
   * **Unroll the `g` loop over `GROUPS_S`** (since `GROUPS_S` is exactly 2).
   * **Unroll the `j` loop by a factor of 4** (using step 4) in both the exponentiation phase and the integer encoding phase, which dramatically reduces host branching overhead and exposes instruction-level parallelism (ILP) for CPU registers.

#include <stdint.h>
#include <stdio.h>
#include <string.h>

// Fully unrolled inner loops of float-free softmax to accelerate host execution
static inline void run_softmax_unrolled(uint64_t S1_local[ATT_S][ATT_S / 4], uint8_t P_q_local[ATT_S][ATT_S], uint8_t P_scales_local[GROUPS_S][ATT_S]) {
    for (int i = 0; i < ATT_S; i++) {
        int64_t q[ATT_S];
        int64_t mq = -((int64_t)1 << 62);
        
        // Unroll j loop (step 4, each step processes 4 elements via l unrolling)
        for (int j = 0; j < ATT_S; j += 4) {
            uint64_t w = S1_local[i][j / 4];
            
            // Fully unrolled l from 0 to 3 with compile-time ATT_CAUSAL support
#if ATT_CAUSAL
            if (j > i) { q[j] = -((int64_t)1 << 62); } else
#endif
            {
                int64_t qq0 = mx_ibf16_to_q((uint16_t)(w & 0xFFFF));
                q[j] = qq0; if (qq0 > mq) mq = qq0;
            }
            
#if ATT_CAUSAL
            if (j + 1 > i) { q[j + 1] = -((int64_t)1 << 62); } else
#endif
            {
                int64_t qq1 = mx_ibf16_to_q((uint16_t)((w >> 16) & 0xFFFF));
                q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
            }
            
#if ATT_CAUSAL
            if (j + 2 > i) { q[j + 2] = -((int64_t)1 << 62); } else
#endif
            {
                int64_t qq2 = mx_ibf16_to_q((uint16_t)((w >> 32) & 0xFFFF));
                q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
            }
            
#if ATT_CAUSAL
            if (j + 3 > i) { q[j + 3] = -((int64_t)1 << 62); } else
#endif
            {
                int64_t qq3 = mx_ibf16_to_q((uint16_t)((w >> 48) & 0xFFFF));
                q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
            }
        }
        
        int64_t p[ATT_S];
        int64_t sm = 0;
        int64_t gmax_arr[2]; // GROUPS_S = 2
        
        // Unroll g = 0
        {
            int64_t gmax = 0;
            for (int j = 0; j < 32; j += 4) {
                int64_t pv0 = mx_iexp(q[j] - mq);     p[j] = pv0;     sm += pv0; if (pv0 > gmax) gmax = pv0;
                int64_t pv1 = mx_iexp(q[j + 1] - mq); p[j + 1] = pv1; sm += pv1; if (pv1 > gmax) gmax = pv1;
                int64_t pv2 = mx_iexp(q[j + 2] - mq); p[j + 2] = pv2; sm += pv2; if (pv2 > gmax) gmax = pv2;
                int64_t pv3 = mx_iexp(q[j + 3] - mq); p[j + 3] = pv3; sm += pv3; if (pv3 > gmax) gmax = pv3;
            }
            gmax_arr[0] = gmax;
        }
        
        // Unroll g = 1
        {
            int64_t gmax = 0;
            for (int j = 32; j < 64; j += 4) {
                int64_t pv0 = mx_iexp(q[j] - mq);     p[j] = pv0;     sm += pv0; if (pv0 > gmax) gmax = pv0;
                int64_t pv1 = mx_iexp(q[j + 1] - mq); p[j + 1] = pv1; sm += pv1; if (pv1 > gmax) gmax = pv1;
                int64_t pv2 = mx_iexp(q[j + 2] - mq); p[j + 2] = pv2; sm += pv2; if (pv2 > gmax) gmax = pv2;
                int64_t pv3 = mx_iexp(q[j + 3] - mq); p[j + 3] = pv3; sm += pv3; if (pv3 > gmax) gmax = pv3;
            }
            gmax_arr[1] = gmax;
        }
        
        int ls = mx_ilog2((uint64_t)sm);
        uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
        
        // Unroll g = 0 encoding
        {
            int64_t gmax = gmax_arr[0];
            if (gmax == 0) {
                P_scales_local[0][i] = 0;
                for (int j = 0; j < 32; j += 4) {
                    P_q_local[i][j] = 0; P_q_local[i][j + 1] = 0; P_q_local[i][j + 2] = 0; P_q_local[i][j + 3] = 0;
                }
            } else {
                int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
                P_scales_local[0][i] = (uint8_t)(e + 127);
                for (int j = 0; j < 32; j += 4) {
                    P_q_local[i][j]     = mx_ienc_fp8(p[j], inv_sm, ls, e);
                    P_q_local[i][j + 1] = mx_ienc_fp8(p[j + 1], inv_sm, ls, e);
                    P_q_local[i][j + 2] = mx_ienc_fp8(p[j + 2], inv_sm, ls, e);
                    P_q_local[i][j + 3] = mx_ienc_fp8(p[j + 3], inv_sm, ls, e);
                }
            }
        }
        
        // Unroll g = 1 encoding
        {
            int64_t gmax = gmax_arr[1];
            if (gmax == 0) {
                P_scales_local[1][i] = 0;
                for (int j = 32; j < 64; j += 4) {
                    P_q_local[i][j] = 0; P_q_local[i][j + 1] = 0; P_q_local[i][j + 2] = 0; P_q_local[i][j + 3] = 0;
                }
            } else {
                int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
                P_scales_local[1][i] = (uint8_t)(e + 127);
                for (int j = 32; j < 64; j += 4) {
                    P_q_local[i][j]     = mx_ienc_fp8(p[j], inv_sm, ls, e);
                    P_q_local[i][j + 1] = mx_ienc_fp8(p[j + 1], inv_sm, ls, e);
                    P_q_local[i][j + 2] = mx_ienc_fp8(p[j + 2], inv_sm, ls, e);
                    P_q_local[i][j + 3] = mx_ienc_fp8(p[j + 3], inv_sm, ls, e);
                }
            }
        }
    }
}
