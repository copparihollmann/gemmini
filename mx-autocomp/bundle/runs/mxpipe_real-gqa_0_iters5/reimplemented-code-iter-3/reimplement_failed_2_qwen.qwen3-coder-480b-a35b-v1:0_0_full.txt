Looking at the error message, the issue is that `cols=64` overflows the 6-bit hardware field, which means we're trying to load too many columns at once. The maximum block length is 4 tiles (64 bytes = 4 * 16 bytes), but we need to ensure `cols` doesn't exceed 63.

Let me analyze the problematic sections:

1. In the first matmul (Q @ K^T): `gemmini_config_ld(ATT_D * sizeof(elem_t))` where ATT_D = 64, so stride is 64 bytes
2. When loading tiles with `bb * DIM` where `bb` can be up to `MAX_BLOCK_LEN = 4`, giving `4 * 16 = 64` columns

The issue is that 64 columns exceeds the 6-bit field limit (max 63). We need to reduce the block batch size.

```c
// AUTO-GENERATED baseline MHA kernel (S=64, D=64, H=4, H_kv=2, causal=0, fp8:e4m3).
void solution(void) {
  const int g_kv = ATT_H / ATT_HKV; // = 2, group size for KV sharing

  // Pre-calculate persistent base addresses for K^T and V
  const int kt_tiles_K = ATT_D / DIM;  // 4
  const int kt_tiles_J = ATT_S / DIM;  // 4
  const int v_tiles_K  = ATT_S / DIM;  // 4
  const int v_tiles_J  = ATT_D / DIM;  // 4
  const int kt_size_rows = kt_tiles_K * kt_tiles_J * DIM; // 4*4*16 = 256
  const int v_size_rows  = v_tiles_K  * v_tiles_J  * DIM; // 4*4*16 = 256

  const uint32_t b_base_KT_persistent = BANK_NUM * BANK_ROWS - kt_size_rows;         // 8192 - 256 = 7936
  const uint32_t b_base_V_persistent  = b_base_KT_persistent - v_size_rows;          // 7936 - 256 = 7680
  // Q/P base remains at 0
  const uint32_t a_base_persistent = 0;

  for (int h = 0; h < ATT_H; h++) {
    int kv = h / g_kv; // Integer division correctly maps h to its KV group
    const elem_t *Q_in = Qs[h];         const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv];      const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];        const uint8_t *V_sc = Vss[kv];

    const int first_of_group = (h % g_kv == 0);

    // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
      uint32_t a_base = a_base_persistent;
      uint32_t b_base = b_base_KT_persistent; // Use persistent KT base

      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0); // A scales for Q
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1); // B scales for KT
      
      gemmini_config_ld(ATT_D * sizeof(elem_t)); // Stride for Q [M][K]
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k += 3) { // MAX_BLOCK_LEN reduced to 3 to avoid cols=64 overflow
          int bb = (tiles_K - k < 3) ? (tiles_K - k) : 3; // Max 3 tiles per block to keep cols <= 48
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      
      // Only load K^T if this is the first head in the group sharing it
      if (first_of_group) {
          gemmini_config_ld(ATT_S * sizeof(elem_t)); // Stride for KT [K][N]
          for (int k = 0; k < tiles_K; k++)
            for (int j = 0; j < tiles_J; j += 3) { // MAX_BLOCK_LEN reduced to 3
              int bb = (tiles_J - j < 3) ? (tiles_J - j) : 3;
              gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                    b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
            }
      }

      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      
      // Pass the END address for B (which is BANK_NUM*BANK_ROWS for the top of spad)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      for (int j = 0; j < ATT_S; j += 4) {                       
        uint64_t w = S1_hw[i][j / 4];
        for (int l = 0; l < 4; l++) {
#if ATT_CAUSAL
          if (j + l > i) { q[j + l] = -((int64_t)1 << 62); continue; }   
#endif
          int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
          q[j + l] = qq; if (qq > mq) mq = qq;
        }
      }
      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];                                
      for (int g = 0; g < GROUPS_S; g++) {                       
        int64_t gmax = 0;                                        
        for (int j = g * 32; j < (g + 1) * 32; j++) { int64_t pv = mx_iexp(q[j] - mq); p[j] = pv; sm += pv; if (pv > gmax) gmax = pv; }
        gmax_arr[g] = gmax;
      }
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;   
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) { P_scales[g][i] = 0; for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0; continue; }
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);
        for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
      }
    }

    // ---- 3) O = P @ V on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
      uint32_t a_base = a_base_persistent;
      uint32_t b_base = b_base_V_persistent; // Use persistent V base

      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0); // A scales for P
      gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);             // B scales for V
      
      gemmini_config_ld(ATT_S * sizeof(elem_t)); // Stride for P [M][K]
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k += 3) { // MAX_BLOCK_LEN reduced to 3
          int bb = (tiles_K - k < 3) ? (tiles_K - k) : 3;
          gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      
      // Only load V if this is the first head in the group sharing it
      if (first_of_group) {
          gemmini_config_ld(ATT_D * sizeof(elem_t)); // Stride for V [K][N]
          for (int k = 0; k < tiles_K; k++)
            for (int j = 0; j < tiles_J; j += 3) { // MAX_BLOCK_LEN reduced to 3
              int bb = (tiles_J - j < 3) ? (tiles_J - j) : 3;
              gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                    b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
            }
      }

      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      
      // Pass the END address for B (which is b_base_KT_persistent, start of KT tiles)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, b_base_KT_persistent, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
  }
}
```
