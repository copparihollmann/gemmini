### Optimization Plan: Loop Unrolling

We will apply **loop unrolling** to eliminate CPU-side branching, instruction overhead, and loop control structures, which represent the main performance bottleneck in the functional simulator.

1. **Unroll the Head Loop ($H=4$):** 
   Since the number of attention heads is small ($H=4$) and the key-value heads group size is fixed ($H_{kv}=2$), we can completely unroll the outer loop `for (int h = 0; h < ATT_H; h++)`. This removes the dynamic tracking of `last_kv` and the branch overhead of `if (kv != last_kv)`. The cache loading of $K^T$ and $V$ is statically resolved:
   - For $h=0$ (first head of $kv=0$), we statically stream $K^T[0]$ and $V[0]$.
   - For $h=1$, we skip loading $K^T$ and $V$ entirely (reusing cached scratchpad).
   - For $h=2$ (first head of $kv=1$), we statically stream $K^T[1]$ and $V[1]$.
   - For $h=3$, we skip loading.

2. **Unroll the Softmax Loop Nest:**
   The CPU-side integer softmax performs unpacking, maximum extraction, and requantization. We can fully unroll and inline these operations:
   - Replace variable-shifted unpacking (`w >> (l * 16)`) with four static 16-bit masks/shifts, avoiding ALU-intensive dynamic shifts on the CPU.
   - Unroll the reduction groups (`GROUPS_S = 2`), executing Group 0 and Group 1 computations explicitly without loop counter updates.

3. **Unroll MVIN Tile Loops:**
   Replace the nested loop structures for loading $Q$, $P_q$, $K^T$, and $V$ with flat, contiguous sequences of `gemmini_extended_mvin` calls. This eliminates nested loop structures, address calculation ALU operations, and array offsets at runtime.

    uint8_t P_scales[GROUPS_S][ATT_S]
) {
    for (int i = 0; i < ATT_S; i++) {
        int64_t q[ATT_S];
        int64_t mq = -((int64_t)1 << 62);
        
        // Unrolled unpacking of S1_hw (4 bf16 values per uint64_t)
        for (int j = 0; j < ATT_S; j += 4) {
            uint64_t w = S1_hw[i][j / 4];
            uint16_t w0 = w & 0xFFFF;
            uint16_t w1 = (w >> 16) & 0xFFFF;
            uint16_t w2 = (w >> 32) & 0xFFFF;
            uint16_t w3 = (w >> 48);
            
            int64_t qq0 = mx_ibf16_to_q(w0);
            int64_t qq1 = mx_ibf16_to_q(w1);
            int64_t qq2 = mx_ibf16_to_q(w2);
            int64_t qq3 = mx_ibf16_to_q(w3);
            
            q[j + 0] = qq0; if (qq0 > mq) mq = qq0;
            q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
            q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
            q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
        }
        
        int64_t p[ATT_S];
        int64_t sm = 0;
        int64_t gmax_arr[GROUPS_S];
        
        // Unrolled Group 0 Reduction
        {
            int64_t gmax = 0;
            for (int j = 0; j < 32; j++) {
                int64_t pv = mx_iexp(q[j] - mq);
                p[j] = pv;
                sm += pv;
                if (pv > gmax) gmax = pv;
            }
            gmax_arr[0] = gmax;
        }
        
        // Unrolled Group 1 Reduction
        {
            int64_t gmax = 0;
            for (int j = 32; j < 64; j++) {
                int64_t pv = mx_iexp(q[j] - mq);
                p[j] = pv;
                sm += pv;
                if (pv > gmax) gmax = pv;
            }
            gmax_arr[1] = gmax;
        }

        uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

        {
            int64_t gmax = gmax_arr[0];
            if (gmax == 0) {
                P_scales[0][i] = 0;
                for (int j = 0; j < 32; j++) P_q[i][j] = 0;
            } else {
                int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
                P_scales[0][i] = (uint8_t)(e + 127);
                for (int j = 0; j < 32; j++) {
                    P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
                }
            }
        }
        
        // Unrolled Requantization for Group 1
        {
            int64_t gmax = gmax_arr[1];
            if (gmax == 0) {
                P_scales[1][i] = 0;
                for (int j = 32; j < 64; j++) P_q[i][j] = 0;

                }
            }
        }
    }

            int64_t gmax = gmax_arr[0];
            if (gmax == 0) {
                P_scales[0][i] = 0;
                for (int j = 0; j < 32; j++) P_q[i][j] = 0;
            } else {
                int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
                P_scales[0][i] = (uint8_t)(e + 127);
                for (int j = 0; j < 32; j++) {
                    P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
                }
            }
        }
        
        // Unrolled Requantization for Group 1
        {
            int64_t gmax = gmax_arr[1];
            if (gmax == 0) {
                P_scales[1][i] = 0;
                for (int j = 32; j < 64; j++) P_q[i][j] = 0;
            } else {
                int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
                P_scales[1][i] = (uint8_t)(e + 127);
                for (int j = 32; j < 64; j++) {
                    P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
                }
            }
        }
    }
}
