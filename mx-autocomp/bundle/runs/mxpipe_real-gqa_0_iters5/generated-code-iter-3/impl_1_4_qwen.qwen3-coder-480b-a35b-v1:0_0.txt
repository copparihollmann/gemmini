// High-performance MHA kernel (S=64, D=64, H=4, H_kv=2, causal=0, fp8:e4m3).
void solution(void) {
  int last_kv = -1;

  // === Hoisted format-invariant configurations ===
  // These settings are constant throughout execution and can be safely issued once at startup.
  gemmini_extended3_config_ex(
    WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
    /*C_stride=*/1, /*A_stride=*/1,
    /*A_transpose=*/0, /*B_transpose=*/0,
    /*set_only_strides=*/false,
    /*act_mx_fmt=*/0, /*wgt_mx_fmt=*/0, /*out_mx_fmt=*/3, /*uselut=*/0
  );

  for (int h = 0; h < ATT_H; h++) {
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];  const uint8_t *V_sc = Vss[kv];

    int tiles_I = ATT_S / DIM;
    int tiles_J = ATT_S / DIM;
    int tiles_K = ATT_D / DIM;

    uint32_t a_base = 0;
    // Partition the high-address scratchpad region for persistent B tiles
    uint32_t b_base_KT = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 8192 - 256 = 7936
    uint32_t b_base_V = b_base_KT - tiles_K * tiles_J * DIM;             // 7936 - 256 = 7680

    // ---- 1) Hoist and reuse K^T and V tiles across heads sharing the same KV group ----
    if (kv != last_kv) {
      // Stream KT once per kv group
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) {
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                b_base_KT + (k * tiles_J + j) * DIM, bb * DIM, DIM);
        }
      }

      // Stream V once per kv group
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) {
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                b_base_V + (k * tiles_J + j) * DIM, bb * DIM, DIM);
        }
      }
      last_kv = kv;
    }

    // ---- 2) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      gemmini_flush(0);
      
      // Configuration moved to top; omitted here
      
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0); // A scales
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1); // B scales

      // Load activation (Q)
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) {
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);
        }
      }

      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      
      // Compute using KT residing in top high-address scratchpad (8192 end address)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 3) FLOAT-FREE integer softmax + fp8 requant ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      for (int j = 0; j < ATT_S; j += 4) {
        uint64_t w = S1_hw[i][j / 4];
        for (int l = 0; l < 4; l++) {
#if ATT_CAUSAL
          if (j + l > i) { q[j + l] = -((int64_t)1 << 62); continue; }
#endif
          int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
          q[j + l] = qq; if (qq > mq) mq = qq;
        }
      }
      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) {
          int64_t pv = mx_iexp(q[j] - mq);
          p[j] = pv;
          sm += pv;
          if (pv > gmax) gmax = pv;
        }
        gmax_arr[g] = gmax;
      }
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) {
          P_scales[g][i] = 0;
          for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0;
          continue;
        }
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);
        for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
      }
    }

    // ---- 4) O = P @ V on Gemmini (bf16 out) ----
    {
      int tiles_J_v = ATT_D / DIM;
      gemmini_flush(0);

      // Configuration moved to top; omitted here

      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0); // A scales
      gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);              // B scales

      // Load activation (P_q)
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) {
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);
        }
      }

      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J_v, tiles_K, 0, 0, 1);

      // Compute using V residing in lower high-address scratchpad (b_base_KT/7936 end address)
      gemmini_loop_ws_spad(tiles_I, tiles_J_v, tiles_K, 0, 0, 0, a_base, b_base_KT, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
  }
}
