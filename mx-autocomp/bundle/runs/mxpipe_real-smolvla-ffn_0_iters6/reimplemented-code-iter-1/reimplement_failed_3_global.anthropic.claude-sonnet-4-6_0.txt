void solution(void) {
  // Move flush and config outside the loop - only needed once
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Use BLK=64 so tiles=4, accumulator holds exactly 4*4*16=256 rows
  const int TILED_BLK = 64;
  const int tiles = TILED_BLK / DIM;         // 4
  const int tiles_K = MATMUL_K / DIM;
  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  const int SPAD_DEST = 128;

  const int blocks_m = (REAL_M + TILED_BLK - 1) / TILED_BLK;
  const int blocks_n = (REAL_N + TILED_BLK - 1) / TILED_BLK;

  for (int i0 = 0; i0 < blocks_m; i0++) {
    for (int j0 = 0; j0 < blocks_n; j0++) {
      int mrows = REAL_M - i0 * TILED_BLK;
      if (mrows > TILED_BLK) mrows = TILED_BLK;
      if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * TILED_BLK;
      if (ncols > TILED_BLK) ncols = TILED_BLK;
      if (ncols <= 0) continue;

      // Copy scales for this block
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * TILED_BLK], TILED_BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * TILED_BLK], TILED_BLK);
      }

      int pad_i = TILED_BLK - mrows;
      int pad_j = TILED_BLK - ncols;

      // Step 2: Load scales (A sel=0, B sel=1)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN A tiles into scratchpad (port 0 only for loop_ws_spad)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * TILED_BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // MVIN B tiles into scratchpad (port 0 only - required for loop_ws_spad)
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * TILED_BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM, DIM, DIM);

      // Step 5: Configure requantized mvout
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles, tiles, tiles_K, 0, 0, 1);

      // Step 6: Compute
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain results from mx_smem to DRAM
      if (mrows == TILED_BLK && ncols == TILED_BLK) {
        gemmini_mx_read_smem(&C_hw[i0 * TILED_BLK][j0 * TILED_BLK / BF16_PER_WORD],
                             SPAD_DEST * 16, TILED_BLK * TILED_BLK);
      } else {
        for (int r = 0; r < mrows; r++)
          gemmini_mx_read_smem(&C_hw[i0 * TILED_BLK + r][(j0 * TILED_BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * TILED_BLK, ncols);
      }
    }
  }

  // Fence after all mx_read_smem operations
  gemmini_fence();
}
