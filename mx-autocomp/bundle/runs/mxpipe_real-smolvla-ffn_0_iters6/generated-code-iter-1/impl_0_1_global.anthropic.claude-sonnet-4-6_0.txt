void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  /* One-time pipeline flush and format configuration */
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    int pad_i = BLK - mrows;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_j = BLK - ncols;

      /* ── Load A and B scale factors directly from source arrays (no memcpy) ── */
      /* A scales: rows are indexed by [g][row], contiguous along the row dim    */
      /* Pass pointer to first element of this row-block's column slice           */
      /* A_scales_row layout: [MATMUL_K/32][REAL_M], stride REAL_M per g-group   */
      /* We load BLK entries per g-group; they are contiguous in memory           */
      gemmini_mx_load_scales((uint64_t)(&A_scales_row[0][i0 * BLK]),
                             A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)(&B_scales_col[0][j0 * BLK]),
                             B_S_BYTES, 1);

      /* ── MVIN A tiles: layout a_base + (i*tiles_K + k)*DIM ── */
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      /* ── MVIN B tiles: layout b_base + (k*tiles + j)*DIM ── */
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM,
            DIM, DIM);

      /* ── Configure output / requantizer ── */
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      /* ── Execute fused tiled matmul ── */
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0,
                           false, 0x38);

      /* ── Drain BF16 results from mx_smem to DRAM ── */
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
