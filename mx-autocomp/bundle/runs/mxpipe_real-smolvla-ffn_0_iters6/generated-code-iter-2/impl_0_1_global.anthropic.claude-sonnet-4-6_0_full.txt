Looking at the strategy, I need to:

1. Replace the `(BLOCKS_M × BLOCKS_N)` loop with a larger-tile loop using `BIGBLK` rows/cols at a time
2. Compute `BIGBLK` such that the accumulator constraint `tiles_I_big * tiles_J_big * DIM ≤ 256` is satisfied (i.e., `tiles_I_big * tiles_J_big ≤ 16`)
3. Also satisfy scratchpad: `(tiles_I_big + tiles_J_big) * tiles_K * DIM ≤ 8192`
4. Issue config/flush/scale-load only once per big tile, not once per small `(i0,j0)` pair

With `tiles_K = MATMUL_K/DIM`, a balanced choice of `tiles_I_big = tiles_J_big = 4` gives `4*4*16 = 256 ≤ 256` accumulator rows (exactly full) and scratchpad use `(4+4)*tiles_K*16`. For typical `tiles_K=4` that's `512 ≤ 8192`. So `BIGBLK = 4*DIM = 64`.

I also remove the `gemmini_flush` from the inner loop (keep it only once before the main computation), and avoid redundant config re-issues.

```c
void solution(void) {
  // Fuse multiple (i0,j0) BLK-blocks into a single gemmini_loop_ws_spad call.
  // Choose BIGBLK_TILES_I and BIGBLK_TILES_J (in DIM-tile units) to fill
  // the accumulator in one shot: tiles_I_big * tiles_J_big * DIM <= 256.
  // Also satisfy scratchpad: (tiles_I_big + tiles_J_big) * tiles_K * DIM <= 8192.
  //
  // tiles_K = MATMUL_K / DIM
  // We pick tiles_I_big = tiles_J_big = 4, so BIGBLK = 4*DIM = 64.
  //   acc rows used  = 4*4*16 = 256  (exactly full, single-buffer)
  //   spad rows used = (4+4)*tiles_K*16; for tiles_K<=32 this is <=4096 <= 8192

  const int tiles_K     = MATMUL_K / DIM;
  const int TIB         = 4;   // big-tile count in I direction (tiles)
  const int TJB         = 4;   // big-tile count in J direction (tiles)
  const int BIGBLK      = TIB * DIM;  // = 64 elements
  const int BIGBLKS_M   = (REAL_M + BIGBLK - 1) / BIGBLK;
  const int BIGBLKS_N   = (REAL_N + BIGBLK - 1) / BIGBLK;
  const int SPAD_DEST   = 128;

  // Scratchpad layout:
  //   A tiles at a_base + (i * tiles_K + k) * DIM,  i in [0, TIB), k in [0, tiles_K)
  //   B tiles at b_base + (j * tiles_K + k) * DIM,  j in [0, TJB), k in [0, tiles_K)
  //   b_base = BANK_NUM*BANK_ROWS - TJB*tiles_K*DIM  (top of scratchpad, growing down)
  const uint32_t a_base = 0;
  const uint32_t b_base = (uint32_t)(BANK_NUM * BANK_ROWS - TJB * tiles_K * DIM);

  // Issue flush once before any accelerator use.
  gemmini_flush(0);

  // CONFIG_EX is state that persists; issue once.
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int I0 = 0; I0 < BIGBLKS_M; I0++) {
    for (int J0 = 0; J0 < BIGBLKS_N; J0++) {
      // Actual output dimensions for this big tile (may be smaller at edges).
      int mrows = REAL_M - I0 * BIGBLK;
      if (mrows > BIGBLK) mrows = BIGBLK;
      if (mrows <= 0) continue;
      int ncols = REAL_N - J0 * BIGBLK;
      if (ncols > BIGBLK) ncols = BIGBLK;
      if (ncols <= 0) continue;

      // Actual tile counts for this big tile (may be less than TIB/TJB at edges).
      int act_tI = (mrows + DIM - 1) / DIM;
      int act_tJ = (ncols + DIM - 1) / DIM;
      int pad_i  = act_tI * DIM - mrows;
      int pad_j  = act_tJ * DIM - ncols;

      // ------------------------------------------------------------------
      // 1. Pack scale factors for this big tile into local staging buffers.
      //    A_s: [tiles_K][act_tI * DIM]  (GK groups, each covering act_tI full tiles)
      //    B_s: [tiles_K][act_tJ * DIM]
      //    We copy act_tI DIM-columns of A scales and act_tJ DIM-columns of B scales.
      // ------------------------------------------------------------------
      // A scale staging: rows indexed by K-group (g = 0..tiles_K-1),
      // columns indexed by M row index within big tile (0..act_tI*DIM-1).
      // Original layout: A_scales_row[g][m_global]
      // B scale staging: B_scales_col[g][n_global]
      for (int g = 0; g < tiles_K; g++) {
        int a_cols = act_tI * DIM;
        if (a_cols > BLK) a_cols = BLK;
        memcpy(&A_s[g][0],
               &A_scales_row[g][I0 * BIGBLK],
               a_cols * sizeof(A_scales_row[0][0]));
        int b_cols = act_tJ * DIM;
        if (b_cols > BLK) b_cols = BLK;
        memcpy(&B_s[g][0],
               &B_scales_col[g][J0 * BIGBLK],
               b_cols * sizeof(B_scales_col[0][0]));
      }

      // ------------------------------------------------------------------
      // 2. Load scale factors (once per big tile).
      //    sel=0 → A (activation), sel=1 → B (weight).
      // ------------------------------------------------------------------
      gemmini_mx_load_scales((uint64_t)A_s,
                             (size_t)(tiles_K * act_tI * DIM) * sizeof(A_s[0][0]),
                             0);
      gemmini_mx_load_scales((uint64_t)B_s,
                             (size_t)(tiles_K * act_tJ * DIM) * sizeof(B_s[0][0]),
                             1);

      // ------------------------------------------------------------------
      // 3. MVIN A tiles into scratchpad.
      //    DRAM layout: A_in[m_global][k_col], row stride = MATMUL_K.
      //    Spad address: a_base + (i * tiles_K + k) * DIM
      // ------------------------------------------------------------------
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < act_tI; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) +
                      (I0 * BIGBLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM, DIM);
        }
      }

      // ------------------------------------------------------------------
      // 4. MVIN B tiles into scratchpad.
      //    DRAM layout: B_in[k_row][n_global], row stride = MATMUL_N.
      //    Spad address: b_base + (j * tiles_K + k) * DIM
      //    (j iterates over J-tiles, k over K-tiles — same stride pattern
      //     as A but transposed: b_base + (k * act_tJ + j) * DIM was the
      //     original; keep the same (k*tiles+j)*DIM mapping)
      // ------------------------------------------------------------------
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < act_tJ; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) +
                      (k * DIM) * MATMUL_N + (J0 * BIGBLK + j * DIM)),
              b_base + (k * act_tJ + j) * DIM,
              DIM, DIM);
        }
      }

      // ------------------------------------------------------------------
      // 5. Configure requantized mvout (once per big tile).
      // ------------------------------------------------------------------
      gemmini_config_st(BIGBLK * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   act_tI, act_tJ, tiles_K,
                                   0, 0, 1);

      // ------------------------------------------------------------------
      // 6. Fused hardware matmul (one call covers the full big tile).
      //    b_spad end address = BANK_NUM*BANK_ROWS (top of spad).
      // ------------------------------------------------------------------
      gemmini_loop_ws_spad(act_tI, act_tJ, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // ------------------------------------------------------------------
      // 7. Drain BF16 results from mx_smem to DRAM.
      //    BF16 results are packed 4 per uint64 word; C_hw row width = REAL_N/BF16_PER_WORD.
      //    smem word offset for row r = SPAD_DEST*16 + r*BIGBLK
      //    (BIGBLK elements per row in smem, each is 1 BF16 word unit).
      // ------------------------------------------------------------------
      if (mrows == BIGBLK && ncols == BIGBLK) {
        // Full big tile: drain in one call.
        gemmini_mx_read_smem(
            &C_hw[I0 * BIGBLK][J0 * BIGBLK / BF16_PER_WORD],
            SPAD_DEST * 16,
            BIGBLK * BIGBLK);
      } else {
        // Partial edge tile: drain row by row to avoid overwriting out-of-bounds.
        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(
              &C_hw[I0 * BIGBLK + r][J0 * BIGBLK / BF16_PER_WORD],
              SPAD_DEST * 16 + r * BIGBLK,
              ncols);
        }
      }
    }
  }

  // Single fence after all tiles are drained.
  gemmini_fence();
}
```
