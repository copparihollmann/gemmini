void solution(void) {
  // One-time accelerator config: format, dataflow, strides
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;

      // Copy per-block scales from the full scale arrays into staging buffers
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      uint32_t a_base   = 0;
      uint32_t b_base   = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int      SPAD_DEST = 128;
      int      pad_i    = BLK - mrows;
      int      pad_j    = BLK - ncols;

      // Load scales for this (i0, j0) block
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles (port 0: mvin) with stride = MATMUL_K bytes per row
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *a_ptr = ((const elem_t *)A_in)
                                + (i0 * BLK + i * DIM) * MATMUL_K
                                + k * DIM;
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
          gemmini_extended_mvin((void *)a_ptr, spad_addr, DIM, DIM);
        }
      }

      // MVIN B tiles (port 1: mvin2) with stride = MATMUL_N bytes per row
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *b_ptr = ((const elem_t *)B_in)
                                + (k * DIM) * MATMUL_N
                                + (j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
          gemmini_extended_mvin2((void *)b_ptr, spad_addr, DIM, DIM);
        }
      }

      // Configure output stride and requantizer bounds
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Execute fused tiled matmul (funct-24 marker is one-shot, macro re-issues it)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,

                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM, one row at a time

          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],

      }

    }
  }
