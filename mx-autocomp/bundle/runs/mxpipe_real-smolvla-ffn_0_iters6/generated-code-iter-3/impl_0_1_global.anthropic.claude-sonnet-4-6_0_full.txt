Looking at the original code, I need to understand what it does and then optimize it. The original code:

1. Iterates over BLOCKS_M × BLOCKS_N output tiles
2. For each tile, copies scales, loads data, computes, and reads results
3. Has `gemmini_flush(0)` inside the loop (expensive!)
4. Uses single-buffered approach with no overlap
5. Reloads scales every iteration even when they could be reused
6. Uses `gemmini_extended_mvin` (port 0) for both A and B instead of separate ports

Key optimizations:
- Remove `gemmini_flush` from inner loop (only need it once at start)
- Use `mvin` for A and `mvin2` for B to overlap loads
- Batch mvin calls using `MAX_BLOCK_LEN=4` blocks at a time
- Avoid redundant `gemmini_extended3_config_ex` re-issue (config persists)
- Avoid redundant `gemmini_config_st` re-issue
- Move scale loading outside the loop when scales cover the full matrix
- Use `set_only_strides` when only stride changes between A and B config
- Precompute addresses and avoid redundant config calls

Note: scales are per-block along K, and per row/col for the tile, so they must be reloaded per (i0,j0) block since they are subsets of the full scale arrays.

```c
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  // Issue flush once before any accelerator work
  gemmini_flush(0);

  // Config that never changes across blocks: dataflow, MX formats, strides
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false,
                               0, 0, 3, 0);

  // Config store never changes
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Copy scale sub-blocks for this (i0, j0) tile
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Load A and B scales into hardware scale memories
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Load A tiles using mvin (port 0): stride = MATMUL_K bytes
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

      for (int i = 0; i < tiles; i++) {
        int k = 0;
        while (k < tiles_K) {
          // Batch up to MAX_BLOCK_LEN=4 consecutive K-tiles for same i
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          // Check that spad rows are consecutive: (i*tiles_K+k)*DIM is base,
          // next batch rows are contiguous
          const elem_t *a_dram = ((const elem_t *)A_in)
                                 + (i0 * BLK + i * DIM) * MATMUL_K
                                 + k * DIM;
          uint32_t a_spad = a_base + (i * tiles_K + k) * DIM;
          // Use block mvin: cols = batch*DIM covers batch tiles in one command
          gemmini_extended_mvin(a_dram, a_spad, batch * DIM, DIM);
          k += batch;
        }
      }

      // Load B tiles using mvin2 (port 1): stride = MATMUL_N bytes
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

      for (int j = 0; j < tiles; j++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          const elem_t *b_dram = ((const elem_t *)B_in)
                                 + (k * DIM) * MATMUL_N
                                 + (j0 * BLK + j * DIM);
          uint32_t b_spad = b_base + (k * tiles + j) * DIM;
          gemmini_extended_mvin2(b_dram, b_spad, batch * DIM, DIM);
          k += batch;
        }
      }

      // Configure requantizer with correct tile bounds
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Fused hardware matmul (funct 24 marker is one-shot, macro re-issues it)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0,
                           false, 0x38);

      // Drain BF16 results from mx_smem to DRAM
      if (mrows == BLK && ncols == BLK) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK][j0 * BLK / BF16_PER_WORD],
                             SPAD_DEST * 16, BLK * BLK);
      } else {
        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * BLK,
                               ncols);
        }
      }
    }
  }

  // Single fence after all tiles are drained
  gemmini_fence();
}
```
