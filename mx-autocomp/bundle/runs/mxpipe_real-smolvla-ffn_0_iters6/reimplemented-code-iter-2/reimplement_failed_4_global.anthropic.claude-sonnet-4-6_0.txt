#include <stdint.h>
#include <string.h>
#include "gemmini.h"
#include "gemmini_params.h"

// Matrix dimensions must be multiples of DIM=16
#define MATMUL_M 64
#define MATMUL_N 64
#define MATMUL_K 64

#define tiles_I (MATMUL_M / DIM)
#define tiles_J (MATMUL_N / DIM)
#define tiles_K (MATMUL_K / DIM)

// Scratchpad layout:
// A tiles at a_base + (i*tiles_K + k)*DIM
// B tiles at b_base + (j*tiles_K + k)*DIM
// b_base = BANK_NUM*BANK_ROWS - tiles_K*tiles_J*DIM (top of scratchpad, growing down)
#define a_base 0
#define b_base (BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM)

// Output destination in mx_smem
#define SPAD_DEST 0

// Input matrices (FP8 E4M3, elem_t = uint8_t)
static elem_t A[MATMUL_M][MATMUL_K] __attribute__((aligned(64)));
static elem_t B[MATMUL_K][MATMUL_N] __attribute__((aligned(64)));

// Output matrix (BF16, stored as uint16_t)
static uint16_t C_hw[MATMUL_M][MATMUL_N] __attribute__((aligned(64)));

// Scale factors: one e8m0 byte per MX block along K
// A_scales_row: [tiles_K][MATMUL_M] - per row, per K-block
// B_scales_col: [tiles_K][MATMUL_N] - per col, per K-block
static uint8_t A_scales_row[tiles_K][MATMUL_M] __attribute__((aligned(64)));
static uint8_t B_scales_col[tiles_K][MATMUL_N] __attribute__((aligned(64)));

void matmul_mx_fp8(
    elem_t A_in[MATMUL_M][MATMUL_K],
    elem_t B_in[MATMUL_K][MATMUL_N],
    uint8_t A_scales[tiles_K][MATMUL_M],
    uint8_t B_scales[tiles_K][MATMUL_N],
    uint16_t C_out[MATMUL_M][MATMUL_N])
{
    // Step 1: Configure execution - FP8 E4M3 inputs, BF16 output
    // act_mx_fmt=0 (FP8 E4M3), wgt_mx_fmt=0 (FP8 E4M3), out_mx_fmt=3 (BF16), uselut=0
    gemmini_extended3_config_ex(
        WEIGHT_STATIONARY,
        /*sys_act*/0,
        /*sys_shift*/0,
        /*sys_acc_scale*/ACC_SCALE_IDENTITY,
        /*C_stride*/1,
        /*A_stride*/1,
        /*A_transpose*/0,
        /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0,
        /*wgt_mx_fmt*/0,
        /*out_mx_fmt*/3,
        /*uselut*/0);

    // Step 2: Load scale factors
    // sel=0 for A (activation) scales, sel=1 for B (weight) scales
    gemmini_mx_load_scales((uint64_t)A_scales, sizeof(uint8_t) * tiles_K * MATMUL_M, 0);
    gemmini_mx_load_scales((uint64_t)B_scales, sizeof(uint8_t) * tiles_K * MATMUL_N, 1);

    // Step 3: No LUT needed for FP8 (uselut=0)

    // Step 4: MVIN tiles of A and B into scratchpad
    // Configure load stride for A (row stride in bytes = MATMUL_K * sizeof(elem_t))
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

    // Load A tiles: a_base + (i*tiles_K + k)*DIM
    for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
            uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
            elem_t *dram_ptr = &A_in[i * DIM][k * DIM];
            gemmini_extended_mvin(
                (uint64_t)dram_ptr,
                spad_addr,
                DIM,  // cols
                DIM); // rows
        }
    }

    // Configure load stride for B (row stride in bytes = MATMUL_N * sizeof(elem_t))
    gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

    // Load B tiles: b_base + (j*tiles_K + k)*DIM
    for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
            uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
            elem_t *dram_ptr = &B_in[k * DIM][j * DIM];
            gemmini_extended_mvin2(
                (uint64_t)dram_ptr,
                spad_addr,
                DIM,  // cols
                DIM); // rows
        }
    }

    // Step 5: Configure the requantized mvout
    gemmini_extended_config_st(sizeof(uint16_t) * MATMUL_N, 0, ACC_SCALE_IDENTITY);
    gemmini_mxquant_config_mvout(
        (uint64_t)0,           // scale_dram (unused for BF16 out)
        /*i_bound*/tiles_I,
        /*j_bound*/tiles_J,
        /*k_bound*/tiles_K,
        /*scale_act_sel*/0,
        /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // Step 6: Compute - weight-stationary tiled matmul
    // Must re-issue LOOP_WS_CONFIG_SPAD_AB (funct 24) before every call (one-shot marker)
    // B spad end address = BANK_NUM*BANK_ROWS (top of scratchpad)
    gemmini_loop_ws_spad(
        tiles_I, tiles_J, tiles_K,
        /*pad_I*/0, /*pad_J*/0, /*pad_K*/0,
        /*A*/a_base,
        /*B*/BANK_NUM * BANK_ROWS,
        /*D*/0,
        /*C*/SPAD_DEST,
        /*A_transpose*/false,
        /*B_transpose*/false,
        /*full_C*/false,
        /*low_D*/false,
        /*ex_accumulate*/false,
        /*act*/NO_ACTIVATION,
        /*a_spad_id*/0,
        /*b_spad_id*/0,
        /*is_resadd*/false,
        /*skips*/0x38);

    // Step 7: Read BF16 results from mx_smem to DRAM
    // BF16 values packed 4 per uint64; smem_off_bf16 = SPAD_DEST * 16
    gemmini_mx_read_smem(
        (uint64_t)C_out,
        SPAD_DEST * 16,
        MATMUL_M * MATMUL_N);

    gemmini_fence();
}

int main(void) {
    // Initialize input matrices with simple test data
    for (int i = 0; i < MATMUL_M; i++) {
        for (int k = 0; k < MATMUL_K; k++) {
            // FP8 E4M3 encoding for 1.0 = 0x38
            A[i][k] = 0x38;
        }
    }
    for (int k = 0; k < MATMUL_K; k++) {
        for (int j = 0; j < MATMUL_N; j++) {
            // FP8 E4M3 encoding for 1.0 = 0x38
            B[k][j] = 0x38;
        }
    }

    // Initialize scale factors to 1.0 (e8m0 encoding: exponent=127 = 0x7F)
    for (int ki = 0; ki < tiles_K; ki++) {
        for (int m = 0; m < MATMUL_M; m++) {
            A_scales_row[ki][m] = 0x7F;
        }
        for (int n = 0; n < MATMUL_N; n++) {
            B_scales_col[ki][n] = 0x7F;
        }
    }

    gemmini_flush(0);

    matmul_mx_fp8(A, B, A_scales_row, B_scales_col, C_hw);

    return 0;
}
