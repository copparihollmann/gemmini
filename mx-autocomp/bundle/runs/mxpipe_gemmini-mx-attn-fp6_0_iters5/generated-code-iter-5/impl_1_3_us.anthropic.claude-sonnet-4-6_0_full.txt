void solution(void) {
  // --- First matmul: Q x KT -> S1 (attention scores) ---
  // tiles_I = ATT_S/DIM = 8, tiles_J = ATT_S/DIM = 8, tiles_K = ATT_D/DIM = 4
  // But original splits by /2 for double-buffer: ti=4, tj=4, tk=4

    int tj = ATT_S / DIM / 2;   // 4
    int tk = ATT_D / DIM;       // 4
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;  // 8192 - 4*4*16 = 7936
    int SD = 128;  // SPAD_DEST word offset for mx_smem

    gemmini_flush(0);

    // Step 1: Configure execution (fp6 e3m2: act=1, wgt=1, out=3=BF16, uselut=1)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,

    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Step 3: Load LUTs (sel=0 for B/KT, sel=1 for A/Q)
    gemmini_mx_load_lut((uint64_t)KT_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)Q_lp, 1, 1);

        gemmini_extended_mvin(
            (void*)(((elem_t*)Q_hw) + i * DIM * ATT_D + k * DIM),
            ab + (i * tk + k) * DIM,
            DIM, DIM);
      }
    }

            bb + (k * tj + j) * DIM,

        ti, tj, tk,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/8);

        ab, BANK_NUM * BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    gemmini_mx_read_smem(&S1_hw[0][0], SD * 16, ATT_S * ATT_S);

  for (int i = 0; i < ATT_S; i++) {
    float row[ATT_S];
    float mx = -1e30f;

      row[j] = bf16f(b);
      if (row[j] > mx) mx = row[j];
    }

    float sm = 0.f;
    for (int j = 0; j < ATT_S; j++) {
      row[j] = expf2(row[j] - mx);
      sm += row[j];
    }
    for (int j = 0; j < ATT_S; j++) {
      row[j] /= sm;
    }

        if (a > gm) gm = a;

    // Step 2: Load scale factors (sel=0 for A/P, sel=1 for B/V)
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Step 3: Load LUTs (sel=0 for B/V, sel=1 for A/P)
    gemmini_mx_load_lut((uint64_t)V_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)P_lp, 1, 1);

    // Step 4: MVIN tiles of P (A) and V (B) into scratchpad
    // P: row stride = ATT_S * sizeof(elem_t) = 128 bytes (but fp4 packed: half)

    // However mvin uses elem_t stride, and P is stored as ATT_S elem_t entries
    // (fp6 dual packing means ATT_S/2 bytes represent ATT_S fp4 elements)
    // The original uses ATT_S*sizeof(elem_t) for stride meaning 128 bytes per row

    for (int i = 0; i < ti; i++) {

        gemmini_extended_mvin(
            (void*)(((elem_t*)P_hw) + i * DIM * ATT_S + k * DIM),
            ab + (i * tk + k) * DIM,
            DIM, DIM);
      }
    }

    // V: row stride = (ATT_D/2) * sizeof(elem_t) = 32 bytes (fp6 packed)
    gemmini_config_ld(ATT_D * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)V_in) + k * DIM * (ATT_D / 2) + j * DIM),
            bb + (k * tj + j) * DIM,

        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/8);

    // Step 6: Compute

        ab, BANK_NUM * BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: Read results from mx_smem to DRAM
    gemmini_mx_read_smem(&O_hw[0][0], SD * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
