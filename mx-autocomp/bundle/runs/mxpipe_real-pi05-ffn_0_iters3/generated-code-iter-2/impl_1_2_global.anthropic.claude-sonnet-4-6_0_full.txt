void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  // Hoist one-time hardware initialization outside all loops.
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  // gemmini_mxquant_config_mvout arguments are identical every iteration:
  // tile counts, scale_factors pointer, selectors and granularity never change.
  // Hoist it here; the hardware re-reads this config at each loop_ws_spad
  // execution, so a single upfront issue covers all (i0,j0) blocks.
  // NOTE: per programming-model rules we must re-issue this before each
  // gemmini_loop_ws_spad, so it stays inside the inner loop below – but the
  // gemmini_config_ld strides and gemmini_extended3_config_ex are truly
  // constant and are safely hoisted.

  // Hoist the constant load-stride configurations.
  // We will switch between A-stride and B-stride inside the loop, but we can
  // set A-stride once here and restore it (or just re-issue) inside the loop.
  // The cleanest approach: hoist the two gemmini_config_ld calls to just
  // before they are needed inside the loop, but issue gemmini_extended3_config_ex
  // and gemmini_flush only once outside.

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Stage per-block scales (these genuinely differ per (i0,j0)).
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Load A and B scale factors into hardware scale memories.
      // sel=0 → A (activation) scales, sel=1 → B (weight) scales.
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // DMA A tiles into scratchpad (stride = full K row of A).
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // DMA B tiles into scratchpad (stride = full N row of B).
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM,
            DIM, DIM);

      // Configure output stride and requantizer.
      // These arguments are constant across all (i0,j0) blocks; kept here
      // because the programming model requires gemmini_mxquant_config_mvout
      // to immediately precede the corresponding gemmini_loop_ws_spad.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, tiles_K);

      // Execute the fused tiled weight-stationary matmul.
      // mx_loop_spad_marker is one-shot: LOOP_WS_CONFIG_SPAD_AB (funct 24)
      // is re-issued automatically inside gemmini_loop_ws_spad before every
      // LOOP_WS (funct 8) invocation.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16-packed results from mx_smem to DRAM, one output row at a time.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      // Full memory barrier: host must not read C_hw until DMA completes.
      gemmini_fence();
    }
  }
}
