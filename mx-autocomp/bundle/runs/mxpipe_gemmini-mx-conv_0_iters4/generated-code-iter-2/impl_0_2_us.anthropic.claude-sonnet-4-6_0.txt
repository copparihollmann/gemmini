#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_data_mx_lut_hw.h"

static elem_t A_buf[MATMUL_M][MATMUL_K] __attribute__((aligned(64)));

static const int tiles_I = MATMUL_M / DIM;
static const int tiles_J = MATMUL_N / DIM;
static const int tiles_K = MATMUL_K / DIM;

static const int a_base = 0;
static const int b_base_val = BANK_NUM * BANK_ROWS - (MATMUL_N / DIM) * (MATMUL_K / DIM) * DIM;

void solution(void) {
    // ------------------------------------------------------------------
    // 1. Im2col: materialize A_buf from X_in
    // ------------------------------------------------------------------
    for (int pi = 0; pi < CONV_H / CONV_KSZ; pi++)
        for (int pj = 0; pj < PATCHES_X; pj++)
            for (int c = 0; c < CONV_C; c++)
                for (int dy = 0; dy < CONV_KSZ; dy++)
                    for (int dx = 0; dx < CONV_KSZ; dx++)
                        A_buf[pi * PATCHES_X + pj][c * CONV_KSZ * CONV_KSZ + dy * CONV_KSZ + dx] =
                            X_in[c][pi * CONV_KSZ + dy][pj * CONV_KSZ + dx];

    // ------------------------------------------------------------------
    // 2. Configure execution (MX FP8, weight-stationary, BF16 output, no LUT)
    // ------------------------------------------------------------------
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                1, 1, 0, 0, false,
                                0, 0, 3, 0);

    // ------------------------------------------------------------------
    // 3. Load scale factors: sel=0 for A (activation), sel=1 for B (weight)
    // ------------------------------------------------------------------
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // ------------------------------------------------------------------
    // 4. Precompute flat pointer arrays for A and B mvin
    //    Decouples multiply-heavy address arithmetic from DMA issuance.
    // ------------------------------------------------------------------
    const int num_a_tiles = tiles_I * tiles_K;
    const int num_b_tiles = tiles_J * tiles_K;

    // A tile pointers: row-major over (i, k)
    elem_t *a_dram[tiles_I * tiles_K];
    uint32_t a_spad_addr[tiles_I * tiles_K];
    for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
            int idx = i * tiles_K + k;
            a_dram[idx]      = (elem_t *)A_buf + i * DIM * MATMUL_K + k * DIM;
            a_spad_addr[idx] = (uint32_t)(a_base + idx * DIM);
        }
    }

    // B tile pointers: row-major over (j, k), matching spad layout b_base+(j*tiles_K+k)*DIM
    elem_t *b_dram[tiles_J * tiles_K];
    uint32_t b_spad_addr[tiles_J * tiles_K];
    for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
            int idx = j * tiles_K + k;
            b_dram[idx]      = (elem_t *)B_in + j * DIM * MATMUL_K + k * DIM;
            b_spad_addr[idx] = (uint32_t)(b_base_val + idx * DIM);
        }
    }

    // ------------------------------------------------------------------
    // 5. Issue A mvins in a tight loop (no arithmetic inside)
    // ------------------------------------------------------------------
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int idx = 0; idx < num_a_tiles; idx++) {
        gemmini_extended_mvin((void *)a_dram[idx], a_spad_addr[idx], DIM, DIM);
    }

    // ------------------------------------------------------------------
    // 6. Issue B mvins in a tight loop (no arithmetic inside)
    // ------------------------------------------------------------------
    for (int idx = 0; idx < num_b_tiles; idx++) {
        gemmini_extended_mvin((void *)b_dram[idx], b_spad_addr[idx], DIM, DIM);
    }

    // ------------------------------------------------------------------
    // 7. Configure output stride and requantizer
    //    i_bound=tiles_I, j_bound=tiles_J, k_bound=tiles_K must match
    //    the subsequent gemmini_loop_ws_spad arguments exactly.
    // ------------------------------------------------------------------
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                 tiles_I, tiles_J, tiles_K,
                                 0, 0, 1);

    // ------------------------------------------------------------------
    // 8. Fused weight-stationary hardware loop
    //    mx_loop_spad_marker is one-shot: LOOP_WS_CONFIG_SPAD_AB (funct 24)
    //    is re-issued inside gemmini_loop_ws_spad before every LOOP_WS.
    // ------------------------------------------------------------------
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K,
                         0, 0, 0,
                         (uint32_t)a_base,          // A spad base address
                         BANK_NUM * BANK_ROWS,       // B spad end address
                         0,                          // D (no bias)
                         SPAD_DEST,
                         false, false,
                         false, false, false,
                         NO_ACTIVATION,
                         0, 0,
                         false,
                         0x38);

    // ------------------------------------------------------------------
    // 9. Drain BF16 results from mx_smem to DRAM, then fence
    // ------------------------------------------------------------------
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
