void solution(void) {
  // Strategy 11: Fuse im2col into mvin — eliminate the large A_buf staging
  // buffer. Fill a small DIM×DIM tile directly from X_in and immediately
  // issue the mvin, removing the separate im2col pre-pass and halving
  // memory traffic for A.

  static elem_t a_tile[DIM][DIM];

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Load all B tiles first using mvin2 (no im2col needed, stride = MATMUL_K)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++)
    for (int k = 0; k < tiles_K; k++)
      gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + j * DIM * MATMUL_K + k * DIM),
          b_base + (j * tiles_K + k) * DIM, DIM, DIM);

  // Configure stride once for a_tile (DIM-wide, row-major) before A tile loop
  gemmini_config_ld(DIM * sizeof(elem_t));

  // Load A tiles: fuse im2col address computation into mvin pointer.
  // a_tile is DIM×DIM and reused for each tile, eliminating A_buf entirely.
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      // Fill a_tile by computing im2col source directly from X_in
      for (int r = 0; r < DIM; r++) {
        int row_idx = i * DIM + r;
        int pi      = row_idx / PATCHES_X;
        int pj      = row_idx % PATCHES_X;
        for (int cl = 0; cl < DIM; cl++) {
          int col_idx = k * DIM + cl;
          int c       = col_idx / (CONV_KSZ * CONV_KSZ);
          int rem     = col_idx % (CONV_KSZ * CONV_KSZ);
          int dy      = rem / CONV_KSZ;
          int dx      = rem % CONV_KSZ;
          a_tile[r][cl] = X_in[c][pi * CONV_KSZ + dy][pj * CONV_KSZ + dx];
        }
      }
      // mvin this tile; stride already configured to DIM * sizeof(elem_t)
      gemmini_extended_mvin((void*)a_tile,
          a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K, 0, 0, tiles_K);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
