void solution(void) {
  // im2col: patch (pi,pj) -> row pi*PATCHES_X+pj; column c*k*k + dy*k + dx
  for (int pi = 0; pi < CONV_H / CONV_KSZ; pi++)
    for (int pj = 0; pj < PATCHES_X; pj++)
      for (int c = 0; c < CONV_C; c++)
        for (int dy = 0; dy < CONV_KSZ; dy++)
          for (int dx = 0; dx < CONV_KSZ; dx++)
            A_buf[pi * PATCHES_X + pj][c * CONV_KSZ * CONV_KSZ + dy * CONV_KSZ + dx] =
                X_in[c][pi * CONV_KSZ + dy][pj * CONV_KSZ + dx];

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure port 0 (mvin) for A: stride = MATMUL_K columns of elem_t
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Configure port 1 (mvin2) for B: stride = MATMUL_K columns of elem_t
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  int total_a = tiles_I * tiles_K;
  int total_b = tiles_J * tiles_K;
  int max_total = (total_a > total_b) ? total_a : total_b;

  // Pre-compute all A tile DRAM pointers and scratchpad addresses
  const elem_t *a_dram_ptrs[total_a];
  uint32_t      a_spad_addrs[total_a];
  {
    const elem_t *a_dram_row = (const elem_t *)A_buf;
    uint32_t a_spad_cur = a_base;
    for (int i = 0; i < tiles_I; i++) {
      const elem_t *a_dram_tile = a_dram_row;
      for (int k = 0; k < tiles_K; k++) {
        int idx = i * tiles_K + k;
        a_dram_ptrs[idx] = a_dram_tile;
        a_spad_addrs[idx] = a_spad_cur;
        a_dram_tile += DIM;
        a_spad_cur  += DIM;
      }
      a_dram_row += (size_t)DIM * MATMUL_K;
    }
  }

  // Pre-compute all B tile DRAM pointers and scratchpad addresses
  const elem_t *b_dram_ptrs[total_b];
  uint32_t      b_spad_addrs[total_b];
  {
    const elem_t *b_dram_row = (const elem_t *)B_in;
    uint32_t b_spad_cur = b_base;
    for (int j = 0; j < tiles_J; j++) {
      const elem_t *b_dram_tile = b_dram_row;
      for (int k = 0; k < tiles_K; k++) {
        int idx = j * tiles_K + k;
        b_dram_ptrs[idx] = b_dram_tile;
        b_spad_addrs[idx] = b_spad_cur;
        b_dram_tile += DIM;
        b_spad_cur  += DIM;
      }
      b_dram_row += (size_t)DIM * MATMUL_K;
    }
  }

  // Issue mvin commands using pre-computed addresses — no arithmetic in the hot loop
  for (int idx = 0; idx < max_total; idx++) {
    if (idx < total_a) {
      gemmini_extended_mvin(
          a_dram_ptrs[idx],
          a_spad_addrs[idx],
          DIM, DIM);
    }
    if (idx < total_b) {
      gemmini_extended_mvin2(
          b_dram_ptrs[idx],
          b_spad_addrs[idx],
          DIM, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K, 0, 0, tiles_K);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
