void solution(void) {
  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM;
    int tiles_J = ATT_S / DIM;
    int tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

    gemmini_flush(0);

    // Step 1: Configure execution (FP8 E4M3 inputs, BF16 output)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        /*C_stride*/1, /*A_stride*/1,
        /*A_transpose*/0, /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

    // Step 2: Load scale factors (sel=0 for A/activation, sel=1 for B/weight)
    gemmini_mx_load_scales((uint64_t)&Q_scales_row,  sizeof(Q_scales_row),  0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Step 4: MVIN tiles of A (Q) into scratchpad via mvin port
    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    for (int i = 0; i < tiles_I; i++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
      }
    }

    // Step 4: MVIN tiles of B (KT) into scratchpad via mvin2 port
    gemmini_extended3_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)KT_in) + k * DIM * ATT_S + j * DIM),
            b_base + (k * tiles_J + j) * DIM,
            DIM, DIM);
      }
    }

    // Step 5: Configure output stride and requantizer
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout(
        (uint64_t)scale_factors,
        /*i_bound*/tiles_I, /*j_bound*/tiles_J, /*k_bound*/tiles_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // Step 6: Compute tiled matmul (one-shot marker re-issued inside macro)
    gemmini_loop_ws_spad(
        tiles_I, tiles_J, tiles_K,
        /*pad_I*/0, /*pad_J*/0, /*pad_K*/0,
        a_base, BANK_NUM * BANK_ROWS,
        /*D*/0, /*C=SPAD_DEST*/SPAD_DEST,
        /*A_transpose*/false, /*B_transpose*/false,
        /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
        /*act*/NO_ACTIVATION,
        /*a_spad_id*/0, /*b_spad_id*/0,
        /*is_resadd*/false,
        /*skips*/0x38);

    // Step 7: Drain BF16 results from mx_smem to DRAM
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2) P = softmax(S1) rows on CPU; quantize to fp8 + per-32-group scales ----
  for (int i = 0; i < ATT_S; i++) {
    float row[ATT_S];

    // Pass 1: Unpack BF16 -> float, find maxv
    float maxv = -1e30f;
    for (int j = 0; j < ATT_S; j++) {
      uint16_t b = (uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF);
      row[j] = bf16_to_f(b);
      if (row[j] > maxv) maxv = row[j];
    }

    // Pass 2: Compute exp(row[j] - maxv), accumulate sum
    float sum = 0.0f;
    for (int j = 0; j < ATT_S; j++) {
      row[j] = exp_f(row[j] - maxv);
      sum += row[j];
    }

    float inv_sum = 1.0f / sum;

    // Pass 3 (fused): normalize + per-group max scan + per-group quantize
    for (int g = 0; g < GROUPS_S; g++) {
      // Fused normalize + group max
      float gmax = 0.0f;
      for (int j = g * 32; j < (g + 1) * 32; j++) {
        row[j] *= inv_sum;
        float a = f_abs(row[j]);
        if (a > gmax) gmax = a;
      }
      // Compute e8m0 scale
      int e = 0;
      if (gmax > 0.0f) e = f_exp(gmax / 16.0f) + 1;
      float inv_scale = pow2i(-e);
      P_scales[g][i] = (uint8_t)(e + 127);
      // Quantize group
      for (int j = g * 32; j < (g + 1) * 32; j++)
        P_q[i][j] = fp8_e4m3_rtz(row[j] * inv_scale);
    }
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM;
    int tiles_J = ATT_D / DIM;
    int tiles_K = ATT_S / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

    gemmini_flush(0);

    // Step 1: Configure execution (FP8 E4M3 inputs, BF16 output)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        /*C_stride*/1, /*A_stride*/1,
        /*A_transpose*/0, /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

    // Step 2: Load scale factors (sel=0 for A/activation, sel=1 for B/weight)
    gemmini_mx_load_scales((uint64_t)&P_scales,     sizeof(P_scales),     0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Step 4: MVIN tiles of A (P_q) into scratchpad via mvin port
    gemmini_extended3_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    for (int i = 0; i < tiles_I; i++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
      }
    }

    // Step 4: MVIN tiles of B (V) into scratchpad via mvin2 port
    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)V_in) + k * DIM * ATT_D + j * DIM),
            b_base + (k * tiles_J + j) * DIM,
            DIM, DIM);
      }
    }

    // Step 5: Configure output stride and requantizer
    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout(
        (uint64_t)scale_factors,
        /*i_bound*/tiles_I, /*j_bound*/tiles_J, /*k_bound*/tiles_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // Step 6: Compute tiled matmul (one-shot marker re-issued inside macro)
    gemmini_loop_ws_spad(
        tiles_I, tiles_J, tiles_K,
        /*pad_I*/0, /*pad_J*/0, /*pad_K*/0,
        a_base, BANK_NUM * BANK_ROWS,
        /*D*/0, /*C=SPAD_DEST*/SPAD_DEST,
        /*A_transpose*/false, /*B_transpose*/false,
        /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
        /*act*/NO_ACTIVATION,
        /*a_spad_id*/0, /*b_spad_id*/0,
        /*is_resadd*/false,
        /*skips*/0x38);

    // Step 7: Drain BF16 results from mx_smem to DRAM
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
