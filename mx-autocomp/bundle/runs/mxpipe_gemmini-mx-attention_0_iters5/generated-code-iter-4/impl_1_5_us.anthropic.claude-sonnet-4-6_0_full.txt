Looking at the original code, I need to understand what it does:
1. Compute S1 = Q @ K^T (FP8, output BF16)
2. CPU softmax on S1, quantize result to FP8 P_q with per-32-group scales
3. Compute O = P @ V (FP8, output BF16)

The original code already follows the correct programming model. The key issues to address for optimization:
- The `gemmini_flush(0)` calls are expensive - only need one at the start
- Config calls persist, so we can avoid redundant re-issue when format/stride unchanged
- The mvin loops can use block mvin (up to MAX_BLOCK_LEN=4 tiles per command)
- Verify scratchpad layout is correct
- The second matmul has same config as first, so we can skip re-issuing config_ex

For S=128, D=64: tiles_I=8, tiles_J=8, tiles_K=4 for QK^T; tiles_I=8, tiles_J=4, tiles_K=8 for PV.

Let me check scratchpad for QK^T: A needs (8*4)*16=512 rows, B needs (8*4)*16=512 rows, total 1024 ≤ 8192. ✓

For PV: A needs (8*8)*16=1024 rows, B needs (4*8)*16=512 rows, total 1536 ≤ 8192. ✓

I can batch mvin calls using block mvin where tiles are contiguous in DRAM and scratchpad.

```c
void solution(void) {
  // Only one flush needed at the very beginning
  gemmini_flush(0);

  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM;   // 8
    int tiles_J = ATT_S / DIM;   // 8
    int tiles_K = ATT_D / DIM;   // 4
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    // b_base = 8192 - 4*8*16 = 8192 - 512 = 7680

    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&Q_scales_row,  sizeof(Q_scales_row),  0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Mvin A (Q): shape [ATT_S][ATT_D], row-stride = ATT_D * sizeof(elem_t)
    // A tile (i,k) -> DRAM: Q + i*DIM*ATT_D + k*DIM, spad: a_base + (i*tiles_K+k)*DIM
    // Tiles are contiguous along k for fixed i, so batch up to MAX_BLOCK_LEN=4 tiles per row of i
    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    for (int i = 0; i < tiles_I; i++) {
      // tiles_K=4 == MAX_BLOCK_LEN, load all k tiles in one block mvin per i
      elem_t *dram_ptr = ((elem_t*)Q_in) + i * DIM * ATT_D;
      uint32_t spad_addr = a_base + i * tiles_K * DIM;
      gemmini_extended_mvin(dram_ptr, spad_addr, tiles_K * DIM, DIM);
    }

    // Mvin B (KT): shape [ATT_D][ATT_S], row-stride = ATT_S * sizeof(elem_t)
    // B tile (k,j) -> DRAM: KT + k*DIM*ATT_S + j*DIM, spad: b_base + (k*tiles_J+j)*DIM
    // For fixed k, tiles along j are contiguous; batch up to MAX_BLOCK_LEN=4 per k row
    gemmini_extended3_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tiles_K; k++) {
      // tiles_J=8, MAX_BLOCK_LEN=4, so issue 2 batches per k
      for (int jb = 0; jb < tiles_J; jb += MAX_BLOCK_LEN) {
        int batch = (tiles_J - jb < MAX_BLOCK_LEN) ? (tiles_J - jb) : MAX_BLOCK_LEN;
        elem_t *dram_ptr = ((elem_t*)KT_in) + k * DIM * ATT_S + jb * DIM;
        uint32_t spad_addr = b_base + (k * tiles_J + jb) * DIM;
        gemmini_extended_mvin2(dram_ptr, spad_addr, batch * DIM, DIM);
      }
    }

    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K,
                                 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                         a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                         false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2) P = softmax(S1) on CPU; quantize to fp8 + per-32-group scales ----
  for (int i = 0; i < ATT_S; i++) {
    float row[ATT_S];
    float maxv = -1e30f;
    for (int j = 0; j < ATT_S; j++) {
      uint16_t b = (uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF);
      row[j] = bf16_to_f(b);
      if (row[j] > maxv) maxv = row[j];
    }
    float sum = 0.0f;
    for (int j = 0; j < ATT_S; j++) { row[j] = exp_f(row[j] - maxv); sum += row[j]; }
    float inv_sum = 1.0f / sum;
    for (int j = 0; j < ATT_S; j++) row[j] *= inv_sum;
    for (int g = 0; g < GROUPS_S; g++) {
      float gmax = 0.0f;
      for (int j = g * 32; j < (g + 1) * 32; j++)
        if (f_abs(row[j]) > gmax) gmax = f_abs(row[j]);
      int e = 0;
      if (gmax > 0.0f) e = f_exp(gmax / 16.0f) + 1;
      float inv_scale = pow2i(-e);
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g * 32; j < (g + 1) * 32; j++)
        P_q[i][j] = fp8_e4m3_rtz(row[j] * inv_scale);
    }
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM;   // 8
    int tiles_J = ATT_D / DIM;   // 4
    int tiles_K = ATT_S / DIM;   // 8
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    // b_base = 8192 - 8*4*16 = 8192 - 512 = 7680

    // Config format unchanged (fp8 in, bf16 out), re-issue to be safe after CPU work
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&P_scales,    sizeof(P_scales),    0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Mvin A (P_q): shape [ATT_S][ATT_S], row-stride = ATT_S * sizeof(elem_t)
    // A tile (i,k): DRAM P_q + i*DIM*ATT_S + k*DIM, spad: a_base + (i*tiles_K+k)*DIM
    // tiles_K=8, MAX_BLOCK_LEN=4 => 2 batches per i row
    gemmini_extended3_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    for (int i = 0; i < tiles_I; i++) {
      for (int kb = 0; kb < tiles_K; kb += MAX_BLOCK_LEN) {
        int batch = (tiles_K - kb < MAX_BLOCK_LEN) ? (tiles_K - kb) : MAX_BLOCK_LEN;
        elem_t *dram_ptr = ((elem_t*)P_q) + i * DIM * ATT_S + kb * DIM;
        uint32_t spad_addr = a_base + (i * tiles_K + kb) * DIM;
        gemmini_extended_mvin(dram_ptr, spad_addr, batch * DIM, DIM);
      }
    }

    // Mvin B (V): shape [ATT_S][ATT_D], row-stride = ATT_D * sizeof(elem_t)
    // B tile (k,j): DRAM V + k*DIM*ATT_D + j*DIM, spad: b_base + (k*tiles_J+j)*DIM
    // tiles_J=4 == MAX_BLOCK_LEN => 1 batch per k row
    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t*)V_in) + k * DIM * ATT_D;
      uint32_t spad_addr = b_base + k * tiles_J * DIM;
      gemmini_extended_mvin2(dram_ptr, spad_addr, tiles_J * DIM, DIM);
    }

    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K,
                                 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                         a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                         false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
