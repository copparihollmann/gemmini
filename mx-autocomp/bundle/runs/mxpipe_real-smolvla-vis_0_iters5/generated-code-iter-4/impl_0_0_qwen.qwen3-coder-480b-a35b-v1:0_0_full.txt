void solution(void) {
  // Only flush once at the very beginning of the program to avoid draining the pipeline
  gemmini_flush(0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  int GK      = MATMUL_K / 32;

  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  
  // Double-buffering bases for A
  uint32_t a_bases[2] = {0, tiles * tiles_K * DIM};
  int SPAD_DEST = 128;

  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&A_scales_T[i0][g * BLK],
             &A_scales_row[g][i0 * BLK],
             BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&B_scales_T[j0][g * BLK],
             &B_scales_col[g][j0 * BLK],
             BLK);
    }
  }

  // Hoist invariant execution configuration
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // Configure A load stride once before starting any A loads
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // Load B scales once per j0
    gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

    // Mvin B tiles once per j0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);
      }
    }

    // Reconfigure A load stride
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));

    // Identify first valid i0 block to pre-load
    int first_i0 = -1;
    for (int i0_check = 0; i0_check < BLOCKS_M; i0_check++) {
      int mrows_check = REAL_M - i0_check * BLK;
      if (mrows_check > 0) {
        first_i0 = i0_check;
        break;
      }
    }

    if (first_i0 == -1) continue;

    // Pre-load scales and data of the first block into a_bases[0]
    gemmini_mx_load_scales((uint64_t)A_scales_T[first_i0], A_S_BYTES, 0);
    for (int i = 0; i < tiles; i++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
          (void*)(((elem_t*)A_in) + (first_i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
          a_bases[0] + (i * tiles_K + k) * DIM, DIM, DIM);
      }
    }

    int curr_buf = 0;
    for (int i0 = first_i0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows <= 0) continue;

      // Identify the next valid i0 block for double-buffered loading
      int next_i0 = -1;
      for (int ni = i0 + 1; ni < BLOCKS_M; ni++) {
        if (REAL_M - ni * BLK > 0) {
          next_i0 = ni;
          break;
        }
      }

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // 1. Issue non-blocking compute and output drain for the current block
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_bases[curr_buf], BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);
      }

      // 2. Issue non-blocking scale and data pre-loads for the next block
      if (next_i0 != -1) {
        gemmini_mx_load_scales((uint64_t)A_scales_T[next_i0], A_S_BYTES, 0);
        for (int i = 0; i < tiles; i++) {
          for (int k = 0; k < tiles_K; k++) {
            gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (next_i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_bases[1 - curr_buf] + (i * tiles_K + k) * DIM, DIM, DIM);
          }
        }
      }

      // Alternate active buffer
      curr_buf = 1 - curr_buf;
    }
  }

  // Issue one final fence to guarantee all asynchronous loads/stores/computes complete
  gemmini_fence();
}
