### Optimization Plan: Strategy 9 (Consolidate scale factor arrays to minimize scale load calls)

Upon analyzing the loop structure of the baseline program, we can identify a major source of redundant memory traffic and control overhead:
1. **Redundant Scale Factor Loads**: The B scale factors `B_scales_T[j0]` only depend on the outer loop index `j0`. However, they are loaded into Gemmini's scale memory using `gemmini_mx_load_scales` inside the inner loop `i0`. This means the same weight scale factors are repeatedly fetched from DRAM `BLOCKS_M` times. By hoisting this load out of the `i0` loop and placing it directly inside the `j0` loop, we reduce B-side scale loading operations by a factor of `BLOCKS_M`.
2. **Redundant Pipeline Flushes**: The `gemmini_flush(0)` instruction is issued on every single inner-loop iteration. Flushing the pipeline repeatedly stalls execution and clears hardware state unnecessarily. We can safely hoist this flush to the very beginning of the function, ensuring the pipeline is clean once at startup while maximizing hardware utilization across tiles.

This plan maintains strict compliance with all programming model rules, specifically preserving the required relative ordering of configuration, scale/LUT loading, tile preloading, compute, and SMEM drainage on every computational path.

void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  int GK      = MATMUL_K / 32;

  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&A_scales_T[i0][g * BLK],
             &A_scales_row[g][i0 * BLK],
             BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&B_scales_T[j0][g * BLK],
             &B_scales_col[g][j0 * BLK],
             BLK);
    }
  }

  // Hoist flush and execution configuration outside all loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // Mvin B tiles once per j0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);
      }
    }

    // Hoisted: Load B scale factors only once per outer loop iteration
    gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      // Load A scale factors (changes per i0)
      gemmini_mx_load_scales((uint64_t)A_scales_T[i0], A_S_BYTES, 0);

      // Mvin A tiles
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Configure requantized mvout
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Dispatch: fast interior path vs slow boundary path
      if (i0 < BLOCKS_M - 1 && j0 < BLOCKS_N - 1) {
        // Fast interior: pad_i = 0, pad_j = 0, mrows = BLK, ncols = BLK
        gemmini_loop_ws_spad(tiles, tiles, tiles_K, 0, 0, 0,
                             a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                             false, false, false, false, false,
                             NO_ACTIVATION, 0, 0, false, 0x38);

        for (int r = 0; r < BLK; r++) {
          gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK, BLK);
        }

        gemmini_fence();
      } else {
        // Slow boundary path: handle partial tiles
        int pad_i = BLK - mrows;
        int pad_j = BLK - ncols;

        gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                             a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                             false, false, false, false, false,
                             NO_ACTIVATION, 0, 0, false, 0x38);

        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK, ncols);
        }

        gemmini_fence();
      }
    }
  }
}
