### Optimization Plan

We will apply **Strategy 4: Compile-Time Tile Bound and Parameter Resolution** to optimize the performance of the kernel. 

In the original code, parameters like `tiles`, `tiles_K`, `GK`, `b_base`, `a_base`, and `SPAD_DEST` are declared as mutable `int` or `uint32_t` variables. Although they depend on compile-time macro constants (such as `BLK`, `DIM`, `MATMUL_K`, `BANK_NUM`, and `BANK_ROWS`), declaring them as mutable variables prevents the compiler from performing aggressive optimizations.

By marking these key tiling and scratchpad layout parameters as `const`:
1. **Constant Propagation**: The compiler can propagate constant values directly into address and stride calculations.
2. **Elimination of Runtime Shift/OR Operations**: RoCC instructions require packing address components into registers via bit-shifts and bitwise-ORs (e.g., `(rows << 48) | (cols << 32) | spad_addr` in `gemmini_extended_mvin`). When these inputs are recognized as compile-time constants, the compiler pre-calculates the packed register value, completely eliminating runtime overhead.
3. **Loop Optimization**: Loop bounds such as `tiles` and `tiles_K` become compile-time constants, enabling loop unrolling and efficient induction variable optimization.

#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Assuming MATMUL_M, MATMUL_N, MATMUL_K, BLK, BLK_OUTC, A_S_BYTES, B_S_BYTES, BLOCKS_M, BLOCKS_N, REAL_M, REAL_N, scale_factors, C_hw, A_in, B_in, A_scales_row, B_scales_col are defined in the context.

void solution(void) {
  // Declare tiling bounds and spad offsets as compile-time constants
  const int tiles   = BLK / DIM;
  const int tiles_K = MATMUL_K / DIM;
  const int GK      = MATMUL_K / 32;

  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  const uint32_t a_base = 0;
  const int SPAD_DEST = 128;

  // Static buffers for pre-transposed scale factors
  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];
  
  // One-time initialization flag
  static int initialized = 0;
  if (!initialized) {
    // Transpose A scales: from [GK][ROWS] to [BLOCKS_M][GK*BLK]
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      for (int g = 0; g < GK; g++) {
        memcpy(&A_scales_T[i0][g * BLK],
               &A_scales_row[g][i0 * BLK],
               BLK);
      }
    }

    // Transpose B scales: from [GK][COLS] to [BLOCKS_N][GK*BLK]
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      for (int g = 0; g < GK; g++) {
        memcpy(&B_scales_T[j0][g * BLK],
               &B_scales_col[g][j0 * BLK],
               BLK);
      }
    }
    initialized = 1;
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // Load B tiles into scratchpad once per j0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);
      }
    }

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      gemmini_flush(0);

      // Step 1: Configure execution unit with MX formats
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load pre-transposed scale factors
      gemmini_mx_load_scales((uint64_t)A_scales_T[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

      // Step 3: Load A tiles into scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Step 4: Configure output requantization
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Dispatch optimized paths based on boundary conditions
      if (i0 < BLOCKS_M - 1 && j0 < BLOCKS_N - 1) {
        // Fast path: full tiles, no padding
        gemmini_loop_ws_spad(tiles, tiles, tiles_K, 0, 0, 0,
                             a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                             false, false, false, false, false,
                             NO_ACTIVATION, 0, 0, false, 0x38);

        for (int r = 0; r < BLK; r++) {
          gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK, BLK);
        }

        gemmini_fence();
      } else {
        // Boundary path: handle partial tiles with padding
        int pad_i = BLK - mrows;
        int pad_j = BLK - ncols;

        gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                             a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                             false, false, false, false, false,
                             NO_ACTIVATION, 0, 0, false, 0x38);

        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK, ncols);
        }

        gemmini_fence();
      }
    }
  }
}
