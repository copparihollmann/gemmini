void solution(void) {
  // Use full matrix dimension tiles to maximize scratchpad & accumulator utilization
  int tiles_I = MATMUL_M / DIM;
  int tiles_J = MATMUL_N / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  int SPAD_DEST = 128;

  // 1. Configure execution for weight-stationary MX fp8
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Load scale factors DIRECTLY from global DRAM arrays (Strategy 5)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. MVIN A tiles directly from A_in
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM;
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *) dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 4. MVIN B tiles directly from B_in
  // FIX: Use gemmini_extended_mvin (port 0) instead of gemmini_extended_mvin2 (port 1)
  // This is required for RTL compatibility with gemmini_loop_ws_spad
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM;
      uint32_t sp_addr = b_base + (j * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *) dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 5. Configure the requantizer writing boundaries
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                               tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 6. Compute using the fused weight-stationary hardware loop
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);

  // 7. Drain BF16 outputs from SMEM to DRAM
  // FIX: Correct the SMEM offset calculation and read all elements in one call
  gemmini_mx_read_smem(
    &C_hw[0][0],
    SPAD_DEST * 16, 
    MATMUL_M * MATMUL_N);

  // Ensure host execution waits for the DMA store operation to finish
  gemmini_fence();
}
