void solution(void) {
  // Hoist CONFIG_EX outside all loops since the format, strides, and dataflow are invariant
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int full_blocks_m = REAL_M / BLK;
  int full_blocks_n = REAL_N / BLK;

  // 1. Fast-path core loop: fully inside the matrix, no padding or boundary checks needed
  for (int i0 = 0; i0 < full_blocks_m; i0++) {
    // A scales only depend on i0: copy once per i0, reuse across all j0.
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

    for (int j0 = 0; j0 < full_blocks_n; j0++) {
      // B scales depend on j0: copy once per (i0,j0).
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

      int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      // B tiles now placed after A tiles since we're using the same port
      uint32_t b_base = a_base + tiles * tiles_K * DIM;
      int SPAD_DEST = 128;

      gemmini_flush(0);
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (j * tiles_K + k) * DIM, DIM, DIM);
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
      
      // Fast-path hardware compute: no dynamic padding calculations
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      for (int r = 0; r < BLK; r++)
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, BLK);
      gemmini_fence();
    }
  }

  // 2. Boundary Loop Nest A: Bottom boundary row blocks (fractional M), all columns
  if (full_blocks_m < BLOCKS_M) {
    int i0 = full_blocks_m;
    int mrows = REAL_M - i0 * BLK;
    if (mrows > 0) {
      int pad_i = BLK - mrows;
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], mrows);

      for (int j0 = 0; j0 < BLOCKS_N; j0++) {
        int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;
        int pad_j = BLK - ncols;

        for (int g = 0; g < MATMUL_K / 32; g++)
          memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], ncols);

        int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
        uint32_t a_base = 0;
        // B tiles now placed after A tiles since we're using the same port
        uint32_t b_base = a_base + tiles * tiles_K * DIM;
        int SPAD_DEST = 128;

        gemmini_flush(0);
        gemmini_mx_load_scales((uint64_t)A_s, (mrows + 31) / 32 * 32, 0);
        gemmini_mx_load_scales((uint64_t)B_s, (ncols + 31) / 32 * 32, 1);
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i = 0; i < tiles; i++)
          for (int k = 0; k < tiles_K; k++)
            gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                  a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        gemmini_config_ld(MATMUL_N * sizeof(elem_t));
        for (int j = 0; j < tiles; j++)
          for (int k = 0; k < tiles_K; k++)
            gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                  b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        gemmini_config_st(BLK_OUTC * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
        gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                             SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
        for (int r = 0; r < mrows; r++)
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * BLK, ncols);
        gemmini_fence();
      }
    }
  }

  // 3. Boundary Loop Nest B: Right boundary column blocks (fractional N), full row blocks
  if (full_blocks_n < BLOCKS_N) {
    int j0 = full_blocks_n;
    int ncols = REAL_N - j0 * BLK;
    if (ncols > 0) {
      int pad_j = BLK - ncols;
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], ncols);

      for (int i0 = 0; i0 < full_blocks_m; i0++) {
        for (int g = 0; g < MATMUL_K / 32; g++)
          memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

        int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
        uint32_t a_base = 0;
        // B tiles now placed after A tiles since we're using the same port
        uint32_t b_base = a_base + tiles * tiles_K * DIM;
        int SPAD_DEST = 128;

        gemmini_flush(0);
        gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
        gemmini_mx_load_scales((uint64_t)B_s, (ncols + 31) / 32 * 32, 1);
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i = 0; i < tiles; i++)
          for (int k = 0; k < tiles_K; k++)
            gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                  a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        gemmini_config_ld(MATMUL_N * sizeof(elem_t));
        for (int j = 0; j < tiles; j++)
          for (int k = 0; k < tiles_K; k++)
            gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                  b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        gemmini_config_st(BLK_OUTC * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
        gemmini_loop_ws_spad(tiles, tiles, tiles_K, 0, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                             SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
        for (int r = 0; r < BLK; r++)
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * BLK, ncols);
        gemmini_fence();
      }
    }
  }
}
