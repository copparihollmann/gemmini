void solution(void) {
  // Early-out if the problem exceeds on-chain memory limits.
  const int tiles_I = (REAL_M + DIM - 1) / DIM;
  const int tiles_J = (REAL_N + DIM - 1) / DIM;
  const int tiles_K = MATMUL_K / DIM;

  // Check scratchpad constraint: (tiles_I*tiles_K + tiles_J*tiles_K)*DIM <= 8192
  if ((tiles_I * tiles_K + tiles_J * tiles_K) * DIM > 8192) return;
  // Check accumulator constraint: tiles_I*tiles_J*DIM <= 256
  if (tiles_I * tiles_J * DIM > 256) return;

  const int total_I = tiles_I * DIM;
  const int total_J = tiles_J * DIM;

  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  const int SPAD_DEST = 0; // Start from beginning of mx_smem

  // Step 1: Configure execution (invariant, hoisted)
  gemmini_extended3_config_ex(
      WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      /*C_stride=*/1, /*A_stride=*/1, /*A_transpose=*/0, /*B_transpose=*/0,
      /*set_only_strides=*/false,
      /*act_mx_fmt=*/0, /*wgt_mx_fmt=*/0, /*out_mx_fmt=*/3, /*uselut=*/0);

  // Step 2: Load scales directly from global memory
  gemmini_mx_load_scales((uint64_t)A_scales_row, sizeof(A_scales_row), /*sel=*/0);
  gemmini_mx_load_scales((uint64_t)B_scales_col, sizeof(B_scales_col), /*sel=*/1);

  // Step 3: Load A tiles into scratchpad (strided)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      int rows = (i == tiles_I - 1 && REAL_M % DIM != 0) ? (REAL_M % DIM) : DIM;
      int cols = (k == tiles_K - 1 && MATMUL_K % DIM != 0) ? (MATMUL_K % DIM) : DIM;
      int addr = (i * DIM) * MATMUL_K + k * DIM;
      
      if (rows > 0 && cols > 0) {
        gemmini_extended_mvin((void *)(((elem_t *)A_in) + addr),
                              a_base + (i * tiles_K + k) * DIM,
                              cols, rows);
      }
    }
  }

  // Step 4: Load B tiles into scratchpad (transposed-strided)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      int rows = (k == tiles_K - 1 && MATMUL_K % DIM != 0) ? (MATMUL_K % DIM) : DIM;
      int cols = (j == tiles_J - 1 && REAL_N % DIM != 0) ? (REAL_N % DIM) : DIM;
      int addr = (k * DIM) * MATMUL_N + (j * DIM);

      if (rows > 0 && cols > 0) {
        gemmini_extended_mvin2((void *)(((elem_t *)B_in) + addr),
                               b_base + (k * tiles_J + j) * DIM,
                               cols, rows);
      }
    }
  }

  // Step 5: Configure output requantization
  gemmini_config_st(total_J * sizeof(out_t));
  gemmini_mxquant_config_mvout(
      (uint64_t)scale_factors,
      /*i_bound=*/tiles_I, /*j_bound=*/tiles_J, /*k_bound=*/tiles_K,
      /*scale_act_sel=*/0, /*scale_w_sel=*/0,
      /*lut_update_granularity=*/1);

  // Step 6: Execute matmul in fused hardware loop
  const int pad_i = total_I - REAL_M;
  const int pad_j = total_J - REAL_N;
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      pad_i, pad_j, 0,
      a_base, b_base, 0, SPAD_DEST,
      /*A_transpose=*/false, /*B_transpose=*/false,
      /*full_C=*/false, /*low_D=*/false, /*ex_accumulate=*/false,
      NO_ACTIVATION, /*a_spad_id=*/0, /*b_spad_id=*/0,
      /*is_resadd=*/false, /*skips=*/0x38);

  // Step 7: Drain final results
  const int num_outputs = REAL_M * REAL_N;
  gemmini_mx_read_smem((uint64_t)C_hw, SPAD_DEST * 16, num_outputs);
  gemmini_fence();
}
