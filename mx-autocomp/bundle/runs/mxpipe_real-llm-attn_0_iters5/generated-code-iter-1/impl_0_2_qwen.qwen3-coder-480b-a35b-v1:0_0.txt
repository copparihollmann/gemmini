// Optimized outer-tiled MX matmul using maximum scratchpad/accumulator utilization
void solution(void) {
  // Precompute tile counts based on matrix dimensions
  const int tiles_I = MATMUL_M / DIM;
  const int tiles_J = MATMUL_N / DIM;
  const int tiles_K = MATMUL_K / DIM;

  // Calculate base addresses for A and B in scratchpad
  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
  const int SPAD_DEST = 0;  // Use start of accumulator for output

  // Ensure memory constraints are satisfied
  if ((tiles_I * tiles_K + tiles_J * tiles_K) * DIM > 8192) return;  // Scratchpad overflow
  if (tiles_I * tiles_J * DIM > 256) return;                         // Accumulator overflow

  // Flush previous state
  gemmini_flush(0);

  // Configure execution once (assuming FP8 E4M3 inputs, BF16 output)
  gemmini_extended3_config_ex(
      WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
      /*set_only_strides*/false,
      /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

  // Load scale factors directly from global arrays (no host staging)
  gemmini_mx_load_scales((uint64_t)A_scales_row, sizeof(A_scales_row), 0);  // A (activations)
  gemmini_mx_load_scales((uint64_t)B_scales_col, sizeof(B_scales_col), 1);  // B (weights)

  // Load A tiles into scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin(
          (void *)(((elem_t *)A_in) + i * tiles_K * DIM * DIM + k * DIM),
          a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // Load B tiles into scratchpad
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin(
          (void *)(((elem_t *)B_in) + k * DIM * MATMUL_N + j * DIM),
          b_base + (k * tiles_J + j) * DIM, DIM, DIM);
    }
  }

  // Configure output and requantization parameters
  gemmini_config_st(MATMUL_N * sizeof(out_t));
  gemmini_mxquant_config_mvout(
      (uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      0, 0, 1);  // scale_act_sel=0, scale_w_sel=0, granularity=1

  // Execute the full matmul in hardware
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      0, 0, 0,  // No padding needed due to exact multiples
      a_base, b_base, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Drain results from shared memory
  for (int i = 0; i < MATMUL_M; i++) {
    gemmini_mx_read_smem(
        &C_hw[i][0],
        SPAD_DEST * 16 + i * MATMUL_N,
        MATMUL_N);
  }

  // Ensure all DMA operations complete before host accesses output
  gemmini_fence();
}
