Looking at the error and the code, I can identify several issues that would cause incorrect results:

1. The `gemmini_mx_read_smem` call has incorrect addressing - it's trying to read `ncols` elements but BF16 values are packed 4 per uint64 word
2. The output buffer addressing is incorrect for the packed BF16 format
3. The fence should be called once after all reads for a tile, not after each row read

Let me fix these issues:

```c
// Optimized MX matmul with global configuration and scale load hoisting.
void solution(void) {
  // Hoist invariant configurations to the very beginning
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST = 128;

  // Hoist output and requantization configurations
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    // Partial last M block: compute only the real rows, pad the rest.
    int mrows = REAL_M - i0 * BLK; 
    if (mrows > BLK) mrows = BLK; 
    if (mrows <= 0) continue;

    // Stage activation (A) scale factors once per row-block i0 (hoisted out of j0 loop)
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
    }
    gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; 
      if (ncols > BLK) ncols = BLK; 
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Stage weight (B) scale factors for the current column-block j0
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles directly from DRAM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // MVIN B tiles directly from DRAM
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (k * tiles + j) * DIM, DIM, DIM);
        }
      }

      // Fused compute loop
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Readout to host memory - fix the addressing for packed BF16 format
      int bf16_words_to_read = (ncols + 3) / 4; // Round up to account for 4 BF16 per word
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK / 4],
                             SPAD_DEST * 16 + r * BLK, bf16_words_to_read * 4); // 4 BF16 per word
      }
      gemmini_fence();
    }
  }
}
```
