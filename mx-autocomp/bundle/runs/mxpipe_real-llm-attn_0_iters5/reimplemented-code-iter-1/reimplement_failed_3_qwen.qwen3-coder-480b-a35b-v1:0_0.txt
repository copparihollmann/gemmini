void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division for actual tiles needed
      int tiles_J = (ncols + DIM - 1) / DIM;  // Ceiling division for actual tiles needed
      int tiles_K = MATMUL_K / DIM;
      
      // Proper scratchpad layout calculation
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;  // Fixed: use tiles_J, not tiles_I
      
      int SPAD_DEST = 0;  // Use beginning of accumulator space
      int pad_i = tiles_I * DIM - mrows;  // Correct padding calculation
      int pad_j = tiles_J * DIM - ncols;  // Correct padding calculation

      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      
      // Load A scales - fixed size calculation (BLK elements, but scales are per row)
      gemmini_mx_load_scales((uint64_t)&A_scales_row[0][i0 * BLK], tiles_I * sizeof(uint8_t), 0);
      // Load B scales - fixed size calculation  
      gemmini_mx_load_scales((uint64_t)&B_scales_col[0][j0 * BLK], tiles_J * sizeof(uint8_t), 1);
      
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      // Load A tiles
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      // Load B tiles  
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (j * tiles_K + k) * DIM, DIM, DIM);  // Fixed B tile layout
      
      gemmini_config_st(MATMUL_N * sizeof(out_t));  // Fixed stride
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      
      // Fixed output reading - account for BF16 packing (4 per uint64)
      for (int r = 0; r < mrows; r++) {
        int bf16_words = (ncols + 3) / 4;  // Ceiling division for BF16 packing
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK], 
                             SPAD_DEST * 16 + r * tiles_J * DIM,  // Fixed offset calculation
                             ncols);  // Number of BF16 elements to read
      }
      
      gemmini_fence();
    }
  }
}
