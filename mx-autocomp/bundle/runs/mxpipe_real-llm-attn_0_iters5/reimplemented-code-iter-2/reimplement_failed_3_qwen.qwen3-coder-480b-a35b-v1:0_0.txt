void solution(void) {
  // Hoisted static setup
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              /*C_stride*/1, /*A_stride*/1,
                              /*A_transpose*/0, /*B_transpose*/0,
                              /*set_only_strides*/false,
                              /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3,
                              /*uselut*/0);

  int tiles_I = BLK / DIM;
  int tiles_J = BLK / DIM; 
  int tiles_K = MATMUL_K / DIM;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    int pad_i = BLK - mrows;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    int SPAD_DEST = 0;

    // Hoisted A matrix and scale loading (invariant across j0)
    for (int g = 0; g < MATMUL_K / 32; g++)
      for (int r = 0; r < mrows; r++)
        A_s[g][r] = A_scales_row[g][i0 * BLK + r];

    gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);

    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, DIM, DIM);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_j = BLK - ncols;

      // B scales and B matrix loading (dependent on j0)
      for (int g = 0; g < MATMUL_K / 32; g++)
        for (int c = 0; c < ncols; c++)
          B_s[g][c] = B_scales_col[g][j0 * BLK + c];

      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                 b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      // Compute and output
      gemmini_config_st(ncols * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 1, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0,
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      int bf16_count = mrows * ncols;
      int words_needed = (bf16_count + 3) / 4; // 4 BF16 per uint64 word
      gemmini_mx_read_smem((uint64_t)C_hw + (i0 * BLK * REAL_N + j0 * BLK) * sizeof(out_t), 
                          SPAD_DEST * 16, bf16_count);
      gemmini_fence();
    }
  }
}
