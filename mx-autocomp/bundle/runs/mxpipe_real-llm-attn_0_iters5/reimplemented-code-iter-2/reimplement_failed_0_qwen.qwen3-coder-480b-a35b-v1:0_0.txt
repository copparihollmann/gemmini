void solution(void) {
  // Hoist invariant configuration outside all loops
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Load the ENTIRE scale factor matrices directly from their source DRAM locations.
  // This utilizes the hardware DMA to stream scales directly into the scale memories (sel 0 = A, sel 1 = B),
  // completely eliminating host-CPU memcpy staging and redundant inner-loop scale loads.
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; 
    if (mrows > BLK) mrows = BLK; 
    if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; 
      if (ncols > BLK) ncols = BLK; 
      if (ncols <= 0) continue;

      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division
      int tiles_J = (ncols + DIM - 1) / DIM;  // Ceiling division
      int tiles_K = MATMUL_K / DIM;
      
      // Properly calculate scratchpad layout
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      int SPAD_DEST = 0;  // Use beginning of accumulator
      int pad_i = tiles_I * DIM - mrows;
      int pad_j = tiles_J * DIM - ncols;

      gemmini_flush(0);

      // MVIN A tiles directly from strided source
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // MVIN B tiles directly from strided source
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                 b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Configure store and requantizer
      gemmini_config_st(MATMUL_N * sizeof(out_t));
      // Fix scale factor addressing - use the correct scale block for this tile
      gemmini_mxquant_config_mvout((uint64_t)scale_factors + (i0 * BLK) * MATMUL_N * sizeof(uint8_t),
                                   tiles_I, tiles_J, tiles_K, 0, 1, 1);

      // Fused WS compute loop over scratchpad
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Read output from shared memory - fix BF16 packing
      int bf16_words_per_row = (ncols + 1) / 2;  // Ceiling division for BF16 packing (2 per word)
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK],
                             SPAD_DEST * 16 + r * tiles_J * DIM, ncols);
      }
      gemmini_fence();
    }
  }
}
