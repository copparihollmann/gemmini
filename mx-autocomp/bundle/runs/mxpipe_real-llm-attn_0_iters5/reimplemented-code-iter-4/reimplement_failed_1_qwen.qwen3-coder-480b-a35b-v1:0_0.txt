void solution(void) {
  // Perform initial flush once at the start, outside all loops
  gemmini_flush(0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

    // A scales only depend on i0: copy once per i0, reuse across all j0.
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // B scales depend on j0: copy once per (i0,j0).
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division
      int tiles_J = (ncols + DIM - 1) / DIM;  // Ceiling division
      int tiles_K = MATMUL_K / DIM;
      
      // Properly calculate the actual dimensions for this block
      int actual_m = ((tiles_I - 1) * DIM + DIM > mrows) ? mrows : tiles_I * DIM;
      int actual_n = ((tiles_J - 1) * DIM + DIM > ncols) ? ncols : tiles_J * DIM;

      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;  // Fixed B base calculation
      
      int SPAD_DEST = 0;  // Start from beginning of accumulator

      // Configure execution
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      
      // Load scale factors
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  // A scales (sel=0)
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  // B scales (sel=1)
      
      // Load A tiles
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          int a_addr = (i0 * BLK + i * DIM) * MATMUL_K + k * DIM;
          int spad_addr = a_base + (i * tiles_K + k) * DIM;
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + a_addr),
                                spad_addr, DIM, DIM);
        }
      }
      
      // Load B tiles
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          int b_addr = (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM);
          int spad_addr = b_base + (j * tiles_K + k) * DIM;  // Fixed B layout
          gemmini_extended_mvin2((void*)(((elem_t*)B_in) + b_addr),
                                 spad_addr, DIM, DIM);
        }
      }
      
      // Configure output
      gemmini_config_st(MATMUL_N * sizeof(elem_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles_I, tiles_J, tiles_K,
                                   0, 0, 1);  // Correct bounds
      
      // Compute matmul
      int pad_i = tiles_I * DIM - mrows;
      int pad_j = tiles_J * DIM - ncols;
      
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 
                           pad_i, pad_j, 0,
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, 
                           NO_ACTIVATION, 0, 0, false, 0x38);
      
      // Read results - fix the addressing
      gemmini_config_st(MATMUL_N * sizeof(out_t));
      for (int i = 0; i < mrows; i++) {
        int c_row_offset = (i0 * BLK + i) * MATMUL_N + (j0 * BLK);
        gemmini_extended_mvout((void*)((out_t*)C_hw + c_row_offset),
                               SPAD_DEST + i * MATMUL_N, 
                               ncols, 1);
      }
      
      gemmini_fence();
    }
  }
}
