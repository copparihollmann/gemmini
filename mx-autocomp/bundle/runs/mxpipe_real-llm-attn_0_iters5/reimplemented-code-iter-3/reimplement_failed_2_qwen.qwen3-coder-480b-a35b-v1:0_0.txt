#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

#define DIM 16
#define ADDR_LEN 32
#define BF16_PER_WORD 4

typedef uint8_t  elem_t;   // fp8:e4m3 input elements
typedef uint8_t  welem_t;  // fp8:e4m3 weight elements
typedef uint64_t out_t;    // 4x bf16 packed per word

void solution(void) {
  int tiles_I = MATMUL_M / DIM;
  int tiles_J = MATMUL_N / DIM;
  int tiles_K = MATMUL_K / DIM;

  int pad_i = MATMUL_M - REAL_M;
  int pad_j = MATMUL_N - REAL_N;
  int pad_k = MATMUL_K - REAL_K;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  int SPAD_DEST = 128;

  uint32_t scale_factors[512] __attribute__((aligned(32))) = {0};

  gemmini_flush(0);

  // 1. Configure Gemmini execution format for weight-stationary MX fp8
  gemmini_extended3_config_ex(
      WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Direct load scale factors from global scale arrays (Avoid host-side staging)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. MVIN A tiles directly from global input DRAM buffer
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t *)A_in) + i * DIM * MATMUL_K + k * DIM;
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 4. MVIN B tiles directly from global input DRAM buffer
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t *)B_in) + k * DIM * MATMUL_N + j * DIM;
      uint32_t sp_addr = b_base + (k * tiles_J + j) * DIM;
      gemmini_extended_mvin((void *)dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 5. Configure output strides and requantizer parameters
  gemmini_config_st(MATMUL_N / BF16_PER_WORD * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 6. Execute high-performance hardware tiled WS loop
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      pad_i, pad_j, pad_k,
      a_base,
      b_base,
      0,
      SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION,
      0, 0,
      false,
      0x38);

  // 7. Drain output row-by-row into destination output buffer
  for (int r = 0; r < REAL_M; r++) {
    gemmini_mx_read_smem(&C_hw[r][0], SPAD_DEST * 16 + r * MATMUL_N, REAL_N);
  }

  gemmini_fence();
}
