Looking at the code and the error message, I can identify several issues that need to be fixed:

1. The programming model sequence is violated - `gemmini_mx_load_scales` is called for B first, then A, but both must be loaded before computation
2. The scratchpad layout for B matrix placement is incorrect
3. The output reading logic has issues with BF16 packing
4. Missing proper configuration of requantization parameters

void solution(void) {
  // 1. Flush and configure Gemmini once (hoisted from loops)
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  // 2. Pre-stage all scale factors to eliminate CPU memcpy overhead during computation
  uint8_t A_s_all[BLOCKS_M][A_S_BYTES];
  uint8_t B_s_all[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_s_all[i0][g * BLK], &A_scales_row[g][i0 * BLK], BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_s_all[j0][g * BLK], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  // 3. Weight-Stationary Tiled Loops (j0 is outer)
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

    int tiles = (ncols + DIM - 1) / DIM;  // Ceiling division
    int tiles_K = MATMUL_K / DIM;
    
    // Calculate proper B base address at the top of scratchpad
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;

    // Load weight (B) tiles once per j0 and reuse across all i0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                               b_base + (j * tiles_K + k) * DIM, DIM, DIM);
      }
    }

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division
      int tiles_J = tiles;
      
      uint32_t a_base = 0;
      int SPAD_DEST = 0;  // Use beginning of scratchpad for output

      // Load activation (A) tiles first
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Now load scale factors in correct order - A first, then B
      gemmini_mx_load_scales((uint64_t)A_s_all[i0], A_S_BYTES, 0);  // A scales first
      gemmini_mx_load_scales((uint64_t)B_s_all[j0], B_S_BYTES, 1);  // B scales second

      // Configure requantization boundaries and run fused loop
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      
      int pad_i = tiles_I * DIM - mrows;
      int pad_j = tiles_J * DIM - ncols;
      
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results to DRAM - correctly handle BF16 packing (4 per uint64)
      int bf16_cols = (ncols + 3) / 4 * 4;  // Round up to nearest multiple of 4
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK],
                             SPAD_DEST * 16 + r * BLK_OUTC, bf16_cols);
      }
      gemmini_fence();
    }
  }
}
