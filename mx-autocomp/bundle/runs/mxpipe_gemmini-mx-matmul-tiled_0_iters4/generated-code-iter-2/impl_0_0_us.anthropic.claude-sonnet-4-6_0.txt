void solution(void) {
  int tiles   = BLK / DIM;        // tiles along I and J
  int tiles_K = MATMUL_K / DIM;   // tiles along K
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST = 128;

  // One-time setup: flush, configure execution, configure load/store strides.
  // These persist across outer iterations.
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);
  // Configure load stride for A (row stride = MATMUL_K bytes, elem_t=1 byte)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Configure load stride for B on mvin2 port (same stride)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
  // Configure store stride
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      // Stage A block (BLK rows × MATMUL_K cols)
      for (int r = 0; r < BLK; r++)
        memcpy(&A_blk[r][0], &A_in[i0 * BLK + r][0], MATMUL_K);

      // Stage B block (BLK rows × MATMUL_K cols)
      for (int r = 0; r < BLK; r++)
        memcpy(&B_blk[r][0], &B_in[j0 * BLK + r][0], MATMUL_K);

      // Stage per-32-group scales for this (i0, j0) pair
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Load A and B scale factors into hardware scale memories
      // sel=0 -> A (activation) scales, sel=1 -> B (weight) scales
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles into scratchpad using mvin port 0, batching up to MAX_BLOCK_LEN=4
      // A tile layout: a_base + (i * tiles_K + k) * DIM
      // Inner loop over k is the fast index; batch consecutive k tiles (up to 4)
      for (int i = 0; i < tiles; i++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          // batch consecutive k-tiles share the same i-row in A, contiguous in DRAM
          elem_t *dram_ptr = ((elem_t *)A_blk) + i * DIM * MATMUL_K + k * DIM;
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
          // Use block mvin: cols = batch*DIM covers batch tiles horizontally,
          // but the scratchpad layout requires each tile at a separate spad row offset.
          // gemmini_extended_mvin with cols=batch*DIM, rows=DIM loads batch consecutive
          // DIM-wide tiles from contiguous DRAM into consecutive spad rows.
          gemmini_extended_mvin((void *)dram_ptr, spad_addr,
                                (uint32_t)(batch * DIM), DIM);
          k += batch;
        }
      }

      // MVIN B tiles into scratchpad using mvin2 port (independent DMA channel)
      // B tile layout: b_base + (j * tiles_K + k) * DIM
      for (int j = 0; j < tiles; j++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          elem_t *dram_ptr = ((elem_t *)B_blk) + j * DIM * MATMUL_K + k * DIM;
          uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
          gemmini_extended_mvin2((void *)dram_ptr, spad_addr,
                                 (uint32_t)(batch * DIM), DIM);
          k += batch;
        }
      }

      // Configure the requantizer output: scale_dram, tile bounds, scale selectors
      // i_bound=tiles, j_bound=tiles, k_bound=tiles_K must match loop_ws_spad dims
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Fused hardware matmul: funct24 marker is one-shot, issued inside the macro
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           0, 0, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM
      // smem_off_bf16 = SPAD_DEST * 16, num_bf16 = BLK * BLK
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      // Unpack BF16 results and store to output accumulator
      for (int r = 0; r < BLK; r++)
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
    }
  }
}
