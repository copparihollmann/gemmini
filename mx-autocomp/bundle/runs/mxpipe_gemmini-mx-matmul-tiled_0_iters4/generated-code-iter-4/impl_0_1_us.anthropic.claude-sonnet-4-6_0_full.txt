Looking at the original code, I need to understand what it does:
- Outer loops over BLOCKS_M × BLOCKS_N blocks
- For each block pair, copies A and B data, loads scales, mvins tiles, computes, reads results
- The original has `gemmini_flush(0)` inside the loop which is expensive
- mvin uses a single port for both A and B tiles sequentially
- No batching of mvin commands (MAX_BLOCK_LEN=4 could batch up to 4 tiles)

Key optimizations I can make:
1. Move `gemmini_flush(0)` and `gemmini_extended3_config_ex` outside the loop (config persists)
2. Use `gemmini_extended_mvin` (port 0) for A and `gemmini_extended_mvin2` (port 1) for B to overlap loads
3. Batch mvin commands using MAX_BLOCK_LEN=4 where possible (load up to 4 consecutive DIM-row tiles per command)
4. The `gemmini_config_st` and format config only need to be set once

Let me check the mvin batching: for A tiles, consecutive k tiles at fixed i are at addresses `a_base + (i*tiles_K + k)*DIM` - these are consecutive in spad. So for each i, we can batch up to 4 k-tiles per mvin. Similarly for B.

For the DRAM side: `A_blk[i*DIM][k*DIM]` - stride is `MATMUL_K`, so consecutive k tiles are `DIM` bytes apart in a row of `MATMUL_K` bytes, not contiguous. Can't batch multiple k-tiles in one mvin from DRAM unless they're contiguous. Actually for fixed i, varying k: DRAM ptr = `A_blk + i*DIM*MATMUL_K + k*DIM` - k tiles are DIM bytes apart but stride is MATMUL_K. Not contiguous across k, so batching across k won't work. We can batch across i only if k is fixed... but the loop order is i then k. Let me keep single-tile mvins but use dual ports.

```c
// AUTO-GENERATED outer-tiled MX matmul baseline (256x256x512).
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST    = 128;

  // One-time configuration: format, dataflow, strides
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {

      // Stage A block (BLK rows x MATMUL_K cols)
      for (int r = 0; r < BLK; r++)
        memcpy(&A_blk[r][0], &A_in[i0 * BLK + r][0], MATMUL_K);

      // Stage B block (BLK rows x MATMUL_K cols)
      for (int r = 0; r < BLK; r++)
        memcpy(&B_blk[r][0], &B_in[j0 * BLK + r][0], MATMUL_K);

      // Stage scale factors
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Load scale factors: sel=0 for A (activation), sel=1 for B (weight)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A tiles via port 0, B tiles via port 1 (interleaved for overlap)
      // For each i-tile, batch up to MAX_BLOCK_LEN=4 consecutive k-tiles
      // A tile (i,k): spad = a_base + (i*tiles_K + k)*DIM  (consecutive for fixed i, varying k)
      // DRAM ptr for A[i][k]: base + i*DIM*MATMUL_K + k*DIM, stride=MATMUL_K (already configured)
      for (int i = 0; i < tiles; i++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          // Check DRAM contiguity: for fixed i, tiles at k..k+batch-1
          // DRAM: row i*DIM, columns k*DIM..(k+batch-1)*DIM+DIM-1
          // Each tile row spans DIM bytes within a MATMUL_K-wide row.
          // gemmini_extended_mvin with cols=batch*DIM, rows=DIM loads batch tiles
          // only if the DRAM layout is contiguous (cols consecutive in a row).
          // Since stride=MATMUL_K and we want batch*DIM cols from one DRAM row,
          // this is valid: the mvin reads rows=DIM rows of cols=batch*DIM each.
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_blk) + i * DIM * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            batch * DIM,
            DIM);
          k += batch;
        }
      }

      // B tiles via port 1
      for (int j = 0; j < tiles; j++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_blk) + j * DIM * MATMUL_K + k * DIM),
            b_base + (j * tiles_K + k) * DIM,
            batch * DIM,
            DIM);
          k += batch;
        }
      }

      // Configure requantizer: bounds must match loop_ws_spad tile counts
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Fused hardware matmul (one-shot marker + loop)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           0, 0, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM
      gemmini_mx_read_smem(&C_blk[0][0], SPAD_DEST * 16, BLK * BLK);
      gemmini_fence();

      // Unpack BF16 results (4 per uint64) into C_acc
      for (int r = 0; r < BLK; r++)
        for (int c = 0; c < BLK; c++) {
          uint16_t b = (uint16_t)((C_blk[r][c / 4] >> ((c % 4) * 16)) & 0xFFFF);
          C_acc[i0 * BLK + r][j0 * BLK + c] = bf16f(b);
        }
    }
  }
}
```
