Looking at the original code, I need to rewrite the Flash Attention implementation for MX-Gemmini. The original code has a comment saying "Error: max retries reached" as the plan, which means I should just produce a correct, clean implementation following all the rules.

Let me analyze the key aspects:
1. Flash Attention with S=128, D=64, tile BT along sequence dimension
2. Two matmuls per tile: Q@K^T (scores) and P@V (output accumulation)
3. Online softmax with running max/sum
4. Must follow exact programming model sequence

```c
void solution(void) {
  // ---- flash attention: tile K/V along sequence, online softmax ----
  for (int i = 0; i < ATT_S; i++) { M_run[i] = -1e30f; L_run[i] = 0.0f; }
  memset(O_acc, 0, ATT_S * ATT_D * sizeof(float));

  for (int t = 0; t < N_TILES; t++) {
    // 1) scores tile S1 = Q @ K_t^T   [S x BT] on Gemmini
    {
      int tiles_I = ATT_S / DIM;   // rows of Q
      int tiles_J = BT / DIM;      // cols of KT tile
      int tiles_K = ATT_D / DIM;   // shared dim
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      int SPAD_DEST = 128;

      // Step 1: config_ex (fp8 input, bf16 output)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
          /*set_only_strides*/false,
          /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: load scales for A (Q) and B (KT tile)
      gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);

      // Repack KT column scales into contiguous [group*BT + n] layout
      static uint8_t KT_sc_tile[(ATT_D / 32) * BT];
      for (int sg = 0; sg < ATT_D / 32; sg++)
        for (int sn = 0; sn < BT; sn++)
          KT_sc_tile[sg * BT + sn] = KT_scales_col[sg][t * BT + sn];
      gemmini_mx_load_scales((uint64_t)KT_sc_tile, (ATT_D / 32) * BT, 1);

      // Step 4: mvin A tiles (Q) into scratchpad
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // mvin B tiles (KT) into scratchpad via mvin2
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + t * BT + j * DIM),
              b_base + (k * tiles_J + j) * DIM, DIM, DIM);

      // Step 5: configure requantized mvout
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // Step 6: fused hardware matmul (one-shot marker + loop)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: drain results to DRAM
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * BT);
      gemmini_fence();
    }

    // 2) online softmax update + fp8 requant of P_t (CPU)
    for (int i = 0; i < ATT_S; i++) {
      float m_new = M_run[i];
      float row[BT];
      for (int j = 0; j < BT; j++) {
        uint16_t b = (uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF);
        row[j] = bf16_to_f(b);
        if (row[j] > m_new) m_new = row[j];
      }
      float corr = exp_f(M_run[i] - m_new);
      float l_new = L_run[i] * corr;
      for (int j = 0; j < BT; j++) { row[j] = exp_f(row[j] - m_new); l_new += row[j]; }
      for (int d = 0; d < ATT_D; d++) O_acc[i][d] *= corr;
      M_run[i] = m_new; L_run[i] = l_new;
      // per-32 group scale + fp8 RTZ of P (unnormalized)
      for (int g = 0; g < GROUPS_BT; g++) {
        float gmax = 0.0f;
        for (int j = g * 32; j < (g + 1) * 32; j++)
          if (f_abs(row[j]) > gmax) gmax = f_abs(row[j]);
        int e = 0;
        if (gmax > 0.0f) e = f_exp(gmax / 16.0f) + 1;
        float inv_scale = pow2i(-e);
        P_s[g][i] = (uint8_t)(e + 127);
        for (int j = g * 32; j < (g + 1) * 32; j++)
          P_q[i][j] = fp8_e4m3_rtz(row[j] * inv_scale);
      }
    }

    // 3) O_tile = P_t @ V_t  [S x D] on Gemmini; accumulate in fp32
    {
      int tiles_I = ATT_S / DIM;   // rows of P
      int tiles_J = ATT_D / DIM;   // cols of V tile
      int tiles_K = BT / DIM;      // shared dim
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      int SPAD_DEST = 128;

      // Step 1: config_ex (fp8 input, bf16 output)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
          /*set_only_strides*/false,
          /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: load scales for A (P_q) and B (V tile)
      // P_s layout: [GROUPS_BT][ATT_S] = [BT/32][S], group-major, row = group index
      // HW expects scale_a_mem[group * M_rows + i], so flat [GROUPS_BT * ATT_S] is correct
      gemmini_mx_load_scales((uint64_t)P_s, GROUPS_BT * ATT_S, 0);

      // V_scales_col[t*GROUPS_BT .. (t+1)*GROUPS_BT-1][0..ATT_D-1]
      // GROUPS_BT groups along K=BT, ATT_D scale values per group (one per col)
      gemmini_mx_load_scales((uint64_t)&V_scales_col[t * GROUPS_BT][0],
          GROUPS_BT * ATT_D, 1);

      // Step 4: mvin A tiles (P_q) into scratchpad
      gemmini_config_ld(BT * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)P_q) + i * DIM * BT + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // mvin B tiles (V) into scratchpad via mvin2
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)V_in) + (t * BT + k * DIM) * ATT_D + j * DIM),
              b_base + (k * tiles_J + j) * DIM, DIM, DIM);

      // Step 5: configure requantized mvout
      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // Step 6: fused hardware matmul (one-shot marker + loop)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: drain results to DRAM
      gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();

      // Accumulate BF16 hardware output into fp32 O_acc
      for (int i = 0; i < ATT_S; i++)
        for (int d = 0; d < ATT_D; d++) {
          uint16_t b = (uint16_t)((O_hw[i][d / 4] >> ((d % 4) * 16)) & 0xFFFF);
          O_acc[i][d] += bf16_to_f(b);
        }
    }
  }

  // final normalization by the online denominator
  for (int i = 0; i < ATT_S; i++) {
    float inv_l = 1.0f / L_run[i];
    for (int d = 0; d < ATT_D; d++) O_acc[i][d] *= inv_l;
  }
}
```
