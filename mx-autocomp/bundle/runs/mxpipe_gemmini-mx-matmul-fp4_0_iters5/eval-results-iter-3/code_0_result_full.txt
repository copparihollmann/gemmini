void solution(void) {
  const int ti = MATMUL_M / DIM / 2;   // 4
  const int tj = MATMUL_N / DIM / 2;   // 4
  const int tk = MATMUL_K / DIM;        // 16
  const int ni = MATMUL_M / DIM / ti;   // 2 I-chunks
  const int nj = MATMUL_N / DIM / tj;   // 2 J-chunks

  const uint32_t ab = 0;
  const uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM; // 8192-1024=7168

  const int SDST = 128;

  // Step 1: Configure (hoisted, invariant)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 2, 2, 3, 0);

  // Step 2: Load scales (hoisted, invariant across all tiles)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  gemmini_config_st(OUT_COLS * sizeof(out_t));

  // Outer loop: J-chunks — B tiles loaded once per J-chunk, reused across I-chunks
  for (int jc = 0; jc < nj; jc++) {
    // Load B tiles for this J-chunk into scratchpad (persists for all I-chunks)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        int gj = jc * tj + j;  // global j tile index
        gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + gj*DIM),
            bb + (k*tj + j)*DIM,
            DIM, DIM);
      }
    }

    // Inner loop: I-chunks — A reloaded per I-chunk, B reused from above
    for (int ic = 0; ic < ni; ic++) {
      // Load A tiles for this I-chunk
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < ti; i++) {
        int gi = ic * ti + i;  // global i tile index
        for (int k = 0; k < tk; k++) {
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in_hw) + gi*DIM*MATMUL_K + k*DIM),
              ab + (i*tk + k)*DIM,
              DIM, DIM);
        }
      }

      // Configure requantizer for this (ic, jc) output block
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 1);

      // Compute: A already in spad at ab, B already in spad at bb
      gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
          ab, BANK_NUM * BANK_ROWS, 0, SDST,
          false, false, false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results for this (ic, jc) output block
      // Output is ti*DIM rows × tj*DIM cols, packed as BF16
      // smem offset: SDST*16 (word offset into mx_smem)
      // Each row of ti*DIM output rows has tj*DIM BF16 values
      // Write to C_hw at row offset ic*ti*DIM, col offset jc*tj*DIM
      // C_hw has OUT_COLS = MATMUL_M/BF16_PER_WORD qwords per row (width MATMUL_N)
      // The read_smem reads a flat block; we map it to the correct C_hw slice.
      gemmini_mx_read_smem(
          &C_hw[ic * ti * DIM][jc * tj * DIM / BF16_PER_WORD],
          SDST * 16,
          ti * DIM * tj * DIM);
    }
  }

  gemmini_fence();
}
