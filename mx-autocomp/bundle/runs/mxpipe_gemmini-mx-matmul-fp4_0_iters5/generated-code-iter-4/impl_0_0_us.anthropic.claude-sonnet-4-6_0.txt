void solution(void) {
  // Full tile counts for 128x128x256 with DIM=16
  int tiles_I = MATMUL_M / DIM; // 8
  int tiles_J = MATMUL_N / DIM; // 8
  int tiles_K = MATMUL_K / DIM; // 16

  // Split I and J to fit accumulator: tiles_I*tiles_J*DIM <= 256
  // 8*8*16 = 1024 > 256, so tile: bi_step*bj_step*DIM <= 256
  // Use bi_step=1, bj_step=16 -> 1*16*16=256 fits exactly (but bj_step<=tiles_J=8)
  // Use bi_step=2, bj_step=8 -> 2*8*16=256 fits exactly
  // Or bi_step=1, bj_step=8 -> 1*8*16=128 <= 256 (safe)
  // Scratchpad: A block = bi_step*tiles_K*DIM, B block = bj_step*tiles_K*DIM
  // bi_step=4, bj_step=4: A=4*16*16=1024, B=4*16*16=1024, total=2048 <= 8192 ok
  // acc: 4*4*16=256 <= 256 ok (exactly fits)

  int bi_step = 4;
  int bj_step = 4;

  // Step 1: Configure execution with FP4 E2M1 format
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 2, 2, 3, 0);

  // Step 2: Load scale factors for A and B (full matrices, loaded once)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 5 (partial): Configure output stride once
  gemmini_config_st(OUT_COLS * sizeof(out_t));

  // Loop over output tiles in blocks that fit in accumulator
  for (int bi = 0; bi < tiles_I; bi += bi_step) {
    int cur_ti = bi_step;
    if (bi + cur_ti > tiles_I) cur_ti = tiles_I - bi;

    for (int bj = 0; bj < tiles_J; bj += bj_step) {
      int cur_tj = bj_step;
      if (bj + cur_tj > tiles_J) cur_tj = tiles_J - bj;

      // Scratchpad layout for this block:
      // A tiles: a_base at 0, needs cur_ti * tiles_K * DIM rows
      // B tiles: b_base at top of scratchpad, needs cur_tj * tiles_K * DIM rows
      uint32_t a_base = 0;
      uint32_t b_base = (uint32_t)(BANK_NUM * BANK_ROWS - cur_tj * tiles_K * DIM);

      // SPAD_DEST for output (use row offset after A tiles, well within smem range)
      int SDST = 0;

      // Step 4: MVIN A tiles for this bi block
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < cur_ti; i++) {
        int k = 0;
        while (k < tiles_K) {
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in_hw) + (bi + i) * DIM * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM * batch,
              DIM);
          k += batch;
        }
      }

      // Step 4b: MVIN B tiles for this bj block
      // FP4 E2M1: 4-bit elements, packed 2 per byte -> stride = MATMUL_N/2 bytes
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
      for (int k = 0; k < tiles_K; k++) {
        int j = 0;
        while (j < cur_tj) {
          int batch = cur_tj - j;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + (bj + j) * DIM),
              b_base + (k * cur_tj + j) * DIM,
              DIM * batch,
              DIM);
          j += batch;
        }
      }

      // Step 5: Configure requantized mvout with correct tile bounds
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          cur_ti, cur_tj, tiles_K,
          0, 0, 1);

      // Step 6: Compute weight-stationary tiled matmul
      gemmini_loop_ws_spad(cur_ti, cur_tj, tiles_K, 0, 0, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SDST,
          false, false, false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain results from mx_smem to DRAM for this output block
      // Results are cur_ti*DIM rows × cur_tj*DIM cols of BF16
      // Output DRAM address: C_hw[bi*DIM][bj*DIM]
      gemmini_mx_read_smem(
          (void*)(&C_hw[bi * DIM][bj * DIM]),
          SDST * 16,
          cur_ti * DIM * cur_tj * DIM);
      gemmini_fence();
    }
  }
}
