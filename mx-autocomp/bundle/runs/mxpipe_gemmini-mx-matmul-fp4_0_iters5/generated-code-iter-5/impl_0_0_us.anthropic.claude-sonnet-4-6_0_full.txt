Looking at the original code, I need to understand the problem:

1. The matrix dimensions are 128×128×256 with FP4 E2M1 format
2. The original uses `ti = MATMUL_M/DIM/2` and `tj = MATMUL_N/DIM/2` — halving the tile counts, suggesting double-buffering or splitting
3. The error "max retries reached" suggests the original may have a correctness issue with the halved tile counts not covering the full matrix

Let me analyze:
- MATMUL_M=128, MATMUL_N=128, MATMUL_K=256
- Full tiles: tiles_I=8, tiles_J=8, tiles_K=16
- Original uses ti=4, tj=4 (half), but only does one pass — this misses half the output rows and columns!

The fix: use full tile counts (ti=8, tj=8, tk=16) and ensure scratchpad fits:
- A tiles: 8*16*16 = 2048 rows ✓
- B tiles (FP4, half width): 16*8*16 = 2048 rows
- b_base = 8192 - 16*8*16 = 8192 - 2048 = 6144
- Total: 2048 + 2048 = 4096 ≤ 8192 ✓
- Accumulator: 8*8*16 = 1024 > 256 rows ✗

So we need to tile in software. Split into quadrants (ti=4, tj=4) and loop:

```c
void solution(void) {
  {
    int tiles_I = MATMUL_M / DIM;   // 8
    int tiles_J = MATMUL_N / DIM;   // 8
    int tiles_K = MATMUL_K / DIM;   // 16

    int ti = 4; // sub-tile count along I (fits accumulator: 4*4*16=256)
    int tj = 4; // sub-tile count along J

    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    int SDST = 128;

    // Step 1: Configure execution with FP4 E2M1 format
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 2, 2, 3, 0);

    // Step 2: Load scale factors for A and B (full matrices)
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // Step 4: MVIN all tiles of A into scratchpad
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    for (int i = 0; i < tiles_I; i++) {
      int k = 0;
      while (k < tiles_K) {
        int batch = tiles_K - k;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM),
            ab + (i * tiles_K + k) * DIM,
            DIM * batch,
            DIM);
        k += batch;
      }
    }

    // Step 4b: MVIN all tiles of B into scratchpad (FP4 packed, stride = MATMUL_N/2)
    gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tiles_K; k++) {
      int j = 0;
      while (j < tiles_J) {
        int batch = tiles_J - j;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM),
            bb + (k * tiles_J + j) * DIM,
            DIM * batch,
            DIM);
        j += batch;
      }
    }

    // Steps 5-7: Loop over output quadrants to stay within accumulator limit
    // ti*tj*DIM = 4*4*16 = 256 <= ACC_ROWS=256
    for (int ii = 0; ii < tiles_I; ii += ti) {
      for (int jj = 0; jj < tiles_J; jj += tj) {
        int cur_ti = ti;
        int cur_tj = tj;

        // Compute a_base and b_base offsets for this sub-tile
        uint32_t a_sub = ab + ii * tiles_K * DIM;
        uint32_t b_sub = bb + jj * DIM; // B is laid out as [k][j], base for this j-block

        // Step 5: Configure requantized mvout for this sub-tile
        gemmini_config_st(OUT_COLS * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors,
            cur_ti, cur_tj, tiles_K,
            0, 0, 1);

        // Step 6: Compute weight-stationary tiled matmul for sub-tile
        // A base: ab + ii*tiles_K*DIM (rows for this i-block, all k)
        // B base: BANK_NUM*BANK_ROWS (top of scratchpad, loop_ws_spad uses end address)
        // B tiles are at bb + (k*tiles_J + jj)*DIM, but the hardware loop
        // uses b_base = BANK_NUM*BANK_ROWS and indexes from there.
        // We need to pass the correct b_end and a_base for the sub-tile.
        // The hardware expects A at a_base + (i*tiles_K + k)*DIM
        // and B at b_base_end - (tiles_K*tiles_J - (k*tiles_J+j) - 1)*DIM
        // Actually b is addressed as bb + (k*tj + j)*DIM for local j,
        // but we loaded globally. We need separate loads per sub-tile for correctness,
        // OR use the full b_base with jj offset baked in.
        //
        // Since B is loaded globally as bb + (k*tiles_J + j)*DIM,
        // for a given jj block, the j-th local tile is at bb + (k*tiles_J + jj+j)*DIM.
        // The hardware loop_ws_spad with B end = BANK_NUM*BANK_ROWS iterates
        // b_addr = b_end - tiles_K*cur_tj*DIM + (k*cur_tj+j)*DIM
        // = BANK_NUM*BANK_ROWS - tiles_K*tiles_J*DIM + k*tiles_J*DIM + jj*DIM + j*DIM (only if cur_tj=tiles_J)
        // This only works when jj=0 and cur_tj=tiles_J. For sub-tiling in J, 
        // we need to reload B for each jj block, or restructure.
        //
        // Simplest correct approach: reload B sub-tiles for each (ii,jj) block.
        // But scales are loaded once — they cover full matrix, indexed by tile position.
        // For sub-tiling, we re-use the already-loaded scales (they persist).
        //
        // Let's restructure: load B sub-tiles fresh for each jj, A sub-tiles for each ii.
        // To keep it clean, do full loads outside and use a single loop_ws_spad call
        // that processes the full tiles_I x tiles_J x tiles_K — but we need acc to fit.
        //
        // CONCLUSION: We must split into 4 quadrants, each 4x4x16 tiles.
        // Each quadrant needs its own mvin of A (rows ii..ii+ti-1) and B (cols jj..jj+tj-1).
        // Scales loaded once (full matrix) - they persist and are indexed correctly by tile.

        // We'll break out of this loop structure and redo below.
        (void)a_sub; (void)b_sub; (void)cur_ti; (void)cur_tj;
        break;
      }
      break;
    }

    // Correct implementation: process 4 quadrants with sub-tile loads
    // Scratchpad layout per quadrant:
    //   A sub-tiles: ab_q = 0, holds ti*tiles_K*DIM = 4*16*16 = 1024 rows
    //   B sub-tiles: bb_q = BANK_NUM*BANK_ROWS - tiles_K*tj*DIM = 8192 - 16*4*16 = 7168
    //   Total A+B = 1024 + 1024 = 2048 rows, well within 8192
    //   Accumulator: ti*tj*DIM = 4*4*16 = 256 rows = ACC_ROWS exactly

    uint32_t ab_q = 0;
    uint32_t bb_q = BANK_NUM * BANK_ROWS - tiles_K * tj * DIM; // 7168

    for (int ii = 0; ii < tiles_I; ii += ti) {
      for (int jj = 0; jj < tiles_J; jj += tj) {

        // Load A sub-tiles for rows [ii, ii+ti)
        gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
        for (int i = 0; i < ti; i++) {
          int k = 0;
          while (k < tiles_K) {
            int batch = tiles_K - k;
            if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
            gemmini_extended_mvin(
                (void*)(((elem_t*)A_in_hw) + (ii + i) * DIM * MATMUL_K + k * DIM),
                ab_q + (i * tiles_K + k) * DIM,
                DIM * batch,
                DIM);
            k += batch;
          }
        }

        // Load B sub-tiles for cols [jj, jj+tj)
        gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
        for (int k = 0; k < tiles_K; k++) {
          int j = 0;
          while (j < tj) {
            int batch = tj - j;
            if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
            gemmini_extended_mvin2(
                (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + (jj + j) * DIM),
                bb_q + (k * tj + j) * DIM,
                DIM * batch,
                DIM);
            j += batch;
          }
        }

        // Configure requantized mvout
        gemmini_config_st(OUT_COLS * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors,
            ti, tj, tiles_K,
            0, 0, 1);

        // Compute
        gemmini_loop_ws_spad(ti, tj, tiles_K, 0, 0, 0,
            ab_q, BANK_NUM * BANK_ROWS, 0, SDST,
            false, false, false, false, false,
            NO_ACTIVATION, 0, 0, false, 0x38);

        // Drain results
        gemmini_mx_read_smem(
            &C_hw[ii * DIM][jj * DIM],
            SDST * 16,
            ti * DIM * tj * DIM);
        gemmini_fence();
      }
    }
  }
}
```
