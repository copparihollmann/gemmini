Let me compute full tiles: `tiles_I = 8`, `tiles_J = 8`, `tiles_K = 16`. Scratchpad check: `(8*16 + 8*16)*16 = 4096` rows ≤ 8192. Accumulator: `8*8*16 = 1024` > 256. So we need to tile the output.

The approach: process output in chunks that fit in the accumulator. With `tiles_I=4, tiles_J=4`: acc = `4*4*16=256` rows exactly. We need to loop over i_outer and j_outer blocks.

```c
void solution(void) {
  const int full_ti = MATMUL_M / DIM;  // 8
  const int full_tj = MATMUL_N / DIM;  // 8
  const int tk      = MATMUL_K / DIM;  // 16

  // Process in 4x4 output tile blocks to fit accumulator (4*4*16=256 rows)
  const int block_i = 4;
  const int block_j = 4;

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 2, 2, 3, 0);

  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  gemmini_config_st(OUT_COLS * sizeof(out_t));

  for (int io = 0; io < full_ti; io += block_i) {
    for (int jo = 0; jo < full_tj; jo += block_j) {
      int ti = block_i;  // 4
      int tj = block_j;  // 4

      uint32_t ab = 0;
      uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;

      // Verify no overlap: A needs ti*tk*DIM = 4*16*16 = 1024 rows
      // B starts at 8192 - 16*4*16 = 8192 - 1024 = 7168, no overlap

      int SDST = 0;

      // Load A tiles for this i-block (all k)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < ti; i++) {
        for (int k = 0; k < tk; k++) {
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in_hw) + (io + i)*DIM*MATMUL_K + k*DIM),
              ab + (i*tk + k)*DIM,
              DIM, DIM);
        }
      }

      // Load B tiles for this j-block (all k)
      // FP4: B stride is MATMUL_N/2 bytes (packed)
      gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
      for (int k = 0; k < tk; k++) {
        for (int j = 0; j < tj; j++) {
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + (jo + j)*DIM),
              bb + (k*tj + j)*DIM,
              DIM, DIM);
        }
      }

      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          ti, tj, tk,
          0, 0, 1);

      gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
          ab, BANK_NUM * BANK_ROWS, 0, SDST,
          false, false, false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Read results: ti*tj tiles = 4*4*16*16 = 4096 BF16 elements
      // Each output row for this block: written to C_hw[io*DIM .. (io+ti)*DIM][jo*DIM .. (jo+tj)*DIM]
      // mx_smem stores results row by row; read out and scatter
      int num_bf16 = ti * tj * DIM * DIM;
      // Use a temporary to read the block, then copy to correct C_hw location
      // Actually gemmini_mx_read_smem writes directly to DRAM address
      // We need contiguous output: write to a temp then scatter, or arrange output pointer
      // C_hw layout: [MATMUL_M][MATMUL_N], so row io*DIM+r, cols jo*DIM...(jo+tj)*DIM-1
      // The smem output is ti*DIM rows x tj*DIM cols, row-major within the block
      // We must write to correct location in C_hw
      // Since C_hw rows are MATMUL_N wide but block is tj*DIM wide, not contiguous
      // Use intermediate buffer approach - but we don't have one declared
      // Instead, read to a temp region and memcpy - but that requires temp storage
      // Better: issue per-row reads for each output row of the block
      for (int i = 0; i < ti * DIM; i++) {
        gemmini_mx_read_smem(
            (void*)(&C_hw[io*DIM + i][jo*DIM]),
            SDST * 16 + i * tj * DIM,
            tj * DIM);
      }
      gemmini_fence();
    }
  }
}
```
