void solution(void) {
  int tiles_K = MATMUL_K / DIM;

  // Step 1: Configure execution ONCE
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  // Find best TI, TJ fitting scratchpad and accumulator constraints
  // Scratchpad: (TI + TJ) * tiles_K * DIM <= BANK_NUM * BANK_ROWS
  // Accumulator: TI * TJ * DIM <= ACC_ROWS  =>  TI * TJ <= ACC_ROWS/DIM = 16
  int TI = 1, TJ = 1;
  {
    int best = 0;
    for (int ti = 1; ti <= 16; ti++) {
      for (int tj = 1; tj <= 16; tj++) {
        if ((ti + tj) * tiles_K * DIM > BANK_NUM * BANK_ROWS) continue;
        if (ti * tj * DIM > ACC_ROWS) continue;
        if (ti * tj > best) {
          best = ti * tj;
          TI = ti;
          TJ = tj;
        }
      }
    }
  }

  int TI_BLK = TI * DIM;
  int TJ_BLK = TJ * DIM;

  // B base: top of scratchpad, growing downward.
  uint32_t b_base = BANK_NUM * BANK_ROWS - TJ * tiles_K * DIM;
  uint32_t a_base = 0;
  // SPAD_DEST in mx_smem: use 0 (word offset for gemmini_mx_read_smem is SPAD_DEST*16)
  int SPAD_DEST = 0;

  int GK = MATMUL_K / 32; // number of scale groups along K

    gemmini_config_ld(MATMUL_K * sizeof(elem_t));

      if (mrows_blk > TI_BLK) mrows_blk = TI_BLK;
      int ti_actual = (mrows_blk + DIM - 1) / DIM;
      if (ti_actual <= 0) continue;

      int pad_i = TI * DIM - mrows_blk;

      // Stage A scales for this I block: A_scales_row[GK][MATMUL_M], need rows [i0_elem..i0_elem+mrows_blk)
      for (int g = 0; g < GK; g++) {
        for (int r = 0; r < mrows_blk; r++) {
          A_s[g][r] = A_scales_row[g][i0_elem + r];
        }
        // Pad remaining entries with 0
        for (int r = mrows_blk; r < TI_BLK; r++) {
          A_s[g][r] = 0;
        }
      }

      for (int g = 0; g < GK; g++) {

                           NO_ACTIVATION, 0, 0, false, 0x38);

      }

      gemmini_fence();
    }
  }
}
