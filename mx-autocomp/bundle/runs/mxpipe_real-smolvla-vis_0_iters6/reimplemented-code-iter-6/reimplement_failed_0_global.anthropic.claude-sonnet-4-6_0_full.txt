Looking at the error, the problem is clear: B tiles are being loaded with `gemmini_extended_mvin2` (port 1), but `gemmini_loop_ws_spad` requires all inputs to be loaded via `gemmini_extended_mvin` (port 0, the single-port scratchpad region). The fix is to use `gemmini_extended_mvin` for both A and B tile loads.

```c
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  int GK      = MATMUL_K / 32;

  // b_base depends only on tile geometry, not on i0/j0 — fixed for all blocks.
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  // --- One-time pre-transposition of scale arrays into block-contiguous layout ---
  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&A_scales_T[i0][g * BLK],
             &A_scales_row[g][i0 * BLK],
             BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&B_scales_T[j0][g * BLK],
             &B_scales_col[g][j0 * BLK],
             BLK);
    }
  }

  // Hoist config_ex outside all loops — format fields never change across iterations.
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  // Hoist config_st outside all loops — BLK_OUTC is constant.
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  // Reorder: j0 outer so B tiles can be hoisted; i0 inner sweeps rows.
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // --- Mvin B tiles ONCE per j0 into the top of the scratchpad ---
    // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility.
    // Hoist config_ld for B outside i0 loop — stride is constant.
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));   // B row stride = N
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);

    // Hoist config_ld for A outside i0 loop — stride is constant.
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));   // A row stride = K

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Step 2: Load A and B scales into hardware directly from pre-transposed buffers.
      // sel=0 → mx_scale_a_mem (activations/A), sel=1 → mx_scale_b_mem (weights/B).
      gemmini_mx_load_scales((uint64_t)A_scales_T[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

      // Step 3: Mvin A tiles for this i0 row-block.
      // B tiles are already resident in the scratchpad from the j0-level hoist.
      // config_ld for A already set above (hoisted outside i0 loop).
      // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility.
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Step 4: Configure requantized mvout.
      // i_bound=tiles, j_bound=tiles, k_bound=tiles_K must match gemmini_loop_ws_spad.
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Step 5: Execute fused hardware matmul.
      // mx_loop_spad_marker (funct 24) is one-shot and re-issued inside the macro
      // before every gemmini_loop_ws_spad call.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);

      // Step 7: Fence — ensures all mx_read_smem DMA writes are visible to the host.
      gemmini_fence();

      // Reload B tiles for the next i0 iteration (they were overwritten by A mvin
      // if a_base and b_base regions overlap, or simply need to be refreshed).
      // Only reload if there are more i0 iterations to process.
      if (i0 + 1 < BLOCKS_M) {
        int next_mrows = REAL_M - (i0 + 1) * BLK;
        if (next_mrows > 0) {
          gemmini_config_ld(MATMUL_N * sizeof(elem_t));
          for (int j = 0; j < tiles; j++)
            for (int k = 0; k < tiles_K; k++)
              gemmini_extended_mvin(
                (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                b_base + (k * tiles + j) * DIM, DIM, DIM);
          gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        }
      }
    }
  }
}
```
