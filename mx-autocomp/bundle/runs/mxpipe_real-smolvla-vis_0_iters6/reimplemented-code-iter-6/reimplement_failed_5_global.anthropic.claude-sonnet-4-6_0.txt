void solution(void) {
  const int tiles_I   = MATMUL_M / DIM;
  const int tiles_J   = MATMUL_N / DIM;
  const int tiles_K   = MATMUL_K / DIM;
  const int GK        = MATMUL_K / 32;  // scale groups along K (one scale per 32 elements)

  // b_base at top of scratchpad
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
  const uint32_t a_base = 0;
  const int SPAD_DEST   = 0;  // output word offset in mx_smem

  // Step 1: Configure execution once
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // Step 2: Load A and B scales (all at once, full matrices)
  gemmini_mx_load_scales((uint64_t)A_scales_row, GK * MATMUL_M, 0);
  gemmini_mx_load_scales((uint64_t)B_scales_col, GK * MATMUL_N, 1);

  // Step 3: Mvin ALL B tiles into scratchpad (weight-stationary: B fixed)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin2(
        (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j * DIM)),
        b_base + (j * tiles_K + k) * DIM,
        DIM, DIM);
    }
  }

  // Step 4: Mvin ALL A tiles into scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin(
        (void*)(((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM),
        a_base + (i * tiles_K + k) * DIM,
        DIM, DIM);
    }
  }

  // Step 5: Configure requantized mvout
  gemmini_config_st(MATMUL_N * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)A_scales_row,
                               tiles_I, tiles_J, tiles_K,
                               0, 0, 1);

  // Step 6: Execute fused hardware matmul
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain BF16 results from mx_smem to DRAM
  gemmini_mx_read_smem(
    &C_hw[0][0],
    SPAD_DEST * 16,
    MATMUL_M * MATMUL_N);

  // Step 8: Fence
  gemmini_fence();
}
