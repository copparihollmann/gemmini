void solution(void) {
  int tiles_K = MATMUL_K / DIM;

  // Step 1: Configure execution ONCE — format and dataflow are loop-invariant.
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // Configure output stride ONCE — BLK_OUTC * sizeof(out_t) is constant.
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  // Compute maximum tile counts TI, TJ (in DIM-row units) that fit in:
  //   Scratchpad: (TI + TJ) * tiles_K * DIM <= BANK_NUM * BANK_ROWS = 8192
  //   Accumulator: TI * TJ * DIM <= ACC_ROWS = 256  =>  TI * TJ <= 16
  // We search for the largest square TI=TJ, then try TI=TJ+1 etc.
  int TI = 1, TJ = 1;
  {
    int best = 0;
    // Search over reasonable small values; tiles_K >= 1
    for (int ti = 1; ti <= 16; ti++) {
      for (int tj = 1; tj <= 16; tj++) {
        // Scratchpad constraint
        if ((ti + tj) * tiles_K * DIM > BANK_NUM * BANK_ROWS) continue;
        // Accumulator constraint: TI*TJ*DIM <= ACC_ROWS
        if (ti * tj * DIM > ACC_ROWS) continue;
        // Maximize product TI*TJ (more output per invocation)
        if (ti * tj > best) {
          best = ti * tj;
          TI = ti;
          TJ = tj;
        }
      }
    }
  }

  // TI, TJ are now tile counts (each covers TI*DIM rows / TJ*DIM cols).
  int TI_BLK = TI * DIM;  // row-block size in elements
  int TJ_BLK = TJ * DIM;  // col-block size in elements

  // B base: top of scratchpad, growing downward.
  uint32_t b_base = BANK_NUM * BANK_ROWS - TJ * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  // Scale staging buffers sized to the larger of BLK or TI_BLK/TJ_BLK.
  // We reuse A_s / B_s but must not exceed their declared sizes.
  // The outer loops step by TI_BLK / TJ_BLK.

  for (int j0_elem = 0; j0_elem < REAL_N; j0_elem += TJ_BLK) {
    int ncols_blk = REAL_N - j0_elem;
    if (ncols_blk > TJ_BLK) ncols_blk = TJ_BLK;
    int tj_actual = (ncols_blk + DIM - 1) / DIM;  // actual J tiles this block
    if (tj_actual <= 0) continue;

    int pad_j = TJ * DIM - ncols_blk;  // padding in last J block

    // Stage B scales for this J block ONCE.
    // B_scales_col is [MATMUL_K/32][MATMUL_N]: pick columns [j0_elem .. j0_elem+TJ_BLK)
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0_elem], ncols_blk);

    // Mvin B tiles ONCE per J block.
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < TJ; j++) {
      if (j0_elem + j * DIM >= REAL_N) break;
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0_elem + j * DIM)),
          b_base + (k * TJ + j) * DIM, DIM, DIM);
      }
    }

    // A-stride config: set once per J block, before the I loop.
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));

    for (int i0_elem = 0; i0_elem < REAL_M; i0_elem += TI_BLK) {
      int mrows_blk = REAL_M - i0_elem;
      if (mrows_blk > TI_BLK) mrows_blk = TI_BLK;
      int ti_actual = (mrows_blk + DIM - 1) / DIM;  // actual I tiles this block
      if (ti_actual <= 0) continue;

      int pad_i = TI * DIM - mrows_blk;

      // Stage A scales for this (I,J) block.
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0_elem], mrows_blk);

      gemmini_flush(0);

      // Step 2: Load A and B scales into hardware.
      // sel=0 => mx_scale_a_mem (A/activation), sel=1 => mx_scale_b_mem (B/weight).
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 3: Mvin A tiles for this I block.
      // A-stride config already set outside i0 loop.
      for (int i = 0; i < TI; i++) {
        if (i0_elem + i * DIM >= REAL_M) break;
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0_elem + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Step 4: Configure requantized mvout with actual tile dimensions.
      // i_bound and j_bound must match TI, TJ used in gemmini_loop_ws_spad.
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   TI, TJ, tiles_K,
                                   0, 0, 1);

      // Step 5: Execute fused hardware matmul.
      // b_base argument is BANK_NUM*BANK_ROWS (end-of-scratchpad sentinel),
      // consistent with gemmini_loop_ws_spad convention.
      gemmini_loop_ws_spad(TI, TJ, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row.
      // Only drain actual (non-padded) rows.
      for (int r = 0; r < mrows_blk; r++) {
        gemmini_mx_read_smem(
          &C_hw[i0_elem + r][(j0_elem) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * TJ_BLK, ncols_blk);
      }

      // Step 7: Fence — one per (I,J) outer tile, after all reads for this tile.
      gemmini_fence();
    }
  }
}
