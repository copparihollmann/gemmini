void solution(void) {
  // BLK=64: tiles=4, tiles_K=16
  // Scratchpad: (4+4)*16*16 = 2048 rows <= 8192 ✓
  // Accumulator: 4*4*16 = 256 rows <= 256 ✓
  const int BLK_NEW   = 64;
  const int tiles     = BLK_NEW / DIM;   // 4
  const int tiles_K   = MATMUL_K / DIM;  // 16
  const int GK        = MATMUL_K / 32;   // scale groups along K

  // B base: top of scratchpad, fixed geometry
  // b_base = 8192 - 4*16*16 = 7168
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  const uint32_t a_base = 0;
  const int SPAD_DEST   = 128;

  const int BLOCKS_N_NEW = (REAL_N + BLK_NEW - 1) / BLK_NEW;
  const int BLOCKS_M_NEW = (REAL_M + BLK_NEW - 1) / BLK_NEW;

  // Configure execution once — format/dataflow unchanged across all tiles
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int j0 = 0; j0 < BLOCKS_N_NEW; j0++) {
    int ncols = REAL_N - j0 * BLK_NEW;
    if (ncols > BLK_NEW) ncols = BLK_NEW;
    if (ncols <= 0) continue;
    int pad_j = BLK_NEW - ncols;

    // --- Stage B scales once per j0 (unchanged across all i0 for this j0) ---
    for (int g = 0; g < GK; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK_NEW], ncols);

    // --- Hoist B mvin to j0 level: B tiles fixed across all i0 iterations ---
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK_NEW + j * DIM)),
          b_base + (j * tiles_K + k) * DIM,
          DIM, DIM);

    for (int i0 = 0; i0 < BLOCKS_M_NEW; i0++) {
      int mrows = REAL_M - i0 * BLK_NEW;
      if (mrows > BLK_NEW) mrows = BLK_NEW;
      if (mrows <= 0) continue;
      int pad_i = BLK_NEW - mrows;

      // Stage A scales for this i0 block (non-contiguous slice, must memcpy)
      for (int g = 0; g < GK; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK_NEW], mrows);

      // PROF accel
      gemmini_flush(0);

      // Step 1: Configure execution (re-issue with set_only_strides=true after
      // first call since format/dataflow fields never change; this avoids full
      // reconfiguration overhead while keeping the accelerator state consistent)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load A and B scales into hardware.
      // sel=0 → mx_scale_a_mem (activations/A), sel=1 → mx_scale_b_mem (weights/B)
      // A_s holds GK rows of BLK_NEW bytes; B_s same layout
      gemmini_mx_load_scales((uint64_t)A_s, GK * BLK_NEW, 0);
      gemmini_mx_load_scales((uint64_t)B_s, GK * BLK_NEW, 1);

      // Step 3: Mvin A tiles (B tiles already resident from j0-level hoist)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));  // A row stride = K cols
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK_NEW + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // Step 4: Configure requantized mvout.
      // i_bound=tiles, j_bound=tiles, k_bound=tiles_K must match loop_ws_spad exactly.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   0, 0, 1);

      // Step 5: Execute fused hardware matmul.
      // mx_loop_spad_marker (funct 24) is one-shot and re-issued inside the macro
      // before every gemmini_loop_ws_spad call.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK_NEW + r][(j0 * BLK_NEW) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK_NEW, ncols);

      // Step 7: Fence — ensures all mx_read_smem DMA writes are visible to host.
      gemmini_fence();
    }
  }
}
