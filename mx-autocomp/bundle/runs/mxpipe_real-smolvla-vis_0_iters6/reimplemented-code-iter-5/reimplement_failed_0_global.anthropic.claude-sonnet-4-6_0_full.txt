Looking at the failing code, I need to identify the issues:

1. **Scale loading selector is wrong**: The documentation says `sel=0` for A (activation) scales and `sel=1` for B (weight) scales. But looking at the `gemmini_mx_load_scales` description more carefully: "sel 0 = A-row scales, 1 = B-col scales" - wait, let me re-check. Rule 11 says sel=0 for A, sel=1 for B. The code has `sel=0` for A and `sel=1` for B - that looks correct.

2. **B tile layout in scratchpad**: The B base address should be `BANK_NUM*BANK_ROWS - tiles_K*tiles_J*DIM` and the loop_ws_spad passes `BANK_NUM*BANK_ROWS` as B end address. The B tile indexing uses `(k * tiles + j)` but should be `(j * tiles_K + k)` to match the documented layout `b_base + (j*tiles_K + k)*DIM`.

3. **B mvin stride**: B is stored column-major in the scratchpad (j-outer, k-inner), but the DRAM layout needs correct stride. The stride for B should be `MATMUL_N * sizeof(elem_t)`.

4. **scale_factors**: This variable is referenced but unclear - looking at the context, it seems like this should be the output scale memory, but the mxquant_config_mvout for BF16 output may not need valid scale_dram since out_fmt=3 (BF16).

5. **B tile indexing in mvin**: `b_base + (k * tiles + j) * DIM` should be `b_base + (j * tiles_K + k) * DIM`.

void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  int GK      = MATMUL_K / 32;

  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  // --- One-time pre-transposition of scale arrays into block-contiguous layout ---
  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&A_scales_T[i0][g * BLK],
             &A_scales_row[g][i0 * BLK],
             BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&B_scales_T[j0][g * BLK],
             &B_scales_col[g][j0 * BLK],
             BLK);
    }
  }

  gemmini_flush(0);

  // Step 1: Configure execution (same for all blocks — issue once).
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // Reorder: j0 outer so B tiles can be hoisted; i0 inner sweeps rows.
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    // --- Mvin B tiles ONCE per j0 into the top of the scratchpad ---
    // B layout: b_base + (j * tiles_K + k) * DIM  (j-outer, k-inner)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));   // B row stride = N cols
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (j * tiles_K + k) * DIM, DIM, DIM);

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Step 2: Load A scales (sel=0) and B scales (sel=1).
      gemmini_mx_load_scales((uint64_t)A_scales_T[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

      // Step 3: Mvin A tiles for this i0 row-block.
      // A layout: a_base + (i * tiles_K + k) * DIM  (i-outer, k-inner)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));  // A row stride = K cols
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Step 4: Configure requantized mvout.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Step 5: Execute fused hardware matmul.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++)

          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);

      // Step 7: Fence
      gemmini_fence();
    }
  }
}
