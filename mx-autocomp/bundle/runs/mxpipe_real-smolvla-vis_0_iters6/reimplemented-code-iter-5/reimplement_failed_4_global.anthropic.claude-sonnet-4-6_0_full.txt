Looking at the error, the issue is that B tiles are being loaded with `gemmini_extended_mvin2` (port 1), but `gemmini_loop_ws_spad` requires all inputs to be loaded via `gemmini_extended_mvin` (port 0, single-port). I need to change all `mvin2` calls to `mvin` calls.

```c
void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  // Flush once, outside all loops.
  gemmini_flush(0);

  // i0 is now the OUTER loop: A tiles are hoisted and reused across all j0.
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;
    int pad_i = BLK - mrows;

    // Stage A scales once per i0 block.
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

    // j0 is now the INNER loop: B tiles and B scales vary per j0.
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;
      int pad_j = BLK - ncols;

      // Stage B scales for this j0 block.
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

      // Step 1: Configure execution (must precede scale loads and mvins).
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load A and B scales into hardware.
      // sel=0 → mx_scale_a_mem (activations/A), sel=1 → mx_scale_b_mem (weights/B).
      // A scales are the same for all j0 within this i0 block.
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 3a: Mvin A tiles for this i0 row-block only on the first j0
      // iteration. A tiles remain resident in scratchpad at a_base for all
      // subsequent j0 iterations, since the b_base region never overlaps a_base.
      if (j0 == 0) {
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));  // A row stride = K
        for (int i = 0; i < tiles; i++)
          for (int k = 0; k < tiles_K; k++)
            gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      }

      // Step 3b: Mvin B tiles for this j0 column-block into top of scratchpad.
      // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility.
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));   // B row stride = N
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM, DIM, DIM);

      // Step 4: Configure requantized mvout.
      // i_bound=tiles, j_bound=tiles, k_bound=tiles_K must match gemmini_loop_ws_spad.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Step 5: Execute fused hardware matmul.
      // mx_loop_spad_marker (funct 24) is one-shot and re-issued inside the macro
      // before every gemmini_loop_ws_spad call.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 6: Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);

      // Step 7: Fence — ensures all mx_read_smem DMA writes are visible to the host.
      gemmini_fence();
    }
  }
}
```
