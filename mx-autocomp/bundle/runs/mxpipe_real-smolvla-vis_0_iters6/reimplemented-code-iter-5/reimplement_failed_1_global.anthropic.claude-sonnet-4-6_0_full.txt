Looking at the failing code, I need to identify the issues:

1. **`gemmini_flush(0)` is being called inside the inner loop** - this is a full accelerator flush that resets state and is very expensive/problematic inside compute loops.

2. **Scale memory layout mismatch** - The `gemmini_mxquant_config_mvout` `dram_addr` parameter is documented as `scale_dram + m * N_blocks + bi` indexing. But `scale_factors` is a global - need to verify this is correct.

3. **The `gemmini_mx_read_smem` offset calculation** - `SPAD_DEST * 16 + r * BLK` - the smem_off_bf16 is in 16-bit word units. `r * BLK` where BLK is in number of elements (BF16 values), so this should be correct if BLK is the number of BF16 per row.

4. **`C_hw` write address** - `C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD]` - the column indexing with `/BF16_PER_WORD` looks suspicious if `C_hw` is indexed in BF16 units directly.

5. **The B mvin is pre-issued before scales are loaded for the next iteration** - this violates the ordering: config_ex → load_scales → mvin → config_mvout → loop_ws_spad.

6. **`gemmini_flush` should only be called once at the start**, not in the loop.

Let me rewrite with correct ordering and fix the issues:

void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  int GK      = MATMUL_K / 32;

  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  static uint8_t A_scales_T[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_T[BLOCKS_N][B_S_BYTES];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&A_scales_T[i0][g * BLK],
             &A_scales_row[g][i0 * BLK],
             BLK);
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < GK; g++) {
      memcpy(&B_scales_T[j0][g * BLK],
             &B_scales_col[g][j0 * BLK],
             BLK);
    }
  }

  gemmini_flush(0);

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Step 1: Configure execution.
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load A and B scales.
      gemmini_mx_load_scales((uint64_t)A_scales_T[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_T[j0], B_S_BYTES, 1);

      // Step 3: Mvin B tiles for this j0 column-block.
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM, DIM, DIM);

      // Step 4: Mvin A tiles for this i0 row-block.
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Step 5: Configure requantized mvout.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K, 0, 0, 1);

      // Step 6: Execute fused hardware matmul.
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][j0 * BLK],
          SPAD_DEST * 16 + r * BLK, ncols);

      // Step 8: Fence.
      gemmini_fence();
    }
  }
}
