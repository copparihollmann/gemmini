void solution(void) {
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    /* ------------------------------------------------------------------ */
    /* Hoist B MVIN outside the i0 loop: B tiles depend only on j0.        */
    /* We need a valid config_ld for B before issuing mvin2.               */
    /* b_base is fixed for this j0 (tiles_I == tiles_J == tiles here).     */
    /* ------------------------------------------------------------------ */
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;

    gemmini_config_ld(MATMUL_N * sizeof(elem_t));   /* B row stride */
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM,
          DIM, DIM);

    /* ------------------------------------------------------------------ */
    /* Inner loop over row-blocks: A tiles change per i0.                  */
    /* ------------------------------------------------------------------ */
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      /* Stage per-group scales for this (i0, j0) block. */
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      uint32_t a_base  = 0;
      int      SPAD_DEST = 128;

      /* Programming-model step 1: configure execution. */
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  1, 1, 0, 0, false,
                                  /*act_mx_fmt*/0, /*wgt_mx_fmt*/0,
                                  /*out_mx_fmt*/3, /*uselut*/0);

      /* Programming-model step 2: load both scale buffers. */
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  /* sel=0 → A */
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  /* sel=1 → B */

      /* Programming-model step 4: MVIN A tiles for this row-block. */
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));          /* A row stride */
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      /* Programming-model step 5: configure requantized mvout. */
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);

      /* Programming-model step 6: fused tiled matmul (B already in spad). */
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0,
                           false, 0x38);

      /* Programming-model step 7: drain BF16 results from mx_smem. */
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
