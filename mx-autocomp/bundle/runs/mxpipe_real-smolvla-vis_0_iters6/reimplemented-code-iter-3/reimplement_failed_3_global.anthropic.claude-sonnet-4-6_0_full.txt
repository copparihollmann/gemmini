Looking at the code, I need to identify the bugs:

1. **B tile layout is wrong**: The layout formula `b_base + (k * tiles + j) * DIM` is incorrect. Per the spec, B tiles should be at `b_base + (j*tiles_K + k)*DIM`. The current code uses `(k * tiles + j)` which swaps the indexing.

2. **Scale loading selectors**: The spec says `sel=0` for A (activation) scales and `sel=1` for B (weight) scales. Looking at the code: `gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0)` - sel=0 for A is correct, and `sel=1` for B is correct. This looks fine.

3. **B DRAM address**: B is stored row-major as `B[K][N]`, so `B_in[k*DIM][j0*BLK + j*DIM]` needs stride `MATMUL_N`. The mvin config sets stride `MATMUL_N * sizeof(elem_t)` which is correct.

4. **B layout in scratchpad**: The `b_base` end address passed to `gemmini_loop_ws_spad` should be the top of scratchpad (`BANK_NUM*BANK_ROWS`), but the actual tile placement must match `b_base + (j*tiles_K + k)*DIM`.

5. **The B mvin spad address**: Currently `b_base + (k * tiles + j) * DIM` but should be `b_base + (j * tiles_K + k) * DIM` to match what the hardware loop expects.

6. **`gemmini_flush` should not be inside the loop** for performance, but more critically the B stride in mvin: B in DRAM is `B[K/DIM tiles][N]` shaped, so row `k*DIM` has stride `MATMUL_N`.

The primary correctness bug is the B scratchpad address formula being transposed relative to what `gemmini_loop_ws_spad` expects.

void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++)
   for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // Stage per-block scales for this (i0, j0) output tile
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      // B placed at top of scratchpad, growing downward
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Step 1: Configure execution
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          1, 1, 0, 0, false,
          /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: Load scale factors (sel=0 for A/activation, sel=1 for B/weight)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN tiles of A into scratchpad
      // A layout: a_base + (i*tiles_K + k)*DIM
      // A is stored row-major: A[M][K], stride = MATMUL_K bytes per row
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
          elem_t *dram_ptr = (elem_t*)A_in + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM;
          gemmini_extended_mvin((void*)dram_ptr, spad_addr, DIM, DIM);
        }
      }

      // Step 4: MVIN tiles of B into scratchpad (via mvin2 port)
      // B layout in spad: b_base + (j*tiles_K + k)*DIM
      // B is stored row-major: B[K][N], stride = MATMUL_N bytes per row
      // So B tile (k,j) starts at DRAM row k*DIM, col j0*BLK+j*DIM
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
          elem_t *dram_ptr = (elem_t*)B_in + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM);
          gemmini_extended_mvin2((void*)dram_ptr, spad_addr, DIM, DIM);
        }
      }

      // Step 5: Configure output stride and requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          /*i_bound*/tiles, /*j_bound*/tiles, /*k_bound*/tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

      // Step 6: Execute fused weight-stationary matmul
      // B end address = BANK_NUM*BANK_ROWS (top of scratchpad)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain BF16 results from mx_smem to DRAM
      if (ncols == BLK) {
        // Full column block: contiguous read for all mrows*BLK elements
        gemmini_mx_read_smem(
            &C_hw[i0 * BLK + 0][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16,
            mrows * BLK);
      } else {
        // Partial last column block: per-row reads with ncols elements each
        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(
              &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
              SPAD_DEST * 16 + r * BLK,
              ncols);
        }
      }
      gemmini_fence();
    }
}
