Looking at the failing code, I need to identify the bugs:

1. **B matrix layout is wrong**: The code uses `b_base + (k * tiles + j) * DIM` but the correct layout is `b_base + (j * tiles_K + k) * DIM` (j is the outer index, k is inner).

2. **Scale loading selectors are swapped**: `gemmini_mx_load_scales` uses sel=0 for A and sel=1 for B. Looking at the code, it has `sel=0` for `A_s` and `sel=1` for `B_s` which seems correct per the convention, but let me re-check: "sel 0 = A (activation), sel 1 = B (weight)" - the code has this right.

3. **`gemmini_config_ld` stride for B**: B is stored row-major with stride `MATMUL_N`. The mvin for B reads `B[k*DIM][j0*BLK + j*DIM]` with stride `MATMUL_N * sizeof(elem_t)` - this looks correct.

4. **`gemmini_mx_read_smem` offset calculation**: `SPAD_DEST * 16 + r * BLK` - the smem word offset should be in 16-bit units. Each row of the output is `BLK` BF16 values wide, so `r * BLK` is correct for the row offset. But the `ncols` parameter should be `ncols` BF16 values, not bytes.

5. **B scratchpad layout**: The `b_base` uses `b_base + (j*tiles_K + k)*DIM` per the architecture docs, but B is being loaded with `(k * tiles + j)` indexing. This is the key bug.

6. **`gemmini_config_ld` for B mvin**: B is accessed as `B[k*DIM, j*DIM]` with matrix stride `MATMUL_N`. The stride passed to `config_ld` should be `MATMUL_N * sizeof(elem_t)`.

void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // Copy per-block scales for this (i0,j0) tile
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;

      // A tiles grow up from row 0; B tiles placed at top of scratchpad
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles * DIM;
      int SPAD_DEST = 0;  // output smem destination word offset (in 16-word units for read_smem)

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // 1. Load scale factors: sel=0 for A (activation), sel=1 for B (weight)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // 2. MVIN A tiles: layout a_base + (i*tiles_K + k)*DIM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM, DIM);

      // 3. MVIN B tiles: layout b_base + (j*tiles_K + k)*DIM
      //    B is stored row-major [K][N], so B[k*DIM][j0*BLK + j*DIM] with stride MATMUL_N
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (j * tiles_K + k) * DIM,
              DIM, DIM);

      // 4. Configure requantized mvout and compute
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles, tiles, tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // 5. Fused hardware matmul (LOOP_WS_CONFIG_SPAD_AB marker re-issued inside macro)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 6. Drain BF16 results from mx_smem to DRAM
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK,
            ncols);
      gemmini_fence();
    }
}
