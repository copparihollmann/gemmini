void solution(void) {
  gemmini_flush(0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      // Copy per-block scales for A and B
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;

      // A tiles start at row 0; B tiles start at top of scratchpad
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - (uint32_t)(tiles * tiles_K * DIM);
      int SPAD_DEST = 128;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Step 1: Configure execution (FP8 E4M3 inputs, BF16 output)
      gemmini_extended3_config_ex(
          WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          /*C_stride*/1, /*A_stride*/1,
          /*A_transpose*/0, /*B_transpose*/0,
          /*set_only_strides*/false,
          /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: Load scales — sel=0 for A (activation), sel=1 for B (weight)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN A tiles into scratchpad
      // A is row-major [M][K], tile (i,k) starts at A_in[(i0*BLK+i*DIM)*MATMUL_K + k*DIM]
      // Scratchpad layout: a_base + (i*tiles_K + k)*DIM
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *a_ptr = ((const elem_t *)A_in)
              + (size_t)(i0 * BLK + i * DIM) * MATMUL_K
              + (size_t)(k * DIM);
          uint32_t spad_addr = a_base + (uint32_t)((i * tiles_K + k) * DIM);
          gemmini_extended_mvin((void *)a_ptr, spad_addr, DIM, DIM);
        }
      }

      // MVIN B tiles into scratchpad
      // B is row-major [K][N], tile (k,j) starts at B_in[k*DIM*MATMUL_N + (j0*BLK+j*DIM)]
      // Scratchpad layout: b_base + (j*tiles_K + k)*DIM  (j outer, k inner)
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *b_ptr = ((const elem_t *)B_in)
              + (size_t)(k * DIM) * MATMUL_N
              + (size_t)(j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (uint32_t)((j * tiles_K + k) * DIM);
          gemmini_extended_mvin2((void *)b_ptr, spad_addr, DIM, DIM);
        }
      }

      // Step 5: Configure requantized mvout
      gemmini_config_st(BLK * sizeof(out_t));
      gemmini_mxquant_config_mvout(
          (uint64_t)scale_factors,
          /*i_bound*/tiles, /*j_bound*/tiles, /*k_bound*/tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // Step 6: Fused hardware matmul (weight-stationary, tiles already in spad)
      gemmini_loop_ws_spad(
          tiles, tiles, tiles_K,
          pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS,
          /*D*/0, /*C=SPAD_DEST*/SPAD_DEST,
          /*A_transpose*/false, /*B_transpose*/false,
          /*full_C*/false, /*low_D*/false,
          /*ex_accumulate*/false,
          /*act*/NO_ACTIVATION,
          /*a_spad_id*/0, /*b_spad_id*/0,
          /*is_resadd*/false,
          /*skips*/0x38);

      // Step 7: Drain BF16 results from mx_smem to DRAM, row by row
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(
            (void *)&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK,
            ncols);
      }
      gemmini_fence();
    }
  }
}
