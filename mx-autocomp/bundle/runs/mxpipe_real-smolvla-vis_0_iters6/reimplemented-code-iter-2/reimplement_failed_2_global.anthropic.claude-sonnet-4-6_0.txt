void solution(void) {
  gemmini_flush(0);

  int tiles_K = MATMUL_K / DIM;
  int tiles   = BLK / DIM;          // tiles_I = tiles_J = tiles (square blocks)

  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 0;           // smem word offset for output

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Stage per-block e8m0 scales for this (i0, j0) output tile.
      // A_scales_row[g][m] — one scale per row-group along K
      // B_scales_col[g][n] — one scale per col-group along K
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // 1. Configure execution format: FP8 E4M3 inputs, BF16 output.
      gemmini_extended3_config_ex(
          WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          /*C_stride*/1, /*A_stride*/1,
          /*A_transpose*/0, /*B_transpose*/0,
          /*set_only_strides*/false,
          /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // 2. Load scale factors: sel=0 for A (activation), sel=1 for B (weight).
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // 4. MVIN A tiles: A is row-major [M][K], stride = MATMUL_K bytes.
      //    A tile (i,k) -> DRAM: A_in[(i0*BLK + i*DIM) * MATMUL_K + k*DIM]
      //    Scratchpad: a_base + (i*tiles_K + k)*DIM
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM, DIM);
        }
      }

      // 4. MVIN B tiles: B is row-major [K][N], stride = MATMUL_N bytes.
      //    B tile (j,k) -> DRAM: B_in[k*DIM * MATMUL_N + (j0*BLK + j*DIM)]
      //    Scratchpad: b_base + (j*tiles_K + k)*DIM  (per spec)
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (j * tiles_K + k) * DIM,
              DIM, DIM);
        }
      }

      // 5. Configure output stride and requantizer.
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout(
          (uint64_t)scale_factors,
          /*i_bound*/tiles, /*j_bound*/tiles, /*k_bound*/tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // 6. Execute fused tiled matmul (one-shot marker re-issued internally by macro).
      gemmini_loop_ws_spad(
          tiles, tiles, tiles_K,
          pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS, /*D*/0, /*C/SPAD_DEST*/SPAD_DEST,
          /*A_transpose*/false, /*B_transpose*/false,
          /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
          NO_ACTIVATION, /*a_spad_id*/0, /*b_spad_id*/0,
          /*is_resadd*/false, /*skips*/0x38);

      // 7. Drain BF16 results from mx_smem to DRAM, row by row.
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK,
            ncols);
      }
      gemmini_fence();
    }
  }
}
