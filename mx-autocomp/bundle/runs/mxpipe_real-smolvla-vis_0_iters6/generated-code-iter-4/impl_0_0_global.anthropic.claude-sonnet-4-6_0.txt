void solution(void) {
  int tiles_K = MATMUL_K / DIM;

  // Blocking: tiles_J_blk=1 column-tile, tiles_I_blk as large as constraints allow.
  // Spad constraint: (tiles_I_blk*tiles_K + tiles_J_blk*tiles_K)*DIM <= 8192
  //   => tiles_I_blk <= (8192/DIM - tiles_J_blk*tiles_K) / tiles_K
  // Acc  constraint: tiles_I_blk * tiles_J_blk * DIM <= 256
  //   => tiles_I_blk <= 256 / (tiles_J_blk * DIM)
  const int tiles_J_blk = 1;
  int tiles_I_blk_spad = (8192 / DIM - tiles_J_blk * tiles_K) / tiles_K;
  int tiles_I_blk_acc  = 256 / (tiles_J_blk * DIM);
  int tiles_I_blk = tiles_I_blk_spad < tiles_I_blk_acc
                      ? tiles_I_blk_spad : tiles_I_blk_acc;
  if (tiles_I_blk < 1) tiles_I_blk = 1;

  // b_base: B occupies top of scratchpad, 1 J-tile wide.
  // b_base = BANK_NUM*BANK_ROWS - tiles_J_blk*tiles_K*DIM
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J_blk * tiles_K * DIM;
  uint32_t a_base = 0;
  int SPAD_DEST = 128;

  // Outer loop: j (one DIM-column tile at a time within each j0 block)
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols_blk = REAL_N - j0 * BLK;
    if (ncols_blk > BLK) ncols_blk = BLK;
    if (ncols_blk <= 0) continue;

    // Iterate over J tiles within this j0 block, one tile at a time (tiles_J_blk=1)
    int tiles_j_in_blk = BLK / DIM;
    for (int jt = 0; jt < tiles_j_in_blk; jt++) {
      int j_global_tile = j0 * tiles_j_in_blk + jt;
      int ncols = REAL_N - j_global_tile * DIM;
      if (ncols > DIM) ncols = DIM;
      if (ncols <= 0) continue;
      int pad_j = DIM - ncols;

      // --- Stage B scales for this J tile (1 tile wide) ---
      // B_scales_col[g][col]: we need column indices j_global_tile*DIM .. +DIM
      // Pack into B_s[g][0..DIM-1]
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&B_s[g][0],
               &B_scales_col[g][j_global_tile * DIM],
               DIM * sizeof(B_scales_col[0][0]));

      // --- Mvin B tiles (1 J-tile) ONCE; reused across all i0 super-blocks ---
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j_global_tile * DIM),
          b_base + k * DIM,   // b_base + (k * tiles_J_blk + 0)*DIM, tiles_J_blk=1
          DIM, DIM);
      }

      // Inner loop: i0 super-blocks of tiles_I_blk rows
      int total_tiles_I = (REAL_M + DIM - 1) / DIM;
      for (int it_base = 0; it_base < total_tiles_I; it_base += tiles_I_blk) {
        int tiles_I_this = tiles_I_blk;
        if (it_base + tiles_I_this > total_tiles_I)
          tiles_I_this = total_tiles_I - it_base;

        int row_start = it_base * DIM;
        int mrows = REAL_M - row_start;
        if (mrows > tiles_I_this * DIM) mrows = tiles_I_this * DIM;
        if (mrows <= 0) continue;
        int pad_i = tiles_I_this * DIM - mrows;

        // --- Stage A scales for this I super-block ---
        // A_scales_row[g][row]: rows it_base*DIM .. +tiles_I_this*DIM
        for (int g = 0; g < MATMUL_K / 32; g++)
          memcpy(&A_s[g][0],
                 &A_scales_row[g][row_start],
                 tiles_I_this * DIM * sizeof(A_scales_row[0][0]));

        gemmini_flush(0);

        // Step 1: Configure execution
        gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                    1, 1, 0, 0, false, 0, 0, 3, 0);

        // Step 2: Load A and B scales
        // A_S_BYTES_THIS = tiles_I_this * DIM * (MATMUL_K/32) bytes (one e8m0 per 32-elem block along K)
        int a_s_bytes_this = (MATMUL_K / 32) * tiles_I_this * DIM
                             * (int)sizeof(A_scales_row[0][0]);
        int b_s_bytes_this = (MATMUL_K / 32) * tiles_J_blk * DIM
                             * (int)sizeof(B_scales_col[0][0]);
        gemmini_mx_load_scales((uint64_t)A_s, a_s_bytes_this, 0);
        gemmini_mx_load_scales((uint64_t)B_s, b_s_bytes_this, 1);

        // Step 3: Mvin A tiles for this I super-block
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i = 0; i < tiles_I_this; i++) {
          for (int k = 0; k < tiles_K; k++) {
            gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (row_start + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM, DIM);
          }
        }

        // Step 4: Configure requantized mvout
        // i_bound=tiles_I_this, j_bound=tiles_J_blk=1, k_bound=tiles_K
        gemmini_config_st(BLK_OUTC * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                     tiles_I_this, tiles_J_blk, tiles_K,
                                     0, 0, 1);

        // Step 5: Execute fused hardware matmul
        // B is at b_base = BANK_NUM*BANK_ROWS - tiles_J_blk*tiles_K*DIM
        // Pass BANK_NUM*BANK_ROWS as the B spad end-address (b_base + tiles_J_blk*tiles_K*DIM)
        gemmini_loop_ws_spad(tiles_I_this, tiles_J_blk, tiles_K,
                             pad_i, pad_j, 0,
                             a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                             false, false, false, false, false,
                             NO_ACTIVATION, 0, 0, false, 0x38);

        // Step 6: Drain BF16 results from mx_smem to DRAM
        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(
            &C_hw[row_start + r][(j_global_tile * DIM) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * tiles_J_blk * DIM,
            ncols);
        }

        // Step 7: Fence
        gemmini_fence();
      }
    }
  }
}
