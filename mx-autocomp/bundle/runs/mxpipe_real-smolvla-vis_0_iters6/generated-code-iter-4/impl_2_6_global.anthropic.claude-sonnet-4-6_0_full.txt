Looking at the problem carefully:

**Constraints analysis:**
- Scratchpad: 8192 rows, each 16 bytes
- Accumulator: 256 rows, each 128 bytes (16 × 8 bytes)
- `tiles_K = 256/16 = 16`
- Accumulator constraint: `tiles_I * tiles_J * DIM ≤ 256` → `tiles_I * tiles_J ≤ 16`
- Best square: `tiles_I = 4, tiles_J = 4` (M_block=64, N_block=64)
- Scratchpad check: `(4*16 + 4*16)*16 = 2048 ≤ 8192` ✓
- Accumulator check: `4*4*16 = 256 ≤ 256` ✓

**Key improvements:**
1. Use `BLOCK_I = 64, BLOCK_J = 64` instead of `BLK` to maximize per-invocation work
2. Move `gemmini_flush` and `config_ex` outside the loop (config persists)
3. Scale staging only copies what's needed per block
4. Amortizes fixed overhead over 16× more compute

```c
void solution(void) {
  // Block dimensions chosen to saturate scratchpad and accumulator:
  //   tiles_I=4, tiles_J=4, tiles_K=16
  //   Scratchpad: (4*16 + 4*16)*16 = 2048 rows <= 8192 OK
  //   Accumulator: 4*4*16 = 256 rows <= 256 OK
  #define BLOCK_I 64
  #define BLOCK_J 64

  const int tiles_K = MATMUL_K / DIM;  // = 16

  // One-time setup: flush and configure format/dataflow outside the tile loop.
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < REAL_M; i0 += BLOCK_I) {
    int mrows_blk = REAL_M - i0;
    if (mrows_blk > BLOCK_I) mrows_blk = BLOCK_I;
    if (mrows_blk <= 0) continue;

    // Number of full+partial DIM tiles in this M-block
    int local_tiles_I = (mrows_blk + DIM - 1) / DIM;
    int pad_i = local_tiles_I * DIM - mrows_blk;

    for (int j0 = 0; j0 < REAL_N; j0 += BLOCK_J) {
      int ncols_blk = REAL_N - j0;
      if (ncols_blk > BLOCK_J) ncols_blk = BLOCK_J;
      if (ncols_blk <= 0) continue;

      int local_tiles_J = (ncols_blk + DIM - 1) / DIM;
      int pad_j = local_tiles_J * DIM - ncols_blk;

      // Verify accumulator won't overflow: local_tiles_I * local_tiles_J * DIM <= 256
      // With BLOCK_I=64, BLOCK_J=64: max 4*4*16=256, which is exactly ACC_ROWS.

      // Stage per-block e8m0 scales (only the rows/cols this block needs).
      // A_s: [tiles_K][local_tiles_I * DIM] row scales for A
      // B_s: [tiles_K][local_tiles_J * DIM] col scales for B
      for (int g = 0; g < tiles_K; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0], local_tiles_I * DIM);
        memcpy(&B_s[g][0], &B_scales_col[g][j0], local_tiles_J * DIM);
      }

      // Scratchpad layout:
      //   A tiles: a_base + (i * tiles_K + k) * DIM
      //   B tiles: b_base + (k * local_tiles_J + j) * DIM  [equiv: b_base + (j*tiles_K+k)*DIM per spec layout]
      uint32_t a_base = 0;
      uint32_t b_base = (uint32_t)(BANK_NUM * BANK_ROWS) - (uint32_t)(local_tiles_J * tiles_K * DIM);
      int SPAD_DEST = 128;

      // Load A scales (sel=0 = activation/A) and B scales (sel=1 = weight/B)
      gemmini_mx_load_scales((uint64_t)A_s,
                             (size_t)tiles_K * local_tiles_I * DIM * sizeof(uint8_t), 0);
      gemmini_mx_load_scales((uint64_t)B_s,
                             (size_t)tiles_K * local_tiles_J * DIM * sizeof(uint8_t), 1);

      // MVIN A tiles: A is row-major [REAL_M][MATMUL_K], stride = MATMUL_K bytes/row
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < local_tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *a_ptr = (const elem_t *)A_in
                                + (size_t)(i0 + i * DIM) * MATMUL_K
                                + (size_t)(k * DIM);
          gemmini_extended_mvin((void *)a_ptr,
                                a_base + (uint32_t)((i * tiles_K + k) * DIM),
                                DIM, DIM);
        }
      }

      // MVIN B tiles using mvin2: B is row-major [MATMUL_K][REAL_N], stride = MATMUL_N bytes/row
      // Layout in spad: b_base + (j * tiles_K + k) * DIM
      // b_base = BANK_NUM*BANK_ROWS - local_tiles_J*tiles_K*DIM (top of spad, growing down)
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < local_tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          const elem_t *b_ptr = (const elem_t *)B_in
                                + (size_t)(k * DIM) * MATMUL_N
                                + (size_t)(j0 + j * DIM);
          gemmini_extended_mvin2((void *)b_ptr,
                                 b_base + (uint32_t)((j * tiles_K + k) * DIM),
                                 DIM, DIM);
        }
      }

      // Configure output: stride in bytes for one BF16-packed output row
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   local_tiles_I, local_tiles_J, tiles_K,
                                   0, 0, 1);

      // Fused hardware matmul: B spad end address = BANK_NUM*BANK_ROWS (top sentinel)
      gemmini_loop_ws_spad(local_tiles_I, local_tiles_J, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain BF16 results from mx_smem to DRAM, one output row at a time.
      // smem layout: row r is at word offset SPAD_DEST*16 + r*local_tiles_J*DIM
      // ncols_blk elements per row (BF16, packed 4 per uint64 word in smem).
      for (int r = 0; r < mrows_blk; r++) {
        gemmini_mx_read_smem(
          &C_hw[i0 + r][(j0) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * local_tiles_J * DIM,
          ncols_blk);
      }
      gemmini_fence();
    }
  }

  #undef BLOCK_I
  #undef BLOCK_J
}
```
