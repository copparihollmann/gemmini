Looking at the error, the problem is clear: B tiles are being loaded with `gemmini_extended_mvin2` (port 1), but `gemmini_loop_ws_spad` requires all inputs to be loaded via `gemmini_extended_mvin` (port 0). I need to change all B tile loads to use `gemmini_extended_mvin` instead of `gemmini_extended_mvin2`.

```c
void solution(void) {
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;

    int tiles   = BLK / DIM;
    int tiles_K = MATMUL_K / DIM;

    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
    int SPAD_DEST   = 128;

    // --- Stage B scale factors (once per j0 block) ---
    for (int g = 0; g < MATMUL_K / 32; g++)
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);

    // --- One-time setup per j0 block ---
    gemmini_flush(0);

    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                1, 1, 0, 0, false,
                                0, 0, 3, 0);

    // Load B (weight) scales into mx_scale_b_mem (sel=1) once per j0.
    // They persist until overwritten, so we do NOT reload them in the i0 loop.
    gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

    // --- Mvin B tiles once per j0 block ---
    // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));   // B row stride = full N
    for (int j = 0; j < tiles; j++)
      for (int k = 0; k < tiles_K; k++)
        gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
          b_base + (k * tiles + j) * DIM, DIM, DIM);

    // ----------------------------------------------------------------
    // Inner loop over i0: A tiles and A scales change, B stays fixed
    // ----------------------------------------------------------------
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // --- Stage A scale factors for this i0 block ---
      for (int g = 0; g < MATMUL_K / 32; g++)
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);

      // Load A (activation) scales into mx_scale_a_mem (sel=0).
      // B scales already in mx_scale_b_mem from the outer j0 setup above.
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);

      // --- Mvin A tiles for this (i0, j0) ---
      // Must use gemmini_extended_mvin (port 0) for loop_ws_spad compatibility
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));  // A row stride = full K
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // --- Configure output / requantizer ---
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);

      // --- Execute fused tiled matmul ---
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           0, SPAD_DEST,
                           false, false,
                           false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // --- Drain BF16 results from mx_smem to DRAM ---
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK, ncols);

      gemmini_fence();
    }
  }
}
```
