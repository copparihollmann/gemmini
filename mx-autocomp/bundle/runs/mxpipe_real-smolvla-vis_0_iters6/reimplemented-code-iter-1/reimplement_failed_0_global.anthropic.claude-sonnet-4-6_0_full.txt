Looking at the code, I need to identify the bugs:

1. **B tile layout is wrong**: The code uses `b_base + (k * tiles + j) * DIM` but the spec says B tiles should be at `b_base + (j * tiles_K + k) * DIM`. The loop variable naming and indexing are swapped.

2. **B mvin stride**: B is stored row-major as `B[K][N]`, so stride should be `MATMUL_N * sizeof(elem_t)` — that part looks correct.

3. **B address calculation**: `B_in[k*DIM][j0*BLK + j*DIM]` — B is treated as shape `[K][N]`, so the DRAM address should be `(k*DIM) * MATMUL_N + (j0*BLK + j*DIM)` — that looks correct.

4. **mvin2 vs mvin**: B should be loaded via `gemmini_extended_mvin2` (separate port for B operand), but the code uses `gemmini_extended_mvin` for both A and B. This means both A and B go through port 0 with the same config_ld stride, causing the second config_ld to overwrite the first for all subsequent mvins on port 0.

5. **config_ld for B**: B mvin should use `gemmini_extended_mvin2` with `gemmini_extended3_config_ld(..., id=1)` to configure port 1 separately.

6. **smem read offset**: `r * BLK` should account for the fact that smem is indexed in BF16 words, and BLK columns worth of BF16 values per row.

Let me trace the B spad layout: `b_base + (j*tiles_K + k)*DIM` per spec. With `tiles = BLK/DIM` and `tiles_K = MATMUL_K/DIM`, this is correct per spec.

  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Configure port 0 (mvin) for A: stride = MATMUL_K columns
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // Configure port 1 (mvin2) for B: stride = MATMUL_N columns
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);

      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);

      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION,
                           0, 0, false, 0x38);

      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      gemmini_fence();
    }
}
