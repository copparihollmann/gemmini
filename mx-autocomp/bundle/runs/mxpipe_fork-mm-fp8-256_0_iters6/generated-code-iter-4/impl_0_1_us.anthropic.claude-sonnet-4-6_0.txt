void solution(void) {

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

#ifdef SPIKE_SIM
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
#else
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_A, (uint8_t *) &A_scales_row, MATMUL_M, MATMUL_K);
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_B, (uint8_t *) &B_scales_col, MATMUL_N, MATMUL_K);
#endif

  // Configure port 0 (mvin) for A tiles: stride = MATMUL_K * sizeof(elem_t)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 0);

  // Configure port 1 (mvin2) for B tiles: stride = MATMUL_N * sizeof(elem_t)
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), SCALE_IDENTITY, false, 1);

  // Interleaved A (mvin port 0) and B (mvin2 port 1) tile loads.
  // Both DMA ports operate concurrently, overlapping A and B transfers.
  //
  // For each k, issue all A rows for that k-slice, then issue all B columns
  // for that k-slice. Since the two ports are independent the hardware can
  // pipeline them; issuing them in alternating fashion keeps both ports busy.
  for (int k = 0; k < tiles_K; k++) {
    // Issue A tiles for this k-slice (one per i-row)
    for (int i = 0; i < tiles_I; i++) {
      elem_t *a_dram_ptr = ((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM;
      uint32_t a_sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)a_dram_ptr, a_sp_addr, DIM, DIM);
    }

    // Issue B tiles for this k-slice using mvin2 port, batching up to MAX_BLOCK_LEN=4
    for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {
      int batch = (tiles_J - j < MAX_BLOCK_LEN) ? (tiles_J - j) : MAX_BLOCK_LEN;
      elem_t *b_dram_ptr = ((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM;
      uint32_t b_sp_addr = b_base + (k * tiles_J + j) * DIM;
      gemmini_extended_mvin2((void *)b_dram_ptr, b_sp_addr, batch * DIM, DIM);
    }
  }

  int SPAD_DEST = 0;

  gemmini_config_st(DIM * sizeof(elem_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // ---- Compute ----
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      0, 0, 0,
      a_base,
      BANK_NUM * BANK_ROWS,
      0,
      SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION,
      0, 0,
      false,
      0x38);

  gemmini_fence();

#ifdef SPIKE_SIM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
#else
  printf("Moving out:\n");
  for (int i = 0; i < tiles_J*tiles_I*2; i++) {
      gemmini_mvout((void*)((uint64_t*) C_hw + i*2*DIM), SPAD_DEST + i*DIM);
  }
  gemmini_fence();
#endif

}
