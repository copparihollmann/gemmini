void solution(void) {

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

#ifdef SPIKE_SIM
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
#else
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_A, (uint8_t *) &A_scales_row, MATMUL_M, MATMUL_K);
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_B, (uint8_t *) &B_scales_col, MATMUL_N, MATMUL_K);
#endif

  // Configure port 0 (mvin) for A: row stride = MATMUL_K bytes
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Configure port 1 (mvin2) for B: row stride = MATMUL_N bytes
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Unified interleaved loop: for each k-slice, issue A tiles on port 0
  // and B tiles on port 1 back-to-back so both DMA ports can overlap.
  for (int k = 0; k < tiles_K; k++) {
    // Load all i-row A tiles for this k-slice via port 0 (mvin), batched
    for (int i = 0; i < tiles_I; i += MAX_BLOCK_LEN) {
      int batch_a = (tiles_I - i < MAX_BLOCK_LEN) ? (tiles_I - i) : MAX_BLOCK_LEN;
      // Each row of A in DRAM: consecutive k-tiles are contiguous within a row,
      // but here we fix k and vary i — stride along i dimension.
      // For batch along i: rows are separated by MATMUL_K elements in DRAM,
      // which matches the configured stride. Each batch element is tile (i+b, k).
      // gemmini_block_mvin uses port 0 and issues cols = batch*DIM, rows = DIM.
      // However, batching along i requires consecutive DRAM rows at stride MATMUL_K,
      // which is exactly what the configured stride provides — valid batching.
      elem_t *dram_a = ((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM;
      uint32_t sp_a  = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void*)dram_a, sp_a, DIM, batch_a * DIM);
    }

    // Load all j-col B tiles for this k-slice via port 1 (mvin2), batched
    for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {
      int batch_b = (tiles_J - j < MAX_BLOCK_LEN) ? (tiles_J - j) : MAX_BLOCK_LEN;
      elem_t *dram_b = ((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM;
      uint32_t sp_b  = b_base + (k * tiles_J + j) * DIM;
      gemmini_extended_mvin2((void*)dram_b, sp_b, DIM, batch_b * DIM);
    }
  }

  // SPAD_DEST = 128 avoids aliasing with a_base = 0 region
  int SPAD_DEST = 128;

  gemmini_config_st(OUT_COLS * sizeof(out_t));

  // lut_update_granularity = tiles_K: refresh LUT only once per (i,j) output
  // tile pair, amortizing requantizer overhead across the full K reduction
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, tiles_K);

  // ---- Compute ----
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      0, 0, 0,
      a_base,
      BANK_NUM * BANK_ROWS,
      0,
      SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION,
      0, 0,
      false,
      0x38);

  gemmini_fence();

#ifdef SPIKE_SIM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
#else
  printf("Moving out:\n");
  for (int i = 0; i < tiles_J*tiles_I*2; i++) {
      gemmini_mvout((void*)((uint64_t*) C_hw + i*2*DIM), SPAD_DEST + i*DIM);
  }
  gemmini_fence();
#endif

  gemmini_fence();
}
