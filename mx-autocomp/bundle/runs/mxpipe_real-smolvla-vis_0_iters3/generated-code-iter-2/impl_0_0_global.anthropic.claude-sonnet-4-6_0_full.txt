void solution(void) {
  // Hoist flush and config_ex outside all loops — they are invariant
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 128;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0)  continue;

    // A scales only depend on i0 — load once per i0 sweep over all j0.
    // A_scales_row is [MATMUL_K/32][REAL_M], layout is row-major so
    // A_scales_row[g][i0*BLK .. i0*BLK+BLK-1] are contiguous in each g-row.
    // We load the entire A_scales_row array once per i0 block; the hardware
    // requantizer uses scale_dram indexing (scale_dram + m*N_blocks + bi),
    // so we pass the pointer to the sub-block that begins at column i0*BLK.
    // Since A_S_BYTES = (MATMUL_K/32)*BLK, loading BLK bytes per g-row is
    // exactly the slice for this i0 block.
    gemmini_mx_load_scales((uint64_t)(&A_scales_row[0][i0 * BLK]), A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0)  continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // B scales depend on j0 — load once per j0 iteration.
      // B_scales_col is [MATMUL_K/32][REAL_N]; the slice for this j0 block
      // starts at B_scales_col[0][j0*BLK] with B_S_BYTES total bytes.
      gemmini_mx_load_scales((uint64_t)(&B_scales_col[0][j0 * BLK]), B_S_BYTES, 1);

      // ---- MVIN A tiles (scratchpad mvin port 0) ----
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // ---- MVIN B tiles (scratchpad mvin port 0, reuse after A config) ----
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM,
            DIM, DIM);

      // ---- Configure requantized mvout ----
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);

      // ---- Fused hardware matmul ----
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // ---- Drain BF16 results from mx_smem to DRAM ----
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
