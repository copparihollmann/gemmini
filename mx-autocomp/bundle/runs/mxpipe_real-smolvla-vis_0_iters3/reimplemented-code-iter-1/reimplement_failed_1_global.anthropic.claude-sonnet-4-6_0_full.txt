Looking at the code, I need to identify the bug. The key issue is in the B matrix layout in the scratchpad.

The documentation states:
- **B tile layout:** `b_base + (j*tiles_K + k)*DIM` — j is the tile column index, k is the tile K index
- But the code loads B as: `b_base + (k * tiles + j) * DIM` — this is `k*tiles_J + j` order

The loop iterates `j` in outer, `k` in inner, but the spad address formula uses `k*tiles + j` which is wrong. It should be `j*tiles_K + k` per the architecture spec.

Also checking the B DRAM address: B is stored row-major as `[K][N]`, so element at row `k*DIM`, col `j0*BLK + j*DIM` has DRAM address `B_in + (k*DIM)*MATMUL_N + (j0*BLK + j*DIM)`. That part looks correct.

The scratchpad address for B should be `b_base + (j*tiles_K + k)*DIM`.

Also, `gemmini_mxquant_config_mvout` is called with `tiles` for both i and j bounds but the loop uses `tiles` for I (rows) and `tiles` for J (cols) — for a non-square problem where `tiles_I != tiles_J` this could be wrong. However since `BLK` is fixed for both dimensions here, this may be okay.

Let me also verify the `scale_factors` — it appears undefined in context. Looking at the original code it uses `scale_factors` which seems to be a pre-existing global. I'll keep it.

The main fix: B spad address should be `b_base + (j * tiles_K + k) * DIM`.

```c
void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++)
   for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // Stage per-block scales for this (i0, j0) block
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles_I = BLK / DIM;
      int tiles_J = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Step 1: Configure execution
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                   1, 1, 0, 0, false, 0, 0, 3, 0);

      // Step 2: Load scale factors (sel=0 for A/activations, sel=1 for B/weights)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN tiles of A into scratchpad
      // A layout: a_base + (i*tiles_K + k)*DIM
      // A is [M][K] row-major, stride = MATMUL_K bytes per row
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 4: MVIN tiles of B into scratchpad
      // B layout: b_base + (j*tiles_K + k)*DIM  (j=col-tile, k=K-tile)
      // B is [K][N] row-major, stride = MATMUL_N bytes per row
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 5: Configure output stride and requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                    tiles_I, tiles_J, tiles_K,
                                    0, 0, 1);

      // Step 6: Compute weight-stationary tiled matmul
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain BF16 results from mx_smem to DRAM
      if (ncols == BLK) {
        // Full-width block: contiguous transfer
        gemmini_mx_read_smem(&C_hw[i0 * BLK][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16, mrows * BLK);
      } else {
        // Partial last column block: transfer row by row
        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * BLK, ncols);
        }
      }
      gemmini_fence();
    }
}
```
