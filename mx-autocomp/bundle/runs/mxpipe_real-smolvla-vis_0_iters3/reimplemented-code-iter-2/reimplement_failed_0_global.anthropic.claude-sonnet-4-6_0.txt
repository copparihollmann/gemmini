void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base   = 0;
  uint32_t b_base   = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int      SPAD_DEST = 0;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0)  continue;

    int pad_i = BLK - mrows;

    gemmini_mx_load_scales(
        (uint64_t)(&A_scales_row[0][i0 * BLK]),
        A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0)  continue;

      int pad_j = BLK - ncols;

      gemmini_mx_load_scales(
          (uint64_t)(&B_scales_col[0][j0 * BLK]),
          B_S_BYTES, 1);

      // ---- MVIN A tiles (mvin port 0) ----
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);

      // ---- MVIN B tiles (mvin2 port 1) ----
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);

      // ---- Configure requantized mvout ----
      gemmini_extended_config_st(BLK_OUTC * sizeof(out_t), 0, ACC_SCALE_IDENTITY);
      gemmini_mxquant_config_mvout(
          (uint64_t)scale_factors,
          tiles, tiles, tiles_K,
          0, 0, 1);

      // ---- Fused hardware matmul ----
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // ---- Drain BF16 results from mx_smem to DRAM ----
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
          &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
          SPAD_DEST * 16 + r * BLK,
          ncols);

      gemmini_fence();
    }
  }
}
