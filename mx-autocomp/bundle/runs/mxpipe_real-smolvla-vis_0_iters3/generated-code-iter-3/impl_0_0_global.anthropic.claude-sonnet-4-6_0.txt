void solution(void) {
  // Hoist all loop-invariant computations and configurations
  int tiles   = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base  = 0;
  uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST    = 128;

  // Hoist one-time hardware initialization
  gemmini_flush(0);

  // Hoist loop-invariant configuration (format, dataflow, strides never change)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
      /*set_only_strides*/false,
      /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

  // Hoist loop-invariant output config (BLK_OUTC, scale_factors, tiles, tiles_K never change)
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      /*i_bound*/tiles, /*j_bound*/tiles, /*k_bound*/tiles_K,
      /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Compute actual rows/cols for this block (handles partial last block)
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Stage per-block scale factors (data-dependent, cannot hoist)
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Per-block scale loads (data changes each block, must stay in loop)
      // Rule: sel=0 for A (activation) scales, sel=1 for B (weight) scales
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Load A tiles into scratchpad (stride = full K row width)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM,
              DIM, DIM);

      // Load B tiles into scratchpad (stride = full N row width)
      // B layout: b_base + (k * tiles + j) * DIM  (k-major within block)
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (k * tiles + j) * DIM,
              DIM, DIM);

      // Execute fused hardware weight-stationary matmul
      // Note: gemmini_loop_ws_spad internally issues funct 24 (LOOP_WS_CONFIG_SPAD_AB)
      // which sets the one-shot mx_loop_spad_marker before the funct 8 (LOOP_WS).
      // This must be called every iteration (marker is one-shot, consumed on execute).
      gemmini_loop_ws_spad(
          tiles, tiles, tiles_K,
          pad_i, pad_j, 0,
          a_base, BANK_NUM * BANK_ROWS,
          /*D*/0, SPAD_DEST,
          /*A_transpose*/false, /*B_transpose*/false,
          /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
          NO_ACTIVATION,
          /*a_spad_id*/0, /*b_spad_id*/0,
          /*is_resadd*/false,
          /*skips*/0x38);

      // Drain BF16-packed results from mx_smem to DRAM (row by row for partial blocks)
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(
            &C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
            SPAD_DEST * 16 + r * BLK,
            ncols);

      // Full memory barrier: host must not read C_hw until fence completes
      gemmini_fence();
    }
  }
}
