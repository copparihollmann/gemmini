Here is the optimization plan and the high-performance implementation.

### Optimization Plan: Softmax Group Loop Unrolling (Strategy 7)

**Goal:** Reduce host CPU execution overhead during the integer softmax and FP8 requantization passes.

1. **Locate CPU Bottleneck:** The host CPU does not have a floating-point unit and handles the integer softmax using the I-BERT approximation (`mx_iexp` and `mx_ienc_fp8`). The inner 32-element group loops inside the main attention-row loop are executed frequently ($\text{ATT\_S} \times \text{GROUPS\_S} = 128 \times 4 = 512$ times per call). 
2. **Apply Loop Unrolling:** We will manually unroll the 32-element loops inside the softmax block by a factor of 8. Since $32$ is a multiple of $8$, the loops can be neatly structured into 4-step strides of 8 elements.
3. **Benefits:** 
   - Eliminates loop branch misprediction penalties on the host processor.
   - Reduces pointer arithmetic and loop-counter update instructions.
   - Exposes Instruction-Level Parallelism (ILP), allowing the compiler to schedule independent memory loads, `mx_iexp` lookups, and scalar accumulations more efficiently.

### Optimized Code Implementation

// Optimized MX attention kernel (S=128, D=64) using Causal Loop Splitting and Group Loop Unrolling.
void solution(void) {
  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k += 2) { 
        int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
      }
    gemmini_config_ld(ATT_S * sizeof(elem_t));   // B=KT[D][S] stride N=S -> slot (k*tiles_J+j)
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j += 2) { 
        int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
      }
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);

#if ATT_CAUSAL
    // Causal Loop Splitting: Eliminate branching in the inner loop
    int boundary_word = i / 4;

    // 1) Process fully unmasked words
    for (int wd = 0; wd < boundary_word; wd++) {
      uint64_t w = S1_hw[i][wd];
      int base_j = wd * 4;
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[base_j + l] = qq;
        if (qq > mq) mq = qq;
      }
    }

    // 2) Process the boundary word
    {
      uint64_t w = S1_hw[i][boundary_word];
      int base_j = boundary_word * 4;
      int active_lanes = (i % 4) + 1;
      for (int l = 0; l < active_lanes; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[base_j + l] = qq;
        if (qq > mq) mq = qq;
      }
      for (int l = active_lanes; l < 4; l++) {
        q[base_j + l] = -((int64_t)1 << 62);
      }
    }

    // 3) Process fully masked elements (future columns)
    for (int j = (boundary_word + 1) * 4; j < ATT_S; j++) {
      q[j] = -((int64_t)1 << 62);
    }
#else
    // Fallback: standard branchless loop when causal masking is disabled
    for (int j = 0; j < ATT_S; j += 4) {
      uint64_t w = S1_hw[i][j / 4];
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[j + l] = qq;
        if (qq > mq) mq = qq;
      }
    }
#endif

    int64_t p[ATT_S];
    int64_t sm = 0;
    int64_t gmax_arr[GROUPS_S];

    // Softmax Group Loop Unrolling by 8 for Exponentiation
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = 0;
      int base_idx = g * 32;

      for (int k = 0; k < 32; k += 8) {
        int idx0 = base_idx + k + 0;
        int64_t pv0 = mx_iexp(q[idx0] - mq); p[idx0] = pv0; sm += pv0; if (pv0 > gmax) gmax = pv0;

        int idx1 = base_idx + k + 1;
        int64_t pv1 = mx_iexp(q[idx1] - mq); p[idx1] = pv1; sm += pv1; if (pv1 > gmax) gmax = pv1;

        int idx2 = base_idx + k + 2;
        int64_t pv2 = mx_iexp(q[idx2] - mq); p[idx2] = pv2; sm += pv2; if (pv2 > gmax) gmax = pv2;

        int idx3 = base_idx + k + 3;
        int64_t pv3 = mx_iexp(q[idx3] - mq); p[idx3] = pv3; sm += pv3; if (pv3 > gmax) gmax = pv3;

        int idx4 = base_idx + k + 4;
        int64_t pv4 = mx_iexp(q[idx4] - mq); p[idx4] = pv4; sm += pv4; if (pv4 > gmax) gmax = pv4;

        int idx5 = base_idx + k + 5;
        int64_t pv5 = mx_iexp(q[idx5] - mq); p[idx5] = pv5; sm += pv5; if (pv5 > gmax) gmax = pv5;

        int idx6 = base_idx + k + 6;
        int64_t pv6 = mx_iexp(q[idx6] - mq); p[idx6] = pv6; sm += pv6; if (pv6 > gmax) gmax = pv6;

        int idx7 = base_idx + k + 7;
        int64_t pv7 = mx_iexp(q[idx7] - mq); p[idx7] = pv7; sm += pv7; if (pv7 > gmax) gmax = pv7;
      }
      gmax_arr[g] = gmax;
    }

    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

    // Softmax Group Loop Unrolling by 8 for Requantization / Encoding
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      int base_idx = g * 32;

      if (gmax == 0) {
        P_scales[g][i] = 0;
        for (int k = 0; k < 32; k += 8) {
          P_q[i][base_idx + k + 0] = 0;
          P_q[i][base_idx + k + 1] = 0;
          P_q[i][base_idx + k + 2] = 0;
          P_q[i][base_idx + k + 3] = 0;
          P_q[i][base_idx + k + 4] = 0;
          P_q[i][base_idx + k + 5] = 0;
          P_q[i][base_idx + k + 6] = 0;
          P_q[i][base_idx + k + 7] = 0;
        }
        continue;
      }

      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales[g][i] = (uint8_t)(e + 127);

      for (int k = 0; k < 32; k += 8) {
        P_q[i][base_idx + k + 0] = mx_ienc_fp8(p[base_idx + k + 0], inv_sm, ls, e);
        P_q[i][base_idx + k + 1] = mx_ienc_fp8(p[base_idx + k + 1], inv_sm, ls, e);
        P_q[i][base_idx + k + 2] = mx_ienc_fp8(p[base_idx + k + 2], inv_sm, ls, e);
        P_q[i][base_idx + k + 3] = mx_ienc_fp8(p[base_idx + k + 3], inv_sm, ls, e);
        P_q[i][base_idx + k + 4] = mx_ienc_fp8(p[base_idx + k + 4], inv_sm, ls, e);
        P_q[i][base_idx + k + 5] = mx_ienc_fp8(p[base_idx + k + 5], inv_sm, ls, e);
        P_q[i][base_idx + k + 6] = mx_ienc_fp8(p[base_idx + k + 6], inv_sm, ls, e);
        P_q[i][base_idx + k + 7] = mx_ienc_fp8(p[base_idx + k + 7], inv_sm, ls, e);
      }
    }
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k += 2) { 
        int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
      }
    gemmini_config_ld(ATT_D * sizeof(elem_t));   // B=V[S][D] stride N=D -> slot (k*tiles_J+j)
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j += 2) { 
        int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
      }
    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
