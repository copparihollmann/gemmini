An elegant and highly effective optimization strategy for this kernel is **Strategy 5: Implement host-coprocessor pipelining**. 

### Optimization Plan

1. **The Inefficiency**:
   In the baseline code, the entire $128 \times 128$ intermediate attention matrix `S1_hw` is transferred in one bulk DMA call (`gemmini_mx_read_smem`) from the accelerator's shared memory (`mx_smem`) to off-chip DRAM. A heavy CPU-side memory fence (`gemmini_fence()`) immediately follows. This forces the host CPU to idle while waiting for the entire matrix transfer to complete. Only then does the CPU begin the computationally expensive, float-free softmax calculation row-by-row.

2. **The Pipelined Solution**:
   Since the RoCC command queue allows the CPU to issue instructions to Gemmini asynchronously, we can overlap the memory transfer of row $i+1$ with the CPU's softmax calculation of row $i$.
   * **Initialization**: Immediately after `gemmini_loop_ws_spad` in Part 1 finishes launching, we asynchronously initiate the DMA read for only the **first row** (row 0) of `S1_hw`.
   * **In the Softmax Loop (for each row $i$)**:
     1. Issue `gemmini_fence()` to block the CPU until row $i$'s asynchronous transfer is guaranteed to be complete.
     2. Immediately issue an asynchronous `gemmini_mx_read_smem` command for the **next row** ($i + 1$), if it exists.
     3. Perform the integer softmax and fp8-requantization on the CPU for the current row $i$ using the already-transferred data in `S1_hw[i]`.
   * **Result**: The latency of transferring row $i+1$ from Gemmini to DRAM is entirely hidden behind the CPU execution of row $i$'s softmax logic, drastically reducing overall execution time.

---

### Optimized Code

Here is the fully optimized implementation of `solution()` using host-coprocessor pipelining:

```c
// Optimized MX attention kernel (S=128, D=64) with host-coprocessor pipelining.
void solution(void) {
  // Hoist the persistent MX format and execution configuration to the very beginning
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);
    
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++) {
      for (int k = 0; k < tiles_K; k += 2) { 
        int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
      }
    }
    
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      for (int j = 0; j < tiles_J; j += 2) { 
        int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
      }
    }
    
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    // Asynchronously prefetch only the 0-th row of S1_hw to kickstart the pipeline
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S);
  }

  // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
  for (int i = 0; i < ATT_S; i++) {
    // 1. Wait until row i's data is fully written to DRAM
    gemmini_fence();

    // 2. Asynchronously request transfer of row i+1 while the CPU processes row i
    if (i + 1 < ATT_S) {
      gemmini_mx_read_smem(&S1_hw[i + 1][0], SPAD_DEST * 16 + (i + 1) * ATT_S, ATT_S);
    }

    // 3. Perform computation on row i
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);
    for (int j = 0; j < ATT_S; j += 4) {
      uint64_t w = S1_hw[i][j / 4];
      for (int l = 0; l < 4; l++) {
#if ATT_CAUSAL
        if (j + l > i) { q[j + l] = -((int64_t)1 << 62); continue; }
#endif
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[j + l] = qq; if (qq > mq) mq = qq;
      }
    }
    
    int64_t p[ATT_S];
    int64_t sm = 0;
    int64_t gmax_arr[GROUPS_S];
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = 0;
      for (int j = g * 32; j < (g + 1) * 32; j++) { 
        int64_t pv = mx_iexp(q[j] - mq); 
        p[j] = pv; 
        sm += pv; 
        if (pv > gmax) gmax = pv; 
      }
      gmax_arr[g] = gmax;
    }
    
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      if (gmax == 0) { 
        P_scales[g][i] = 0; 
        for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0; 
        continue; 
      }
      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
    }
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);
    
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++) {
      for (int k = 0; k < tiles_K; k += 2) { 
        int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
      }
    }
    
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      for (int j = 0; j < tiles_J; j += 2) { 
        int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
      }
    }
    
    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
