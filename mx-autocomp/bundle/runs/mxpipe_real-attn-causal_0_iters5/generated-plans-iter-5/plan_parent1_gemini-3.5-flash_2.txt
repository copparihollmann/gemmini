An elegant and highly effective optimization strategy for this kernel is **Causal-Aware Softmax and Requantization Bypassing with Batch-3 Scratchpad DMA Coalescing**.

### How the Strategy Works

1. **Causal-Aware Softmax and Requantization Bypassing**: When causal attention (`#if ATT_CAUSAL`) is enabled, the upper triangle elements (where $j > i$) are masked out and mathematically guaranteed to produce a softmax probability of zero. We exploit this on the host CPU by rewriting the loops to determine the active boundaries and completely bypassing the memory loads of `S1_hw`, the call to `mx_ibf16_to_q`, the `mx_iexp` math block, and the `mx_ienc_fp8` requantization for all indices where $j > i$. This eliminates roughly half of all CPU math and memory accesses, reducing host pipeline stalls and branch mispredictions.
2. **Batch-3 Scratchpad DMA Coalescing**: The original loading of weight matrices ($K^T$ and $P_q$) batches loads into groups of 2 tiles. However, the hardware DMA controller is capable of loading up to 3 tiles at a time without exceeding the maximum column constraint (since $3 \times 16 = 48 \le 63$). By restructuring the loading to fetch tiles in groups of 3 (e.g., $3, 3, 2$), we reduce the number of issued `mvin` instructions for $K^T$ from 16 to 12, and for $P_q$ from 32 to 24.
3. **Software Cache Prefetching**: We insert `__builtin_prefetch` instructions to pull the upcoming rows of `S1_hw` and `P_q` into the CPU's cache hierarchy ahead of time while the CPU computes the softmax of the current row.

### Optimized Code

```c
// Optimized MX attention kernel (S=128, D=64) with hoisted configuration.
void solution(void) {
  // Hoist the persistent MX format and execution configuration to the very beginning
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);
    
    // Load Q with balanced batching of 2 (since tiles_K = 4)
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++) {
      gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + 0 * DIM),
                            a_base + (i * tiles_K + 0) * DIM, 2 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + 2 * DIM),
                            a_base + (i * tiles_K + 2) * DIM, 2 * DIM, DIM);
    }
    
    // Load KT with batching of 3 (reduces total mvin instructions from 16 to 12)
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + 0 * DIM),
                            b_base + (k * tiles_J + 0) * DIM, 3 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + 3 * DIM),
                            b_base + (k * tiles_J + 3) * DIM, 3 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + 6 * DIM),
                            b_base + (k * tiles_J + 6) * DIM, 2 * DIM, DIM);
    }
    
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
  for (int i = 0; i < ATT_S; i++) {
    // Software Prefetching for the next row
    if (i + 1 < ATT_S) {
      __builtin_prefetch(&S1_hw[i+1][0], 0, 3);
      __builtin_prefetch(&S1_hw[i+1][8], 0, 3);
      __builtin_prefetch(&S1_hw[i+1][16], 0, 3);
      __builtin_prefetch(&S1_hw[i+1][24], 0, 3);
      __builtin_prefetch(&P_q[i+1][0], 1, 3);
      __builtin_prefetch(&P_q[i+1][64], 1, 3);
    }

    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);

#if ATT_CAUSAL
    // Causal optimized unpacking: skip reading upper-triangle S1_hw and calling mx_ibf16_to_q
    int j_unpack = 0;
    int limit = (i / 4) * 4;
    for (; j_unpack < limit; j_unpack += 4) {
      uint64_t w = S1_hw[i][j_unpack / 4];
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[j_unpack + l] = qq;
        if (qq > mq) mq = qq;
      }
    }
    if (j_unpack < ATT_S) {
      uint64_t w = S1_hw[i][j_unpack / 4];
      for (int l = 0; l < 4; l++) {
        if (j_unpack + l > i) {
          q[j_unpack + l] = -((int64_t)1 << 62);
        } else {
          int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
          q[j_unpack + l] = qq;
          if (qq > mq) mq = qq;
        }
      }
      j_unpack += 4;
    }
    for (; j_unpack < ATT_S; j_unpack++) {
      q[j_unpack] = -((int64_t)1 << 62);
    }
#else
    for (int j = 0; j < ATT_S; j += 4) {
      uint64_t w = S1_hw[i][j / 4];
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[j + l] = qq;
        if (qq > mq) mq = qq;
      }
    }
#endif

    int64_t p[ATT_S];
    int64_t sm = 0;
    int64_t gmax_arr[GROUPS_S];

#if ATT_CAUSAL
    // Causal optimized iexp: skip calling mx_iexp for upper triangle elements
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = 0;
      int g_start = g * 32;
      int g_end = (g + 1) * 32;
      for (int j = g_start; j < g_end; j++) {
        if (j > i) {
          p[j] = 0;
        } else {
          int64_t pv = mx_iexp(q[j] - mq);
          p[j] = pv;
          sm += pv;
          if (pv > gmax) gmax = pv;
        }
      }
      gmax_arr[g] = gmax;
    }
#else
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = 0;
      for (int j = g * 32; j < (g + 1) * 32; j++) {
        int64_t pv = mx_iexp(q[j] - mq);
        p[j] = pv;
        sm += pv;
        if (pv > gmax) gmax = pv;
      }
      gmax_arr[g] = gmax;
    }
#endif

    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

#if ATT_CAUSAL
    // Causal optimized requantization: skip mx_ienc_fp8 for upper triangle elements
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      if (gmax == 0) {
        P_scales[g][i] = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0;
        continue;
      }
      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales[g][i] = (uint8_t)(e + 127);
      int g_start = g * 32;
      int g_end = (g + 1) * 32;
      for (int j = g_start; j < g_end; j++) {
        if (j > i) {
          P_q[i][j] = 0;
        } else {
          P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
        }
      }
    }
#else
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      if (gmax == 0) {
        P_scales[g][i] = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0;
        continue;
      }
      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g * 32; j < (g + 1) * 32; j++) {
        P_q[i][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
      }
    }
#endif
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);
    
    // Load P_q with batching of 3 (reduces total mvin instructions from 32 to 24)
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++) {
      gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + 0 * DIM),
                            a_base + (i * tiles_K + 0) * DIM, 3 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + 3 * DIM),
                            a_base + (i * tiles_K + 3) * DIM, 3 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + 6 * DIM),
                            a_base + (i * tiles_K + 6) * DIM, 2 * DIM, DIM);
    }
    
    // Load V with balanced batching of 2 (since tiles_J = 4)
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + 0 * DIM),
                            b_base + (k * tiles_J + 0) * DIM, 2 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + 2 * DIM),
                            b_base + (k * tiles_J + 2) * DIM, 2 * DIM, DIM);
    }
    
    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
