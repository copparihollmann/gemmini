### Optimization Plan

We will apply **Strategy 2** to eliminate redundant operations per element inside the hot inner loop of the softmax quantization phase. 

1. **Analyze the Inefficiency**:
   In the original softmax quantization loop, the function `mx_ienc_fp8(p[j], inv_sm, ls, e)` is called for every element $j$. Under the hood, this function derives the bitwise shift amount for quantization from `ls` and `e` using the formula:
   $$\text{shift} = ls + \text{MX\_FRAC} + e - \text{MX\_P\_TARGET\_LOG2}$$
   Since both `ls` and `e` are constant across all elements within a group $g$, calculating this shift amount for every single element $j$ inside the inner loop is redundant and wastes valuable CPU cycles on a host processor that may lack an FPU or have limited instruction throughput.

2. **Formulate the Solution**:
   - We will define a specialized helper function `mx_ienc_fp8_opt(int64_t p, uint64_t inv_sm, int shift)` which delegates directly to the original `mx_ienc_fp8`. To bypass the internal shift derivation inside `mx_ienc_fp8` while maintaining exact bitwise equivalence, we pass `shift - MX_FRAC + MX_P_TARGET_LOG2` as the `ls` parameter and `0` as the `e` parameter.
   - When inlined by the compiler, the constant-folding pass will simplify the internal expression `(shift - MX_FRAC + MX_P_TARGET_LOG2) + MX_FRAC + 0 - MX_P_TARGET_LOG2` back to the pre-computed `shift`, completely eliminating redundant addition and subtraction operations inside the loop.
   - We hoist the `shift` calculation out of the inner loop over $j$ and pass the pre-computed `shift` directly to `mx_ienc_fp8_opt`.

This optimization preserves exact bitwise results, avoids re-implementing any of the complex rounding and clamping logic of FP8 encoding, and significantly speeds up the host-side execution of the integer softmax.

### Optimized Code

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

#define DIM 16
#define ADDR_LEN 32

// Specialized helper to eliminate redundant shift computations inside the inner loop
static inline uint8_t mx_ienc_fp8_opt(int64_t p, uint64_t inv_sm, int shift) {
  // Leverage compiler inlining and constant folding to pass pre-computed shift
  return mx_ienc_fp8(p, inv_sm, shift - MX_FRAC + MX_P_TARGET_LOG2, 0);
}

// Optimized MX attention kernel (S=128, D=64) using Causal Loop Splitting and Hoisted Shifts.
void solution(void) {
  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k += 2) { int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); }
    gemmini_config_ld(ATT_S * sizeof(elem_t));   // B=KT[D][S] stride N=S -> slot (k*tiles_J+j)
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j += 2) { int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); }
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);

#if ATT_CAUSAL
    // Causal Loop Splitting: Eliminate branching in the inner loop
    int boundary_word = i / 4;

    // 1) Process fully unmasked words
    for (int wd = 0; wd < boundary_word; wd++) {
      uint64_t w = S1_hw[i][wd];
      int base_j = wd * 4;
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[base_j + l] = qq;
        if (qq > mq) mq = qq;
      }
    }

    // 2) Process the boundary word
    {
      uint64_t w = S1_hw[i][boundary_word];
      int base_j = boundary_word * 4;
      int active_lanes = (i % 4) + 1;
      for (int l = 0; l < active_lanes; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[base_j + l] = qq;
        if (qq > mq) mq = qq;
      }
      for (int l = active_lanes; l < 4; l++) {
        q[base_j + l] = -((int64_t)1 << 62);
      }
    }

    // 3) Process fully masked elements (future columns)
    for (int j = (boundary_word + 1) * 4; j < ATT_S; j++) {
      q[j] = -((int64_t)1 << 62);
    }
#else
    // Fallback: standard branchless loop when causal masking is disabled
    for (int j = 0; j < ATT_S; j += 4) {
      uint64_t w = S1_hw[i][j / 4];
      for (int l = 0; l < 4; l++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
        q[j + l] = qq;
        if (qq > mq) mq = qq;
      }
    }
#endif

    int64_t p[ATT_S];
    int64_t sm = 0;
    int64_t gmax_arr[GROUPS_S];                                // fused group-max: capture per-group max IN the iexp pass
    for (int g = 0; g < GROUPS_S; g++) {                       // (group-nested, register-scalar gmax) -> removes the
      int64_t gmax = 0;                                        // separate 16384-elem max scan. Output bit-identical.
      for (int j = g * 32; j < (g + 1) * 32; j++) { int64_t pv = mx_iexp(q[j] - mq); p[j] = pv; sm += pv; if (pv > gmax) gmax = pv; }
      gmax_arr[g] = gmax;
    }
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;   // one rounded reciprocal/row
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      if (gmax == 0) { P_scales[g][i] = 0; for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0; continue; }
      // group e8m0 scale so the group-max lands at ~2^MX_P_TARGET_LOG2 (moderate; avoids e4 overflow)
      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales[g][i] = (uint8_t)(e + 127);
      
      // Hoist the shift calculation out of the quantization loop
      int shift = ls + MX_FRAC + e - MX_P_TARGET_LOG2;
      for (int j = g * 32; j < (g + 1) * 32; j++) {
        P_q[i][j] = mx_ienc_fp8_opt(p[j], inv_sm, shift);
      }
    }
  }

  // ---- 3) O = P @ V on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k += 2) { int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); }
    gemmini_config_ld(ATT_D * sizeof(elem_t));   // B=V[S][D] stride N=D -> slot (k*tiles_J+j)
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j += 2) { int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
        gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                              b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); }
    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
