Looking at the failing code, I need to identify the issues:

1. **B matrix mvin uses wrong port** - B should use `gemmini_extended_mvin2` (mvin2 port), not `gemmini_extended_mvin` (mvin port). The scratchpad layout requires A via mvin and B via mvin2.

2. **B matrix stride calculation is wrong** - B is stored in DRAM as `[K][N]` layout. The stride should be `MATMUL_N * sizeof(elem_t)` which is what we have, but let me verify the address calculation: `B_in[k*DIM][j0*BLK + j*DIM]` - this needs stride = `MATMUL_N`.

3. **B scratchpad layout** - The B base should be `BANK_NUM*BANK_ROWS - tiles_K*tiles*DIM` (tiles here is tiles_J), and B tile at `b_base + (j*tiles_K + k)*DIM` (not `k*tiles + j`). The layout in the spec says `b_base + (j*tiles_K + k)*DIM`.

4. **scale_factors** - The `gemmini_mxquant_config_mvout` passes `scale_factors` but this should match the output scale used in requantization. For BF16 output (out_fmt=3), we can pass 0 or a valid address.

5. **The B mvin address indexing** - `b_base + (k * tiles + j) * DIM` should be `b_base + (j * tiles_K + k) * DIM` to match the layout spec.

void solution(void) {
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles_I_blk = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // tiles_I and tiles_J are both tiles_I_blk (BLK/DIM tiles per block)
      int tiles_I = tiles_I_blk;
      int tiles_J = tiles_I_blk;

      // Copy relevant scale rows into contiguous scratch arrays
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      uint32_t a_base = 0;
      // B placed at top of scratchpad, b_base grows downward
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Step 2: Load scale factors — A=sel0, B=sel1
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 4: MVIN A tiles (rows of A block) via mvin port
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 4: MVIN B tiles (cols of B block) via mvin2 port
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (j * tiles_K + k) * DIM,
            DIM, DIM);
        }
      }

      // Step 5: Configure output stride and requantizer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          tiles_I, tiles_J, tiles_K,
          /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

      // Step 6: Execute fused weight-stationary matmul
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false,
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain results from mx_smem to DRAM
      if (mrows == BLK && ncols == BLK) {
        // Fast path: full BLK×BLK block — single contiguous smem read
        gemmini_mx_read_smem(&C_hw[i0 * BLK][j0 * BLK / BF16_PER_WORD],
                             SPAD_DEST * 16, BLK * BLK);
      } else {
        // Slow path: partial boundary block — per-row reads with ncols truncation
        for (int r = 0; r < mrows; r++)
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                               SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
}
