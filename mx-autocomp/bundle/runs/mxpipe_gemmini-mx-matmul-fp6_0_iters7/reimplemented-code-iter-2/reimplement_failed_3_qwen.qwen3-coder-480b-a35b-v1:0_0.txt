// Optimized high-performance MX matmul kernel (128x128x256, fp6:e3m2).
void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2, tj = MATMUL_N / DIM / 2, tk = MATMUL_K / DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;

    // 1. Configure execution (WS mode, formats, lut enabled)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 1, 1, 3, 1);

    // 2. Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // 3. Load quantization LUTs
    gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
    gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);

    // 4. Coalesced MVIN for A (respecting MAX_BLOCK_LEN=4 limit)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      // Break down the transfer into blocks of at most 4 tiles (64 columns)
      // Since tk=16, we need to do 16/4=4 separate transfers
      for (int k_block = 0; k_block < tk; k_block += 4) {
        int remaining_k = tk - k_block;
        int batch_size = (remaining_k > 4) ? 4 : remaining_k;
        
        gemmini_extended_mvin(
          (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k_block * DIM), 
          ab + i * tk * DIM + k_block * DIM, 
          batch_size * DIM, 
          DIM
        );
      }
    }

    // 5. Coalesced MVIN for B (respecting MAX_BLOCK_LEN=4 limit)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      // Break down the transfer into blocks of at most 4 tiles (64 columns)
      // Since tj=4, we can do this in a single transfer (4*16=64 columns, which is at the limit)
      gemmini_extended_mvin(
        (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2)), 
        bb + k * tj * DIM, 
        tj * DIM, 
        DIM
      );
    }

    // 6. Configure store and MX-quantized mvout
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 9);

    // 7. Execute hardware fused weight-stationary compute loop
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // 8. Drain BF16 results from shared memory to DRAM
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
