Looking at the error message, the issue is that `cols=64` overflows the 6-bit hardware num_cols field. The maximum value for the 6-bit field is 63, so we need to reduce the batch size from 4 tiles (64 columns) to 3 tiles (48 columns) or fewer.

Let me fix the code by adjusting the MVIN operations to respect the MAX_BLOCK_LEN limitation:

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// OPTIMIZED MX matmul kernel (128x128x256, fp6:e3m2).
void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2, tj = MATMUL_N / DIM / 2, tk = MATMUL_K / DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;
    
    // 1. Configure execution formats (FP6 inputs, BF16 output, LUT enabled)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 1, 1, 3, 1);
    
    // 2. Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
    
    // 3. Load quant LUTs
    gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
    gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);
    
    // 4. Batch MVIN for A: Load 3 tiles (48 columns) at once to stay within 6-bit limit
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k += 3) {
        int remaining_k = (tk - k < 3) ? (tk - k) : 3;
        gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i*DIM*MATMUL_K + k*DIM), ab + (i*tk + k)*DIM, remaining_k * DIM, DIM);
      }
    }
    
    // 5. Batch MVIN for B: Load 3 tiles (48 columns) at once to stay within 6-bit limit
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      gemmini_extended_mvin2((void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2)), bb + (k*tj)*DIM, 3 * DIM, DIM);
    }
    
    // 6. Configure requant writeback and set update granularity to tk (16)
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, tk);
    
    // 7. Compute weight-stationary tiled matmul over scratchpad
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    // 8. Drain SMEM and synchronize
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
```
