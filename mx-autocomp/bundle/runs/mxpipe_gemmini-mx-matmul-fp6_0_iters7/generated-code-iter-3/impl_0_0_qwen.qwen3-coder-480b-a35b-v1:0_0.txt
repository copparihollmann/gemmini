#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// OPTIMIZED MX matmul kernel (128x128x256, fp6:e3m2) with double-buffered output.
void solution(void) {
  {
    // Partitions: 2 blocks of 64x128 along the M dimension
    int ti = 2;  // 64 rows / DIM (per block)
    int tj = 8;  // 128 cols / DIM 
    int tk = 16; // 256 reduction / DIM

    uint32_t ab_0 = 0;
    uint32_t ab_1 = ti * tk * DIM; // 2 * 16 * 16 = 512
    uint32_t bb   = BANK_NUM * BANK_ROWS - tk * tj * DIM; // 8192 - 16 * 8 * 16 = 6144
    
    int SDST_0 = 128;
    int SDST_1 = 256;
    
    // 1. One-time Setup and Scale/LUT Configuration
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 1, 1, 3, 1);
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
    gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
    gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);
    
    // 2. Load the ENTIRE A matrix (128x256) into double-buffered scratchpad slots (ab_0 and ab_1)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < 8; i++) {  // 128/16 = 8 tiles in I dimension
      uint32_t target_sp_addr = (i < 4) ? (ab_0 + (i * tk) * DIM) : (ab_1 + ((i - 4) * tk) * DIM);
      for (int k = 0; k < tk; k += 2) {
        gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM), target_sp_addr + k * DIM, 2 * DIM, DIM);
      }
    }
    
    // 3. Load the ENTIRE B matrix (256x128) into the fixed high-address region
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j += 2) {
        gemmini_extended_mvin((void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM), bb + (k * tj + j) * DIM, 2 * DIM, DIM);
      }
    }
    
    // Configure output formats and scale factor limits
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, 8, tj, tk, 0, 0, 9); // Full matrix dimensions for scale bounds
    
    // 4. Compute Block 0 -> writes output to SDST_0
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab_0, BANK_NUM * BANK_ROWS, 0, SDST_0, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    // 5. Asynchronously stream Block 0 output to DRAM
    gemmini_mx_read_smem(&C_hw[0][0], SDST_0 * 16, ti * tj * DIM * DIM);
    
    // 6. Compute Block 1 -> writes output to SDST_1 (Overlaps with DRAM write of Block 0)
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab_1, BANK_NUM * BANK_ROWS, 0, SDST_1, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    // 7. Stream Block 1 output to DRAM
    gemmini_mx_read_smem(&C_hw[ti * DIM][0], SDST_1 * 16, ti * tj * DIM * DIM);
    
    // 8. Final Synchronization
    gemmini_fence();
  }
}
