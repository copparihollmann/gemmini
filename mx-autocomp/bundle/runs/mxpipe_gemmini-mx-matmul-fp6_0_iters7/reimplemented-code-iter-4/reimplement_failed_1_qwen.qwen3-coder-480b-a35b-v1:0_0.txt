#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Optimized MX matmul kernel (128x128x256, fp6:e3m2).
void solution(void) {
  {
    const int ti = MATMUL_M / DIM / 2;  // 128/16/2 = 4
    const int tj = MATMUL_N / DIM / 2;  // 128/16/2 = 4  
    const int tk = MATMUL_K / DIM;      // 256/16 = 16
    const uint32_t ab = 0;  // A base address
    const uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * 2 * DIM;  // B base address - corrected calculation
    const int SDST = 0;  // Use beginning of shared memory

    // 1. Configure execution for FP6 weight-stationary matmul
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 1, 1, 3, 1);

    // 2. Load scale factors for A and B
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // 3. Load format-invariant quantization LUTs
    gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
    gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);

    // 4. Batch MVIN for A: Load 2x2 tile blocks at once (32x32 elements)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    const elem_t *a_ptr_i = (const elem_t*)A_in_hw;
    uint32_t ab_i = ab;
    for (int i = 0; i < ti; i++) {
      const elem_t *a_ptr_k = a_ptr_i;
      uint32_t ab_k = ab_i;
      for (int k = 0; k < tk; k++) {
        // Load a 2x1 tile block (32x16 elements) - within MAX_BLOCK_LEN=4 constraint
        gemmini_extended_mvin((void*)a_ptr_k, ab_k, 2 * DIM, DIM);
        a_ptr_k += DIM * MATMUL_K;
        ab_k += 2 * DIM;  // Move by 2 tiles horizontally
      }
      a_ptr_i += 2 * DIM * MATMUL_K;  // Move by 2 tiles vertically
      ab_i += tk * 2 * DIM;  // Update base address for next row of tiles
    }

    // 5. Batch MVIN for B: Load 2x2 tile blocks
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    const elem_t *b_ptr_k = (const elem_t*)B_in;
    uint32_t bb_k = bb;
    for (int k = 0; k < tk; k++) {
      const elem_t *b_ptr_j = b_ptr_k;
      uint32_t bb_j = bb_k;
      for (int j = 0; j < tj; j++) {
        // Load a 2x2 tile block (32x32 elements)
        gemmini_extended_mvin2((void*)b_ptr_j, bb_j, 2 * DIM, 2 * DIM);
        b_ptr_j += 2 * DIM;  // Move horizontally
        bb_j += 2 * DIM;     // Update scratchpad address
      }
      b_ptr_k += 2 * DIM * MATMUL_N;  // Move to next K slice
      bb_k += tj * 2 * DIM;           // Update K-level scratchpad address
    }

    // 6. Configure output store + MX requantizer
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti*2, tj*2, tk, 0, 0, 9);  // Correct tile bounds

    // 7. Compute weight-stationary loop in hardware
    gemmini_loop_ws_spad(ti*2, tj*2, tk, 0, 0, 0, ab, bb, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // 8. Read results and synchronize
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
