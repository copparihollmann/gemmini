#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp4_64x64.h"

#define DIM 16
#define BANK_NUM 4
#define BANK_ROWS 2048

void solution(void) {
  // Problem size: 128x128x256, tiling into 2x2 blocks of 64x64x256
  const int BLOCK_M = 64;
  const int BLOCK_N = 64;
  const int BLOCK_K = 256;
  
  const int tiles_I = BLOCK_M / DIM;  // 4
  const int tiles_J = BLOCK_N / DIM;  // 4
  const int tiles_K = BLOCK_K / DIM;  // 16
  
  const int num_blocks_M = MATMUL_M / BLOCK_M;  // 2
  const int num_blocks_N = MATMUL_N / BLOCK_N;  // 2
  
  // Scratchpad layout
  const uint32_t ab = 0;  // A base address
  const uint32_t bb = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;  // B base address at top
  const int SDST = 128;   // Shared memory destination
  
  static uint32_t scale_factors[512] __attribute__((aligned(32))) = {0};

  // 1. One-time global configurations
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 
                              1, 1, 0, 0, false, 1, 1, 3, 1);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
  gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
  gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);
  gemmini_config_st(MATMUL_N * sizeof(out_t) / 4);  // BF16 output: 4 vals per uint64

  // 2. Software tiling loops over 64x64x256 blocks
  for (int block_i = 0; block_i < num_blocks_M; block_i++) {
    for (int block_j = 0; block_j < num_blocks_N; block_j++) {
      
      // 3. Fully unrolled MVIN for Matrix A (all 4×16=64 tiles)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      
      // Unroll i dimension (4 iterations)
      // i=0
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 0 * DIM) * MATMUL_K + 0 * DIM), 
                            ab + (0 * tiles_K + 0) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 0 * DIM) * MATMUL_K + 4 * DIM), 
                            ab + (0 * tiles_K + 4) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 0 * DIM) * MATMUL_K + 8 * DIM), 
                            ab + (0 * tiles_K + 8) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 0 * DIM) * MATMUL_K + 12 * DIM), 
                            ab + (0 * tiles_K + 12) * DIM, 4 * DIM, DIM);
      
      // i=1
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 1 * DIM) * MATMUL_K + 0 * DIM), 
                            ab + (1 * tiles_K + 0) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 1 * DIM) * MATMUL_K + 4 * DIM), 
                            ab + (1 * tiles_K + 4) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 1 * DIM) * MATMUL_K + 8 * DIM), 
                            ab + (1 * tiles_K + 8) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 1 * DIM) * MATMUL_K + 12 * DIM), 
                            ab + (1 * tiles_K + 12) * DIM, 4 * DIM, DIM);
      
      // i=2
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 2 * DIM) * MATMUL_K + 0 * DIM), 
                            ab + (2 * tiles_K + 0) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 2 * DIM) * MATMUL_K + 4 * DIM), 
                            ab + (2 * tiles_K + 4) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 2 * DIM) * MATMUL_K + 8 * DIM), 
                            ab + (2 * tiles_K + 8) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 2 * DIM) * MATMUL_K + 12 * DIM), 
                            ab + (2 * tiles_K + 12) * DIM, 4 * DIM, DIM);
      
      // i=3
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 3 * DIM) * MATMUL_K + 0 * DIM), 
                            ab + (3 * tiles_K + 0) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 3 * DIM) * MATMUL_K + 4 * DIM), 
                            ab + (3 * tiles_K + 4) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 3 * DIM) * MATMUL_K + 8 * DIM), 
                            ab + (3 * tiles_K + 8) * DIM, 4 * DIM, DIM);
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (block_i * BLOCK_M + 3 * DIM) * MATMUL_K + 12 * DIM), 
                            ab + (3 * tiles_K + 12) * DIM, 4 * DIM, DIM);

      // 4. Fully unrolled MVIN for Matrix B (all 16×4=64 tiles)
      gemmini_config_ld((MATMUL_N / 2) * sizeof(elem_t));
      
      // Unroll k dimension (16 iterations)
      for (int k = 0; k < tiles_K; k++) {
        elem_t *dram_ptr = ((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + (block_j * BLOCK_N) / 2;
        uint32_t sp_addr = bb + (k * tiles_J) * DIM;
        gemmini_extended_mvin((void*)dram_ptr, sp_addr, tiles_J * DIM, DIM);
      }

      // 5. Block-specific requantizer configuration & compute
      uint32_t *block_scale_factors = scale_factors + (block_i * BLOCK_M) * (MATMUL_N / 32) + (block_j * BLOCK_N) / 32;
      gemmini_mxquant_config_mvout((uint64_t)block_scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 9);

      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, 
                          ab, BANK_NUM * BANK_ROWS, 0, SDST,
                          false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 6. Direct strided SMEM -> DRAM draining
      for (int r = 0; r < BLOCK_M; r++) {
        gemmini_mx_read_smem(&C_hw[block_i * BLOCK_M + r][(block_j * BLOCK_N) / 4], 
                            SDST * 16 + r * BLOCK_N, 
                            BLOCK_N);
      }
    }
  }

  // 7. Amortized CPU barrier at the very end
  gemmini_fence();
}
