Looking at the original code, I need to identify and fix the issues:

1. The MVIN A stride is wrong: it uses `MATMUL_M * sizeof(elem_t)` but should use `MATMUL_K * sizeof(elem_t)` (A is M×K, row stride = K elements)
2. The MVIN B stride is wrong: it uses `MATMUL_M * sizeof(elem_t)` but B is N×K, row stride = K elements  
3. The DRAM pointer arithmetic for A is wrong: `A_in[i*DIM*MATMUL_M + k*DIM]` should be `A_in[i*DIM*MATMUL_K + k*DIM]` (row-major A: row i starts at i*DIM*MATMUL_K, tile k starts at k*DIM)
4. The DRAM pointer arithmetic for B is wrong similarly: B is N×K, so `B_in[j*DIM*MATMUL_K + k*DIM]`
5. `lut_update_granularity` is set to `tiles_J` which is reasonable, keeping it
6. The `SPAD_DEST` should be a word offset into `mx_smem`; using 128 is valid as long as it doesn't overlap with scratchpad data

```c
void solution(void) {

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

#ifdef SPIKE_SIM
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
#else
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_A, (uint8_t *) &A_scales_row, MATMUL_M, MATMUL_K);
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_B, (uint8_t *) &B_scales_col, MATMUL_N, MATMUL_K);
#endif

  // ---- MVIN A: tile (i,k) -> a_base + (i*tiles_K + k)*DIM ----
  // A is row-major M x K; row stride = MATMUL_K * sizeof(elem_t)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

  for (int i = 0; i < tiles_I; i++) {
    int k = 0;
    while (k < tiles_K) {
      int batch = tiles_K - k;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      // A[i*DIM, k*DIM]: row-major, row stride = MATMUL_K, so offset = i*DIM*MATMUL_K + k*DIM
      elem_t *dram_ptr = ((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM;
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *) dram_ptr, sp_addr, batch * DIM, DIM);
      k += batch;
    }
  }

  // ---- MVIN B: tile (j,k) -> b_base + (j*tiles_K + k)*DIM ----
  // B is row-major N x K; row stride = MATMUL_K * sizeof(elem_t)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  for (int j = 0; j < tiles_J; j++) {
    int k = 0;
    while (k < tiles_K) {
      int batch = tiles_K - k;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      // B[j*DIM, k*DIM]: row-major, row stride = MATMUL_K, so offset = j*DIM*MATMUL_K + k*DIM
      elem_t *dram_ptr = ((elem_t*)B_in) + j * DIM * MATMUL_K + k * DIM;
      uint32_t sp_addr = b_base + (j * tiles_K + k) * DIM;
      gemmini_extended_mvin2((void *) dram_ptr, sp_addr, batch * DIM, DIM);
      k += batch;
    }
  }

  int SPAD_DEST = 128;

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, tiles_J);

  // ---- Compute ----
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      0, 0, 0,
      a_base,
      BANK_NUM * BANK_ROWS,
      0,
      SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION,
      0, 0,
      false,
      0x38);

  gemmini_fence();
#ifndef SPIKE_SIM
  gemmini_poll_until_ready();
#endif
  gemmini_fence();

#ifdef SPIKE_SIM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
#else
  uint64_t* smem_start_addr = ((uint64_t*)SMEM) + SPAD_DEST * 2;
  printf("Address: %p \n", smem_start_addr);
  for (int i = 0; i < MATMUL_M; i ++) {
    for (int j = 0; j < OUT_COLS; j++) {
        C_hw[i][j] = *(smem_start_addr + (i*OUT_COLS + j));
    }
  }
#endif

  gemmini_fence();
}
```
