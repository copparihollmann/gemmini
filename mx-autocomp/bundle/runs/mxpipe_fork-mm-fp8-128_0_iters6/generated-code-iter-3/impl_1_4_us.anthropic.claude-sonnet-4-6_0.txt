void solution(void) {

  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

#ifdef SPIKE_SIM
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
#else
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_A, (uint8_t *) &A_scales_row, MATMUL_M, MATMUL_K);
  load_scale_factors((volatile uint64_t *) GEMMINI_SF_MEM_B, (uint8_t *) &B_scales_col, MATMUL_N, MATMUL_K);
#endif

  // Configure load port 0 for A: row stride = MATMUL_K * sizeof(elem_t) = MATMUL_K bytes
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

  // MVIN A tiles: A is M x K row-major; tile (i,k) -> scratchpad row a_base + (i*tiles_K + k)*DIM
  for (int i = 0; i < tiles_I; i++) {
    int k = 0;
    while (k < tiles_K) {
      int batch = tiles_K - k;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      elem_t *dram_ptr = ((elem_t *)A_in) + (size_t)i * DIM * MATMUL_K + (size_t)k * DIM;
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)dram_ptr, sp_addr, batch * DIM, DIM);
      k += batch;
    }
  }

  // Configure load port 1 for B: row stride = MATMUL_K * sizeof(elem_t) = MATMUL_K bytes
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // MVIN B tiles: B is N x K row-major; tile (j,k) -> scratchpad row b_base + (j*tiles_K + k)*DIM
  for (int j = 0; j < tiles_J; j++) {
    int k = 0;
    while (k < tiles_K) {
      int batch = tiles_K - k;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      elem_t *dram_ptr = ((elem_t *)B_in) + (size_t)j * DIM * MATMUL_K + (size_t)k * DIM;
      uint32_t sp_addr = b_base + (j * tiles_K + k) * DIM;
      gemmini_extended_mvin2((void *)dram_ptr, sp_addr, batch * DIM, DIM);
      k += batch;
    }
  }

  int SPAD_DEST = 128;

  // Configure output stride and requantizer
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      /*scale_act_sel*/0, /*scale_w_sel*/0,
      /*lut_update_granularity*/tiles_J);

  // funct 24 (LOOP_WS_CONFIG_SPAD_AB) is issued inside gemmini_loop_ws_spad; marker is one-shot
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      /*pad_I*/0, /*pad_J*/0, /*pad_K*/0,
      a_base,
      BANK_NUM * BANK_ROWS,
      /*D*/0,
      SPAD_DEST,
      /*A_transpose*/false, /*B_transpose*/false,
      /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
      NO_ACTIVATION,
      /*a_spad_id*/0, /*b_spad_id*/0,
      /*is_resadd*/false,
      0x38);

  gemmini_fence();
#ifndef SPIKE_SIM
  gemmini_poll_until_ready();
#endif
  gemmini_fence();

#ifdef SPIKE_SIM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
#else
  uint64_t *smem_start_addr = ((uint64_t *)SMEM) + SPAD_DEST * 2;
  printf("Address: %p \n", smem_start_addr);
  for (int i = 0; i < MATMUL_M; i++) {
    for (int j = 0; j < OUT_COLS; j++) {
      C_hw[i][j] = *(smem_start_addr + (i * OUT_COLS + j));
    }
  }
#endif

  gemmini_fence();
}
