  // Port 0 (mvin): A matrix — row stride is MATMUL_K elements
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Port 1 (mvin2): B matrix — row stride is MATMUL_N elements
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Step 4a: MVIN A tiles into scratchpad using port 0, batched along K
  // Layout: a_base + (i*tiles_K + k)*DIM — k-consecutive tiles are contiguous in spad
  // DRAM: row i, col k*DIM — consecutive k-tiles are contiguous in DRAM (stride = MATMUL_K)
  // Batch up to MAX_BLOCK_LEN=4 tiles per instruction (cols = blk*DIM, rows = DIM)
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {

      gemmini_extended_mvin(
          (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
          a_base + (i * tiles_K + k) * DIM,
          blk * DIM, DIM);
    }
  }

  // Step 4b: MVIN B tiles into scratchpad using port 1 (mvin2), batched along K
  // Layout: b_base + (j*tiles_K + k)*DIM — k-consecutive tiles are contiguous in spad
  // DRAM: row k*DIM, col j*DIM — consecutive k-tiles are contiguous (stride = MATMUL_N)
  // Batch up to MAX_BLOCK_LEN=4 tiles per instruction (cols = blk*DIM, rows = DIM)
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {
      int blk = (k + MAX_BLOCK_LEN <= tiles_K) ? MAX_BLOCK_LEN : (tiles_K - k);
      gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM),
          b_base + (j * tiles_K + k) * DIM,
          blk * DIM, DIM);
    }
  }

  // Step 5: Configure output stride and requantizer
  // i_bound/j_bound/k_bound must exactly match tiles_I/tiles_J/tiles_K used below
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // Step 6: Fused hardware weight-stationary matmul over scratchpad
  // B end address = BANK_NUM*BANK_ROWS (top of scratchpad, where B is placed)
  // mx_loop_spad_marker is one-shot: LOOP_WS_CONFIG_SPAD_AB (funct 24) is
  // re-issued automatically inside gemmini_loop_ws_spad before LOOP_WS (funct 8)
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain BF16 results from mx_smem to DRAM, then fence
  // smem_off_bf16 = SPAD_DEST*16 (16-bit word units), num_bf16 = M*N elements
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
