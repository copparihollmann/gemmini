void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Load A tiles via mvin port 1: stride = MATMUL_K cols
  // A[i][k] -> spad: a_base + (i*tiles_K + k)*DIM
  // Batch up to MAX_BLOCK_LEN=4 consecutive k tiles per i row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  for (int i = 0; i < tiles_I; i++) {
    int k = 0;
    while (k < tiles_K) {
      int batch = tiles_K - k;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      gemmini_extended_mvin(
        (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
        a_base + (i * tiles_K + k) * DIM,
        batch * DIM, DIM);
      k += batch;
    }
  }

  // Load B tiles via mvin port 2: stride = MATMUL_N cols
  // B[k][j] -> spad: b_base + (k*tiles_J + j)*DIM
  // Batch up to MAX_BLOCK_LEN=4 consecutive j tiles per k row
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
  for (int k = 0; k < tiles_K; k++) {
    int j = 0;
    while (j < tiles_J) {
      int batch = tiles_J - j;
      if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
      gemmini_extended_mvin2(
        (void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM),
        b_base + (k * tiles_J + j) * DIM,
        batch * DIM, DIM);
      j += batch;
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
