void solution(void) {
  // Step 1: Configure execution format (fp8 e4m3 inputs, bf16 output)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  // Step 2: Load scale factors (sel=0 for A activations, sel=1 for B weights)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 4: MVIN A tiles into scratchpad
  // A layout: a_base + (i*tiles_K + k)*DIM
  // A stride in DRAM: MATMUL_K elements per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {
      int blk = tiles_K - k;
      if (blk > MAX_BLOCK_LEN) blk = MAX_BLOCK_LEN;
      // Each block of 'blk' tiles shares the same i-row in A
      // but they are not contiguous in DRAM (stride MATMUL_K apart),
      // so we must issue one mvin per tile to handle non-contiguous DRAM layout
      for (int kb = 0; kb < blk; kb++) {
        int kk = k + kb;
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + (size_t)(i * DIM) * MATMUL_K + kk * DIM),
            a_base + (i * tiles_K + kk) * DIM,
            DIM, DIM);
      }
    }
  }

  // MVIN B tiles into scratchpad
  // B layout: b_base + (k*tiles_J + j)*DIM
  // B stride in DRAM: MATMUL_N elements per row
  // b_base = BANK_NUM*BANK_ROWS - tiles_K*tiles_J*DIM
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {
      int blk = tiles_J - j;
      if (blk > MAX_BLOCK_LEN) blk = MAX_BLOCK_LEN;
      for (int jb = 0; jb < blk; jb++) {
        int jj = j + jb;
        gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (size_t)(k * DIM) * MATMUL_N + jj * DIM),
            b_base + (k * tiles_J + jj) * DIM,
            DIM, DIM);
      }
    }
  }

  // Step 5: Configure output stride and requantizer
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      /*scale_act_sel*/0, /*scale_w_sel*/0,
      /*lut_update_granularity*/1);

  // Step 6: Execute fused weight-stationary tiled matmul
  // B end address = BANK_NUM*BANK_ROWS (top of scratchpad)
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K,
      0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS,
      0, SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain BF16 results from mx_smem to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
