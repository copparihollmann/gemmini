#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"

#define DIM 16
#define BANK_NUM 4
#define BANK_ROWS 2048
#define USE_LUT 0
#define QUANT_LUT_UPDATE_GRANULARITY 1

typedef uint8_t elem_t;
typedef uint8_t out_t;

static uint32_t scale_factors[MATMUL_M * MATMUL_N / 32] __attribute__((aligned(32)));

void solution(void) {
    int tiles_I = MATMUL_M / DIM;
    int tiles_J = MATMUL_N / DIM;
    int tiles_K = MATMUL_K / DIM;

    uint32_t a_base  = 0;
    uint32_t b_base  = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    int      SPAD_DEST = 128;

    // Step 1: Configure execution
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 0, 0, 3, 0);

    // Step 2: Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // Step 3: Configure a single load port (port 0) for both A and B
    // (both have the same DRAM row stride of MATMUL_K bytes)
    // Eliminates the redundant port-1 config instruction
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

    // Step 4: MVIN all A and B tiles via port 0 only
    // Interleave A and B per k-slice to maximize locality
    for (int k = 0; k < tiles_K; k++) {
        for (int i = 0; i < tiles_I; i++)
            gemmini_extended_mvin(
                (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
                a_base + (i * tiles_K + k) * DIM,
                DIM, DIM);
        for (int j = 0; j < tiles_J; j++)
            gemmini_extended_mvin(
                (void*)(((elem_t*)B_in) + j * DIM * MATMUL_K + k * DIM),
                b_base + (j * tiles_K + k) * DIM,
                DIM, DIM);
    }

    // Step 5: Configure output and requantizer
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        tiles_I, tiles_J, tiles_K, 0, 0, 1);

    // Step 6: Fused weight-stationary matmul
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
        a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: Read BF16 results from shared memory
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
