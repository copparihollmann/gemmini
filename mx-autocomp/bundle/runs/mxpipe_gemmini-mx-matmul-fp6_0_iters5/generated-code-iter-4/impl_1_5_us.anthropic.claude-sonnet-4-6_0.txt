void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2;   // 4
    int tj = MATMUL_N / DIM / 2;   // 4
    int tk = MATMUL_K / DIM;       // 16
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;

    // Step 1: Configure execution with FP6 E3M2 formats
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 1, 1, 3, 1);

    // Step 2: Load scale factors (A sel=0, B sel=1)
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // Step 3: Load LUTs (B sel=0, A sel=1)
    gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
    gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);

    // Step 4a: MVIN A tiles into scratchpad using block mvin (up to MAX_BLOCK_LEN=4)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      int k = 0;
      while (k < tk) {
        int blk = tk - k;
        if (blk > MAX_BLOCK_LEN) blk = MAX_BLOCK_LEN;
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM),
            ab + (i * tk + k) * DIM,
            blk * DIM, DIM);
        k += blk;
      }
    }

    // Step 4b: MVIN B tiles into scratchpad using block mvin (up to MAX_BLOCK_LEN=4)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      int j = 0;
      while (j < tj) {
        int blk = tj - j;
        if (blk > MAX_BLOCK_LEN) blk = MAX_BLOCK_LEN;
        gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM),
            bb + (k * tj + j) * DIM,
            blk * DIM, DIM);
        j += blk;
      }
    }

    // Step 5: Configure requantized mvout
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk,
        0, 0, 9);

    // Step 6: Compute tiled matmul over scratchpad
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM * BANK_ROWS, 0, SDST,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: Drain results from mx_smem to DRAM
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
