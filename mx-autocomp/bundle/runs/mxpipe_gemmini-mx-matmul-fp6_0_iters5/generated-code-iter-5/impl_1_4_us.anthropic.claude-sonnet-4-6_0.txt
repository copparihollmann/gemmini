void solution(void) {
  const int tiles_I = MATMUL_M / DIM;   // 8
  const int tiles_J = MATMUL_N / DIM;   // 8
  const int tiles_K = MATMUL_K / DIM;   // 16

  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  // a needs tiles_I * tiles_K * DIM = 8*16*16 = 2048 rows
  // b needs tiles_J * tiles_K * DIM = 8*16*16 = 2048 rows
  // b_base = 8192 - 2048 = 6144, a uses [0..2047], no overlap
  const int SDST = 0;

  // Step 1: Configure execution - fp6 e3m2, uselut=1
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
      /*set_only_strides*/false,
      /*act_mx_fmt*/1, /*wgt_mx_fmt*/1, /*out_mx_fmt*/3, /*uselut*/1);

  // Step 2: Load scale factors (sel=0 for A, sel=1 for B)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 3: Load LUTs (sel=1 for A LUT, sel=0 for B LUT)
  gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);
  gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);

  // Step 4: MVIN tiles of A into scratchpad
  // A layout: a_base + (i*tiles_K + k)*DIM
  // A DRAM stride: MATMUL_K * sizeof(elem_t) bytes
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {
      int blk = (k + MAX_BLOCK_LEN <= tiles_K) ? MAX_BLOCK_LEN : (tiles_K - k);
      for (int b = 0; b < blk; b++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + (k + b) * DIM),
            a_base + (i * tiles_K + k + b) * DIM,
            DIM, DIM);
      }
    }
  }

  // MVIN tiles of B into scratchpad
  // B is stored as half-width (FP6 packed), stride: MATMUL_N/2 * sizeof(elem_t)
  // B layout: b_base + (k*tiles_J + j)*DIM
  gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j++) {
      gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM),
          b_base + (k * tiles_J + j) * DIM,
          DIM, DIM);
    }
  }

  // Step 5: Configure requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      /*scale_act_sel*/0, /*scale_w_sel*/0,
      /*lut_update_granularity*/16);

  // Step 6: Compute - fused hardware matmul
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SDST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain results from mx_smem to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
