void solution(void) {
  int tiles_I = MATMUL_M / DIM;  // 8
  int tiles_J = MATMUL_N / DIM;  // 8
  int tiles_K = MATMUL_K / DIM;  // 16

  // B fits at top of spad: need tiles_K * tiles_J * DIM = 16*8*16 = 2048 rows
  // A needs tiles_I * tiles_K * DIM = 8*16*16 = 2048 rows
  // Total = 4096 rows <= 8192 rows. OK.
  uint32_t ab = 0;
  uint32_t bb = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 8192 - 2048 = 6144
  int SDST = 0;

  // Step 1: Configure execution (fp6 E3M2: act_mx_fmt=1, wgt_mx_fmt=1, out_mx_fmt=3 BF16, uselut=1)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 1, 1, 3, 1);

  // Step 2: Load scales for A and B
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 3: Load LUTs (fp6 requires LUT, sel 0=B, sel 1=A)
  gemmini_mx_load_lut((uint64_t)B_lut_p, 1, 0);
  gemmini_mx_load_lut((uint64_t)A_lut_p, 1, 1);

  // Step 4: MVIN all A tiles into scratchpad
  // A is row-major: A[i*DIM rows][K cols], stride = MATMUL_K bytes
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++)
    for (int k = 0; k < tiles_K; k++)
      gemmini_extended_mvin(
          (void*)(((elem_t*)A_in_hw) + i*DIM*MATMUL_K + k*DIM),
          ab + (i*tiles_K + k)*DIM,
          DIM, DIM);

  // MVIN all B tiles into scratchpad
  // B is stored as [K/2][N] in elem_t (fp6 packed), stride = MATMUL_N/2 bytes
  gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
  for (int k = 0; k < tiles_K; k++)
    for (int j = 0; j < tiles_J; j++)
      gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + j*DIM),
          bb + (k*tiles_J + j)*DIM,
          DIM, DIM);

  // Step 5: Configure requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      0, 0, 9);

  // Step 6: Compute full tiled matmul
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      ab, BANK_NUM * BANK_ROWS, 0, SDST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain results from mx_smem to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
