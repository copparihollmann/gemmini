Looking at the error message, the issue is that `gemmini_extended_mvin` is being called with `rows=1024`, but the hardware only supports a maximum of 63 rows due to the 6-bit field limitation. The problem occurs in this line:

```c
gemmini_extended_mvin(dram_ptr, sp_addr, DIM, tiles_K * DIM);
```

Where `tiles_K * DIM = (3072/16) * 16 = 3072` rows, which exceeds the 63-row limit.

The fix is to break down the large B matrix loading into smaller chunks that respect the MAX_BLOCK_LEN=4 constraint for scratchpad operations.

```c
// Optimized tiled MX matmul (1x1024x3072).
void solution(void) {
  // Hoist global configurations outside the execution loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Partial last M/N block: compute only the real rows/cols, pad the rest.
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      // Stage only the tiny per-32-group scales
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // Load scale factors
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // MVIN A: Strided straight from DRAM row-by-row
      gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // A rows: stride K
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // MVIN B: Break into chunks respecting MAX_BLOCK_LEN=4 constraint
      gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // B rows: stride N
      for (int j = 0; j < tiles; j++) {
        for (int k_chunk = 0; k_chunk < tiles_K; k_chunk += MAX_BLOCK_LEN) {
          int chunk_size = (tiles_K - k_chunk < MAX_BLOCK_LEN) ? (tiles_K - k_chunk) : MAX_BLOCK_LEN;
          void *dram_ptr = (void*)(((elem_t*)B_in) + (j0 * BLK + j * DIM) * MATMUL_N + k_chunk * DIM);
          uint32_t sp_addr = b_base + (j * tiles_K + k_chunk) * DIM;
          gemmini_extended_mvin(dram_ptr, sp_addr, DIM, chunk_size * DIM);
        }
      }

      // Configure store + MX requantizer for the mvout
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);

      // Execute weight-stationary tiled loop
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Readout packed results row-by-row
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
  }
}
```
