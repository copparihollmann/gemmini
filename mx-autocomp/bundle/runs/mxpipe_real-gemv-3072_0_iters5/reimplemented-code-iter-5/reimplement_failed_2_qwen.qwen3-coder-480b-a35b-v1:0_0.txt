#include <stdint.h>
#include <string.h>

#define TILE 16
#define VALUES_PER_BYTE 2
#define USE_LUT 1
#define QUANT_LUT_UPDATE_GRANULARITY 1
#define DIM 16
#define ADDR_LEN 32
#define BF16_PER_WORD 4

#undef gemmini_fence
#define gemmini_fence() asm volatile("fence")

typedef uint8_t elem_t;
typedef uint64_t out_t; // Changed to match BF16 output (packed 4 per uint64)

// Assume these are defined elsewhere or passed in
extern elem_t A_in[][/*MATMUL_K*/];
extern elem_t B_in[][/*MATMUL_N*/];
extern uint8_t A_scales_row[][/*MATMUL_M or REAL_M padded*/];
extern uint8_t B_scales_col[][/*MATMUL_N or REAL_N padded*/];
extern out_t C_hw[][/*MATMUL_N or REAL_N padded, adjusted for BF16 packing*/];
extern void *scale_factors; // Placeholder, not directly used here but for interface consistency

// Hardware parameters assumed to be defined in gemmini_params.h or similar
#ifndef BANK_NUM
#define BANK_NUM 4
#endif
#ifndef BANK_ROWS
#define BANK_ROWS 2048
#endif
#ifndef ACC_ROWS
#define ACC_ROWS 256
#endif
#ifndef MAX_BLOCK_LEN
#define MAX_BLOCK_LEN 4
#endif
#ifndef MAX_BLOCK_LEN_ACC
#define MAX_BLOCK_LEN_ACC 1
#endif

// Constants for MX format fields
#define MX_FMT_FP8_E4M3 0
#define MX_FMT_BF16 3
#define NO_ACTIVATION 0
#define WEIGHT_STATIONARY 1
#define ACC_SCALE_IDENTITY 1.0f

// RoCC instruction macros (simplified versions)
#define XCUSTOM_ACC 2
#define CONFIG_EX 0
#define k_MX_LOAD_SCALES 27
#define k_MX_LOAD_LUT 29
#define k_MX_READ_SMEM 28
#define k_LOOP_WS_CONFIG_BOUNDS 10
#define k_LOOP_WS_CONFIG_SPAD_AB 24
#define k_LOOP_WS 8
#define k_MVIN 1
#define k_MVIN2 2
#define k_MVIN3 3
#define k_CONFIG 12

#define ROCC_INSTRUCTION_RS1_RS2(x, rs1, rs2, funct) \
    asm volatile(".word (0b0000000 << 25) | (%0 << 20) | (%1 << 15) | (0b0000001 << 12) | (0b0000000 << 7) | (0b1110011 << 0)" :: "r"(rs1), "r"(rs2))

#define gemmini_extended3_config_ex(dataflow, sys_act, sys_shift, sys_acc_scale, C_stride, A_stride, A_transpose, B_transpose, set_only_strides, act_mx_fmt, wgt_mx_fmt, out_mx_fmt, uselut) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, \
        ((uint64_t)(A_stride) << 16) | \
        ((uint64_t)(out_mx_fmt) << 14) | \
        ((uint64_t)(wgt_mx_fmt) << 12) | \
        ((uint64_t)(act_mx_fmt) << 10) | \
        ((uint64_t)(B_transpose) << 9) | \
        ((uint64_t)(A_transpose) << 8) | \
        ((uint64_t)(set_only_strides) << 7) | \
        ((uint64_t)(uselut) << 5) | \
        ((uint64_t)(sys_act) << 3) | \
        ((uint64_t)(dataflow) << 2) | \
        CONFIG_EX, \
        ((uint64_t)(C_stride) << 48) | (uint64_t)(uint32_t)(sys_shift), \
        k_CONFIG)

#define gemmini_mx_load_scales(dram_addr, len, sel) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, (uint64_t)(dram_addr), \
        ((uint64_t)(sel) << 32) | ((uint64_t)(len) & 0xFFFFFFFFu), \
        k_MX_LOAD_SCALES)

#define gemmini_config_ld(stride) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, stride, CONFIG_EX, k_CONFIG)

#define gemmini_extended_mvin(dram_addr, spad_addr, cols, rows) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, (uint64_t)(dram_addr), \
        ((uint64_t)(rows) << (ADDR_LEN + 16)) | ((uint64_t)(cols) << ADDR_LEN) | (spad_addr), \
        k_MVIN)

#define gemmini_config_st(stride) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, stride, CONFIG_EX, k_CONFIG)

#define gemmini_mxquant_config_mvout(dram_addr, i_bound, j_bound, k_bound, scale_act_sel, scale_w_sel, lut_update_granularity) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, \
        ((uint64_t)(scale_w_sel) << 61) | ((uint64_t)(scale_act_sel) << 60) | \
        ((uint64_t)(k_bound) << 51) | ((uint64_t)(j_bound) << 42) | ((uint64_t)(i_bound) << 33) | (uint64_t)(dram_addr), \
        (uint64_t)(lut_update_granularity) & 0xFFFF, k_CONFIG)

#define gemmini_loop_ws_spad(I, J, K, pad_I, pad_J, pad_K, A, B, D, C, A_transpose, B_transpose, full_C, low_D, ex_accumulate, act, a_spad_id, b_spad_id, is_resadd, skips) \
    { \
        ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, ((uint64_t)(pad_K) << 32) | ((uint64_t)(pad_J) << 16) | (uint64_t)(pad_I), \
            ((uint64_t)(K) << 32) | ((uint64_t)(J) << 16) | (uint64_t)(I), k_LOOP_WS_CONFIG_BOUNDS); \
        ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, A, B, k_LOOP_WS_CONFIG_SPAD_AB); \
        ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, \
            ((uint64_t)(a_spad_id) << 18) | ((uint64_t)(b_spad_id) << 16) | ((uint64_t)(act) << 8) | \
            ((low_D) << 2) | ((full_C) << 1) | (ex_accumulate), \
            ((uint64_t)(C) << 32) | 0x200U | (skips) | ((is_resadd) << 2) | ((B_transpose) << 1) | (A_transpose), \
            k_LOOP_WS); \
    }

#define gemmini_mx_read_smem(dram_addr, smem_off_bf16, num_bf16) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, (uint64_t)(dram_addr), \
        ((uint64_t)(num_bf16) << 32) | ((uint64_t)(smem_off_bf16) & 0xFFFFFFFFu), \
        k_MX_READ_SMEM)

#define gemmini_flush(skip) \
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, skip, 0, 0)

void solution(void) {
    // Use a large, efficient block size to maximize accumulator usage
    const int BLK = 64;
    const int BLOCKS_M = (REAL_M + BLK - 1) / BLK;
    const int BLOCKS_N = (REAL_N + BLK - 1) / BLK;

    gemmini_flush(0);

    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
        for (int j0 = 0; j0 < BLOCKS_N; j0++) {
            // Calculate actual dimensions for the current block, handling edge cases
            int mrows = REAL_M - i0 * BLK; 
            if (mrows > BLK) mrows = BLK; 
            if (mrows <= 0) continue;
            
            int ncols = REAL_N - j0 * BLK; 
            if (ncols > BLK) ncols = BLK; 
            if (ncols <= 0) continue;

            // Calculate tiling parameters for the current block
            int tiles_I = (mrows + DIM - 1) / DIM; // Number of vertical tiles in M
            int tiles_J = (ncols + DIM - 1) / DIM; // Number of horizontal tiles in N
            int tiles_K_total = MATMUL_K / DIM;    // Total number of tiles in K

            // Chunking K-dimension to fit within scratchpad memory
            const int chunk_K_tiles = 16; // Represents 256 elements along K

            // Padding for partial tiles at the edges of the block
            int pad_i = tiles_I * DIM - mrows;
            int pad_j = tiles_J * DIM - ncols;
            
            // Map the destination to a fixed location in the accumulator (row 0)
            const int spad_dest = 0; 

            // Configure execution and MX formats once per block
            gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false,
                                        MX_FMT_FP8_E4M3, MX_FMT_FP8_E4M3, MX_FMT_BF16, 0);

            // Loop over K-dimension in chunks
            for (int k_start_tile = 0; k_start_tile < tiles_K_total; k_start_tile += chunk_K_tiles) {
                // Determine the number of K tiles for this chunk
                int k_tiles = tiles_K_total - k_start_tile;
                if (k_tiles > chunk_K_tiles) k_tiles = chunk_K_tiles;

                // Determine which group of scale factors (every 2 K tiles share a scale) this chunk belongs to
                int g_start = k_start_tile / 2;
                int g_count = (k_tiles + 1) / 2; // Ceiling division to cover all relevant groups

                // Stage scale factors for the current K chunk into temporary aligned buffers
                static uint8_t chunk_A_s[32 * 64] __attribute__((aligned(32))); // Max 32 groups * 64 elements
                static uint8_t chunk_B_s[32 * 64] __attribute__((aligned(32))); // Max 32 groups * 64 elements
                
                // Clear buffers first
                memset(chunk_A_s, 0, sizeof(chunk_A_s));
                memset(chunk_B_s, 0, sizeof(chunk_B_s));
                
                for (int g = 0; g < g_count; g++) {
                    // Copy the relevant slice of A scales for the current M block and K group
                    int a_copy_len = (g * BLK + mrows > sizeof(chunk_A_s)) ? (sizeof(chunk_A_s) - g * BLK) : mrows;
                    if (a_copy_len > 0 && (i0 * BLK < /* appropriate bound */)) {
                        memcpy(&chunk_A_s[g * BLK], &A_scales_row[g_start + g][i0 * BLK], 
                               a_copy_len < mrows ? a_copy_len : mrows);
                    }
                    // Copy the relevant slice of B scales for the current N block and K group
                    int b_copy_len = (g * BLK + ncols > sizeof(chunk_B_s)) ? (sizeof(chunk_B_s) - g * BLK) : ncols;
                    if (b_copy_len > 0 && (j0 * BLK < /* appropriate bound */)) {
                        memcpy(&chunk_B_s[g * BLK], &B_scales_col[g_start + g][j0 * BLK], 
                               b_copy_len < ncols ? b_copy_len : ncols);
                    }
                }
                // Load the staged A scale factors into the hardware's scale memory
                gemmini_mx_load_scales((uint64_t)chunk_A_s, g_count * BLK, 0); // sel=0 for A
                // Load the staged B scale factors into the hardware's scale memory
                gemmini_mx_load_scales((uint64_t)chunk_B_s, g_count * BLK, 1); // sel=1 for B

                // MVIN A chunk into the scratchpad
                uint32_t a_base = 0; // Base address for A tiles in scratchpad
                gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Set stride for A matrix rows
                // Loop over M tiles and K tiles for the A matrix
                for (int i = 0; i < tiles_I; i++) {
                    for (int k = 0; k < k_tiles; k++) {
                        // Calculate source DRAM address and destination scratchpad address for A tile
                        elem_t* dram_src = &A_in[i0 * BLK + i * DIM][(k_start_tile + k) * DIM];
                        uint32_t spad_dst = a_base + (i * k_tiles + k) * DIM;
                        // Perform the memory move-in operation
                        gemmini_extended_mvin((void*)dram_src, spad_dst, DIM, DIM);
                    }
                }

                // MVIN B chunk into the scratchpad (allocated at the top, growing downward)
                uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * k_tiles * DIM; // Base address for B tiles
                gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // Set stride for B matrix rows
                // Loop over K tiles and N tiles for the B matrix
                for (int j = 0; j < tiles_J; j++) {
                    for (int k = 0; k < k_tiles; k++) {
                        // Calculate source DRAM address and destination scratchpad address for B tile
                        elem_t* dram_src = &B_in[(k_start_tile + k) * DIM][j0 * BLK + j * DIM];
                        uint32_t spad_dst = b_base + (k * tiles_J + j) * DIM;
                        // Perform the memory move-in operation
                        gemmini_extended_mvin((void*)dram_src, spad_dst, DIM, DIM);
                    }
                }

                // Configure output stride and requantizer (mvout) parameters before computation
                gemmini_config_st(ncols * sizeof(uint16_t)); // Set output stride based on block width (BF16)
                // Configure requantization parameters including bounds and scale selectors
                gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, k_tiles, 0, 0, 1);

                // Determine control flags for the hardware loop
                // Accumulate only after the first chunk (the first chunk initializes the accumulator)
                int ex_accumulate = (k_start_tile == 0) ? 0 : 1;
                // Skip storing to SMEM for all chunks except the last one to keep results in the accumulator
                int skips = (k_start_tile + k_tiles >= tiles_K_total) ? 0x38 : 0xB8; // 0xB8 sets skip bit 7

                // Issue the fused weight-stationary systolic array loop over the scratchpad data
                // Uses ex_accumulate for K-chunk chaining and skips to control intermediate stores
                gemmini_loop_ws_spad(
                    tiles_I, tiles_J, k_tiles,       // Actual tile counts for this iteration
                    pad_i, pad_j, 0,                 // Padding for edge tiles
                    a_base,                         // Scratchpad base addr for A (calculated above)
                    BANK_NUM * BANK_ROWS,           // End addr for B (required argument for spad variant)
                    0,                              // Addr for D/bias (0 = none)
                    spad_dest,                      // Accumulator/spmem destination address offset
                    0, 0,                           // Transposes (false = no transpose)
                    0, 0, ex_accumulate,            // Full C, low D, ex_accumulate flags
                    NO_ACTIVATION,                  // Activation function (none)
                    0, 0,                           // Spad reuse IDs (0 = no reuse)
                    0,                              // Residual add (false = not used)
                    skips                           // Skip mask to control loop behavior
                );
                
            } // End loop over K chunks

            // After all K chunks are processed, read the final accumulated result from SMEM
            // Calculate the number of BF16 values to read (ncols)
            int bf16_count = ncols; // Each BF16 is 16 bits, 4 per 64-bit word
            
            // Calculate destination DRAM address and source SMEM offset
            uint64_t dram_dst = (uint64_t)&C_hw[i0 * BLK][j0 * BLK / BF16_PER_WORD];
            uint32_t smem_src_off = spad_dest * 16; // 16 is DIM
            
            // Perform the memory move-out operation from SMEM to DRAM
            gemmini_mx_read_smem(dram_dst, smem_src_off, bf16_count);
            
            // Ensure all previous accelerator operations are complete before host accesses DRAM
            gemmini_fence();
            
        } // End loop over N blocks (j0)
    } // End loop over M blocks (i0)
}
