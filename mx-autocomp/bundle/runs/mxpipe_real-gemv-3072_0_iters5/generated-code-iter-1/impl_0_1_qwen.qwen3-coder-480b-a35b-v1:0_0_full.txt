// Optimized version of the MX-Gemmini matmul using out-of-line configuration helpers
// to reduce register pressure and improve I-cache locality.

#include <stdint.h>
#include <string.h>

// Assume these are defined elsewhere appropriately
extern elem_t A_in[], B_in[];
extern out_t C_hw[][];
extern scale_t A_scales_row[][...], B_scales_col[][...];
extern uint8_t A_s[...][...], B_s[...][...];
extern void* scale_factors;

#define BLOCKS_M ...
#define BLOCKS_N ...
#define BLK ...
#define DIM 16
#define BANK_NUM 4
#define BANK_ROWS 2048
#define ACC_ROWS 256
#define BF16_PER_WORD 4
#define MATMUL_K ...
#define MATMUL_N ...
#define A_S_BYTES ...
#define B_S_BYTES ...
#define BLK_OUTC ...

#define WEIGHT_STATIONARY 1
#define ACC_SCALE_IDENTITY 1.0f
#define NO_ACTIVATION 0

// Helper function declarations
static void __attribute__((noinline)) config_gemmini_ex(void);
static void __attribute__((noinline)) config_gemmini_ld_A(void);
static void __attribute__((noinline)) config_gemmini_ld_B(void);
static void __attribute__((noinline)) config_gemmini_st_quant(uint32_t SPAD_DEST, int tiles);

// Main optimized solution function
void solution(void) {
  const int tiles_K = MATMUL_K / DIM;
  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - (BLK/DIM) * tiles_K * DIM;
  const int SPAD_DEST = 128;

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Calculate real dimensions for partial blocks
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;
      int tiles = BLK / DIM;

      // Stage scales per group
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      // Flush and configure execution
      gemmini_flush(0);
      config_gemmini_ex();

      // Load scales
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Load A tiles
      config_gemmini_ld_A();
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          uint32_t dram_offset = (i0 * BLK + i * DIM) * MATMUL_K + k * DIM;
          uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + dram_offset),
                                spad_addr, DIM, DIM);
        }
      }

      // Load B tiles
      config_gemmini_ld_B();
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          uint32_t dram_offset = (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (k * tiles + j) * DIM;
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + dram_offset),
                                spad_addr, DIM, DIM);
        }
      }

      // Configure store and quantization
      config_gemmini_st_quant(SPAD_DEST, tiles);

      // Compute
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
  }
}

// Helper implementations

static void __attribute__((noinline)) config_gemmini_ex(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
                              /*set_only_strides*/false,
                              /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);
}

static void __attribute__((noinline)) config_gemmini_ld_A(void) {
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
}

static void __attribute__((noinline)) config_gemmini_ld_B(void) {
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
}

static void __attribute__((noinline)) config_gemmini_st_quant(uint32_t SPAD_DEST, int tiles) {
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, MATMUL_K/DIM, 0, 0, 1);
}
