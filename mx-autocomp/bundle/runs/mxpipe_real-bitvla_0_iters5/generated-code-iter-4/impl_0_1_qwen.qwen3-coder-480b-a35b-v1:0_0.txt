#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

#define DIM 16
#define ADDR_LEN 32
#define BF16_PER_WORD 4
#define OUT_COLS (MATMUL_M / BF16_PER_WORD)

typedef uint8_t  elem_t;   // fp8:e4m3 input elements
typedef uint8_t  welem_t;  // fp8:e4m3 weight elements
typedef uint64_t out_t;    // 4x bf16 packed per word

#pragma GCC optimize("align-loops=64")

// Optimized MX matmul kernel (32x256x256, fp8:e4m3).
void __attribute__((aligned(64))) solution(void) {
  // Calculate scratchpad partition bases
  const int a_base = 0;
  const int b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  // Ensure no overlap between A and B regions
  if (a_base + tiles_I * tiles_K * DIM > b_base) {
    return;
  }

  // 1. Configure execution
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  
  // 2. Load scale factors
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. Load A tiles (Fully Unrolled)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  
  // i = 0
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 0 * DIM), a_base + (0 * tiles_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 1 * DIM), a_base + (0 * tiles_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 2 * DIM), a_base + (0 * tiles_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 3 * DIM), a_base + (0 * tiles_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 4 * DIM), a_base + (0 * tiles_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 5 * DIM), a_base + (0 * tiles_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 6 * DIM), a_base + (0 * tiles_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 7 * DIM), a_base + (0 * tiles_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 8 * DIM), a_base + (0 * tiles_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 9 * DIM), a_base + (0 * tiles_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 10 * DIM), a_base + (0 * tiles_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 11 * DIM), a_base + (0 * tiles_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 12 * DIM), a_base + (0 * tiles_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 13 * DIM), a_base + (0 * tiles_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 14 * DIM), a_base + (0 * tiles_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 15 * DIM), a_base + (0 * tiles_K + 15) * DIM, DIM, DIM);

  // i = 1
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 0 * DIM), a_base + (1 * tiles_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 1 * DIM), a_base + (1 * tiles_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 2 * DIM), a_base + (1 * tiles_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 3 * DIM), a_base + (1 * tiles_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 4 * DIM), a_base + (1 * tiles_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 5 * DIM), a_base + (1 * tiles_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 6 * DIM), a_base + (1 * tiles_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 7 * DIM), a_base + (1 * tiles_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 8 * DIM), a_base + (1 * tiles_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 9 * DIM), a_base + (1 * tiles_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 10 * DIM), a_base + (1 * tiles_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 11 * DIM), a_base + (1 * tiles_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 12 * DIM), a_base + (1 * tiles_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 13 * DIM), a_base + (1 * tiles_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 14 * DIM), a_base + (1 * tiles_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 15 * DIM), a_base + (1 * tiles_K + 15) * DIM, DIM, DIM);

  // 4. Load B tiles (Inner Loop Fully Unrolled)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));

  // Align the main loop to host cache lines for optimal fetch performance
  asm volatile(".balign 64");
  for (int j = 0; j < tiles_J; j++) {
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 0 * DIM * MATMUL_N + j * DIM), b_base + (0 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 1 * DIM * MATMUL_N + j * DIM), b_base + (1 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 2 * DIM * MATMUL_N + j * DIM), b_base + (2 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 3 * DIM * MATMUL_N + j * DIM), b_base + (3 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 4 * DIM * MATMUL_N + j * DIM), b_base + (4 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 5 * DIM * MATMUL_N + j * DIM), b_base + (5 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 6 * DIM * MATMUL_N + j * DIM), b_base + (6 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 7 * DIM * MATMUL_N + j * DIM), b_base + (7 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 8 * DIM * MATMUL_N + j * DIM), b_base + (8 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 9 * DIM * MATMUL_N + j * DIM), b_base + (9 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 10 * DIM * MATMUL_N + j * DIM), b_base + (10 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 11 * DIM * MATMUL_N + j * DIM), b_base + (11 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 12 * DIM * MATMUL_N + j * DIM), b_base + (12 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 13 * DIM * MATMUL_N + j * DIM), b_base + (13 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 14 * DIM * MATMUL_N + j * DIM), b_base + (14 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 15 * DIM * MATMUL_N + j * DIM), b_base + (15 * tiles_J + j) * DIM, DIM, DIM);
  }

  // 5. Configure the requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  
  // 6. Run weight-stationary compute loop
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
                       
  // 7. Drain output and synchronize
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * MATMUL_N);
  gemmini_fence();
}
