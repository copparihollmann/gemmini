void solution(void) {
  // 1. Flush and configure Gemmini MX weight-stationary mode
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 2. Load all B tiles into the scratchpad once (hoisted out of the computation loop)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));      // B rows: stride N
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                            b_base + (k * tiles_J + j) * DIM, DIM, DIM);
    }
  }

  // Define double-buffered SMEM destinations
  const uint32_t SPAD_DEST_0 = SPAD_DEST;           // 128
  const uint32_t SPAD_DEST_1 = SPAD_DEST + 256;     // 384

  // Calculate dynamic rows to read for the two double-buffered chunks
  int rows_to_read_0 = (REAL_M > 16) ? 16 : REAL_M;
  int rows_to_read_1 = (REAL_M > 16) ? (REAL_M - 16) : 0;

  // 3. Software-pipelined double-buffered execution loop
  for (int i0 = 0; i0 < 2; i0++) {
    // MVIN A tiles for the current row block (block_I = 1)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));      // A rows: stride K
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * DIM) * MATMUL_K + k * DIM),
                            a_base + k * DIM, DIM, DIM);
    }

    // Alternate SMEM output destinations
    uint32_t cur_spad_dest = (i0 == 0) ? SPAD_DEST_0 : SPAD_DEST_1;
    int cur_pad_i = (i0 == 0) ? 0 : PAD_I;

    // Configure requantization parameters for the current block_I = 1
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, 1, tiles_J, tiles_K, 0, 0, 1);

    // Compute block i0
    gemmini_loop_ws_spad(1, tiles_J, tiles_K, cur_pad_i, PAD_J, PAD_K, a_base, BANK_NUM * BANK_ROWS, 0,
                         cur_spad_dest, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Overlap: Drain the PREVIOUS block's SMEM buffer to DRAM while the current block computes
    if (i0 > 0) {
      gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST_0 * 16, rows_to_read_0 * MATMUL_N);
    }
  }

  // 4. Drain the last block (block 1) to DRAM
  if (rows_to_read_1 > 0) {
    gemmini_mx_read_smem(&C_hw[16][0], SPAD_DEST_1 * 16, rows_to_read_1 * MATMUL_N);
  } else {
    // If there's no second chunk, drain the first one now
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST_0 * 16, rows_to_read_0 * MATMUL_N);
  }

  gemmini_fence();
}
