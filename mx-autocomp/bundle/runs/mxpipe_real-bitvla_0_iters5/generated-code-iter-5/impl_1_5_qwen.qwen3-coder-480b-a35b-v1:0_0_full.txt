Looking at the optimization plan and the baseline code, I need to implement a double-buffered tiling strategy along the I dimension to overlap data movement and compute while avoiding accumulator overflow.

Let me analyze the key parameters:
- Matrix size: 32×256×256 (M×N×K)
- DIM = 16, so tiles_I = 2, tiles_J = 16, tiles_K = 16
- Each tile is 16×16 elements
- Accumulator limitation: 256 rows, but we need 2×16×16 = 512 rows for the full computation

The optimization approach is to split the I dimension into 2 blocks, process them sequentially with double buffering to overlap computation and data movement.

```c
// Optimized MX matmul kernel (32x256x256, fp8:e4m3).
void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Define memory layout for double buffering
  const uint32_t a_base_0 = a_base;  // First A buffer at start of scratchpad
  const uint32_t a_base_1 = a_base + 1 * tiles_K * DIM;  // Second A buffer (256 rows offset)
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;  // B at top of scratchpad (4096)
  const uint32_t SPAD_DEST_0 = SPAD_DEST;  // First output area
  const uint32_t SPAD_DEST_1 = SPAD_DEST + 1 * tiles_J * DIM;  // Second output area (256 rows offset)

  // Load B matrix once into the top of scratchpad (weight-stationary reuse)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                            b_base + (k * tiles_J + j) * DIM, DIM, DIM);
    }
  }

  // Load first block of A (I=0)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    gemmini_extended_mvin((void*)(((elem_t*)A_in) + 0 * DIM * MATMUL_K + k * DIM),
                          a_base_0 + k * DIM, DIM, DIM);
  }

  // Compute first block (I=0) - using only 1 I-tile, fits in accumulator (1*16*16 = 256 rows)
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, 1, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(1, tiles_J, tiles_K, 0, PAD_J, PAD_K, a_base_0, b_base, 0,
                       SPAD_DEST_0, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Overlap: Load second block of A (I=1) while first block computes
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    gemmini_extended_mvin((void*)(((elem_t*)A_in) + 1 * DIM * MATMUL_K + k * DIM),
                          a_base_1 + k * DIM, DIM, DIM);
  }

  // Compute second block (I=1) - handle any remaining padding in I dimension
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)(scale_factors + 1 * tiles_J), 1, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(1, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K, a_base_1, b_base, 0,
                       SPAD_DEST_1, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Drain results from shared memory to DRAM
  int rows_0 = REAL_M < 16 ? REAL_M : 16;
  if (rows_0 > 0) {
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST_0 * 16, rows_0 * MATMUL_N);
  }

  int rows_1 = REAL_M > 16 ? REAL_M - 16 : 0;
  if (rows_1 > 0) {
    gemmini_mx_read_smem(&C_hw[16][0], SPAD_DEST_1 * 16, rows_1 * MATMUL_N);
  }

  // Synchronize all operations
  gemmini_fence();
}
```
