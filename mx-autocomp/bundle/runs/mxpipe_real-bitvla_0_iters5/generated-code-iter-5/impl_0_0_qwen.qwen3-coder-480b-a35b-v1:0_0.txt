#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

// Optimized MX matmul kernel (32x256x256, fp8:e4m3).
void solution(void) {
  // Utilize host-side CPU registers to cache scratchpad base addresses
  register const int a_base asm("s0") = 0;
  register const int b_base asm("s1") = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  // Ensure no overlap between A and B regions
  if (a_base + tiles_I * tiles_K * DIM > b_base) {
    return;
  }

  // Pin bounds, loop markers, strides, and pads to host registers (s2 - s10)
  register const int t_I asm("s2") = tiles_I;
  register const int t_J asm("s3") = tiles_J;
  register const int t_K asm("s4") = tiles_K;
  register const int out_cols_reg asm("s5") = OUT_COLS;
  register const int spad_dest_reg asm("s6") = SPAD_DEST;
  register const int pad_i_reg asm("s8") = PAD_I;
  register const int pad_j_reg asm("s9") = PAD_J;
  register const int pad_k_reg asm("s10") = PAD_K;

  // 1. Configure execution
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  
  // 2. Load scale factors
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. Load A tiles (Fully Unrolled)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  
  // i = 0
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 0 * DIM), a_base + (0 * t_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 1 * DIM), a_base + (0 * t_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 2 * DIM), a_base + (0 * t_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 3 * DIM), a_base + (0 * t_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 4 * DIM), a_base + (0 * t_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 5 * DIM), a_base + (0 * t_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 6 * DIM), a_base + (0 * t_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 7 * DIM), a_base + (0 * t_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 8 * DIM), a_base + (0 * t_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 9 * DIM), a_base + (0 * t_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 10 * DIM), a_base + (0 * t_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 11 * DIM), a_base + (0 * t_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 12 * DIM), a_base + (0 * t_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 13 * DIM), a_base + (0 * t_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 14 * DIM), a_base + (0 * t_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 0 * DIM * MATMUL_K + 15 * DIM), a_base + (0 * t_K + 15) * DIM, DIM, DIM);

  // i = 1
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 0 * DIM), a_base + (1 * t_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 1 * DIM), a_base + (1 * t_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 2 * DIM), a_base + (1 * t_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 3 * DIM), a_base + (1 * t_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 4 * DIM), a_base + (1 * t_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 5 * DIM), a_base + (1 * t_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 6 * DIM), a_base + (1 * t_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 7 * DIM), a_base + (1 * t_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 8 * DIM), a_base + (1 * t_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 9 * DIM), a_base + (1 * t_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 10 * DIM), a_base + (1 * t_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 11 * DIM), a_base + (1 * t_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 12 * DIM), a_base + (1 * t_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 13 * DIM), a_base + (1 * t_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 14 * DIM), a_base + (1 * t_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)A_in) + 1 * DIM * MATMUL_K + 15 * DIM), a_base + (1 * t_K + 15) * DIM, DIM, DIM);

  // 4. Load B tiles (Inner Loop Fully Unrolled, Loop index pinned to register s7)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (register int j asm("s7") = 0; j < t_J; j++) {
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 0 * DIM * MATMUL_N + j * DIM), b_base + (0 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 1 * DIM * MATMUL_N + j * DIM), b_base + (1 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 2 * DIM * MATMUL_N + j * DIM), b_base + (2 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 3 * DIM * MATMUL_N + j * DIM), b_base + (3 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 4 * DIM * MATMUL_N + j * DIM), b_base + (4 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 5 * DIM * MATMUL_N + j * DIM), b_base + (5 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 6 * DIM * MATMUL_N + j * DIM), b_base + (6 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 7 * DIM * MATMUL_N + j * DIM), b_base + (7 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 8 * DIM * MATMUL_N + j * DIM), b_base + (8 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 9 * DIM * MATMUL_N + j * DIM), b_base + (9 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 10 * DIM * MATMUL_N + j * DIM), b_base + (10 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 11 * DIM * MATMUL_N + j * DIM), b_base + (11 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 12 * DIM * MATMUL_N + j * DIM), b_base + (12 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 13 * DIM * MATMUL_N + j * DIM), b_base + (13 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 14 * DIM * MATMUL_N + j * DIM), b_base + (14 * t_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)B_in) + 15 * DIM * MATMUL_N + j * DIM), b_base + (15 * t_J + j) * DIM, DIM, DIM);
  }

  // 5. Configure the requantized mvout
  gemmini_config_st(out_cols_reg * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, t_I, t_J, t_K, 0, 0, 1);
  
  // 6. Run weight-stationary compute loop
  gemmini_loop_ws_spad(t_I, t_J, t_K, pad_i_reg, pad_j_reg, pad_k_reg,
                       a_base, BANK_NUM * BANK_ROWS, 0, spad_dest_reg,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
                       
  // 7. Drain output and synchronize
  gemmini_mx_read_smem(&C_hw[0][0], spad_dest_reg * 16, REAL_M * MATMUL_N);
  gemmini_fence();
}
