#include <stdlib.h>
#include <string.h>

// Optimized MX matmul kernel (32x256x256, fp8:e4m3) with hugepage DRAM buffers.
void solution(void) {
  // Calculate scratchpad partition bases
  const int a_base = 0;
  const int b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  // Ensure no overlap between A and B regions
  if (a_base + tiles_I * tiles_K * DIM > b_base) {
    return;
  }

  // Define sizes for DRAM allocations
  const size_t size_A = MATMUL_M * MATMUL_K * sizeof(elem_t);
  const size_t size_B = MATMUL_K * MATMUL_N * sizeof(elem_t);
  const size_t size_C = REAL_M * MATMUL_N * sizeof(out_t);
  const size_t size_scales = sizeof(scale_factors);
  const size_t size_A_scales = sizeof(A_scales_row);
  const size_t size_B_scales = sizeof(B_scales_col);

  // Use 2MB alignment boundary for Hugepage mappings
  const size_t hp_align = 2 * 1024 * 1024;

  void *huge_A = NULL;
  void *huge_B = NULL;
  void *huge_C = NULL;
  void *huge_scales = NULL;
  void *huge_A_scales = NULL;
  void *huge_B_scales = NULL;

  // Allocate 2MB-aligned memory buffers
  posix_memalign(&huge_A, hp_align, (size_A + hp_align - 1) & ~(hp_align - 1));
  posix_memalign(&huge_B, hp_align, (size_B + hp_align - 1) & ~(hp_align - 1));
  posix_memalign(&huge_C, hp_align, (size_C + hp_align - 1) & ~(hp_align - 1));
  posix_memalign(&huge_scales, hp_align, (size_scales + hp_align - 1) & ~(hp_align - 1));
  posix_memalign(&huge_A_scales, hp_align, (size_A_scales + hp_align - 1) & ~(hp_align - 1));
  posix_memalign(&huge_B_scales, hp_align, (size_B_scales + hp_align - 1) & ~(hp_align - 1));

  // If allocations succeeded, stage inputs to hugepages
  if (huge_A && huge_B && huge_C && huge_scales && huge_A_scales && huge_B_scales) {
    memcpy(huge_A, A_in, size_A);
    memcpy(huge_B, B_in, size_B);
    memcpy(huge_scales, scale_factors, size_scales);
    memcpy(huge_A_scales, &A_scales_row, size_A_scales);
    memcpy(huge_B_scales, &B_scales_col, size_B_scales);
  } else {
    // Fallback if allocation fails in restricted environment
    huge_A = (void *)A_in;
    huge_B = (void *)B_in;
    huge_C = (void *)C_hw;
    huge_scales = (void *)scale_factors;
    huge_A_scales = (void *)&A_scales_row;
    huge_B_scales = (void *)&B_scales_col;
  }

  // 1. Configure execution
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  
  // 2. Load scale factors
  gemmini_mx_load_scales((uint64_t)huge_A_scales, size_A_scales, 0);
  gemmini_mx_load_scales((uint64_t)huge_B_scales, size_B_scales, 1);

  // 3. Load A tiles (Fully Unrolled)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  
  // i = 0
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 0 * DIM), a_base + (0 * tiles_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 1 * DIM), a_base + (0 * tiles_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 2 * DIM), a_base + (0 * tiles_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 3 * DIM), a_base + (0 * tiles_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 4 * DIM), a_base + (0 * tiles_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 5 * DIM), a_base + (0 * tiles_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 6 * DIM), a_base + (0 * tiles_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 7 * DIM), a_base + (0 * tiles_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 8 * DIM), a_base + (0 * tiles_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 9 * DIM), a_base + (0 * tiles_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 10 * DIM), a_base + (0 * tiles_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 11 * DIM), a_base + (0 * tiles_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 12 * DIM), a_base + (0 * tiles_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 13 * DIM), a_base + (0 * tiles_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 14 * DIM), a_base + (0 * tiles_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 0 * DIM * MATMUL_K + 15 * DIM), a_base + (0 * tiles_K + 15) * DIM, DIM, DIM);

  // i = 1
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 0 * DIM), a_base + (1 * tiles_K + 0) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 1 * DIM), a_base + (1 * tiles_K + 1) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 2 * DIM), a_base + (1 * tiles_K + 2) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 3 * DIM), a_base + (1 * tiles_K + 3) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 4 * DIM), a_base + (1 * tiles_K + 4) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 5 * DIM), a_base + (1 * tiles_K + 5) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 6 * DIM), a_base + (1 * tiles_K + 6) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 7 * DIM), a_base + (1 * tiles_K + 7) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 8 * DIM), a_base + (1 * tiles_K + 8) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 9 * DIM), a_base + (1 * tiles_K + 9) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 10 * DIM), a_base + (1 * tiles_K + 10) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 11 * DIM), a_base + (1 * tiles_K + 11) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 12 * DIM), a_base + (1 * tiles_K + 12) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 13 * DIM), a_base + (1 * tiles_K + 13) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 14 * DIM), a_base + (1 * tiles_K + 14) * DIM, DIM, DIM);
  gemmini_extended_mvin((void *)(((elem_t *)huge_A) + 1 * DIM * MATMUL_K + 15 * DIM), a_base + (1 * tiles_K + 15) * DIM, DIM, DIM);

  // 4. Load B tiles (Inner Loop Fully Unrolled)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 0 * DIM * MATMUL_N + j * DIM), b_base + (0 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 1 * DIM * MATMUL_N + j * DIM), b_base + (1 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 2 * DIM * MATMUL_N + j * DIM), b_base + (2 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 3 * DIM * MATMUL_N + j * DIM), b_base + (3 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 4 * DIM * MATMUL_N + j * DIM), b_base + (4 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 5 * DIM * MATMUL_N + j * DIM), b_base + (5 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 6 * DIM * MATMUL_N + j * DIM), b_base + (6 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 7 * DIM * MATMUL_N + j * DIM), b_base + (7 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 8 * DIM * MATMUL_N + j * DIM), b_base + (8 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 9 * DIM * MATMUL_N + j * DIM), b_base + (9 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 10 * DIM * MATMUL_N + j * DIM), b_base + (10 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 11 * DIM * MATMUL_N + j * DIM), b_base + (11 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 12 * DIM * MATMUL_N + j * DIM), b_base + (12 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 13 * DIM * MATMUL_N + j * DIM), b_base + (13 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 14 * DIM * MATMUL_N + j * DIM), b_base + (14 * tiles_J + j) * DIM, DIM, DIM);
    gemmini_extended_mvin((void *)(((elem_t *)huge_B) + 15 * DIM * MATMUL_N + j * DIM), b_base + (15 * tiles_J + j) * DIM, DIM, DIM);
  }

  // 5. Configure the requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)huge_scales, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  
  // 6. Run weight-stationary compute loop
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
                       
  // 7. Drain output and synchronize
  gemmini_mx_read_smem(huge_C, SPAD_DEST * 16, REAL_M * MATMUL_N);
  gemmini_fence();

  // Copy result back if hugepages were used
  if (huge_C != (void *)C_hw) {
    memcpy(C_hw, huge_C, size_C);
  }

  // Free the allocated memory buffers
  if (huge_A != (void *)A_in) free(huge_A);
  if (huge_B != (void *)B_in) free(huge_B);
  if (huge_C != (void *)C_hw) free(huge_C);
  if (huge_scales != (void *)scale_factors) free(huge_scales);
  if (huge_A_scales != (void *)&A_scales_row) free(huge_A_scales);
  if (huge_B_scales != (void *)&B_scales_col) free(huge_B_scales);
}
