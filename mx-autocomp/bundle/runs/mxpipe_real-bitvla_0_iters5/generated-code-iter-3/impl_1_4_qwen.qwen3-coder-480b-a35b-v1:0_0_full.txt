// Optimized MX matmul kernel (32x256x256, fp8:e4m3).
void solution(void) {
  // Calculate scratchpad partition bases
  const int a_base = 0;
  const int b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  // Ensure no overlap between A and B regions
  if (a_base + tiles_I * tiles_K * DIM > b_base) {
    // Handle error or adjust tiling strategy
    return;
  }

  // Hoisted configuration: Configure execution
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
                              /*set_only_strides*/false,
                              /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

  // Hoisted configuration: Load scale factors
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0); // sel=0 for A/activation scales
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1); // sel=1 for B/weight scales

  // Load A tiles
  gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Set stride for A matrix
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *src_addr = &A_in[i * DIM * MATMUL_K + k * DIM];
      uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)src_addr, spad_addr, DIM, DIM);
    }
  }

  // Load B tiles
  gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // Set stride for B matrix
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *src_addr = &B_in[k * DIM * MATMUL_N + j * DIM];
      uint32_t spad_addr = b_base + (k * tiles_J + j) * DIM;
      gemmini_extended_mvin((void *)src_addr, spad_addr, DIM, DIM);
    }
  }

  // Hoisted configuration: Configure requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                               /*i_bound*/tiles_I, /*j_bound*/tiles_J, /*k_bound*/tiles_K,
                               /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

  // Compute the weight-stationary tiled matmul over the scratchpad
  // Note: funct 24 (LOOP_WS_CONFIG_SPAD_AB) is implicitly issued by gemmini_loop_ws_spad
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       /*A_transpose*/false, /*B_transpose*/false,
                       /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
                       NO_ACTIVATION, /*a_spad_id*/0, /*b_spad_id*/0,
                       /*is_resadd*/false, /*skips*/0x38);

  // Read the result out of shared memory
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * MATMUL_N);
  gemmini_fence(); // Synchronize after reading results
}
