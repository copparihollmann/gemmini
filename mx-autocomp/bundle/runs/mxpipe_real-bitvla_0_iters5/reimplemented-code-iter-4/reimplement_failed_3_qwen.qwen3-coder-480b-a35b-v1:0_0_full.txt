Looking at the error message, the issue is that `cols=64` overflows the 6-bit hardware field, which can only represent values 0-63. This happens when `MAX_BLOCK_LEN=4` and we're trying to transfer 4 tiles of 16 columns each (4×16=64).

According to the documentation, `MAX_BLOCK_LEN = 4` but due to hardware limitations, the actual maximum batch size for transfers is limited by the 6-bit `num_cols` field in the instruction. Since each tile is `DIM=16` columns, the maximum `cols` value is 63, meaning we can only transfer at most 3 tiles per command (3×16=48 ≤ 63).

Let me fix the code to respect this limitation:

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
// Assuming header files like matmul_data.h are included via gemmini_testutils.h or project setup

void solution(void) {
  // 1. Flush and configure MX FP8 weight-stationary mode
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Load per-block scale factors (sel=0 for A, sel=1 for B)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. MVIN A tiles: Coalesced block transfer along K (up to 3 tiles per command due to 6-bit cols field limitation)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // A rows: stride K
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += 3) {  // Changed from MAX_BLOCK_LEN to 3
      int len = (tiles_K - k > 3) ? 3 : (tiles_K - k);  // Changed from MAX_BLOCK_LEN to 3
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, len * DIM, DIM);
    }
  }

  // 4. MVIN B tiles: Coalesced block transfer along K (up to 3 tiles per command due to 6-bit cols field limitation)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // B rows: stride N
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k += 3) {  // Changed from MAX_BLOCK_LEN to 3
      int len = (tiles_K - k > 3) ? 3 : (tiles_K - k);  // Changed from MAX_BLOCK_LEN to 3
      gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                             b_base + (k * tiles_J + j) * DIM, DIM, len * DIM);
    }
  }

  // 5. Configure output stride and MX requantizer
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 6. Execute fused weight-stationary hardware loop
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // 7. Read results from shared memory & enforce memory barrier before CPU reads
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * MATMUL_N);
  gemmini_fence();
}
```
