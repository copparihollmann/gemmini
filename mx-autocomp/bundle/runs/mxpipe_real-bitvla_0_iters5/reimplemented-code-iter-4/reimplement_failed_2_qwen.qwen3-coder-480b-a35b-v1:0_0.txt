// Optimized MX matmul kernel (32x256x256, fp8:e4m3).
void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure load controller for Matrix A (stride K) and Matrix B (stride N)
  // Use single-port mvin for both A and B to satisfy RTL constraint
  gemmini_extended_config_ld(MATMUL_K * sizeof(elem_t)); // For A
  gemmini_extended_config_ld(MATMUL_N * sizeof(elem_t)); // For B

  // Precompute base addresses to reduce computation in inner loops
  const int a_spad_base = a_base;
  const int b_spad_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  // Load all A tiles first
  for (int k = 0; k < tiles_K; k++) {
    for (int i = 0; i < tiles_I; i++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
                            a_spad_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // Then load all B tiles
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j++) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                            b_spad_base + (k * tiles_J + j) * DIM, DIM, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K, 
                       a_spad_base, b_spad_base, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Read only the REAL rows (smem row stride is the padded MATMUL_N)
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * MATMUL_N);
  gemmini_fence();
}
