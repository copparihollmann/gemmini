#include <stdint.h>
#include "include/gemmini_testutils.h"

void solution(void) {
  // 1. Configure Gemmini execution format and dataflow
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  
  // 2. Load scale factors (retained once since they persist)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. Load entire B matrix once (Weight-Stationary)
  const int b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                            b_base + (j * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // Double-buffering bases for A
  const int a_base_0 = 0;
  const int a_base_1 = tiles_K * DIM; // 16 * 16 = 256 rows

  // --- Block 0 (i = 0) ---
  // Load A for Block 0
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    gemmini_extended_mvin((void*)(((elem_t*)A_in) + 0 * DIM * MATMUL_K + k * DIM),
                          a_base_0 + k * DIM, DIM, DIM);
  }

  // Compute Block 0
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, 1, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(1, tiles_J, tiles_K, 0, PAD_J, PAD_K, a_base_0, b_base, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 1, 1, false, 0x38);

  // --- Block 1 (i = 1) Pipelining ---
  // Load A for Block 1 (overlaps with Block 0 compute/read-back)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    gemmini_extended_mvin((void*)(((elem_t*)A_in) + 1 * DIM * MATMUL_K + k * DIM),
                          a_base_1 + k * DIM, DIM, DIM);
  }

  // Read out Block 0 results from SMEM to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, 16 * MATMUL_N);

  // Compute Block 1
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)(((uint8_t*)scale_factors) + 1 * tiles_J * sizeof(uint8_t)), 1, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(1, tiles_J, tiles_K, PAD_I, PAD_J, PAD_K, a_base_1, b_base, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 2, 1, false, 0x38);

  // Read out Block 1 results from SMEM to DRAM
  gemmini_mx_read_smem(&C_hw[16][0], SPAD_DEST * 16, (REAL_M - 16) * sizeof(out_t) / sizeof(uint16_t));

  // Synchronize memory before host read
  gemmini_fence();
}
