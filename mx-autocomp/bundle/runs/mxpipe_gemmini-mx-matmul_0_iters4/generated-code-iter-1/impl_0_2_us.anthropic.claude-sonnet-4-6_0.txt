void solution(void) {
  // Step 1: Configure execution (format invariant across entire matmul)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  // Step 2: Load scale factors for A (sel=0) and B (sel=1)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 3: Configure both mvin ports once, independently
  // Port 0 (mvin)  -> A tiles, stride = MATMUL_K bytes/row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Port 1 (mvin2) -> B tiles, stride = MATMUL_K bytes/row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Step 4: Interleaved A+B mvin in a single combined (i, j, k) loop
  // A layout: a_base + (i*tiles_K + k)*DIM
  // B layout: b_base + (j*tiles_K + k)*DIM
  for (int i = 0; i < tiles_I; i++) {
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        // A tile: row i, k-block k
        elem_t *a_ptr = ((elem_t *)A_in) + i * DIM * MATMUL_K + k * DIM;
        uint32_t a_spad = a_base + (i * tiles_K + k) * DIM;

        // B tile: col-block j, k-block k
        elem_t *b_ptr = ((elem_t *)B_in) + j * DIM * MATMUL_K + k * DIM;
        uint32_t b_spad = b_base + (j * tiles_K + k) * DIM;

        // Issue both mvins back-to-back -> hardware overlaps DMA transfers
        gemmini_extended_mvin ((void *)a_ptr, a_spad, DIM, DIM);
        gemmini_extended_mvin2((void *)b_ptr, b_spad, DIM, DIM);
      }
    }
  }

  // Step 5: Configure store and requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // Step 6: Fused hardware WS loop over scratchpad tiles
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain BF16 results from shared memory to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
