void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure port 0 (mvin)  for A tiles
  // Configure port 1 (mvin2) for B tiles
  // Both share the same DRAM row stride: MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Interleave A (port 0) and B (port 1) tile loads.
  // Each block is capped at MAX_BLOCK_LEN=4 tiles to respect the DMA limit.
  // A tiles occupy scratchpad rows [a_base .. a_base + tiles_I*tiles_K*DIM - 1].
  // B tiles occupy scratchpad rows [b_base .. BANK_NUM*BANK_ROWS - 1], growing downward.
  // These regions are disjoint, so both ports can be in flight simultaneously.

  const int CHUNK = MAX_BLOCK_LEN; // 4

  int max_tiles = (tiles_I > tiles_J) ? tiles_I : tiles_J;

  for (int idx = 0; idx < max_tiles; idx++) {
    // Issue A mvin commands for row tile `idx` (port 0)
    if (idx < tiles_I) {
      elem_t *a_dram = (elem_t*)A_in + idx * DIM * MATMUL_K;
      uint32_t a_sp  = a_base + idx * tiles_K * DIM;
      for (int k = 0; k < tiles_K; k += CHUNK) {
        int blk = ((tiles_K - k) < CHUNK) ? (tiles_K - k) : CHUNK;
        gemmini_extended_mvin(
            (void*)(a_dram + k * DIM),
            a_sp + k * DIM,
            blk * DIM, DIM);
      }
    }
    // Issue B mvin2 commands for col tile `idx` (port 1)
    if (idx < tiles_J) {
      elem_t *b_dram = (elem_t*)B_in + idx * DIM * MATMUL_K;
      uint32_t b_sp  = b_base + idx * tiles_K * DIM;
      for (int k = 0; k < tiles_K; k += CHUNK) {
        int blk = ((tiles_K - k) < CHUNK) ? (tiles_K - k) : CHUNK;
        gemmini_extended_mvin2(
            (void*)(b_dram + k * DIM),
            b_sp + k * DIM,
            blk * DIM, DIM);
      }
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0,
      0, false, false, false, false, false,
      NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], 0, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
