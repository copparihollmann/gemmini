void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure both DMA channels with the same stride
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0); // id=0: A
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1); // id=1: B

  // Interleave A (mvin) and B (mvin2) loads to exploit both DMA channels in parallel
  int max_tiles = tiles_I > tiles_J ? tiles_I : tiles_J;
  for (int t = 0; t < max_tiles; t++) {
    if (t < tiles_I) {
      elem_t *A_row = (elem_t*)A_in + t * DIM * MATMUL_K;
      gemmini_extended_mvin((void*)A_row,
                            a_base + t * tiles_K * DIM,
                            DIM * tiles_K, DIM);
    }
    if (t < tiles_J) {
      elem_t *B_row = (elem_t*)B_in + t * DIM * MATMUL_K;
      gemmini_extended_mvin2((void*)B_row,
                             b_base + t * tiles_K * DIM,
                             DIM * tiles_K, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       0, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], 0, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
