Looking at the original code, I need to understand what it does:

1. **First matmul**: Q × KT → S1 (attention scores), fp6 format, S=128, D=64
2. **Softmax**: CPU-side integer softmax on S1, producing P_hw (fp4-packed) and P_scales
3. **Second matmul**: P × V → O (output), fp6 format

The original code has:
- `ti=ATT_S/DIM/2=4`, `tj=ATT_S/DIM/2=4`, `tk=ATT_D/DIM=4` for first matmul
- `ti=ATT_S/DIM/2=4`, `tj=ATT_D/DIM/2=2`, `tk=ATT_S/DIM=8` for second matmul

The scratchpad params: BANK_NUM=4, BANK_ROWS=2048 → 8192 rows total. The original uses `SD=128` as SPAD_DEST for both. Let me verify the layout fits: for first matmul, `ab=0`, `bb=8192-4*4*16=7936`. A tiles: `4*4*16=256` rows, B tiles from 7936 to 8192 (256 rows). No overlap. ✓

Key observations for optimization:
- The `gemmini_config_st` is placed before `gemmini_mxquant_config_mvout` in original, which is fine
- The mvin loops can potentially be batched using MAX_BLOCK_LEN=4
- The programming sequence order must be strictly followed
- LUT loads persist, so they don't need reloading if unchanged

I'll optimize by batching mvins where possible (up to MAX_BLOCK_LEN=4 consecutive tiles), which reduces instruction count significantly.

```c
void solution(void) {
  // ── First matmul: Q × KT → S1  (FP6 E3M2, S=128, D=64) ──────────────────
  {
    const int ti = ATT_S / DIM / 2;   // 4
    const int tj = ATT_S / DIM / 2;   // 4
    const int tk = ATT_D / DIM;       // 4
    const uint32_t ab = 0;
    const uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM; // 8192-256=7936
    const int SD = 128;

    gemmini_flush(0);

    // Step 1: configure execution (fp6 e3m2 act=1, wgt=1, out=3 BF16, uselut=1)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 1, 1, 3, 1);

    // Step 2: load scales for A (Q) and B (KT)
    gemmini_mx_load_scales((uint64_t)&Q_scales_row,  sizeof(Q_scales_row),  0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Step 3: load LUTs (uselut=1), sel 0=B(KT), sel 1=A(Q)
    gemmini_mx_load_lut((uint64_t)KT_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)Q_lp,  1, 1);

    // Step 4a: mvin A tiles (Q) — stride = ATT_D * sizeof(elem_t) = 64
    // Layout: ab + (i*tk + k)*DIM, batch consecutive k tiles up to MAX_BLOCK_LEN=4
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      int k = 0;
      while (k < tk) {
        int batch = tk - k;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        // batch consecutive k tiles for this i row share contiguous spad rows
        // but DRAM rows are strided by ATT_D, not contiguous across k
        // so we must issue one mvin per tile (k tiles are columns, not contiguous in DRAM)
        for (int b = 0; b < batch; b++) {
          int kk = k + b;
          gemmini_extended_mvin(
              (void*)(((elem_t*)Q_hw) + i * DIM * ATT_D + kk * DIM),
              ab + (i * tk + kk) * DIM,
              DIM, DIM);
        }
        k += batch;
      }
    }

    // Step 4b: mvin B tiles (KT) — stride = ATT_S/2 * sizeof(elem_t) = 64
    // KT_in rows: k*DIM rows of width ATT_S/2, so stride = ATT_S/2 * sizeof(elem_t)
    // Layout: bb + (k*tj + j)*DIM
    // For fixed k, consecutive j tiles ARE contiguous in spad → can batch up to MAX_BLOCK_LEN
    gemmini_config_ld(ATT_S * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      int j = 0;
      while (j < tj) {
        int batch = tj - j;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        // Check if DRAM source for consecutive j is contiguous:
        // ptr = KT_in + k*DIM*(ATT_S/2) + j*DIM
        // next j: ptr + DIM → yes, contiguous in DRAM row direction
        // spad: bb + (k*tj + j)*DIM, bb + (k*tj + j+1)*DIM → contiguous ✓
        gemmini_extended_mvin2(
            (void*)(((elem_t*)KT_in) + k * DIM * (ATT_S / 2) + j * DIM),
            bb + (k * tj + j) * DIM,
            batch * DIM, DIM);
        j += batch;
      }
    }

    // Step 5: configure output stride and requant
    gemmini_config_st((ATT_S / BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk, 0, 0, 8);

    // Step 6: fused tiled matmul
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM * BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: drain results
    gemmini_mx_read_smem(&S1_hw[0][0], SD * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ── CPU softmax: S1 → P_hw (FP4-packed), P_scales ────────────────────────
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);
    for (int j = 0; j < ATT_S; j++) {
      int64_t qq = mx_ibf16_to_q(
          (uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF));
      q[j] = qq;
      if (qq > mq) mq = qq;
    }
    int64_t p[ATT_S], sm = 0;
    for (int j = 0; j < ATT_S; j++) {
      int64_t pv = mx_iexp(q[j] - mq);
      p[j] = pv;
      sm += pv;
    }
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) /
                   (uint64_t)sm;
    for (int g = 0; g < GKP; g++) {
      int64_t gm = 0;
      for (int j = g * 32; j < g * 32 + 32; j++)
        if (p[j] > gm) gm = p[j];
      int e = (gm > 0) ? (mx_ilog2((uint64_t)gm) - ls + 2) : 0;
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g * 32; j < g * 32 + 32; j++) {
        uint64_t R = (((uint64_t)p[j] * inv) + ((uint64_t)1 << (ls - 1))) >> ls;
        uint64_t Rt = (e >= 0) ? (R >> e) : (R << (-e));
        uint8_t idx = mx_enc_P(Rt);
        if ((i & 1) == 0)
          P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0xF0) | idx;
        else
          P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0x0F) | (idx << 4);
      }
    }
  }

  // ── Second matmul: P × V → O  (FP6 E3M2, S=128, D=64) ───────────────────
  {
    const int ti = ATT_S / DIM / 2;   // 4
    const int tj = ATT_D / DIM / 2;   // 2
    const int tk = ATT_S / DIM;       // 8
    const uint32_t ab = 0;
    const uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM; // 8192-256=7936
    const int SD = 128;

    gemmini_flush(0);

    // Step 1: configure execution (fp6 e3m2 act=1, wgt=1, out=3 BF16, uselut=1)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 1, 1, 3, 1);

    // Step 2: load scales for A (P) and B (V)
    gemmini_mx_load_scales((uint64_t)&P_scales,     sizeof(P_scales),     0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Step 3: load LUTs (uselut=1), sel 0=B(V), sel 1=A(P)
    gemmini_mx_load_lut((uint64_t)V_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)P_lp, 1, 1);

    // Step 4a: mvin A tiles (P) — stride = ATT_S * sizeof(elem_t) = 128
    // Layout: ab + (i*tk + k)*DIM
    // For fixed i, consecutive k tiles: DRAM ptr advances by DIM*sizeof(elem_t)=16 bytes
    // across k dimension; with stride=ATT_S=128, rows are NOT contiguous → 1 per mvin
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      int k = 0;
      while (k < tk) {
        int batch = tk - k;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        for (int b = 0; b < batch; b++) {
          int kk = k + b;
          gemmini_extended_mvin(
              (void*)(((elem_t*)P_hw) + i * DIM * ATT_S + kk * DIM),
              ab + (i * tk + kk) * DIM,
              DIM, DIM);
        }
        k += batch;
      }
    }

    // Step 4b: mvin B tiles (V) — stride = ATT_D/2 * sizeof(elem_t) = 32
    // V_in rows: k*DIM rows of width ATT_D/2
    // For fixed k, consecutive j tiles ARE contiguous in DRAM → batch
    gemmini_config_ld(ATT_D * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      int j = 0;
      while (j < tj) {
        int batch = tj - j;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        gemmini_extended_mvin2(
            (void*)(((elem_t*)V_in) + k * DIM * (ATT_D / 2) + j * DIM),
            bb + (k * tj + j) * DIM,
            batch * DIM, DIM);
        j += batch;
      }
    }

    // Step 5: configure output stride and requant
    gemmini_config_st((ATT_D / BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk, 0, 0, 8);

    // Step 6: fused tiled matmul
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM * BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: drain results
    gemmini_mx_read_smem(&O_hw[0][0], SD * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
