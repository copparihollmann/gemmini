Looking at the original code, I need to understand what it does:
1. First matmul: Q × KT → S1 (attention scores), S=128, D=64, fp6 format
2. Softmax computation on CPU
3. Second matmul: P × V → O (attention output), S=128, D=64, fp6 format

The original code has `ti=ATT_S/DIM/2` for both matmuls, which means it's only processing half the rows. Let me analyze the tiling carefully and ensure correctness while optimizing.

Key observations:
- ATT_S=128, ATT_D=64, DIM=16
- First matmul: ti=4, tj=4, tk=4 (128/16/2=4, 64/16=4) — Q(64×64) × KT(64×128) → S1(64×128) but ATT_S*ATT_S=128*128 is read... wait, SD=128 is SPAD_DEST
- The `gemmini_mx_read_smem` reads ATT_S*ATT_S=16384 BF16 values starting at SD*16=128*16=2048 word offset
- For fp6 E3M2: act_mx_fmt=1, wgt_mx_fmt=1 based on the format table, but original uses 1,1,3,1

The scratchpad/accumulator sizes given in the rules (256KB scratchpad, 64KB accumulator) differ from the architecture summary (128KB, 32KB). I'll use the rule values: BANK_NUM=4, BANK_ROWS=2048 but with larger rows, or just trust BANK_NUM*BANK_ROWS=8192 rows as stated in rule 7.

void solution(void) {
  // First matmul: Q x KT -> S1 (attention scores)
  // Q: [ATT_S x ATT_D] fp6, KT: [ATT_D x ATT_S] fp6 (stored transposed)
  // tiles: ti=ATT_S/DIM/2=4, tj=ATT_S/DIM/2=4, tk=ATT_D/DIM=4
  {
    int ti = ATT_S/DIM/2;
    int tj = ATT_S/DIM/2;
    int tk = ATT_D/DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM*BANK_ROWS - tk*tj*DIM;
    int SD = 128;

    gemmini_flush(0);

    // Step 1: Configure execution (fp6 E3M2: act_mx_fmt=1, wgt_mx_fmt=1, out_mx_fmt=3 BF16, uselut=1)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false,
        1, 1, 3, 1);

    // Step 2: Load scale factors (A=Q scales sel=0, B=KT scales sel=1)
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Step 3: Load LUTs (uselut=1, sel=0 for B/KT, sel=1 for A/Q)
    gemmini_mx_load_lut((uint64_t)KT_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)Q_lp, 1, 1);

    // Step 4: MVIN tiles of Q (A) and KT (B) into scratchpad
    // Load Q tiles: A layout at ab + (i*tk + k)*DIM
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)Q_hw) + i*DIM*ATT_D + k*DIM),
            ab + (i*tk + k)*DIM,
            DIM, DIM);
      }
    }

    // Load KT tiles: B layout at bb + (j*tk + k)*DIM
    // KT stored as [ATT_D/2 rows x ATT_S cols] due to fp6 packing (2 elems per byte)
    gemmini_config_ld(ATT_S * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)KT_in) + k*DIM*(ATT_S/2) + j*DIM),
            bb + (k*tj + j)*DIM,
            DIM, DIM);
      }
    }

    // Step 5: Configure requantized mvout
    gemmini_config_st((ATT_S/BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk,
        0, 0, 8);

    // Step 6: Compute (funct 24 marker is one-shot, issued inside macro)
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM*BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: Read results from mx_smem to DRAM
    gemmini_mx_read_smem(&S1_hw[0][0], SD*16, ATT_S*ATT_S);
    gemmini_fence();
  }

  // CPU softmax + fp6 requantization
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);
    for (int j = 0; j < ATT_S; j++) {
      int64_t qq = mx_ibf16_to_q((uint16_t)((S1_hw[i][j/4] >> ((j%4)*16)) & 0xFFFF));
      q[j] = qq;
      if (qq > mq) mq = qq;
    }
    int64_t p[ATT_S], sm = 0;
    for (int j = 0; j < ATT_S; j++) {
      int64_t pv = mx_iexp(q[j] - mq);
      p[j] = pv;
      sm += pv;
    }
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
    for (int g = 0; g < GKP; g++) {
      int64_t gm = 0;
      for (int j = g*32; j < g*32+32; j++)
        if (p[j] > gm) gm = p[j];
      int e = (gm > 0) ? (mx_ilog2((uint64_t)gm) - ls + 2) : 0;
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g*32; j < g*32+32; j++) {
        uint64_t R = (((uint64_t)p[j]*inv) + ((uint64_t)1 << (ls-1))) >> ls;
        uint64_t Rt = (e >= 0) ? (R >> e) : (R << (-e));
        uint8_t idx = mx_enc_P(Rt);
        if ((i & 1) == 0)
          P_hw[i>>1][j] = (P_hw[i>>1][j] & 0xF0) | idx;
        else
          P_hw[i>>1][j] = (P_hw[i>>1][j] & 0x0F) | (idx << 4);
      }
    }
  }

  // Second matmul: P x V -> O (attention output)
  // P: [ATT_S x ATT_S] fp6, V: [ATT_S x ATT_D] fp6
  // tiles: ti=ATT_S/DIM/2=4, tj=ATT_D/DIM/2=2, tk=ATT_S/DIM=8
  {
    int ti = ATT_S/DIM/2;
    int tj = ATT_D/DIM/2;
    int tk = ATT_S/DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM*BANK_ROWS - tk*tj*DIM;
    int SD = 128;

    gemmini_flush(0);

    // Step 1: Configure execution (fp6 E3M2: act_mx_fmt=1, wgt_mx_fmt=1, out_mx_fmt=3 BF16, uselut=1)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false,
        1, 1, 3, 1);

    // Step 2: Load scale factors (A=P scales sel=0, B=V scales sel=1)
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Step 3: Load LUTs (uselut=1, sel=0 for B/V, sel=1 for A/P)
    gemmini_mx_load_lut((uint64_t)V_lp, 1, 0);
    gemmini_mx_load_lut((uint64_t)P_lp, 1, 1);

    // Step 4: MVIN tiles of P (A) and V (B) into scratchpad
    // Load P tiles: A layout at ab + (i*tk + k)*DIM
    // P stored as [ATT_S/2 rows x ATT_S cols] due to fp6 packing
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)P_hw) + i*DIM*ATT_S + k*DIM),
            ab + (i*tk + k)*DIM,
            DIM, DIM);
      }
    }

    // Load V tiles: B layout at bb + (k*tj + j)*DIM
    // V stored as [ATT_S x ATT_D/2] due to fp6 packing
    gemmini_config_ld(ATT_D * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)V_in) + k*DIM*(ATT_D/2) + j*DIM),
            bb + (k*tj + j)*DIM,
            DIM, DIM);
      }
    }

    // Step 5: Configure requantized mvout
    gemmini_config_st((ATT_D/BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk,
        0, 0, 8);

    // Step 6: Compute (funct 24 marker is one-shot, issued inside macro)
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM*BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: Read results from mx_smem to DRAM
    gemmini_mx_read_smem(&O_hw[0][0], SD*16, ATT_S*ATT_D);
    gemmini_fence();
  }
}
