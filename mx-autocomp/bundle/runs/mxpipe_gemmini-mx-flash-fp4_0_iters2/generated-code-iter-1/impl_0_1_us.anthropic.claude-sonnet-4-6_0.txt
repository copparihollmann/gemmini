void solution(void) {
  // Initialize accumulators
  for(int i = 0; i < ATT_S; i++) {
    Mr[i] = -((int64_t)1 << 62);
    Lr[i] = 0;
    for(int d = 0; d < ATT_D; d++) O_acc[i][d] = 0;
  }

  // Tile dimensions for QK matmul: Q(ATT_S x ATT_D) * KT(ATT_D x BT)
  // FP4: 2 elements per byte, so effective row width halved
  const int qk_ti = ATT_S / DIM / 2;   // = 128/16/2 = 4
  const int qk_tj = BT / DIM / 2;       // = BT/16/2
  const int qk_tk = ATT_D / DIM;        // = 64/16 = 4

  // Tile dimensions for PV matmul: P(ATT_S x BT) * V(BT x ATT_D)
  const int pv_ti = ATT_S / DIM / 2;   // = 4
  const int pv_tj = ATT_D / DIM / 2;   // = 64/16/2 = 2
  const int pv_tk = BT / DIM;           // = BT/16

  // Scratchpad base addresses
  const uint32_t qk_ab = 0;
  const uint32_t qk_bb = BANK_NUM * BANK_ROWS - qk_tk * qk_tj * DIM;
  const uint32_t pv_ab = 0;
  const uint32_t pv_bb = BANK_NUM * BANK_ROWS - pv_tk * pv_tj * DIM;

  // Both matmuls use the same format config (FP4 E2M1 in, BF16 out, no LUT)
  // Hoist outside loop since config is identical for both matmuls
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false,
      /*act_mx_fmt*/2, /*wgt_mx_fmt*/2, /*out_mx_fmt*/3, /*uselut*/0);

  // Hoist config_st for QK output (stride = BT/BF16_PER_WORD * sizeof(out_t))
  // Will reconfigure before PV matmul

  gemmini_flush(0);

  for(int t = 0; t < NT; t++) {
    // ----------------------------------------------------------------
    // Step 1: QK matmul: S = Q * KT  (FP4, output BF16)
    // ----------------------------------------------------------------

    // Build KT scale array for this block
    for(int g = 0; g < ATT_D/32; g++)
      for(int n = 0; n < BT; n++)
        KTsc[g * BT + n] = KT_scales_col[g][t * BT + n];

    // Load scales: A=Q_scales_row (sel=0), B=KT scales (sel=1)
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)KTsc, (ATT_D / 32) * BT, 1);

    // Configure output stride for QK result
    gemmini_config_st((BT / BF16_PER_WORD) * sizeof(out_t));

    // Configure requantizer bounds
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        qk_ti, qk_tj, qk_tk,
        /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

    // MVIN Q tiles (A matrix): layout a_base + (i*tk + k)*DIM
    // Q is ATT_S rows x ATT_D cols (fp4: stride = ATT_D bytes for packed fp4)
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for(int i = 0; i < qk_ti; i++) {
      for(int k = 0; k < qk_tk; k += MAX_BLOCK_LEN) {
        int blk = (qk_tk - k < MAX_BLOCK_LEN) ? (qk_tk - k) : MAX_BLOCK_LEN;
        for(int kb = 0; kb < blk; kb++) {
          int kk = k + kb;
          gemmini_extended_mvin(
              (void*)(((elem_t*)Q_hw) + i * DIM * ATT_D + kk * DIM),
              qk_ab + (i * qk_tk + kk) * DIM,
              DIM, DIM);
        }
      }
    }

    // MVIN KT tiles (B matrix): layout b_base + (k*tj + j)*DIM
    // KT is stored transposed: shape (ATT_D x ATT_S), fp4 packed
    // stride along j-axis: ATT_S/2 bytes (fp4 packed)
    gemmini_config_ld(ATT_S * sizeof(elem_t) / 2);
    for(int k = 0; k < qk_tk; k++) {
      for(int j = 0; j < qk_tj; j += MAX_BLOCK_LEN) {
        int blk = (qk_tj - j < MAX_BLOCK_LEN) ? (qk_tj - j) : MAX_BLOCK_LEN;
        for(int jb = 0; jb < blk; jb++) {
          int jj = j + jb;
          gemmini_extended_mvin2(
              (void*)(((elem_t*)KT_in) + k * DIM * (ATT_S / 2) + (t * BT) / 2 + jj * DIM),
              qk_bb + (k * qk_tj + jj) * DIM,
              DIM, DIM);
        }
      }
    }

    // Compute QK matmul
    gemmini_loop_ws_spad(qk_ti, qk_tj, qk_tk, 0, 0, 0,
        qk_ab, BANK_NUM * BANK_ROWS, 0, 128,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Drain QK results to S1_hw
    gemmini_mx_read_smem(&S1_hw[0][0], 128 * 16, ATT_S * BT);
    gemmini_fence();

    // ----------------------------------------------------------------
    // Step 2: Online softmax on CPU + encode P in FP4
    // ----------------------------------------------------------------
    for(int i = 0; i < ATT_S; i++) {
      int64_t q[BT];
      int64_t mn = Mr[i];

      // Extract BF16 scores and find max
      for(int j = 0; j < BT; j++) {
        int64_t qq = mx_ibf16_to_q((uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF));
        q[j] = qq;
        if(qq > mn) mn = qq;
      }

      // Correction factor for previous max
      int64_t corr = mx_iexp(Mr[i] - mn);

      // Compute softmax probabilities
      int64_t p[BT];
      int64_t st = 0;
      for(int j = 0; j < BT; j++) {
        int64_t pv = mx_iexp(q[j] - mn);
        p[j] = pv;
        st += pv;
      }

      // Update running sum and output accumulator
      Lr[i] = (Lr[i] * corr) / MX_IEXP0 + st;
      for(int d = 0; d < ATT_D; d++)
        O_acc[i][d] = (O_acc[i][d] * corr) / MX_IEXP0;
      Mr[i] = mn;

      // Quantize probabilities to FP4 and store P_hw, P_scales
      for(int g = 0; g < GBT; g++) {
        int64_t gm = 0;
        for(int j = g * 32; j < g * 32 + 32; j++)
          if(p[j] > gm) gm = p[j];

        int e = (gm > 0) ? (mx_ilog2((uint64_t)gm) - 2) : 0;
        P_scales[g][i] = (uint8_t)(e + 127);

        for(int j = g * 32; j < g * 32 + 32; j++) {
          uint64_t Rt = (e >= 0) ?
              (((uint64_t)p[j] << MX_FRAC) >> e) :
              ((uint64_t)p[j] << (MX_FRAC - e));
          uint8_t idx = mx_enc_P(Rt);
          if((i & 1) == 0)
            P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0xF0) | idx;
          else
            P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0x0F) | (idx << 4);
        }
      }
    }

    // ----------------------------------------------------------------
    // Step 3: PV matmul: O_tile = P * V  (FP4, output BF16)
    // ----------------------------------------------------------------

    // Load scales: A=P_scales (sel=0), B=V_scales_col for this block (sel=1)
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col[t * GBT][0], GBT * ATT_D, 1);

    // Configure output stride for PV result
    gemmini_config_st((ATT_D / BF16_PER_WORD) * sizeof(out_t));

    // Configure requantizer bounds
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        pv_ti, pv_tj, pv_tk,
        /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

    // MVIN P tiles (A matrix): layout a_base + (i*tk + k)*DIM
    // P is (ATT_S/2 x BT) in bytes (fp4 packed, 2 rows per byte-row)
    gemmini_config_ld(BT * sizeof(elem_t));
    for(int i = 0; i < pv_ti; i++) {
      for(int k = 0; k < pv_tk; k += MAX_BLOCK_LEN) {
        int blk = (pv_tk - k < MAX_BLOCK_LEN) ? (pv_tk - k) : MAX_BLOCK_LEN;
        for(int kb = 0; kb < blk; kb++) {
          int kk = k + kb;
          gemmini_extended_mvin(
              (void*)(((elem_t*)P_hw) + i * DIM * BT + kk * DIM),
              pv_ab + (i * pv_tk + kk) * DIM,
              DIM, DIM);
        }
      }
    }

    // MVIN V tiles (B matrix): layout b_base + (k*tj + j)*DIM
    // V is (BT x ATT_D/2) in bytes (fp4 packed)
    gemmini_config_ld(ATT_D * sizeof(elem_t) / 2);
    for(int k = 0; k < pv_tk; k++) {
      for(int j = 0; j < pv_tj; j += MAX_BLOCK_LEN) {
        int blk = (pv_tj - j < MAX_BLOCK_LEN) ? (pv_tj - j) : MAX_BLOCK_LEN;
        for(int jb = 0; jb < blk; jb++) {
          int jj = j + jb;
          gemmini_extended_mvin2(
              (void*)(((elem_t*)V_in) + (t * BT + k * DIM) * (ATT_D / 2) + jj * DIM),
              pv_bb + (k * pv_tj + jj) * DIM,
              DIM, DIM);
        }
      }
    }

    // Compute PV matmul
    gemmini_loop_ws_spad(pv_ti, pv_tj, pv_tk, 0, 0, 0,
        pv_ab, BANK_NUM * BANK_ROWS, 0, 128,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Drain PV results to O_hw
    gemmini_mx_read_smem(&O_hw[0][0], 128 * 16, ATT_S * ATT_D);
    gemmini_fence();

    // Accumulate BF16 outputs into O_acc
    for(int i = 0; i < ATT_S; i++)
      for(int d = 0; d < ATT_D; d++) {
        uint16_t b = (uint16_t)((O_hw[i][d / 4] >> ((d % 4) * 16)) & 0xFFFF);
        O_acc[i][d] += mx_bf16_to_fx(b, MX_O_F);
      }
  }

  // Final normalization
  for(int i = 0; i < ATT_S; i++) {
    int64_t l = Lr[i];
    if(l == 0) l = 1;
    for(int d = 0; d < ATT_D; d++)
      O_acc[i][d] = O_acc[i][d] / l;
  }
}
