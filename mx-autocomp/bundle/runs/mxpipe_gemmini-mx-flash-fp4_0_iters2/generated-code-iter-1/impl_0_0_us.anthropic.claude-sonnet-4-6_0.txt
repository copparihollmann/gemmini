void solution(void) {
  // Initialize accumulators
  for(int i = 0; i < ATT_S; i++) {
    Mr[i] = -((int64_t)1 << 62);
    Lr[i] = 0;
    for(int d = 0; d < ATT_D; d++) O_acc[i][d] = 0;
  }

  for(int t = 0; t < NT; t++) {
    // ── QK^T matmul ──────────────────────────────────────────────────────────
    {
      int ti = ATT_S / DIM / 2;   // = 4
      int tj = BT / DIM / 2;      // = BT/32
      int tk = ATT_D / DIM;       // = 4
      uint32_t ab = 0;
      uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;

      // Pack KT scales for this token block
      for(int g = 0; g < ATT_D / 32; g++)
        for(int n = 0; n < BT; n++)
          KTsc[g * BT + n] = KT_scales_col[g][t * BT + n];

      gemmini_flush(0);

      // Step 1: Configure execution (fp4 E2M1 for both A and B, BF16 output)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          1, 1, 0, 0, false,
          /*act_mx_fmt*/2, /*wgt_mx_fmt*/2, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: Load scale factors (A=0, B=1)
      gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
      gemmini_mx_load_scales((uint64_t)KTsc, (ATT_D / 32) * BT, 1);

      // Step 4: MVIN tiles of Q (A) into scratchpad
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for(int i = 0; i < ti; i++)
        for(int k = 0; k < tk; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)Q_hw) + i * DIM * ATT_D + k * DIM),
              ab + (i * tk + k) * DIM,
              DIM, DIM);

      // Step 4: MVIN tiles of KT (B) into scratchpad
      // KT is stored transposed; fp4 packs 2 elements per byte -> stride /2
      gemmini_config_ld(ATT_S * sizeof(elem_t) / 2);
      for(int k = 0; k < tk; k++)
        for(int j = 0; j < tj; j++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)KT_in) + k * DIM * (ATT_S / 2) + (t * BT) / 2 + j * DIM),
              bb + (k * tj + j) * DIM,
              DIM, DIM);

      // Step 5: Configure requantized mvout
      gemmini_config_st((BT / BF16_PER_WORD) * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          ti, tj, tk,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // Step 6: Compute (funct 24 is re-issued inside gemmini_loop_ws_spad)
      gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
          ab, BANK_NUM * BANK_ROWS, 0, 128,
          false, false, false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain results
      gemmini_mx_read_smem(&S1_hw[0][0], 128 * 16, ATT_S * BT);
      gemmini_fence();
    }

    // ── CPU-side online softmax + fp4 requant ────────────────────────────────
    for(int i = 0; i < ATT_S; i++) {
      int64_t q[BT];
      int64_t mn = Mr[i];

      // Find local max
      for(int j = 0; j < BT; j++) {
        int64_t qq = mx_ibf16_to_q(
            (uint16_t)((S1_hw[i][j / 4] >> ((j % 4) * 16)) & 0xFFFF));
        q[j] = qq;
        if(qq > mn) mn = qq;
      }

      // Compute exp values and partial sum
      int64_t corr = mx_iexp(Mr[i] - mn);
      int64_t p[BT];
      int64_t st = 0;
      for(int j = 0; j < BT; j++) {
        int64_t pv = mx_iexp(q[j] - mn);
        p[j] = pv;
        st += pv;
      }

      // Update running statistics
      Lr[i] = (Lr[i] * corr) / MX_IEXP0 + st;
      for(int d = 0; d < ATT_D; d++)
        O_acc[i][d] = (O_acc[i][d] * corr) / MX_IEXP0;
      Mr[i] = mn;

      // Requantize p into fp4 P_hw with per-group scales
      for(int g = 0; g < GBT; g++) {
        int64_t gm = 0;
        for(int j = g * 32; j < g * 32 + 32; j++)
          if(p[j] > gm) gm = p[j];

        int e = (gm > 0) ? (mx_ilog2((uint64_t)gm) - 2) : 0;
        P_scales[g][i] = (uint8_t)(e + 127);

        for(int j = g * 32; j < g * 32 + 32; j++) {
          uint64_t Rt = (e >= 0)
              ? (((uint64_t)p[j] << MX_FRAC) >> e)
              : ((uint64_t)p[j] << (MX_FRAC - e));
          uint8_t idx = mx_enc_P(Rt);
          if((i & 1) == 0)
            P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0xF0) | idx;
          else
            P_hw[i >> 1][j] = (P_hw[i >> 1][j] & 0x0F) | (idx << 4);
        }
      }
    }

    // ── PV matmul ─────────────────────────────────────────────────────────────
    {
      int ti = ATT_S / DIM / 2;    // = 4
      int tj = ATT_D / DIM / 2;    // = 2
      int tk = BT / DIM;           // = BT/16
      uint32_t ab = 0;
      uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;

      gemmini_flush(0);

      // Step 1: Configure execution (fp4 E2M1, BF16 output)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
          1, 1, 0, 0, false,
          /*act_mx_fmt*/2, /*wgt_mx_fmt*/2, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: Load scale factors (A=P_scales, B=V_scales)
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
      gemmini_mx_load_scales((uint64_t)&V_scales_col[t * GBT][0],
          GBT * ATT_D, 1);

      // Step 4: MVIN tiles of P (A) into scratchpad
      gemmini_config_ld(BT * sizeof(elem_t));
      for(int i = 0; i < ti; i++)
        for(int k = 0; k < tk; k++)
          gemmini_extended_mvin(
              (void*)(((elem_t*)P_hw) + i * DIM * BT + k * DIM),
              ab + (i * tk + k) * DIM,
              DIM, DIM);

      // Step 4: MVIN tiles of V (B) into scratchpad
      // V is fp4, packed 2 elements per byte -> stride /2
      gemmini_config_ld(ATT_D * sizeof(elem_t) / 2);
      for(int k = 0; k < tk; k++)
        for(int j = 0; j < tj; j++)
          gemmini_extended_mvin2(
              (void*)(((elem_t*)V_in) + (t * BT + k * DIM) * (ATT_D / 2) + j * DIM),
              bb + (k * tj + j) * DIM,
              DIM, DIM);

      // Step 5: Configure requantized mvout
      gemmini_config_st((ATT_D / BF16_PER_WORD) * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          ti, tj, tk,
          /*scale_act_sel*/0, /*scale_w_sel*/0,
          /*lut_update_granularity*/1);

      // Step 6: Compute
      gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
          ab, BANK_NUM * BANK_ROWS, 0, 128,
          false, false, false, false, false,
          NO_ACTIVATION, 0, 0, false, 0x38);

      // Step 7: Drain results
      gemmini_mx_read_smem(&O_hw[0][0], 128 * 16, ATT_S * ATT_D);
      gemmini_fence();

      // Accumulate O_hw (BF16) into O_acc (fixed-point)
      for(int i = 0; i < ATT_S; i++)
        for(int d = 0; d < ATT_D; d++) {
          uint16_t b = (uint16_t)(
              (O_hw[i][d / 4] >> ((d % 4) * 16)) & 0xFFFF);
          O_acc[i][d] += mx_bf16_to_fx(b, MX_O_F);
        }
    }
  } // end token loop

  // Final normalization
  for(int i = 0; i < ATT_S; i++) {
    int64_t l = Lr[i];
    if(l == 0) l = 1;
    for(int d = 0; d < ATT_D; d++)
      O_acc[i][d] = O_acc[i][d] / l;
  }
}
