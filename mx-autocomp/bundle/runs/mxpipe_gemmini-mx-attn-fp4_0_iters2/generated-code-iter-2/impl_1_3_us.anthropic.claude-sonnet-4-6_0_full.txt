Looking at the original code, I need to understand the structure:

1. **Phase 1**: Q × KT → S1_hw (FP4 matmul, ti=4, tj=4, tk=4)
2. **Softmax + FP4 requant**: CPU-side computation
3. **Phase 2**: P × V → O_hw (FP4 matmul, ti=4, tj=2, tk=8)

The original code has V tiles pre-loaded before softmax runs, but then Phase 2 does a `gemmini_flush(0)` which would invalidate scratchpad state. The V pre-load in the original is between Phase 1 and Phase 2, and Phase 2 re-issues `gemmini_extended3_config_ex` and loads P tiles - but notably does NOT reload V tiles, relying on the pre-loaded state. However, `gemmini_flush(0)` resets state, so V must be reloaded. Let me trace carefully: Phase 2 loads P (A matrix) but not V (B matrix) - the pre-load block loaded V. But then `gemmini_flush(0)` is called in Phase 2... This means the original code may rely on flush not clearing scratchpad data, only the pipeline state.

The key optimization opportunity: overlapping the V DMA with the CPU softmax computation is already done. The main issue is correctness and following the programming model.

Let me rewrite preserving semantics while ensuring correct ordering:

```c
void solution(void) {
  // Phase 1: Q x KT -> S1_hw
  {
    const int ti = ATT_S/DIM/2;
    const int tj = ATT_S/DIM/2;
    const int tk = ATT_D/DIM;
    const uint32_t ab = 0;
    const uint32_t bb = BANK_NUM*BANK_ROWS - tk*tj*DIM;
    const int SD = 128;

    gemmini_flush(0);

    // Step 1: config_ex with FP4 format
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 2, 2, 3, 0);

    // Step 2: load scales (A=Q scales sel=0, B=KT scales sel=1)
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);

    // Step 4: mvin A tiles (Q) using mvin port, B tiles (KT) using mvin2 port
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)Q_hw) + i*DIM*ATT_D + k*DIM),
            ab + (i*tk + k)*DIM,
            DIM, DIM);
      }
    }

    gemmini_extended3_config_ld(ATT_S * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)KT_in) + k*DIM*(ATT_S/2) + j*DIM),
            bb + (k*tj + j)*DIM,
            DIM, DIM);
      }
    }

    // Step 5: config_st and mxquant_config_mvout
    gemmini_config_st((ATT_S / BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk, 0, 0, 1);

    // Step 6: compute
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM*BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: drain results
    gemmini_mx_read_smem(&S1_hw[0][0], SD*16, ATT_S*ATT_S);
    gemmini_fence();
  }

  // Preload V tiles into scratchpad while CPU runs softmax
  // Phase 2 parameters: ti2=ATT_S/DIM/2=4, tj2=ATT_D/DIM/2=2, tk2=ATT_S/DIM=8
  {
    const int tj2 = ATT_D/DIM/2;
    const int tk2 = ATT_S/DIM;
    const uint32_t bb2 = BANK_NUM*BANK_ROWS - tk2*tj2*DIM;

    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tk2; k++) {
      for (int j = 0; j < tj2; j++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)V_in) + k*DIM*(ATT_D/2) + j*DIM),
            bb2 + (k*tj2 + j)*DIM,
            DIM, DIM);
      }
    }
  }

  // Host CPU softmax + FP4 requant (runs concurrently with V DMA above)
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);
    for (int j = 0; j < ATT_S; j++) {
      int64_t qq = mx_ibf16_to_q((uint16_t)((S1_hw[i][j/4] >> ((j%4)*16)) & 0xFFFF));
      q[j] = qq;
      if (qq > mq) mq = qq;
    }
    int64_t p[ATT_S];
    int64_t sm = 0;
    for (int j = 0; j < ATT_S; j++) {
      int64_t pv = mx_iexp(q[j] - mq);
      p[j] = pv;
      sm += pv;
    }
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
    for (int g = 0; g < GKP; g++) {
      int64_t gm = 0;
      for (int j = g*32; j < g*32+32; j++)
        if (p[j] > gm) gm = p[j];
      int e = (gm > 0) ? (mx_ilog2((uint64_t)gm) - ls + 0) : 0;
      P_scales[g][i] = (uint8_t)(e + 127);
      for (int j = g*32; j < g*32+32; j++) {
        uint64_t R = (((uint64_t)p[j] * inv) + ((uint64_t)1 << (ls-1))) >> ls;
        uint64_t Rt = (e >= 0) ? (R >> e) : (R << (-e));
        uint8_t idx = mx_enc_P(Rt);
        if ((i & 1) == 0)
          P_hw[i>>1][j] = (P_hw[i>>1][j] & 0xF0) | idx;
        else
          P_hw[i>>1][j] = (P_hw[i>>1][j] & 0x0F) | (idx << 4);
      }
    }
  }

  // Phase 2: P x V -> O_hw (V tiles already in scratchpad from preload above)
  {
    const int ti = ATT_S/DIM/2;
    const int tj = ATT_D/DIM/2;
    const int tk = ATT_S/DIM;
    const uint32_t ab = 0;
    const uint32_t bb = BANK_NUM*BANK_ROWS - tk*tj*DIM;
    const int SD = 128;

    gemmini_flush(0);

    // Step 1: config_ex with FP4 format
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        1, 1, 0, 0, false, 2, 2, 3, 0);

    // Step 2: load scales (A=P scales sel=0, B=V scales sel=1)
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Step 4: mvin A tiles (P) - V tiles are already in scratchpad at bb
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)P_hw) + i*DIM*ATT_S + k*DIM),
            ab + (i*tk + k)*DIM,
            DIM, DIM);
      }
    }

    // Reload V tiles (flush cleared scratchpad pipeline; must reload after flush)
    gemmini_extended3_config_ld(ATT_D * sizeof(elem_t) / 2, MVIN_SCALE_IDENTITY, false, 1);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)V_in) + k*DIM*(ATT_D/2) + j*DIM),
            bb + (k*tj + j)*DIM,
            DIM, DIM);
      }
    }

    // Step 5: config_st and mxquant_config_mvout
    gemmini_config_st((ATT_D / BF16_PER_WORD) * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        ti, tj, tk, 0, 0, 1);

    // Step 6: compute
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0,
        ab, BANK_NUM*BANK_ROWS, 0, SD,
        false, false, false, false, false,
        NO_ACTIVATION, 0, 0, false, 0x38);

    // Step 7: drain results
    gemmini_mx_read_smem(&O_hw[0][0], SD*16, ATT_S*ATT_D);
    gemmini_fence();
  }
}
```
