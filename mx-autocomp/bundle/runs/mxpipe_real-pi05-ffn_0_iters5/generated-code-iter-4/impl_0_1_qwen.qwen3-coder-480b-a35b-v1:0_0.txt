#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Assuming external symbols are defined elsewhere as in the baseline:
// BLK, REAL_M, REAL_N, MATMUL_K, DIM, BANK_NUM, BANK_ROWS, BF16_PER_WORD, BLK_OUTC
// elem_t, out_t, A_s, B_s, A_scales_row, B_scales_col, A_in, B_in, C_hw, scale_factors

void solution(void) {
  // Hoist invariant Gemmini configuration outside of all loops to prevent redundant command dispatch
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Check if the entire matrix can fit in accumulator memory for single-pass optimization
  // Constraint: tiles_I * tiles_J * DIM <= 256 rows (ACC_ROWS)
  int tiles_I = (REAL_M + DIM - 1) / DIM;  // Ceiling division
  int tiles_J = (REAL_N + DIM - 1) / DIM;  // Ceiling division
  
  if (tiles_I * tiles_J <= 16) {  // 16 = 256/16 (ACC_ROWS/DIM)
    // Single-pass optimization: avoid CPU staging of scale factors
    int exact_tiles_I = REAL_M / DIM;
    int exact_tiles_J = REAL_N / DIM;
    int tiles_K = MATMUL_K / DIM;
    
    // Handle padding for non-multiple dimensions
    int pad_I = REAL_M % DIM ? 1 : 0;
    int pad_J = REAL_N % DIM ? 1 : 0;
    
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - (exact_tiles_J + pad_J) * tiles_K * DIM;
    int SPAD_DEST = 128;

    // Direct streaming of scale factors - no CPU memcpy needed
    // A_scales_row has shape [MATMUL_K/32][REAL_M]
    gemmini_mx_load_scales((uint64_t)A_scales_row, (MATMUL_K / 32) * REAL_M, 0);
    // B_scales_col has shape [MATMUL_K/32][REAL_N]
    gemmini_mx_load_scales((uint64_t)B_scales_col, (MATMUL_K / 32) * REAL_N, 1);

    // Stream A tiles directly into the scratchpad
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < exact_tiles_I; i++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
            (void *)(((elem_t *)A_in) + (i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      }
    }
    
    // Handle partial M tile if needed
    if (REAL_M % DIM) {
      int i = exact_tiles_I;
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
            (void *)(((elem_t *)A_in) + (i * DIM) * MATMUL_K + k * DIM),
            a_base + (i * tiles_K + k) * DIM, REAL_M % DIM, DIM);
      }
    }

    // Stream B tiles directly into the scratchpad
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      for (int j = 0; j < exact_tiles_J; j++) {
        gemmini_extended_mvin(
            (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j * DIM)),
            b_base + (k * (exact_tiles_J + pad_J) + j) * DIM, DIM, DIM);
      }
    }
    
    // Handle partial N tile if needed
    if (REAL_N % DIM) {
      int j = exact_tiles_J;
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(
            (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j * DIM)),
            b_base + (k * (exact_tiles_J + pad_J) + j) * DIM, DIM, REAL_N % DIM);
      }
    }

    // Configure the output writer
    gemmini_config_st(REAL_N * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, 
                                 exact_tiles_I + pad_I, 
                                 exact_tiles_J + pad_J, 
                                 tiles_K, 0, 0, 1);

    // Compute the tiled matrix multiplication entirely in hardware
    gemmini_loop_ws_spad(exact_tiles_I + pad_I, exact_tiles_J + pad_J, tiles_K, 
                         REAL_M % DIM ? DIM - (REAL_M % DIM) : 0,
                         REAL_N % DIM ? DIM - (REAL_N % DIM) : 0,
                         0, a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST, 
                         false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Stream results from mx_smem directly to DRAM
    for (int r = 0; r < REAL_M; r++) {
      gemmini_mx_read_smem(&C_hw[r][0], SPAD_DEST * 16 + r * REAL_N, REAL_N);
    }
    gemmini_fence();

  } else {
    // Fallback to the original robust blocked loop for oversized matrices
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;

      int tiles = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;

      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], mrows);
      }
      gemmini_mx_load_scales((uint64_t)A_s, (MATMUL_K / 32) * mrows, 0);

      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          int rows = DIM;
          if (i0 * BLK + i * DIM + DIM > REAL_M) {
            rows = REAL_M - (i0 * BLK + i * DIM);
          }
          if (rows <= 0) continue;
          
          gemmini_extended_mvin(
              (void *)(((elem_t *)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, rows);
        }
      }

      for (int j0 = 0; j0 < BLOCKS_N; j0++) {
        int ncols = REAL_N - j0 * BLK;
        if (ncols > BLK) ncols = BLK;
        if (ncols <= 0) continue;

        for (int g = 0; g < MATMUL_K / 32; g++) {
          memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], ncols);
        }

        int pad_i = BLK - mrows;
        int pad_j = BLK - ncols;

        gemmini_flush(0);
        gemmini_mx_load_scales((uint64_t)B_s, (MATMUL_K / 32) * ncols, 1);

        gemmini_config_ld(MATMUL_N * sizeof(elem_t));
        for (int j = 0; j < tiles; j++) {
          for (int k = 0; k < tiles_K; k++) {
            int cols = DIM;
            if (j0 * BLK + j * DIM + DIM > REAL_N) {
              cols = REAL_N - (j0 * BLK + j * DIM);
            }
            if (cols <= 0) continue;
            
            gemmini_extended_mvin(
                (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                b_base + (k * tiles + j) * DIM, cols, DIM);
          }
        }

        gemmini_config_st(BLK_OUTC * sizeof(out_t));
        gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
        gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base,
                             BANK_NUM * BANK_ROWS, 0, SPAD_DEST, false, false, false, false,
                             false, NO_ACTIVATION, 0, 0, false, 0x38);

        for (int r = 0; r < mrows; r++) {
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK],
                               SPAD_DEST * 16 + r * BLK, ncols);
        }
        gemmini_fence();
      }
    }
  }
}
