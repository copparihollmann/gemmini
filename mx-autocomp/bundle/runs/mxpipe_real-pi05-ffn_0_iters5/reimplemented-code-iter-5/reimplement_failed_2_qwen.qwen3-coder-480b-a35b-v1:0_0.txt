#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Assuming the following definitions are provided by the environment/headers:
// #define DIM 16
// #define MATMUL_M 256
// #define MATMUL_N 256
// #define MATMUL_K 512
// #define BLK 64
// #define BLOCKS_M (REAL_M / BLK)
// #define BLOCKS_N (REAL_N / BLK)
// #define BF16_PER_WORD 4
// typedef uint8_t elem_t;
// typedef uint64_t out_t;

void solution(void) {
  // Global Accel Init
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles = BLK / DIM;  // 64/16 = 4 tiles per block
  int tiles_K = MATMUL_K / DIM;  // 512/16 = 32 tiles along K
  
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;  // Place B at top of scratchpad
  int SPAD_DEST = 128;

  // Outer loop: iterate over B block columns
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK; 
    if (ncols > BLK) ncols = BLK; 
    if (ncols <= 0) continue;

    int pad_j = BLK - ncols;
    int tiles_J = (ncols + DIM - 1) / DIM;  // Ceiling division for actual tiles needed

    // Load B scales once per outer block column
    gemmini_mx_load_scales((uint64_t)&B_scales_col[0][j0 * BLK], B_S_BYTES, 1);

    // MVIN B: Load B into the top partition of scratchpad
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));      // B rows: stride N
    for (int k = 0; k < tiles_K; k++) {
      for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {  // Use MAX_BLOCK_LEN for batching
        int block_len = (j + MAX_BLOCK_LEN > tiles_J) ? (tiles_J - j) : MAX_BLOCK_LEN;
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                               b_base + (j * tiles_K + k) * DIM, block_len * DIM, DIM);
      }
    }

    // Inner loop: iterate over A block rows
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK; 
      if (mrows > BLK) mrows = BLK; 
      if (mrows <= 0) continue;

      int pad_i = BLK - mrows;
      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division for actual tiles needed

      // Load A scales per iteration
      gemmini_mx_load_scales((uint64_t)&A_scales_row[0][i0 * BLK], A_S_BYTES, 0);

      // MVIN A: Load A into the bottom partition of scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));      // A rows: stride K
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {  // Use MAX_BLOCK_LEN for batching
          int block_len = (k + MAX_BLOCK_LEN > tiles_K) ? (tiles_K - k) : MAX_BLOCK_LEN;
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, block_len * DIM, DIM);
        }
      }

      // Configure output strides and quantization bounds for this specific tile
      gemmini_config_st(MATMUL_N * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // Execute Compute
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Output Readout directly from smem
      gemmini_mx_read_smem(&C_hw[i0 * BLK][j0 * BLK], SPAD_DEST * 16, mrows * ncols);
      gemmini_fence();
    }
  }
}
