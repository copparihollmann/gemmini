#include <stdint.h>
#include <stdbool.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Assuming external symbols are defined elsewhere as in the baseline:
// BLK, REAL_M, REAL_N, MATMUL_K, DIM, BANK_NUM, BANK_ROWS, BF16_PER_WORD, BLK_OUTC
// elem_t, out_t, A_scales_row, B_scales_col, A_in, B_in, C_hw, scale_factors

void solution(void) {
  // Hoist invariant Gemmini configuration outside of all loops to prevent redundant command dispatch
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Setup double-buffering for mx_smem
  // Buffer 0 is at row 0, Buffer 1 is at row 256 (64 * 64 / 16 = 256 rows per tile)
  int spad_dests[2] = {0, 256};
  int buf_idx = 0;

  bool has_prev = false;
  int prev_i0 = 0, prev_j0 = 0, prev_mrows = 0, prev_ncols = 0, prev_spad_dest = 0;

  int tiles_I = (BLK + DIM - 1) / DIM;  // Ceiling division
  int tiles_J = (BLK + DIM - 1) / DIM;  // Ceiling division
  int tiles_K = (MATMUL_K + DIM - 1) / DIM;  // Ceiling division

  for (int i0 = 0; i0 < (REAL_M + BLK - 1) / BLK; i0++) {
    // Partial last M block: compute only the real rows, pad the rest.
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    int curr_tiles_I = (mrows + DIM - 1) / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;

    // Load activation scale factors to Gemmini hardware scale memory
    // Scale layout: scale_dram + m * N_blocks + bi (row-major, block-indexed)
    gemmini_mx_load_scales((uint64_t)&A_scales_row[0][i0 * BLK], 
                          ((MATMUL_K + 31) / 32) * 32, 0); // Round up to nearest 32 for proper alignment

    // Load A tiles into the scratchpad once per row block
    gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // A rows: stride K (full block-aligned tiles)
    for (int i = 0; i < curr_tiles_I; i++) {
      for (int k = 0; k < tiles_K; k++) {
        int a_row = i0 * BLK + i * DIM;
        if (a_row < REAL_M) {
          gemmini_extended_mvin(
              (void *)(((elem_t *)A_in) + a_row * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        } else {
          // Handle padding rows - load zeros or skip
          gemmini_extended_mvin(
              (void *)(((elem_t *)A_in) + (REAL_M - 1) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }
    }

    for (int j0 = 0; j0 < (REAL_N + BLK - 1) / BLK; j0++) {
      // Partial last N block: compute only the real cols, pad the rest.
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      int curr_tiles_J = (ncols + DIM - 1) / DIM;
      int pad_i = curr_tiles_I * DIM - mrows;
      int pad_j = curr_tiles_J * DIM - ncols;

      // Accelerator operations for each j0 block
      gemmini_flush(0);
      
      // Load weight scale factors for the current column block
      gemmini_mx_load_scales((uint64_t)&B_scales_col[0][j0 * BLK], 
                            ((MATMUL_K + 31) / 32) * 32, 1);

      gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // B rows: stride N
      for (int j = 0; j < curr_tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
              (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      int curr_spad_dest = spad_dests[buf_idx];

      gemmini_config_st(BLK * sizeof(out_t)); // Set stride properly
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, 
                                  curr_tiles_I, curr_tiles_J, tiles_K, 0, 0, 1);
      
      gemmini_loop_ws_spad(curr_tiles_I, curr_tiles_J, tiles_K, 
                           pad_i, pad_j, 0, a_base,
                           b_base, 0, curr_spad_dest, 
                           false, false, false, false,
                           false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Overlap: Issue the read out of SMEM to DRAM for the PREVIOUS block
      if (has_prev) {
        for (int r = 0; r < prev_mrows; r++) {
          gemmini_mx_read_smem((uint64_t)&C_hw[prev_i0 * BLK + r][j0 * BLK / BF16_PER_WORD],
                               prev_spad_dest * 16 + r * BLK, prev_ncols);
        }
        // Fence only the previous read transfer
        gemmini_fence();
      }

      // Record current block parameters as the previous block for the next iteration
      prev_i0 = i0;
      prev_j0 = j0;
      prev_mrows = mrows;
      prev_ncols = ncols;
      prev_spad_dest = curr_spad_dest;
      has_prev = true;

      // Toggle active buffer
      buf_idx = 1 - buf_idx;
    }
  }

  // Drain the final block out of SMEM to DRAM
  if (has_prev) {
    for (int r = 0; r < prev_mrows; r++) {
      gemmini_mx_read_smem((uint64_t)&C_hw[prev_i0 * BLK + r][prev_j0 * BLK / BF16_PER_WORD],
                           prev_spad_dest * 16 + r * BLK, prev_ncols);
    }
    gemmini_fence();
  }
}
