#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Ensure correct parameters based on the architecture summary
#define DIM 16
#define BF16_PER_WORD 4

void solution(void) {
  // Pre-allocated packaged scale buffers
  // BLK_M = 128, BLK_N = 32
  // BLOCKS_M = 256/128 = 2, BLOCKS_N = 256/32 = 8
  // GK = MATMUL_K / 32 = 512/32 = 16
  static uint8_t A_scales_packaged[2][16][128];
  static uint8_t B_scales_packaged[8][16][32];

  // 1. Pre-package scale factors once to eliminate CPU-bound copying in the outer loop
  for (int i0 = 0; i0 < 2; i0++) {
    for (int g = 0; g < 16; g++) {
      memcpy(&A_scales_packaged[i0][g][0], &A_scales_row[g][i0 * 128], 128);
    }
  }
  for (int j0 = 0; j0 < 8; j0++) {
    for (int g = 0; g < 16; g++) {
      memcpy(&B_scales_packaged[j0][g][0], &B_scales_col[g][j0 * 32], 32);
    }
  }

  int tiles_I = 8;   // 128 / DIM
  int tiles_J = 2;   // 32 / DIM
  int tiles_K = 32;  // 512 / DIM

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 4 * 2048 - 32 * 2 * 16 = 8192 - 1024 = 7168
  int SPAD_DEST = 128;

  // Renamed variables to avoid macro conflicts
  int a_scale_bytes = 16 * 128;  // 16 groups * 128 elements
  int b_scale_bytes = 16 * 32;   // 16 groups * 32 elements

  gemmini_flush(0);

  // Configure weight-stationary MX execution format once
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < 2; i0++) {
    for (int j0 = 0; j0 < 8; j0++) {
      int mrows = 128;
      int ncols = 32;
      
      // Handle partial blocks at boundaries
      if (i0 * 128 + mrows > 256) mrows = 256 - i0 * 128;
      if (j0 * 32 + ncols > 256) ncols = 256 - j0 * 32;
      
      if (mrows <= 0 || ncols <= 0) continue;

      // 2. Load pre-packaged scale factors directly from contiguous memory
      gemmini_mx_load_scales((uint64_t)&A_scales_packaged[i0][0][0], a_scale_bytes, 0);
      gemmini_mx_load_scales((uint64_t)&B_scales_packaged[j0][0][0], b_scale_bytes, 1);

      // 3. MVIN Matrix A
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          if (i0 * 128 + i * DIM >= 256) continue;
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * 128 + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // 4. MVIN Matrix B
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          if (j0 * 32 + j * DIM >= 256) continue;
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * 32 + j * DIM)),
                                b_base + (k * tiles_J + j) * DIM, DIM, DIM);
        }
      }

      // 5. Configure store and requantizer
      gemmini_config_st((32 / BF16_PER_WORD) * sizeof(uint64_t)); // 8 bytes per row segment
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // 6. Compute fused WS tiled matmul
      int pad_i = 128 - mrows;
      int pad_j = 32 - ncols;
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 7. Drain SMEM directly to output
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * 128 + r][(j0 * 32) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * 32, ncols);
      }
      gemmini_fence();
    }
  }
}
