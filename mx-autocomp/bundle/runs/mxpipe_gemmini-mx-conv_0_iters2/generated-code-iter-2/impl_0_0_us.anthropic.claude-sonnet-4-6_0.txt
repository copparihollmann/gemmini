void solution(void) {
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // Configure port 0 (mvin, id=0) for A tiles
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    // Configure port 1 (mvin2, id=1) for B tiles
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

    // Tile im2col over K-blocks and immediately DMA each block:
    // col = c * CONV_KSZ^2 + dy * CONV_KSZ + dx maps to k-block k = col / DIM
    for (int k = 0; k < tiles_K; k++) {
        int col_start = k * DIM;

        // im2col: compute A_buf columns [col_start, col_start+DIM) for all patch rows
        for (int col = col_start; col < col_start + DIM; col++) {
            int tmp = col;
            int dx = tmp % CONV_KSZ; tmp /= CONV_KSZ;
            int dy = tmp % CONV_KSZ; tmp /= CONV_KSZ;
            int c  = tmp;
            for (int pi = 0; pi < CONV_H / CONV_KSZ; pi++)
                for (int pj = 0; pj < PATCHES_X; pj++)
                    A_buf[pi * PATCHES_X + pj][col] =
                        X_in[c][pi * CONV_KSZ + dy][pj * CONV_KSZ + dx];
        }

        // Immediately DMA A tiles for this k-block (cache-hot data)
        for (int i = 0; i < tiles_I; i++)
            gemmini_extended_mvin(
                (void*)(((elem_t*)A_buf) + i * DIM * MATMUL_K + col_start),
                a_base + (i * tiles_K + k) * DIM, DIM, DIM);

        // DMA B tiles for this k-block on port 2 (overlaps with next im2col iteration)
        for (int j = 0; j < tiles_J; j++)
            gemmini_extended_mvin2(
                (void*)(((elem_t*)B_in) + j * DIM * MATMUL_K + col_start),
                b_base + (j * tiles_K + k) * DIM, DIM, DIM);
    }

    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
