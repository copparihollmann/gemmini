// Optimized MX matmul kernel (128x128x256, fp4:e2m1) using batched MVIN strategy.
void solution(void) {
  {
    // Apply Strategy 6: Maximize tile bounds to pack the entire matrix contiguously
    int ti = MATMUL_M / DIM, tj = MATMUL_N / DIM, tk = MATMUL_K / DIM;
    uint32_t ab = 0;
    // Place B matrix at the top of scratchpad as required
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 0; // Use beginning of mx_smem for output
    
    // Configure format and execution for fp4:e2m1 -> bf16
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);
    
    // Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
    
    // Configure store and requantizer
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 1);
    
    // Batched MVIN for A: Load up to 4 tiles (MAX_BLOCK_LEN) at once along K-dimension
    // For fp4, each byte contains 2 elements, so stride needs adjustment
    gemmini_config_ld(MATMUL_K * sizeof(elem_t) / 2); // Divide by 2 for fp4 packing
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k += 4) { // Process up to 4 tiles at a time (MAX_BLOCK_LEN)
        int tiles_to_load = (tk - k) > 4 ? 4 : (tk - k);
        gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i*DIM*(MATMUL_K/2) + k*DIM/2), 
                              ab + (i*tk + k)*DIM, 
                              tiles_to_load * DIM, 
                              DIM);
      }
    }
    
    // Batched MVIN for B: Load up to 4 tiles at once along J-dimension
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2); // Divide by 2 for fp4 packing
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j += 4) { // Process up to 4 tiles at a time (MAX_BLOCK_LEN)
        int tiles_to_load = (tj - j) > 4 ? 4 : (tj - j);
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + j*DIM/2), 
                               bb + (k*tj + j)*DIM, 
                               tiles_to_load * DIM, 
                               DIM);
      }
    }
    
    // Compute weight-stationary loop in hardware (fully-covered fused loop)
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, bb, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    
    // Read BF16 results from SMEM and sync
    gemmini_mx_read_smem(&C_hw[0][0], SDST, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
