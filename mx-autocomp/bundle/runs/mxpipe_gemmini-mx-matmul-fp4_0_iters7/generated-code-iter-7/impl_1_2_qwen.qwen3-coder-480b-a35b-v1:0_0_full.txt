// Optimized MX matmul kernel (128x128x256, fp4:e2m1).
void solution(void) {
  {
    // Declare loop limits as const to enable compile-time induction and shift optimizations
    const int ti = 4;  // MATMUL_M / DIM / 2 = 128 / 16 / 2
    const int tj = 4;  // MATMUL_N / DIM / 2 = 128 / 16 / 2
    const int tk = 16; // MATMUL_K / DIM = 256 / 16

    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM; // Scratchpad top for B placement
    int SDST = 128; // C output address in shared memory

    // Configure execution with A_stride = 2 to match the batched scratchpad layout
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 2, 0, 0, false, 2, 2, 3, 0);

    // Load scale factors for activations (sel=0) and weights (sel=1)
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // Configure output stride for shared memory writeback
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 1);

    // MVIN A: Batched by 2 along the K dimension (cols = 2 * DIM)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Stride matches full K dimension
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k += 2) {
        // Compile-time constant arithmetic eliminates runtime multiplies
        elem_t* addr = &A_in_hw[i * 16 * MATMUL_K + k * 16];
        uint32_t spad_addr = ab + (i * tk + k) * 16;
        gemmini_extended_mvin((void*)addr, spad_addr, 2 * DIM, DIM);
      }
    }

    // MVIN B: Unbatched due to fixed B stride in weight-stationary mode
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2); // Stride for half-N blocks
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        // Compile-time constant arithmetic optimizes address generation
        elem_t* addr = &B_in[k * 16 * (MATMUL_N / 2) + j * 16];
        uint32_t spad_addr = bb + (k * tj + j) * 16;
        gemmini_extended_mvin((void*)addr, spad_addr, DIM, DIM);
      }
    }

    // Execute the weight-stationary fused loop over preloaded tiles
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Drain BF16 results from shared memory to DRAM
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);

    // Full memory barrier before host accesses output
    gemmini_fence();
  }
}
