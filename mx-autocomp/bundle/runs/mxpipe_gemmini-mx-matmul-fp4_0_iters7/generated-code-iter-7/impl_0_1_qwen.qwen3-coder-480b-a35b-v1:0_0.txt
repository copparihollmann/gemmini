#include <stdint.h>
#include "include/gemmini_testutils.h"

// Optimized MX matmul kernel (128x128x256, fp4:e2m1) using loop tiling & pipelining.
void solution(void) {
  // Utilizing compile-time constants for dimensions and tiles
  const int ti = MATMUL_M / DIM; // 8 tiles
  const int tj = MATMUL_N / DIM; // 8 tiles
  const int tk = MATMUL_K / DIM; // 16 tiles
  
  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tk * tj * DIM; // 8192 - 16 * 8 * 16 = 6144
  
  // 1. Configure format and execution (FP4: 2, 2, 3, 0)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);
  
  // 2. Load all scale factors once
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
  
  // 3. Batched MVIN for A: Load the entire A matrix (8 x 16 tiles)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < ti; i++) {
    for (int k = 0; k < tk; k += 4) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM), 
                            a_base + (i * tk + k) * DIM, 
                            4 * DIM, 
                            DIM);
    }
  }
  
  // 4. Batched MVIN for B: Load the entire B matrix (16 x 8 tiles)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
  for (int k = 0; k < tk; k++) {
    for (int j = 0; j < tj; j += 4) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM), 
                            b_base + (k * tj + j) * DIM, 
                            4 * DIM, 
                            DIM);
    }
  }
  
  // 5. Configure store stride once for BF16 output
  gemmini_config_st(MATMUL_N * sizeof(uint16_t));
  
  static uint32_t scale_factors[MATMUL_M * MATMUL_N / 32] __attribute__((aligned(32))) = {0};
  uint16_t *C_ptr = (uint16_t*)C_hw;
  
  // 6. Loop over block rows of C (block_I = 2, block_J = 8)
  for (int b_i = 0; b_i < ti; b_i += 2) {
    // Alternate destination rows to overlap execution with memory transfer (double-buffering)
    int sdst = (b_i & 2) ? 192 : 128;
    
    uint32_t *block_scale_factors = scale_factors + (b_i * DIM) * (MATMUL_N / 32);
    gemmini_mxquant_config_mvout((uint64_t)block_scale_factors, 2, 8, tk, 0, 0, 1);
    
    uint32_t a_base_block = a_base + b_i * tk * DIM;
    gemmini_loop_ws_spad(2, 8, tk, 0, 0, 0, 
                         a_base_block, BANK_NUM * BANK_ROWS, 0, sdst, 
                         false, false, false, false, false, NO_ACTIVATION, 
                         0, 0, false, 0x38);
    
    // Read the fully contiguous block directly into its target destination in DRAM
    gemmini_mx_read_smem(C_ptr + b_i * DIM * MATMUL_N, sdst * 16, 2 * DIM * MATMUL_N);
  }
  
  // 7. Final synchronization barrier
  gemmini_fence();
}
