#include <stdint.h>
#include "include/gemmini_testutils.h"

// Optimized MX matmul kernel using DMA block coalescing
void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2, tj = MATMUL_N / DIM / 2, tk = MATMUL_K / DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;

    // 1. Configure execution formats (FP4 input, BF16 output)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);

    // 2. Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // 3. Configure output and requantization bounds
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 1);

    // 4. Coalesced loading for A (block size = 3 tiles to stay within MAX_BLOCK_LEN=4 limit)
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k += 3) {
        int block_len = (tk - k) > 3 ? 3 : (tk - k);
        gemmini_extended_mvin(
          (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM), 
          ab + (i * tk + k) * DIM, 
          block_len * DIM, 
          DIM
        );
      }
    }

    // 5. Coalesced loading for B (block size = tj = 4 tiles, but limited to 3 due to hardware constraint)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      // Process B tiles in blocks of 3 to respect MAX_BLOCK_LEN constraint
      for (int j = 0; j < tj; j += 3) {
        int block_len = (tj - j) > 3 ? 3 : (tj - j);
        gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM), 
          bb + (k * tj + j) * DIM, 
          block_len * DIM, 
          DIM
        );
      }
    }

    // 6. Execute WS hardware tiled matmul loop
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // 7. Drain results to DRAM and synchronize
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
