// Optimized MX matmul kernel (128x128x256, fp4:e2m1) using loop restructuring and blocking.
void solution(void) {
  const int DIM = 16;
  const int BANK_NUM = 4;
  const int TOTAL_I_TILES = MATMUL_M / DIM; // 128 / 16 = 8
  const int TOTAL_J_TILES = MATMUL_N / DIM; // 128 / 16 = 8
  const int TOTAL_K_TILES = MATMUL_K / DIM; // 256 / 16 = 16

  // Block sizes for outer tiling
  const int ti = 2;  // Process 2 tiles along M per block (32 rows)
  const int tj = 8;  // Process full width along N (128 cols = 8 tiles)
  const int tk = 16; // Full K dimension in inner loop

  // Scratchpad addresses
  uint32_t ab = 0; // Start of A tiles in scratchpad
  uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM; // Place B at top of scratchpad
  int SDST = 0; // Shared memory destination offset (will be updated per block)

  // One-time configuration
  gemmini_extended3_config_ex(
    WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
    /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
    /*set_only_strides*/false,
    /*act_mx_fmt*/2, /*wgt_mx_fmt*/2, /*out_mx_fmt*/3, /*uselut*/0);

  // Load scale factors once since they apply to the whole computation
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure output store stride once
  gemmini_config_st(OUT_COLS * sizeof(out_t));

  // --- Outer block loop over M dimension ---
  const int num_blocks_i = TOTAL_I_TILES / ti; // 8 / 2 = 4 blocks
  for (int block_i = 0; block_i < num_blocks_i; block_i++) {
    // --- Load weights B once per block_i ---
    gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // Stride for full-width B
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        void* dram_ptr = (void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM);
        uint32_t spad_addr = bb + (k * tj + j) * DIM;
        gemmini_extended_mvin2(dram_ptr, spad_addr, DIM, DIM);
      }
    }

    // --- Load activations A for this block_i ---
    gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Stride for full-height A
    for (int i = 0; i < ti; i++) {
      for (int k = 0; k < tk; k++) {
        int global_i = block_i * ti + i;
        void* dram_ptr = (void*)(((elem_t*)A_in_hw) + global_i * DIM * MATMUL_K + k * DIM);
        uint32_t spad_addr = ab + (i * tk + k) * DIM;
        gemmini_extended_mvin(dram_ptr, spad_addr, DIM, DIM);
      }
    }

    // --- Compute and drain output for this block ---
    
    // Update requantization config with correct scale factor pointer for this block's rows
    // A_scales_row is [GK][M], so we index by block start row: (block_i * ti * DIM)
    // Scale group granularity GK is assumed to be K/32 = 256/32 = 8, so scale index is row/GK
    // But the intrinsic loads all scales, so pointer doesn't change per block_i in this simple case
    // If per-block scales were needed, this would be adjusted accordingly.
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                 /*i_bound*/ti, /*j_bound*/tj, /*k_bound*/tk,
                                 /*scale_act_sel*/0, /*scale_w_sel*/0, /*lut_update_granularity*/1);

    // Execute the fused hardware loop for this block's tiles
    gemmini_loop_ws_spad(
      ti, tj, tk, 0, 0, 0,
      ab, bb, 0, SDST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Drain the results to the correct contiguous block in DRAM
    gemmini_mx_read_smem(&C_hw[block_i * ti * DIM][0], SDST * 16, ti * DIM * tj * DIM);
    gemmini_fence();

    // Advance shared memory destination for next block if needed (though SDST=0 works here due to separate DRAM locations)
    // SDST remains fixed as we write to distinct DRAM regions each time.
  }
}
