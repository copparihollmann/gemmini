#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp4_64x64.h"

#define DIM 16
#define ADDR_LEN 32
#define BF16_PER_WORD 4
#define OUT_COLS (MATMUL_M / BF16_PER_WORD)

typedef uint8_t  elem_t;
typedef uint64_t out_t;

// Compile-time constants for tile counts and scratchpad layout
#define TI   (MATMUL_M / DIM / 2)
#define TJ   (MATMUL_N / DIM / 2)
#define TK   (MATMUL_K / DIM)
#define AB   0
#define BB   (BANK_NUM * BANK_ROWS - TK * TJ * DIM)
#define SDST 128

// Precomputed DRAM strides and tile sizes
#define A_STRIDE (MATMUL_K * sizeof(elem_t))
#define B_STRIDE (MATMUL_N * sizeof(elem_t) / 2)

// Precomputed scale lengths
#define A_SCALE_LEN (TI * TK * sizeof(uint8_t))
#define B_SCALE_LEN (TJ * TK * sizeof(uint8_t))

// Optimized MX matmul kernel using compile-time constant folding of instruction fields
void solution(void) {
  // Configure format and execution (MX FP4 format, weight-stationary)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);
  
  // Load scale factors
  gemmini_mx_load_scales((uint64_t)&A_scales_row, A_SCALE_LEN, 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, B_SCALE_LEN, 1);
  
  // Configure store and requantizer bounds (fully folded at compile-time)
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, TI, TJ, TK, 0, 0, 1);
  
  // Batched MVIN for A with static loop bounds
  gemmini_config_ld(A_STRIDE);
  for (int i = 0; i < TI; i++) {
    for (int k = 0; k < TK; k += 2) {
      const elem_t* const dram_addr = &A_in_hw[i * DIM * MATMUL_K + k * DIM];
      const uint32_t spad_addr = AB + (i * TK + k) * DIM;
      gemmini_extended_mvin((void*)dram_addr, spad_addr, 2 * DIM, DIM);
    }
  }
  
  // Batched MVIN for B with static loop bounds
  gemmini_config_ld(B_STRIDE);
  for (int k = 0; k < TK; k++) {
    for (int j = 0; j < TJ; j += 2) {
      const elem_t* const dram_addr = &B_in[k * DIM * (MATMUL_N / 2) + j * DIM];
      const uint32_t spad_addr = BB + (k * TJ + j) * DIM;
      gemmini_extended_mvin((void*)dram_addr, spad_addr, 2 * DIM, DIM);
    }
  }
  
  // Compute weight-stationary loop in hardware (fully folded at compile-time)
  gemmini_loop_ws_spad(TI, TJ, TK, 0, 0, 0, AB, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  
  // Read BF16 results from SMEM and sync
  gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
