// Optimized MX matmul kernel (128x128x256, fp4:e2m1).
void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2; // 4 tiles (64 rows per block)
    int tj = MATMUL_N / DIM / 2; // 4 tiles (128 packed elements / 64 bytes)
    int tk = MATMUL_K / DIM;     // 16 tiles (256 elements)
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;

    // 1. Configure execution format to FP4 inputs and BF16 output
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);

    // 2. Load scale factors once
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // 3. Load B (weight) tiles ONCE into the scratchpad (Strategy 6)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin((void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + j*DIM), bb + (k*tj + j)*DIM, DIM, DIM);
      }
    }

    // 4. Loop over row blocks of M (outer loop tiling)
    for (int b_i = 0; b_i < 2; b_i++) {
      // Load A tiles corresponding to the current row block
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < ti; i++) {
        for (int k = 0; k < tk; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (b_i*ti + i)*DIM*MATMUL_K + k*DIM), ab + (i*tk + k)*DIM, DIM, DIM);
        }
      }

      // Configure the requantization output for current block
      gemmini_config_st(OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)&scale_factors[b_i * ti * DIM][0], ti, tj, tk, 0, 0, 1);

      // Execute fused systolic loop over the current A block and the preloaded B block
      gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, ab, BANK_NUM * BANK_ROWS, 0, SDST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Drain results from SMEM back to DRAM for current block
      gemmini_mx_read_smem(&C_hw[b_i * ti * DIM][0], SDST * 16, ti * DIM * MATMUL_N);
    }

    // Defer the memory fence until all blocks are completed to minimize CPU stalls
    gemmini_fence();
  }
}
