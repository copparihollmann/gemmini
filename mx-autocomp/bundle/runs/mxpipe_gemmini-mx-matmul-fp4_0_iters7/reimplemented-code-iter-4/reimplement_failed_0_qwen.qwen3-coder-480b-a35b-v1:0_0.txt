// Optimized MX matmul kernel (128x128x256, fp4:e2m1) using double-buffered MVIN strategy.
void solution(void) {
  {
    int tiles_I = MATMUL_M / DIM; // 8
    int tiles_J = MATMUL_N / DIM; // 8  
    int tiles_K = MATMUL_K / DIM; // 16
    
    int ti_block = 4; // Process 4 I tiles at a time
    uint32_t ab0 = 0;
    uint32_t ab1 = ti_block * tiles_K * DIM; // 4 * 16 * 16 = 1024
    uint32_t bb = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 4*2048 - 16*8*16 = 8192 - 2048 = 6144
    int SDST0 = 0;
    int SDST1 = ti_block * tiles_J * 16; // 4 * 8 * 16 = 512
    
    // Configure format and execution (FP4 E2M1 for both act and wgt, BF16 output)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 1);
    
    // Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
    
    // Load LUT for FP4 format
    gemmini_mx_load_lut((uint64_t)LUT_AB, 16, 1); // A LUT
    gemmini_mx_load_lut((uint64_t)LUT_BB, 16, 0); // B LUT
    
    // Load B tiles (stationary)
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int k = 0; k < tiles_K; k++) {
      for (int j = 0; j < tiles_J; j++) {
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + k*DIM*MATMUL_N + j*DIM), 
                              bb + (j*tiles_K + k)*DIM, 
                              DIM, 
                              DIM);
      }
    }
    
    // Double buffered execution loop
    for (int i_outer = 0; i_outer < tiles_I; i_outer += ti_block) {
      uint32_t current_ab = (i_outer / ti_block % 2 == 0) ? ab0 : ab1;
      uint32_t next_ab = (i_outer / ti_block % 2 == 0) ? ab1 : ab0;
      int current_sdst = (i_outer / ti_block % 2 == 0) ? SDST0 : SDST1;
      
      // Load A tiles for current block
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i_inner = 0; i_inner < ti_block && i_outer + i_inner < tiles_I; i_inner++) {
        int i = i_outer + i_inner;
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i*DIM*MATMUL_K + k*DIM), 
                                current_ab + (i_inner*tiles_K + k)*DIM, 
                                DIM, 
                                DIM);
        }
      }
      
      // Start loading A for next block in background (if not last iteration)
      if (i_outer + ti_block < tiles_I) {
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i_inner = 0; i_inner < ti_block; i_inner++) {
          int i = i_outer + ti_block + i_inner;
          if (i < tiles_I) {
            for (int k = 0; k < tiles_K; k++) {
              gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i*DIM*MATMUL_K + k*DIM), 
                                    next_ab + (i_inner*tiles_K + k)*DIM, 
                                    DIM, 
                                    DIM);
            }
          }
        }
      }
      
      // Configure store and requantizer for the current block
      gemmini_config_st(OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
          (ti_block < tiles_I - i_outer) ? ti_block : tiles_I - i_outer, 
          tiles_J, tiles_K, 0, 0, 1);
      
      // Compute weight-stationary loop in hardware
      gemmini_loop_ws_spad(
          (ti_block < tiles_I - i_outer) ? ti_block : tiles_I - i_outer, 
          tiles_J, tiles_K, 
          0, 0, 0, 
          current_ab, bb, 0, current_sdst, 
          false, false, false, false, false, NO_ACTIVATION, 
          0, 0, false, 0x38);
    }
    
    // Read BF16 results from SMEM (fully contiguous) and sync
    gemmini_mx_read_smem(&C_hw[0][0], SDST0 * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
