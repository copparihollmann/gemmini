#include <stdint.h>
#include "include/gemmini_testutils.h"

#define DIM 16
#define MATMUL_M 128
#define MATMUL_N 128
#define MATMUL_K 256
#define OUT_COLS 128

typedef uint8_t elem_t;
typedef uint16_t out_t;  // BF16 output is 16-bit

void solution(void) {
    // Double buffering: split N dimension into two 64-element chunks
    const int ti = MATMUL_M / DIM;           // 128/16 = 8
    const int tj_total = MATMUL_N / DIM;     // 128/16 = 8  
    const int tj_half = tj_total / 2;        // 8/2 = 4
    const int tk = MATMUL_K / DIM;           // 256/16 = 16
    
    // Scratchpad partitioning (8192 rows total, split into two 4096-row buffers)
    const uint32_t ab_buf0 = 0;              // Buffer 0 A base
    const uint32_t ab_buf1 = 4096;           // Buffer 1 A base
    const uint32_t bb_buf0 = 4096 - tk * tj_half * DIM;  // Buffer 0 B base (top of first half)
    const uint32_t bb_buf1 = 8192 - tk * tj_half * DIM;  // Buffer 1 B base (top of second half)
    
    const int SDST = 0;  // Shared memory destination offset
    
    // 1. Configure execution - Fix MX format to match data (assuming FP8 E4M3 input, BF16 output)
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    
    // 2. Load scale factors once
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);
    
    // 3. Configure output stride
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    
    // ===== First sub-block (N=0..63) =====
    
    // Load A tiles for first sub-block into buffer 0
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti; i++) {
        for (int k = 0; k < tk; k++) {
            gemmini_extended_mvin(
                (void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM),
                ab_buf0 + (i * tk + k) * DIM,
                DIM, DIM
            );
        }
    }
    
    // Load B tiles for first sub-block into buffer 0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
        for (int j = 0; j < tj_half; j++) {
            gemmini_extended_mvin2(
                (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + j * DIM),
                bb_buf0 + (j * tk + k) * DIM,  // Correct B tile layout
                DIM, DIM
            );
        }
    }
    
    // Configure and compute first sub-block (buffer 0)
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj_half, tk, 0, 0, 1);
    gemmini_loop_ws_spad(ti, tj_half, tk, 0, 0, 0, 
                        ab_buf0, bb_buf0, 0, SDST,
                        false, false, false, false, false, NO_ACTIVATION, 1, 1, false, 0x38);
    
    // Drain first sub-block results (columns 0-63)
    gemmini_mx_read_smem((uint64_t)C_hw, SDST * 16, MATMUL_M * MATMUL_N / 2);
    
    // ===== Second sub-block (N=64..127) =====
    
    // Load B tiles for second sub-block into buffer 1
    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {
        for (int j = 0; j < tj_half; j++) {
            gemmini_extended_mvin2(
                (void*)(((elem_t*)B_in) + k * DIM * (MATMUL_N / 2) + (j + tj_half) * DIM),
                bb_buf1 + (j * tk + k) * DIM,  // Correct B tile layout
                DIM, DIM
            );
        }
    }
    
    // Configure and compute second sub-block (buffer 1)
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj_half, tk, 0, 0, 1);
    gemmini_loop_ws_spad(ti, tj_half, tk, 0, 0, 0,
                        ab_buf0, bb_buf1, 0, SDST + (MATMUL_M * MATMUL_N / 4),  // Offset for second half
                        false, false, false, false, false, NO_ACTIVATION, 1, 2, false, 0x38);
    
    // Drain second sub-block results (columns 64-127)
    gemmini_mx_read_smem((uint64_t)((uint16_t*)C_hw + MATMUL_M * MATMUL_N / 2), 
                         SDST * 16 + (MATMUL_M * MATMUL_N / 4), 
                         MATMUL_M * MATMUL_N / 2);
    
    // Final synchronization
    gemmini_fence();
}
