void solution(void) {

    int tj = MATMUL_N / DIM / 2; // 4
    int tk = MATMUL_K / DIM;     // 16

    gemmini_config_ld(MATMUL_N * sizeof(elem_t) / 2);
    for (int k = 0; k < tk; k++) {

        gemmini_extended_mvin((void*)(((elem_t*)B_in) + k*DIM*(MATMUL_N/2) + j*DIM), 
                              bb + (k*tj + j)*DIM, 
                              2 * DIM, 
                              DIM);
      }
    }
    
    // Prologue: Load A block 0
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));
    for (int i = 0; i < ti_block; i++) {
      for (int k = 0; k < tk; k += 2) {
        gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i*DIM*MATMUL_K + k*DIM), 
                              ab0 + (i*tk + k)*DIM, 
                              2 * DIM, 
                              DIM);
      }
    }
    
    // Double buffered execution loop
    for (int b = 0; b < 2; b++) {
      uint32_t current_ab = (b % 2 == 0) ? ab0 : ab1;
      uint32_t next_ab = (b % 2 == 0) ? ab1 : ab0;
      int current_sdst = (b % 2 == 0) ? SDST0 : SDST1;
      
      // Start loading A for next block in background
      if (b + 1 < 2) {
        int next_i_offset = (b + 1) * ti_block;
        gemmini_config_ld(MATMUL_K * sizeof(elem_t));
        for (int i = 0; i < ti_block; i++) {

            gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + (next_i_offset + i)*DIM*MATMUL_K + k*DIM), 
                                  next_ab + (i*tk + k)*DIM, 
                                  2 * DIM, 
                                  DIM);
          }
        }
      }
      
      // Configure store and requantizer for the current block
      gemmini_config_st(OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti_block, tj, tk, 0, 0, 1);
      
      // Compute weight-stationary loop in hardware
      gemmini_loop_ws_spad(ti_block, tj, tk, 0, 0, 0, current_ab, BANK_NUM * BANK_ROWS, 0, current_sdst, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    }
    
    // Read BF16 results from SMEM (fully contiguous) and sync
    gemmini_mx_read_smem(&C_hw[0][0], SDST0 * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
