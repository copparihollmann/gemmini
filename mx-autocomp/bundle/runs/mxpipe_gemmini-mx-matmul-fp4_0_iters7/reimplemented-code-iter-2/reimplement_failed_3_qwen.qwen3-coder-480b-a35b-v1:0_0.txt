#include <stdint.h>
#include "include/gemmini_testutils.h"

void solution(void) {
  {
    int ti = MATMUL_M / DIM / 2, tj = MATMUL_N / DIM / 2, tk = MATMUL_K / DIM;
    uint32_t ab = 0;
    uint32_t bb = BANK_NUM * BANK_ROWS - tk * tj * DIM;
    int SDST = 128;
    
    // Calculate the proper A base address to avoid overlap with B
    uint32_t a_base = 0;
    // B is placed at the top of scratchpad, growing downward
    // A should be placed at the bottom, growing upward
    // But we need to ensure no overlap: A uses rows [0, ti*tk*DIM), B uses rows [bb, BANK_NUM*BANK_ROWS)
    
    // Actually, let's recalculate the layout properly:
    // Total rows needed for A: ti * tk * DIM
    // Total rows needed for B: tk * tj * DIM
    // Need: ti * tk * DIM + tk * tj * DIM <= BANK_NUM * BANK_ROWS = 8192
    
    // 1. Configure execution format
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 2, 2, 3, 0);

    // 2. Load scale factors
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // 3. Configure requantizer write-back bounds
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, ti, tj, tk, 0, 0, 1);

    // 4. Configure DMA channel for both A and B (using port 0 for both)
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), 1.0f, false, 0);

    // 5. Load A and B tiles using gemmini_extended_mvin (port 0) for both
    for (int k = 0; k < tk; k++) {
      for (int i = 0; i < ti; i++) {
        gemmini_extended_mvin((void*)(((elem_t*)A_in_hw) + i * DIM * MATMUL_K + k * DIM), 
                              a_base + (i * tk + k) * DIM, DIM, DIM);
      }
      for (int j = 0; j < tj; j++) {
        gemmini_extended_mvin((void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM), 
                              bb + (j * tk + k) * DIM, DIM, DIM);
      }
    }

    // 6. Execute hardware-accelerated tiled GEMM loop
    gemmini_loop_ws_spad(ti, tj, tk, 0, 0, 0, a_base, bb, 0, SDST, 
                         false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // 7. Drain the packed BF16 results from SMEM to DRAM
    gemmini_mx_read_smem(&C_hw[0][0], SDST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
  }
}
