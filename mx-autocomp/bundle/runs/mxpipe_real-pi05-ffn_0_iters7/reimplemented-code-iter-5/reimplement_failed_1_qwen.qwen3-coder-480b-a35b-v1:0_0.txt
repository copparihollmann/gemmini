#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

#define GEMMINI_SF_MEM 0x40088000
#define GEMMINI_SF_MEM_A (GEMMINI_SF_MEM + 0x2000)
#define GEMMINI_SF_MEM_B GEMMINI_SF_MEM
#define SMEM 0x40000000

#define DIM 16
#define ADDR_LEN 32

#define BF16_PER_WORD 4
#define OUT_COLS (MATMUL_N / BF16_PER_WORD) // Corrected from M to N for C matrix

#define XCUSTOM_ACC 2
#define WEIGHT_STATIONARY 1
#define NO_ACTIVATION 0
#define ACC_SCALE_IDENTITY 1.0f

#define k_LOOP_WS_CONFIG_BOUNDS 0
#define k_LOOP_WS_CONFIG_SPAD_AB 24
#define k_LOOP_WS 8
#define k_CONFIG 12

typedef uint8_t  elem_t;   // fp8:e4m3 input elements
typedef uint8_t  welem_t;  // fp8:e4m3 weight elements
typedef uint64_t out_t;    // 4x bf16 packed per word

// Helper function to convert float to acc_scale_t bits
static inline uint32_t acc_scale_t_to_acc_scale_t_bits(float f) {
    union { float f; uint32_t i; } u = { .f = f };
    return u.i;
}

void solution(void) {
  // 1. Reset/Flush Gemmini state
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 0, 0, 0); // gemmini_flush(0)

  // 2. Configure execution: Weight-Stationary, FP8 inputs, BF16 output
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)acc_scale_t_to_acc_scale_t_bits(ACC_SCALE_IDENTITY) << 32) | 
      ((uint64_t)(1) << 16) | 
      ((uint64_t)(3) << 14) | 
      ((uint64_t)(0) << 12) | 
      ((uint64_t)(0) << 10) | 
      ((uint64_t)(0) << 7) | 
      ((uint64_t)(0) << 5) | 
      ((uint64_t)(WEIGHT_STATIONARY) << 2) | 
      0, 
      ((uint64_t)(1) << 48) | (uint64_t)(uint32_t)(0), 
      k_CONFIG);

  // 3. Batch load scale factors once for the entire matrix (Strategy 4)
  //    Note: Scale memory expects layout [K_groups][Rows_or_Cols]
  //    So for A (M rows), it's [K/32][M] and for B (N cols), it's [K/32][N]
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      (uint64_t)A_scales_row, 
      ((uint64_t)(0) << 32) | ((uint64_t)((MATMUL_K / 32) * MATMUL_M) & 0xFFFFFFFFu), 
      27);
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      (uint64_t)B_scales_col, 
      ((uint64_t)(1) << 32) | ((uint64_t)((MATMUL_K / 32) * MATMUL_N) & 0xFFFFFFFFu), 
      27);

  // 4. Calculate tiling and scratchpad base offsets
  const int tiles_I = MATMUL_M / DIM; // 64/16 = 4
  const int tiles_J = MATMUL_N / DIM; // 64/16 = 4
  const int tiles_K = MATMUL_K / DIM; // 64/16 = 4

  const uint32_t a_base = 0;
  // Place B tiles at the top of the scratchpad, growing downwards
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 4*2048 - 4*4*16 = 8192 - 256 = 7936
  const int SPAD_DEST = 0; // Use beginning of accumulator for output

  // 5. MVIN A tiles to scratchpad (batched 2-at-a-time for efficiency)
  // Configure stride for A (M x K matrix)
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 32) | (uint64_t)(MATMUL_K * sizeof(elem_t)), 
      0, 
      12); // CONFIG_LD
  
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += 2) { // Process up to 2 tiles in K-dimension
      const int k_tiles_to_load = (k + 2 > tiles_K) ? (tiles_K - k) : 2;
      ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
          (uint64_t)(((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM), 
          ((uint64_t)(DIM) << (ADDR_LEN + 16)) | ((uint64_t)(k_tiles_to_load * DIM) << ADDR_LEN) | (a_base + (i * tiles_K + k) * DIM), 
          0); // k_MVIN
    }
  }

  // 6. MVIN B tiles to scratchpad (batched 2-at-a-time for efficiency)
  // Configure stride for B (K x N matrix)
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 32) | (uint64_t)(MATMUL_N * sizeof(elem_t)), 
      0, 
      12); // CONFIG_LD
  
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += 2) { // Process up to 2 tiles in J-dimension
      const int j_tiles_to_load = (j + 2 > tiles_J) ? (tiles_J - j) : 2;
      ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
          (uint64_t)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM), 
          ((uint64_t)(DIM) << (ADDR_LEN + 16)) | ((uint64_t)(j_tiles_to_load * DIM) << ADDR_LEN) | (b_base + (k * tiles_J + j) * DIM), 
          1); // k_MVIN2
    }
  }

  // 7. Configure output stride and the requantizer
  // Configure stride for packed BF16 output
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 32) | (uint64_t)(OUT_COLS * sizeof(out_t)), 
      0, 
      12); // CONFIG_ST
  
  uint32_t scale_factors[512] = {0}; // Local variable instead of static
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 61) | ((uint64_t)(0) << 60) | ((uint64_t)(tiles_K) << 51) | 
      ((uint64_t)(tiles_J) << 42) | ((uint64_t)(tiles_I) << 33) | (uint64_t)(uint64_t)scale_factors, 
      (uint64_t)(1) & 0xFFFF, 
      13); // CONFIG_SCALE_MEM

  // 8. Execute fused hardware matrix multiplication loop
  // Issue the one-shot marker configuration for scratchpad loop
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 32) | ((uint64_t)(0) << 16) | (uint64_t)(0), // pad_K, pad_J, pad_I
      ((uint64_t)(tiles_K) << 32) | ((uint64_t)(tiles_J) << 16) | (uint64_t)(tiles_I), // K, J, I
      k_LOOP_WS_CONFIG_BOUNDS);
      
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      a_base, b_base, 
      k_LOOP_WS_CONFIG_SPAD_AB);
  
  // Execute the main loop
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      ((uint64_t)(0) << 18) | ((uint64_t)(0) << 16) | // a_spad_id=0, b_spad_id=0
      ((uint64_t)(NO_ACTIVATION) << 8) | // act
      ((0) << 2) | ((0) << 1) | (0),     // low_D=0, full_C=0, ex_accumulate=0
      ((uint64_t)(SPAD_DEST) << 32) | 0x200U | (0x38) | // C_spad_offset, marker, skips
      ((0) << 2) | ((0) << 1) | (0),     // is_resadd=0, B_transpose=0, A_transpose=0
      k_LOOP_WS);

  // 9. Read packed BF16 results from SMEM and synchronize
  // Results are in mx_smem at SPAD_DEST * 16 word offset
  ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, 
      (uint64_t)&C_hw[0][0],        // DRAM destination
      ((uint64_t)(MATMUL_M * MATMUL_N) << 32) | ((uint64_t)(SPAD_DEST * 16) & 0xFFFFFFFFu), 
      28); // k_MX_READ_SMEM
      
  asm volatile("fence"); // gemmini_fence()
}
