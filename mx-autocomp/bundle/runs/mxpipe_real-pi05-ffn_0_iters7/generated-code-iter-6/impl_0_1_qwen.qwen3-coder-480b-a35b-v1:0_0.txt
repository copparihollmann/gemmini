#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

void solution(void) {
  // Constants for tiling based on problem size (64x64x64) and DIM=16
  const int tiles_I = MATMUL_M / DIM; // 64 / 16 = 4
  const int tiles_J = MATMUL_N / DIM; // 64 / 16 = 4
  const int tiles_K = MATMUL_K / DIM; // 64 / 16 = 4

  // Scratchpad addressing scheme according to hardware constraints (Rule 12)
  const uint32_t a_base = 0;
  const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM; // 8192 - 4*4*16 = 8192 - 256 = 7936
  const int SPAD_DEST = 128; // Output location in shared memory

  // Aligned scale_factors buffer for hardware access
  uint32_t scale_factors[512] __attribute__((aligned(32))) = {0};

  // 1. Initial flush to ensure clean state before starting
  gemmini_flush(0);

  // 2. Configure execution: Weight-stationary FP8 MX with BF16 output
  gemmini_extended3_config_ex(
      WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
      /*set_only_strides*/false,
      /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

  // 3. Load scale factors directly from global arrays
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0); // sel=0 for A/activation scales
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1); // sel=1 for B/weight scales

  // 4. Load all A tiles into the lower part of scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Row-major stride for A (MxK matrix)
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      const elem_t *dram_ptr = &A_in[i * DIM][k * DIM]; // Address of A[i*DIM : (i+1)*DIM][k*DIM : (k+1)*DIM]
      uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)dram_ptr, spad_addr, DIM, DIM);
    }
  }

  // 5. Load all B tiles into the upper part of scratchpad, respecting Rule 12 layout
  gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // Row-major stride for B (KxN matrix)
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      const elem_t *dram_ptr = &B_in[k * DIM][j * DIM]; // Address of B[k*DIM : (k+1)*DIM][j*DIM : (j+1)*DIM]
      uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)dram_ptr, spad_addr, DIM, DIM);
    }
  }

  // 6. Configure store and requantization parameters
  const int OUT_COLS = MATMUL_N / BF16_PER_WORD; // Each word holds 4 BF16 values
  gemmini_config_st(OUT_COLS * sizeof(out_t));   // Stride in bytes for output matrix
  gemmini_mxquant_config_mvout(
      (uint64_t)scale_factors,
      /*i_bound*/tiles_I, /*j_bound*/tiles_J, /*k_bound*/tiles_K,
      /*scale_act_sel*/0, /*scale_w_sel*/0,
      /*lut_update_granularity*/1);

  // 7. Launch fused compute loop over all tiles already in scratchpad
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      /*pad_I*/0, /*pad_J*/0, /*pad_K*/0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      /*A_transpose*/false, /*B_transpose*/false,
      /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
      NO_ACTIVATION, /*a_spad_id*/0, /*b_spad_id*/0, /*is_resadd*/false,
      /*skips*/0x38); // Skip redundant ops like preload D

  // 8. Read final BF16 results from shared memory to main DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  
  // 9. Ensure memory consistency before host accesses result
  gemmini_fence();
}
