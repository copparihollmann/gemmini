Looking at the error message, the issue is that `cols=64` overflows the 6-bit hardware field, which can only represent values 0-63. The problem occurs when trying to load chunks of `MAX_BLOCK_LEN=4` tiles, where each tile is `DIM=16` columns, resulting in `4*16=64` columns.

The fix is to limit the block length to 3 tiles maximum when `DIM=16`, since `3*16=48` fits within the 6-bit field (0-63), but `4*16=64` gets truncated to 0.

```c
#include <stdint.h>
#include "include/gemmini_testutils.h"

// Strategy 12: Load the scale factors for the entire A and B matrices into their 
// respective Gemmini scale memories once at the beginning of the kernel.
void solution(void) {
  // 1. Configure execution for weight-stationary FP8 matmul with BF16 output
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 
                              1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Load the entire A and B scale factors once
  gemmini_mx_load_scales((uint64_t)A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)B_scales_col, sizeof(B_scales_col), 1);

  // Define full-matrix tile parameters
  int tiles_I = MATMUL_M / DIM;
  int tiles_J = MATMUL_N / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  int SPAD_DEST = 0;

  // 3. Load entire A matrix in coalesced DMA sweeps (maximum 3 tiles wide to avoid 6-bit overflow)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += 3) {  // Limit to 3 tiles to avoid 6-bit field overflow
      int chunk_tiles = (tiles_K - k) < 3 ? (tiles_K - k) : 3;
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, chunk_tiles * DIM, DIM);
    }
  }

  // 4. Load entire B matrix in coalesced DMA sweeps (maximum 3 tiles wide to avoid 6-bit overflow)
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += 3) {  // Limit to 3 tiles to avoid 6-bit field overflow
      int chunk_tiles = (tiles_J - j) < 3 ? (tiles_J - j) : 3;
      gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                             b_base + (k * tiles_J + j) * DIM, chunk_tiles * DIM, DIM);
    }
  }

  // 5. Configure requantizer write-back scale memory and bounds
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 6. Execute full weight-stationary tiled matmul in hardware
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // 7. Drain entire BF16 result matrix to DRAM in a single DMA transfer
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
```
