An elegant and highly effective way to optimize this MX-Gemmini kernel is to apply **Strategy 3: Double Buffering** to the B-matrix tiles and the shared memory destination (`SPAD_DEST`). 

### Optimization Plan
1. **Scratchpad Partitioning**: Divide the upper region of the scratchpad into two distinct B-tile buffers (`b_base[0]` and `b_base[1]`), growing downwards from the top of the scratchpad (`BANK_NUM * BANK_ROWS`). This fits safely alongside the stationary A-tile region in the lower scratchpad.
2. **SMEM Partitioning**: Partition the MX shared memory (`mx_smem`) into two destination buffers (`spad_dest[0]` and `spad_dest[1]`) to overlap the draining of the previous tile's results with the computation of the current tile.
3. **Pipelining Input and Compute**:
   - Extract the B-tile loading logic into a helper function `load_B_tile_only`.
   - In the inner loop, asynchronously prefetch the B-tiles for iteration `j0 + 1` into the alternate scratchpad buffer.
   - Synchronously load the B scale factors right before triggering the compute of the current iteration `j0`. This ensures we do not overwrite active scale factors in `mx_scale_b_mem` prematurely.
   - Launch `gemmini_loop_ws_spad` and `gemmini_mx_read_smem` asynchronously.
   - Use a single `gemmini_fence()` at the end of the iteration to block only when absolutely necessary, allowing next-tile loading, current-tile computing, and current-tile SMEM draining to execute concurrently on the hardware.

### Rewritten Solution

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Helper function to stream B tiles asynchronously into the designated double buffer
static void load_B_tile_only(int j0, int buf, int tiles, int tiles_K, uint32_t *b_base, const elem_t *B_in) {
  int ncols = REAL_N - j0 * BLK;
  if (ncols > BLK) ncols = BLK;
  if (ncols <= 0) return;

  gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // B rows: stride N
  const elem_t *B_col_base = (const elem_t*)B_in + (j0 * BLK);
  uint32_t dst_base = b_base[buf];

  for (int k = 0; k < tiles_K; k++) {
    const elem_t *B_tile_ptr = B_col_base + (k * DIM * MATMUL_N);
    for (int j = 0; j < tiles; j += 2) {
      gemmini_extended_mvin((void*)B_tile_ptr,
                            dst_base + (k * tiles + j) * DIM, 2 * DIM, DIM);
      B_tile_ptr += 2 * DIM;
    }
  }
}

// AUTO-GENERATED outer-tiled MX matmul optimized via Double Buffering.
void solution(void) {
  // Pre-format scale factor matrices to eliminate CPU-side staging in the hot loop
  static uint8_t A_scales_perf[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_perf[BLOCKS_N][MATMUL_K / 32][BLK];

  // One-time pre-formatting before main loops start
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      
      // Only copy actual data, handle partial blocks correctly
      for (int i = 0; i < mrows; i++) {
        A_scales_perf[i0][g][i] = A_scales_row[g][i0 * BLK + i];
      }
      // Pad with zeros for partial blocks if needed
      for (int i = mrows; i < BLK; i++) {
        A_scales_perf[i0][g][i] = 0;
      }
    }
  }

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;
      
      // Only copy actual data, handle partial blocks correctly
      for (int j = 0; j < ncols; j++) {
        B_scales_perf[j0][g][j] = B_scales_col[g][j0 * BLK + j];
      }
      // Pad with zeros for partial blocks if needed
      for (int j = ncols; j < BLK; j++) {
        B_scales_perf[j0][g][j] = 0;
      }
    }
  }

  // Hoist invariant Gemmini execution configuration outside all loops
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; 
    if (mrows > BLK) mrows = BLK; 
    if (mrows <= 0) continue;

    int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
    uint32_t a_base = 0;

    // Load stationary A scale factors once per row block
    gemmini_mx_load_scales((uint64_t)&A_scales_perf[i0][0][0], A_S_BYTES, 0);

    // Keep A tiles stationary in the lower half of the scratchpad
    gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // A rows: stride K
    const elem_t *A_row_base = (const elem_t*)A_in + (i0 * BLK * MATMUL_K);
    for (int i = 0; i < tiles; i++) {
      const elem_t *A_tile_ptr = A_row_base + (i * DIM * MATMUL_K);
      for (int k = 0; k < tiles_K; k += 2) {
        gemmini_extended_mvin((void*)A_tile_ptr,
                              a_base + (i * tiles_K + k) * DIM, 2 * DIM, DIM);
        A_tile_ptr += 2 * DIM;
      }
    }

    // Set up Double-Buffering configurations for B scratchpad and SMEM destinations
    uint32_t b_base[2];
    b_base[0] = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
    b_base[1] = BANK_NUM * BANK_ROWS - 2 * tiles * tiles_K * DIM;

    uint32_t b_end[2];
    b_end[0] = BANK_NUM * BANK_ROWS;
    b_end[1] = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;

    int spad_dest[2];
    spad_dest[0] = 128;
    spad_dest[1] = 128 + 256; // 4096 / 16 = 256 SMEM rows

    // Prologue: Pre-load the first B-tile (j0 = 0) asynchronously into buffer 0
    int buf = 0;
    load_B_tile_only(0, buf, tiles, tiles_K, b_base, B_in);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; 
      if (ncols > BLK) ncols = BLK; 
      if (ncols <= 0) continue;

      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Flush to synchronize state
      gemmini_flush(0);

      // Load B scale factors synchronously right before computing
      gemmini_mx_load_scales((uint64_t)&B_scales_perf[j0][0][0], B_S_BYTES, 1);

      // 1. Asynchronously load the NEXT B-tile into the alternate double buffer
      int next_buf = 1 - buf;
      if (j0 + 1 < BLOCKS_N) {
        load_B_tile_only(j0 + 1, next_buf, tiles, tiles_K, b_base, B_in);
      }

      // 2. Configure output and run compute on current buffer
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
      
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, b_end[buf], 0,
                           spad_dest[buf], false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 3. Drain results from SMEM to DRAM
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             spad_dest[buf] * 16 + r * BLK, ncols);
      }

      // 4. Fence: block only at the end of the iteration, overlapping DMA loads & compute
      gemmini_fence();

      // Alternate double buffer index
      buf = next_buf;
    }
  }
}
```
