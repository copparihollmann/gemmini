#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_data_mx_lut_hw.h"

void solution(void) {
  // Hoist global/invariant initializations and flushes
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  // Fast stack buffers for pre-packed scales
  uint8_t A_s_packed[BLOCKS_M][A_S_BYTES] __attribute__((aligned(64)));
  uint8_t B_s_packed[BLOCKS_N][B_S_BYTES] __attribute__((aligned(64)));

  // One-time scale factor packing (eliminates inner-loop CPU memcpy)
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_s_packed[i0][g * BLK], &A_scales_row[g][i0 * BLK], BLK);
    }
  }
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_s_packed[j0][g * BLK], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  // Reordered loops: j0 (weight columns) is now outer to enable weight-stationary reuse
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK; 
    if (ncols > BLK) ncols = BLK; 
    if (ncols <= 0) continue;
    int pad_j = BLK - ncols;

    int tiles_J = (ncols + DIM - 1) / DIM;  // Round up and adjust for padding
    int tiles_K = MATMUL_K / DIM;
    uint32_t a_base = 0;
    // Calculate b_base correctly - place B tiles at the top of scratchpad
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
    int SPAD_DEST = (j0 * BLOCKS_M) % (ACC_ROWS/2);  // Ensure unique destination per block

    // Load B scales once per outer-loop iteration
    gemmini_mx_load_scales((uint64_t)B_s_packed[j0], B_S_BYTES, 1);

    // MVIN B tiles into high-address scratchpad region once and reuse across i0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                               b_base + (j * tiles_K + k) * DIM, DIM, DIM);
      }
    }

    // Inner loop: iterate over row-blocks (A activations and output C)
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK; 
      if (mrows > BLK) mrows = BLK; 
      if (mrows <= 0) continue;
      int pad_i = BLK - mrows;

      int tiles_I = (mrows + DIM - 1) / DIM;  // Round up and adjust for padding

      // Load A scales for the current row block
      gemmini_mx_load_scales((uint64_t)A_s_packed[i0], A_S_BYTES, 0);

      // MVIN A tiles directly from DRAM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Configure requantizer with correct tile dimensions
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 1, 1);

      // Compute weight-stationary tiled gemm using the pre-loaded B from scratchpad
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Read output results from shared memory - correct BF16 packing calculation
      int output_words = (ncols + 1) / 2;  // BF16 values are packed 2 per word
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem((void*)(((uint64_t*)&C_hw[i0 * BLK + r][j0 * BLK]) & ~0x7ULL),
                             SPAD_DEST * 16 + r * BLK_OUTC, ncols);
      }
      gemmini_fence();
    }
  }
}
