    gemmini_extended3_config_ld(MATMUL_K * sizeof(uint8_t), MVIN_SCALE_IDENTITY, false, 1);
    
    // Load all B tiles into scratchpad using mvin2
    for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
            uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
            uint8_t *dram_ptr = &B[j * DIM][k * DIM];
            gemmini_extended_mvin2(
                (uint64_t)dram_ptr,
                spad_addr,
                DIM,   // cols
                DIM    // rows
            );
        }
    }

    // =========================================================
    // Step 5: Configure the requantized mvout
    // scale_dram points to scale factors for output requantization
    // For BF16 output, we use the loaded A and B scales
    // scale_act_sel=0 (use A scale buffer), scale_w_sel=0 (use B scale buffer)
    // =========================================================
    gemmini_extended_config_st(MATMUL_N * sizeof(uint16_t), 0, ACC_SCALE_IDENTITY);
    
    gemmini_mxquant_config_mvout(
        (uint64_t)B_scales,     // scale_dram (used for output requantization reference)
        tiles_I,                // i_bound
        tiles_J,                // j_bound  
        tiles_K,                // k_bound
        0,                      // scale_act_sel: use A scale buffer (sel=0)
        0,                      // scale_w_sel: use B scale buffer (sel=0)
        1                       // lut_update_granularity
    );

    // =========================================================
    // Step 6: Compute - weight-stationary tiled matmul
    // funct 24 (LOOP_WS_CONFIG_SPAD_AB) is issued inside gemmini_loop_ws_spad
    // The mx_loop_spad_marker is one-shot, so this must be re-issued each time
    // B end address = BANK_NUM*BANK_ROWS (top of scratchpad)
    // skips=0x38 suppresses redundant operations
    // =========================================================
    gemmini_loop_ws_spad(
        tiles_I,            // I tile count
        tiles_J,            // J tile count
        tiles_K,            // K tile count
        0,                  // pad_I
        0,                  // pad_J
        0,                  // pad_K
        a_base,             // A scratchpad base address
        TOTAL_SPAD_ROWS,    // B scratchpad end address (top of scratchpad)
        0,                  // D (bias) - not used
        SPAD_DEST,          // C output destination in mx_smem
        false,              // A_transpose
        false,              // B_transpose
        false,              // full_C
        false,              // low_D
        false,              // ex_accumulate
        NO_ACTIVATION,      // activation
        0,                  // a_spad_id (no reuse)
        0,                  // b_spad_id (no reuse)
        false,              // is_resadd
        0x38                // skips bitmask
    );

    // =========================================================
    // Step 7: Read results from mx_smem to DRAM
    // BF16 packed 4 per uint64 word
    // smem_off_bf16 = SPAD_DEST * 16
    // num_bf16 = total number of BF16 elements
    // =========================================================
    gemmini_mx_read_smem(
        (uint64_t)C,
        SPAD_DEST * 16,         // smem word offset (16-bit words)
        MATMUL_M * MATMUL_N     // number of BF16 values
    );
    
    gemmini_fence();
}
