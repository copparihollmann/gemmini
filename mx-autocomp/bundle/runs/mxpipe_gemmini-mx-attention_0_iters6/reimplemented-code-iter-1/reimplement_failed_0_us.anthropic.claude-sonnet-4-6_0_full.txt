Looking at the error:

```
GEMMINI MVIN ILLEGAL: block mvin with block_stride(0) < rows(16) (cols=32, blocks=2)
-- RTL LoadController.scala:138 assert(block_stride>=rows).
Configure gemmini_extended4_config_ld(..., block_mvin_stride>=rows, ...).
```

The problem is that when using `gemmini_extended_mvin` with `cols=bb*DIM` (e.g., 32 for bb=2), the hardware interprets this as a block mvin with `blocks=2` and `block_stride=0` (from `gemmini_extended3_config_ld` which sets `block_mvin_stride=DIM=16`). But the stride must be ≥ rows (16). Wait - actually the default `block_mvin_stride=DIM=16` and rows=DIM=16, so that should be fine... 

Re-reading: `gemmini_extended3_config_ld` sets `block_mvin_stride=DIM`. But `gemmini_config_ld` maps to `gemmini_extended2_config_ld` which sets `block_mvin_stride=DIM`. The issue is that `gemmini_config_ld` uses `gemmini_extended2_config_ld` → `gemmini_extended3_config_ld(stride, scale, shrunk, 0)` where id=0 means it uses `gemmini_extended4_config_ld(..., DIM, id)`. So `block_mvin_stride=DIM=16` and rows=DIM=16. That should be fine.

Wait - looking more carefully: for the B-matrix mvin using `gemmini_extended_mvin` (not mvin2) with `bb*DIM` cols. The B tiles use `gemmini_extended_mvin` (k_MVIN) but B should use `gemmini_extended_mvin2`. Actually in the original code for step 1, B is loaded with `gemmini_extended_mvin` (not mvin2). The fix is to use `gemmini_extended_mvin2` for B tiles. But actually the real issue is that `block_mvin_stride` needs to be set to at least `DIM` rows via `gemmini_extended4_config_ld`. Let me use single-tile mvins (bb=1) to avoid the block mvin issue entirely, which is the safest fix.

```c
void solution(void) {
  // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
  {
    int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&Q_scales_row, sizeof(Q_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&KT_scales_col, sizeof(KT_scales_col), 1);
    // Load A = Q tiles (one tile at a time to avoid block-mvin stride issues)
    gemmini_extended4_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, DIM, 0);
    for (int i = 0; i < tiles_I; i++)
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      }
    // Load B = K^T tiles (one tile at a time)
    gemmini_extended4_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, DIM, 0);
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j++) {
        gemmini_extended_mvin2((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                               b_base + (k * tiles_J + j) * DIM, DIM, DIM);
      }
    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- 2+3 fused) softmax row-block + immediate mvin for P @ V ----
  {
    int tiles_I = ATT_S / DIM;   // = 8
    int tiles_J = ATT_D / DIM;   // = 4
    int tiles_K = ATT_S / DIM;   // = 8
    uint32_t a_base = 0;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)&V_scales_col, sizeof(V_scales_col), 1);

    // Load B = V tiles (one tile at a time)
    gemmini_extended4_config_ld(ATT_D * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, DIM, 0);
    for (int k = 0; k < tiles_K; k++)
      for (int j = 0; j < tiles_J; j++) {
        gemmini_extended_mvin2((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                               b_base + (k * tiles_J + j) * DIM, DIM, DIM);
      }

    // Configure mvin port for A = P tiles (stride = ATT_S bytes)
    gemmini_extended4_config_ld(ATT_S * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, DIM, 0);

    // Fused loop: softmax for one i-block, then mvin those rows
    for (int i = 0; i < tiles_I; i++) {
      // -- Softmax for rows [i*DIM .. (i+1)*DIM) --
      for (int row = i * DIM; row < (i + 1) * DIM; row++) {
        int64_t q[ATT_S];
        int64_t mq = -((int64_t)1 << 62);
        for (int j = 0; j < ATT_S; j += 4) {
          uint64_t w = S1_hw[row][j / 4];
          for (int l = 0; l < 4; l++) {
            int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
            q[j + l] = qq;
            if (qq > mq) mq = qq;
          }
        }
        int64_t p[ATT_S];
        int64_t sm = 0;
        for (int j = 0; j < ATT_S; j++) {
          int64_t pv = mx_iexp(q[j] - mq);
          p[j] = pv;
          sm += pv;
        }
        int ls = mx_ilog2((uint64_t)sm);
        uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;
        for (int g = 0; g < GROUPS_S; g++) {
          int64_t gmax = 0;
          for (int j = g * 32; j < (g + 1) * 32; j++)
            if (p[j] > gmax) gmax = p[j];
          if (gmax == 0) {
            P_scales[g][row] = 0;
            for (int j = g * 32; j < (g + 1) * 32; j++) P_q[row][j] = 0;
            continue;
          }
          int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
          P_scales[g][row] = (uint8_t)(e + 127);
          for (int j = g * 32; j < (g + 1) * 32; j++)
            P_q[row][j] = mx_ienc_fp8(p[j], inv_sm, ls, e);
        }
      }

      // -- Mvin the DIM rows just written (one tile at a time) --
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
      }
    }

    // Reload A scales now that P_scales is fully computed
    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);

    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_D);
    gemmini_fence();
  }
}
```
