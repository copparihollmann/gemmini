#include <stdint.h>
#include <string.h>
#include "gemmini.h"
#include "gemmini_params.h"

// Matrix dimensions - must be multiples of DIM=16
#define MATMUL_M 64
#define MATMUL_N 64
#define MATMUL_K 64

#define DIM 16
#define BANK_NUM 4
#define BANK_ROWS 2048
#define TOTAL_SPAD_ROWS (BANK_NUM * BANK_ROWS)  // 8192
#define ACC_ROWS 256
#define MAX_BLOCK_LEN 4

// MX block size along K (number of elements per scale group)
#define MX_BLOCK_SIZE 32

// Output destination in mx_smem (word offset)
#define SPAD_DEST 0

// Tile counts
#define tiles_I (MATMUL_M / DIM)   // 4
#define tiles_J (MATMUL_N / DIM)   // 4
#define tiles_K (MATMUL_K / DIM)   // 4

// Number of scale blocks along K
#define K_BLOCKS ((MATMUL_K + MX_BLOCK_SIZE - 1) / MX_BLOCK_SIZE)  // 2

// Scale arrays: A_scales[K_BLOCKS][tiles_I], B_scales[K_BLOCKS][tiles_J]
// e8m0 scale factors (one per MX block per row/col group)
static uint8_t A_scales_row[K_BLOCKS][MATMUL_M] __attribute__((aligned(64)));
static uint8_t B_scales_col[K_BLOCKS][MATMUL_N] __attribute__((aligned(64)));

// Input matrices in FP8 E4M3 format (uint8_t)
static uint8_t A_fp8[MATMUL_M][MATMUL_K] __attribute__((aligned(64)));
static uint8_t B_fp8[MATMUL_N][MATMUL_K] __attribute__((aligned(64)));  // weight-stationary: B is K-major per tile

// Output matrix in BF16 format (uint16_t, 2 bytes per element)
// BF16 packed 4 per uint64 word
static uint16_t C_hw[MATMUL_M][MATMUL_N] __attribute__((aligned(64)));

// Reference output for verification
static float C_ref[MATMUL_M][MATMUL_N];

// Scratchpad layout:
// A tiles: a_base + (i*tiles_K + k)*DIM
// B tiles: b_base + (j*tiles_K + k)*DIM, b_base = TOTAL_SPAD_ROWS - tiles_K*tiles_J*DIM
#define a_base 0
#define b_base (TOTAL_SPAD_ROWS - tiles_K * tiles_J * DIM)  // 8192 - 4*4*16 = 8192 - 256 = 7936

// Verify no overlap: a needs tiles_I*tiles_K*DIM = 4*4*16 = 256 rows
// a occupies [0, 255], b occupies [7936, 8191] - no overlap

static void init_data(void) {
    // Initialize A with simple FP8 values (using value 0x3C = 1.0 in FP8 E4M3)
    for (int i = 0; i < MATMUL_M; i++) {
        for (int k = 0; k < MATMUL_K; k++) {
            A_fp8[i][k] = 0x3C; // FP8 E4M3 representation of 1.0
        }
    }
    
    // Initialize B with simple FP8 values
    for (int j = 0; j < MATMUL_N; j++) {
        for (int k = 0; k < MATMUL_K; k++) {
            B_fp8[j][k] = 0x3C; // FP8 E4M3 representation of 1.0
        }
    }
    
    // Initialize scales: e8m0 value 127 = 2^0 = 1.0 (bias=127 for e8m0)
    for (int bk = 0; bk < K_BLOCKS; bk++) {
        for (int m = 0; m < MATMUL_M; m++) {
            A_scales_row[bk][m] = 127; // 2^(127-127) = 2^0 = 1.0
        }
        for (int n = 0; n < MATMUL_N; n++) {
            B_scales_col[bk][n] = 127;
        }
    }
}

void matmul_mx_fp8(
    uint8_t A[MATMUL_M][MATMUL_K],
    uint8_t B[MATMUL_N][MATMUL_K],
    uint16_t C[MATMUL_M][MATMUL_N],
    uint8_t A_scales[K_BLOCKS][MATMUL_M],
    uint8_t B_scales[K_BLOCKS][MATMUL_N]
) {
    // =========================================================
    // Step 1: Configure execution - FP8 E4M3 format
    // act_mx_fmt=0 (FP8 E4M3), wgt_mx_fmt=0 (FP8 E4M3), out_mx_fmt=3 (BF16)
    // uselut=0 (FP8 doesn't need LUT)
    // =========================================================
    gemmini_extended3_config_ex(
        WEIGHT_STATIONARY,  // dataflow
        0,                  // sys_act (no activation)
        0,                  // sys_shift
        ACC_SCALE_IDENTITY, // sys_acc_scale
        1,                  // C_stride
        1,                  // A_stride (in units of DIM rows)
        0,                  // A_transpose
        0,                  // B_transpose
        false,              // set_only_strides
        0,                  // act_mx_fmt: FP8 E4M3
        0,                  // wgt_mx_fmt: FP8 E4M3
        3,                  // out_mx_fmt: BF16
        0                   // uselut: not needed for FP8
    );

    // =========================================================
    // Step 2: Load scale factors
    // sel=0 for A (activation) scales, sel=1 for B (weight) scales
    // =========================================================
    gemmini_mx_load_scales((uint64_t)A_scales, sizeof(uint8_t) * K_BLOCKS * MATMUL_M, 0);
    gemmini_mx_load_scales((uint64_t)B_scales, sizeof(uint8_t) * K_BLOCKS * MATMUL_N, 1);

    // =========================================================
    // Step 3: MVIN tiles of A and B into scratchpad
    // A tiles layout: a_base + (i*tiles_K + k)*DIM
    // B tiles layout: b_base + (j*tiles_K + k)*DIM
    // B placed at top of scratchpad: b_base = TOTAL_SPAD_ROWS - tiles_K*tiles_J*DIM
    // =========================================================
    
    // Configure load for A matrix: stride = MATMUL_K bytes (row stride of A in DRAM)
    gemmini_extended3_config_ld(MATMUL_K * sizeof(uint8_t), MVIN_SCALE_IDENTITY, false, 0);
    
    // Load all A tiles into scratchpad
    for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
            uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
            uint8_t *dram_ptr = &A[i * DIM][k * DIM];
            gemmini_extended_mvin(
                (uint64_t)dram_ptr,
                spad_addr,
                DIM,   // cols
                DIM    // rows
            );
        }
    }

    // Configure load for B matrix: stride = MATMUL_K bytes (row stride of B in DRAM)
    // B is stored as [N][K], so each row of B tile has stride MATMUL_K
    gemmini_extended3_config_ld(MATMUL_K * sizeof(uint8_t), MVIN_SCALE_IDENTITY, false, 1);
    
    // Load all B tiles into scratchpad using mvin2
    for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
            uint32_t spad_addr = b_base + (j * tiles_K + k) * DIM;
            uint8_t *dram_ptr = &B[j * DIM][k * DIM];
            gemmini_extended_mvin2(
                (uint64_t)dram_ptr,
                spad_addr,
                DIM,   // cols
                DIM    // rows
            );
        }
    }

    // =========================================================
    // Step 5: Configure the requantized mvout
    // scale_dram points to scale factors for output requantization
    // For BF16 output, we use the loaded A and B scales
    // scale_act_sel=0 (use A scale buffer), scale_w_sel=0 (use B scale buffer)
    // =========================================================
    gemmini_extended_config_st(MATMUL_N * sizeof(uint16_t), 0, ACC_SCALE_IDENTITY);
    
    gemmini_mxquant_config_mvout(
        (uint64_t)B_scales,     // scale_dram (used for output requantization reference)
        tiles_I,                // i_bound
        tiles_J,                // j_bound  
        tiles_K,                // k_bound
        0,                      // scale_act_sel: use A scale buffer (sel=0)
        0,                      // scale_w_sel: use B scale buffer (sel=0)
        1                       // lut_update_granularity
    );

    // =========================================================
    // Step 6: Compute - weight-stationary tiled matmul
    // funct 24 (LOOP_WS_CONFIG_SPAD_AB) is issued inside gemmini_loop_ws_spad
    // The mx_loop_spad_marker is one-shot, so this must be re-issued each time
    // B end address = BANK_NUM*BANK_ROWS (top of scratchpad)
    // skips=0x38 suppresses redundant operations
    // =========================================================
    gemmini_loop_ws_spad(
        tiles_I,            // I tile count
        tiles_J,            // J tile count
        tiles_K,            // K tile count
        0,                  // pad_I
        0,                  // pad_J
        0,                  // pad_K
        a_base,             // A scratchpad base address
        TOTAL_SPAD_ROWS,    // B scratchpad end address (top of scratchpad)
        0,                  // D (bias) - not used
        SPAD_DEST,          // C output destination in mx_smem
        false,              // A_transpose
        false,              // B_transpose
        false,              // full_C
        false,              // low_D
        false,              // ex_accumulate
        NO_ACTIVATION,      // activation
        0,                  // a_spad_id (no reuse)
        0,                  // b_spad_id (no reuse)
        false,              // is_resadd
        0x38                // skips bitmask
    );

    // =========================================================
    // Step 7: Read results from mx_smem to DRAM
    // BF16 packed 4 per uint64 word
    // smem_off_bf16 = SPAD_DEST * 16
    // num_bf16 = total number of BF16 elements
    // =========================================================
    gemmini_mx_read_smem(
        (uint64_t)C,
        SPAD_DEST * 16,         // smem word offset (16-bit words)
        MATMUL_M * MATMUL_N     // number of BF16 values
    );
    
    gemmini_fence();
}

int main(void) {
#ifndef BAREMETAL
    if (mlockall(MCL_CURRENT | MCL_FUTURE) != 0) {
        perror("mlockall failed");
        return 1;
    }
#endif

    gemmini_flush(0);

    init_data();

    matmul_mx_fp8(A_fp8, B_fp8, C_hw, A_scales_row, B_scales_col);

    // Verify: with all FP8 values = 1.0 and all scales = 1.0,
    // each output element should be K * 1.0 * 1.0 = 64.0
    // In BF16: 64.0 = 0x4280
    int errors = 0;
    for (int i = 0; i < MATMUL_M && errors < 10; i++) {
        for (int j = 0; j < MATMUL_N && errors < 10; j++) {
            uint16_t expected_bf16 = 0x4280; // 64.0 in BF16
            if (C_hw[i][j] != expected_bf16) {
                errors++;
            }
        }
    }

    if (errors > 0) {
        return 1;
    }

    return 0;
}
