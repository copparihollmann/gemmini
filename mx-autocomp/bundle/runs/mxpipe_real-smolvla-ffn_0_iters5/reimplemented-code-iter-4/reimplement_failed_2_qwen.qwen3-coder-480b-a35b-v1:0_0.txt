// Optimized MX matmul (50x256x512)
void solution(void) {
  // 1. Flush and configure Gemmini once at the beginning
  gemmini_flush(0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Consolidate and pack B scale factors using optimized 64-bit word copies (avoiding memcpy)
  static uint8_t B_scales_contig[BLOCKS_N][B_S_BYTES];
  for (int g = 0; g < 16; g++) {
    for (int j0 = 0; j0 < 4; j0++) {
      uint64_t *src = (uint64_t*)&B_scales_col[g][j0 * 64];
      uint64_t *dst = (uint64_t*)&B_scales_contig[j0][g * 64];
      dst[0] = src[0];
      dst[1] = src[1];
      dst[2] = src[2];
      dst[3] = src[3];
      dst[4] = src[4];
      dst[5] = src[5];
      dst[6] = src[6];
      dst[7] = src[7];
    }
  }

  // 3. Load consolidated activation scales once (no CPU packing needed!)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);

  // Static tiling parameters
  int i0 = 0;
  int mrows = REAL_M; 
  int tiles_I = BLK / DIM; 
  int tiles_J = BLK / DIM; 
  int tiles_K = MATMUL_K / DIM; 
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM; // Place B at top of scratchpad
  int SPAD_DEST = 0; // Use beginning of mx_smem for output
  int pad_i = BLK - mrows; 

  // 4. Hoist A MVIN: Load A tiles directly from DRAM to scratchpad exactly once
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // 5. Execution loop over N-blocks
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK; 
    if (ncols > BLK) ncols = BLK; 
    if (ncols <= 0) continue;
    int pad_j = BLK - ncols;

    // Load weight scale factors for the current column block j0
    gemmini_mx_load_scales((uint64_t)&B_scales_contig[j0][0], B_S_BYTES, 1);

    // MVIN B tiles directly from source DRAM
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                              b_base + (j * tiles_K + k) * DIM, DIM, DIM);
      }
    }

    // Requantizer setup & Compute execution
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
    
    // Issue the one-shot marker before the compute loop
    ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, a_base, b_base, k_LOOP_WS_CONFIG_SPAD_AB);
    
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, a_base, b_base, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // Store outputs directly from SMEM to DRAM
    int output_words = (mrows * ncols + 3) / 4; // 4 BF16 values per uint64 word
    gemmini_mx_read_smem(&C_hw[i0 * BLK][(j0 * BLK)], SPAD_DEST * 16, output_words * 4);
    gemmini_fence();
  }
}
