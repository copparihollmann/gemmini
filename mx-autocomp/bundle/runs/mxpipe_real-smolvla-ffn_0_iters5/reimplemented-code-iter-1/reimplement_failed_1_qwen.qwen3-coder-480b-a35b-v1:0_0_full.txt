2. The accumulator destination address calculation is wrong
3. The padding parameters in `gemmini_loop_ws_spad` don't account for the actual matrix dimensions
4. The scale loading and memory layout has issues

void solution(void) {
  // Hoisted gemmini_flush(0) call to outside of all loops
  gemmini_flush(0);

  for (int i0 = 0; i0 < BLOCKS_M; i0++)
   for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles_I = (mrows + DIM - 1) / DIM;  // Ceiling division
      int tiles_J = (ncols + DIM - 1) / DIM;  // Ceiling division
      int tiles_K = MATMUL_K / DIM;
      
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;
      int SPAD_DEST = 0;  // Use bottom of accumulator
      int pad_i = tiles_I * DIM - mrows;
      int pad_j = tiles_J * DIM - ncols;

      // Configure execution
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      
      // Load scale factors
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);  // sel=0 for A/activation scales
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);  // sel=1 for B/weight scales

      // Load A tiles into scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);

      // Load B tiles into scratchpad
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++)
        for (int k = 0; k < tiles_K; k++)
          gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                 b_base + (j * tiles_K + k) * DIM, DIM, DIM);

      // Configure output
      gemmini_config_st(MATMUL_N * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // Compute the matmul
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, 
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Read results
      if (mrows == BLK && ncols == BLK && (BLK % DIM) == 0) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK][j0 * BLK], SPAD_DEST * 16, BLK * BLK);
      } else {
        for (int r = 0; r < mrows; r++)
          gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK],
                               SPAD_DEST * 16 + r * tiles_J * DIM, ncols);
      }
    }
  gemmini_fence();
}
