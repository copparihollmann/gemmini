void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK;
    if (mrows > BLK) mrows = BLK;
    if (mrows <= 0) continue;

    // Hoist invariant A scale preparation and loading
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
    }

    gemmini_flush(0);
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                /*C_stride*/1, /*A_stride*/1,
                                /*A_transpose*/0, /*B_transpose*/0,
                                /*set_only_strides*/false,
                                /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3,
                                /*uselut*/0);
    gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, /*sel*/0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK;
      if (ncols > BLK) ncols = BLK;
      if (ncols <= 0) continue;

      // Load variant B scales for current j0 block
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, /*sel*/1);

      int tiles_I = (mrows + DIM - 1) / DIM;  // Round up to nearest tile
      int tiles_J = (ncols + DIM - 1) / DIM;  // Round up to nearest tile
      int tiles_K = MATMUL_K / DIM;
      
      // Calculate proper padding
      int pad_i = tiles_I * DIM - mrows;
      int pad_j = tiles_J * DIM - ncols;
      
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      int SPAD_DEST = 0;  // Use start of accumulator for output

      // Load A tiles
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
              (void *)(((elem_t *)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Load B tiles
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2(
              (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
              b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Configure output and launch compute
      gemmini_config_st(MATMUL_N * sizeof(out_t));  // Full row stride
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   /*i_bound*/tiles_I, /*j_bound*/tiles_J, /*k_bound*/tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);

      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K,
                           pad_i, pad_j, /*pad_K*/0,
                           a_base, b_base, /*D*/0, SPAD_DEST,
                           /*A_transpose*/false, /*B_transpose*/false,
                           /*full_C*/false, /*low_D*/false, /*ex_accumulate*/false,
                           NO_ACTIVATION, /*a_spad_id*/0, /*b_spad_id*/0,
                           /*is_resadd*/false, /*skips*/0x38);

      // Drain results - handle both full and partial tiles
      int total_elems = tiles_I * tiles_J * DIM * DIM;
      gemmini_mx_read_smem(&C_hw[i0 * BLK][j0 * BLK], SPAD_DEST * 16, total_elems);
    }
  }
  gemmini_fence();
}
