// Optimized MX matmul (50x256x512)
void solution(void) {
  // Hoist invariant configurations outside of all loops
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Buffer scales contiguously in DRAM to completely bypass host-side nested loop memcpys
  static uint8_t A_scales_contig[BLOCKS_M][A_S_BYTES];
  static uint8_t B_scales_contig[BLOCKS_N][B_S_BYTES];

  // Pre-pack A scales: iterate over groups first for spatial locality
  for (int g = 0; g < MATMUL_K / 32; g++) {
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      memcpy(&A_scales_contig[i0][g * BLK], &A_scales_row[g][i0 * BLK], BLK);
    }
  }

  // Pre-pack B scales: iterate over groups first for spatial locality
  for (int g = 0; g < MATMUL_K / 32; g++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      memcpy(&B_scales_contig[j0][g * BLK], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    // Determine exact row count and padding for current M-block
    int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

    // Hoist: Load activation scale factors once for the entire row block i0
    gemmini_mx_load_scales((uint64_t)&A_scales_contig[i0][0], A_S_BYTES, 0);

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Determine exact column count and padding for current N-block
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Load weight scale factors for the current column block j0
      gemmini_mx_load_scales((uint64_t)&B_scales_contig[j0][0], B_S_BYTES, 1);

      gemmini_flush(0);

      // MVIN A tiles directly from source DRAM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // MVIN B tiles directly from source DRAM
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (k * tiles + j) * DIM, DIM, DIM);
        }
      }

      // Requantizer setup & Compute execution
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
      // Issue the one-shot marker before the loop
      ROCC_INSTRUCTION_RS1_RS2(XCUSTOM_ACC, a_base, BANK_NUM * BANK_ROWS, k_LOOP_WS_CONFIG_SPAD_AB);
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Store outputs directly from SMEM to DRAM
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
  }
}
