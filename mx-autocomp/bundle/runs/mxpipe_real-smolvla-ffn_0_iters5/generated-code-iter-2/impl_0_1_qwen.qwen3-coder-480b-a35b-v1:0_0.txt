#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_data_mx_lut_hw.h"

// Optimized MX matmul solution with minimal data movement and host-staging eliminated.
void solution(void) {
  // 1. Hoist invariant global configurations completely outside
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // 2. Load scale factors directly from DRAM, eliminating memcpy staging overhead
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. Configure tiling dimensions for the entire matrix
  int tiles_I = REAL_M / DIM;
  int tiles_J = REAL_N / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  int SPAD_DEST = 128;

  static uint32_t scale_factors[512] __attribute__((aligned(32))) = {0};

  gemmini_flush(0);

  // 4. MVIN all A tiles directly from strided DRAM into scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // 5. MVIN all B tiles directly from strided DRAM into scratchpad
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j * DIM)),
                            b_base + (k * tiles_J + j) * DIM, DIM, DIM);
    }
  }

  // 6. Configure output store stride and requantizer bounds for the whole matrix
  gemmini_config_st(REAL_N * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 7. Single-shot fused hardware loop computation
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // 8. Continuous block readout directly from shared memory to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * REAL_N / BF16_PER_WORD);
  gemmini_fence();
}
