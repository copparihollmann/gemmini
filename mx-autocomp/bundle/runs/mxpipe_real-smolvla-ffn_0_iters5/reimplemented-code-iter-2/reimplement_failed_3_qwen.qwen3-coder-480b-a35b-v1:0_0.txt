// Optimized outer-tiled MX matmul with weight-stationary B reuse
void solution(void) {
  // Hoist invariant configurations outside all loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              /*C_stride*/1, /*A_stride*/1, /*A_transpose*/0, /*B_transpose*/0,
                              /*set_only_strides*/false,
                              /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  const int tiles_K = MATMUL_K / DIM;

  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    // Compute only the real columns, pad the rest.
    int ncols = REAL_N - j0 * BLK;
    if (ncols > BLK) ncols = BLK;
    if (ncols <= 0) continue;
    int pad_j = BLK - ncols;

    // Prepare and load B scales (column scales) once per j0
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
    }
    gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1); // sel=1 for B (weight) scales

    // Calculate B tile layout: place at top of scratchpad
    const int tiles = BLK / DIM;
    const uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
    const int SPAD_DEST_BASE = 128;

    // Load B tiles into high scratchpad region once per j0
    gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // B rows: stride N
    for (int j = 0; j < tiles; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin(  // Changed from mvin2 to mvin
            (void *)(((elem_t *)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
            b_base + (k * tiles + j) * DIM, DIM, DIM);
      }
    }

    // Iterate over A tiles (inner loop for weight-stationary reuse)
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      // Compute only the real rows, pad the rest.
      int mrows = REAL_M - i0 * BLK;
      if (mrows > BLK) mrows = BLK;
      if (mrows <= 0) continue;
      int pad_i = BLK - mrows;

      // Prepare and load A scales (row scales) per i0
      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
      }
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0); // sel=0 for A (activation) scales

      // Place A tiles at bottom of scratchpad
      const uint32_t a_base = 0;
      const int SPAD_DEST = SPAD_DEST_BASE + i0; // Unique dest per i0 to avoid clobbering

      // Load A tiles into scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // A rows: stride K
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin(
              (void *)(((elem_t *)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
              a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // Configure requantized mvout and execute computation
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                           false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Read results from shared memory
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence(); // Ensure all outputs are written before next iteration
    }
  }
}
