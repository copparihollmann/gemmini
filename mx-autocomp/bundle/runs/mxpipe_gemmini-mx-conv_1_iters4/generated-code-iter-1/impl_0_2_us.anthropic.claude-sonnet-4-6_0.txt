void solution(void) {
  // --- Hoist all configuration before im2col ---
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                               1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure A load stride (invariant across all A tiles)
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));

  // --- Fused im2col + mvin A: issue mvin as soon as each DIM×DIM tile is ready ---
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      // Fill the DIM rows x DIM cols of A_buf for tile (i,k)
      for (int row = i * DIM; row < (i + 1) * DIM; row++) {
        int pi = row / PATCHES_X;
        int pj = row % PATCHES_X;
        for (int col = k * DIM; col < (k + 1) * DIM; col++) {
          int c  = col / (CONV_KSZ * CONV_KSZ);
          int dy = (col % (CONV_KSZ * CONV_KSZ)) / CONV_KSZ;
          int dx = col % CONV_KSZ;
          A_buf[row][col] = X_in[c][pi * CONV_KSZ + dy][pj * CONV_KSZ + dx];
        }
      }
      // Immediately DMA this tile to the scratchpad
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin(
          (void*)(((elem_t*)A_buf) + i * DIM * MATMUL_K + k * DIM),
          sp_addr, DIM, DIM);
      // CPU proceeds to next tile while DMA runs
    }
  }

  // --- MVIN B (separate config_ld for B's stride) ---
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++)
    for (int k = 0; k < tiles_K; k++)
      gemmini_extended_mvin(
          (void*)(((elem_t*)B_in) + j * DIM * MATMUL_K + k * DIM),
          b_base + (j * tiles_K + k) * DIM, DIM, DIM);

  // --- Configure output and compute ---
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
                       false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
