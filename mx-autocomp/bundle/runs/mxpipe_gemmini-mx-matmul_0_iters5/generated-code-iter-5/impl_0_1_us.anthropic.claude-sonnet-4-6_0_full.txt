```c
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

// Verify compile-time tile counts match the unroll factor below
static_assert(MATMUL_M / DIM == 4 && MATMUL_K / DIM == 4,
              "unroll count mismatch: expected tiles_I=4 and tiles_K=4");

void solution(void) {
    // Tile counts (all compile-time constants for this 64×64×64 problem)
    const int tiles_I = MATMUL_M / DIM;  // 4
    const int tiles_J = MATMUL_N / DIM;  // 4
    const int tiles_K = MATMUL_K / DIM;  // 4

    // Scratchpad layout
    const uint32_t a_base    = 0;
    const uint32_t b_base    = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    const uint32_t SPAD_DEST = 0;

    // -----------------------------------------------------------------------
    // 1. Configure execution (fp8 e4m3 inputs, BF16 output, weight-stationary)
    // -----------------------------------------------------------------------
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        /*C_stride*/1, /*A_stride*/1,
        /*A_transpose*/0, /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

    // -----------------------------------------------------------------------
    // 2. Load scale factors (A: sel=0, B: sel=1)
    // -----------------------------------------------------------------------
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // -----------------------------------------------------------------------
    // 3. MVIN A tiles — fully unrolled, 4 iterations, each len=tiles_K=4
    //    Stride = MATMUL_K * sizeof(elem_t) = 64 bytes (set once, covers B too
    //    since MATMUL_N == MATMUL_K == 64).
    //    Each gemmini_block_mvin transfers tiles_K*DIM=64 cols × DIM=16 rows
    //    = one complete row-group of A (exactly 1024 bytes, within MAX_BLOCK_LEN).
    // -----------------------------------------------------------------------
    gemmini_config_ld(MATMUL_K * sizeof(elem_t));

    {
        const elem_t * const A_ptr = (const elem_t *)A_in;

        gemmini_block_mvin((const void *)(A_ptr + 0 * DIM * MATMUL_K),
                           a_base + 0 * tiles_K * DIM, tiles_K);
        gemmini_block_mvin((const void *)(A_ptr + 1 * DIM * MATMUL_K),
                           a_base + 1 * tiles_K * DIM, tiles_K);
        gemmini_block_mvin((const void *)(A_ptr + 2 * DIM * MATMUL_K),
                           a_base + 2 * tiles_K * DIM, tiles_K);
        gemmini_block_mvin((const void *)(A_ptr + 3 * DIM * MATMUL_K),
                           a_base + 3 * tiles_K * DIM, tiles_K);
    }

    // -----------------------------------------------------------------------
    // 4. MVIN B tiles — fully unrolled, 4 iterations, each len=tiles_J=4
    //    Stride remains MATMUL_N * sizeof(elem_t) = 64 bytes (same config_ld).
    //    Each gemmini_block_mvin transfers tiles_J*DIM=64 cols × DIM=16 rows
    //    = one complete row-group of B (exactly 1024 bytes, within MAX_BLOCK_LEN).
    //    B placed at top of scratchpad (b_base), growing downward, no overlap
    //    with A tiles at a_base.
    // -----------------------------------------------------------------------
    {
        const elem_t * const B_ptr = (const elem_t *)B_in;

        gemmini_block_mvin((const void *)(B_ptr + 0 * DIM * MATMUL_N),
                           b_base + 0 * tiles_J * DIM, tiles_J);
        gemmini_block_mvin((const void *)(B_ptr + 1 * DIM * MATMUL_N),
                           b_base + 1 * tiles_J * DIM, tiles_J);
        gemmini_block_mvin((const void *)(B_ptr + 2 * DIM * MATMUL_N),
                           b_base + 2 * tiles_J * DIM, tiles_J);
        gemmini_block_mvin((const void *)(B_ptr + 3 * DIM * MATMUL_N),
                           b_base + 3 * tiles_J * DIM, tiles_J);
    }

    // -----------------------------------------------------------------------
    // 5. Configure output store and requantizer
    //    i_bound/j_bound/k_bound exactly match tiles_I/tiles_J/tiles_K used
    //    in the subsequent gemmini_loop_ws_spad call (required for correctness).
    // -----------------------------------------------------------------------
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        tiles_I, tiles_J, tiles_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // -----------------------------------------------------------------------
    // 6. Execute fused weight-stationary tiled matmul
    //    LOOP_WS_CONFIG_SPAD_AB (funct 24) is re-issued inside the macro
    //    (mx_loop_spad_marker is one-shot — must be set before every call).
    //    skip_mask=0x38: skip ldA + ldB + ldD (already in scratchpad).
    //    SPAD_DEST=0: results written to mx_smem at word offset 0.
    // -----------------------------------------------------------------------
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
        a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // -----------------------------------------------------------------------
    // 7. Drain BF16 results from shared memory to DRAM, then fence
    //    smem_off_bf16 = SPAD_DEST*16 = 0 (minimum possible offset).
    //    num_bf16 = MATMUL_M * MATMUL_N = 4096 16-bit values.
    // -----------------------------------------------------------------------
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
```
