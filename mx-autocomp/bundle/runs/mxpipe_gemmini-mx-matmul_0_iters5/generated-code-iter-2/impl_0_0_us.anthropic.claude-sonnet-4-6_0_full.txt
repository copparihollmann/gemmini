void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure mvin port 0 (id=0) for A tiles: stride = MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Configure mvin2 port 1 (id=1) for B tiles: stride = MATMUL_N bytes per row
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Interleave A (mvin port 0) and B (mvin2 port 1) loads so the hardware
  // can execute both DMA streams in parallel, hiding DRAM latency of one
  // behind the other.  The two ports are independent, so issuing them
  // back-to-back lets the DMA engines run concurrently.
  int common = (tiles_I < tiles_K) ? tiles_I : tiles_K;
  int i, k;

  // Issue paired A and B commands together up to the shorter dimension.
  for (i = 0, k = 0; i < common; i++, k++) {
    elem_t *a_ptr = ((elem_t*)A_in) + i * DIM * MATMUL_K;
    uint32_t a_sp  = a_base + (i * tiles_K) * DIM;
    gemmini_extended_mvin((void*)a_ptr, a_sp, tiles_K * DIM, DIM);

    elem_t *b_ptr = ((elem_t*)B_in) + k * DIM * MATMUL_N;
    uint32_t b_sp  = b_base + (k * tiles_J) * DIM;
    gemmini_extended_mvin2((void*)b_ptr, b_sp, tiles_J * DIM, DIM);
  }

  // Drain any remaining A tile rows (tiles_I > tiles_K).
  for (; i < tiles_I; i++) {
    elem_t *a_ptr = ((elem_t*)A_in) + i * DIM * MATMUL_K;
    uint32_t a_sp  = a_base + (i * tiles_K) * DIM;
    gemmini_extended_mvin((void*)a_ptr, a_sp, tiles_K * DIM, DIM);
  }

  // Drain any remaining B k-slabs (tiles_K > tiles_I).
  for (; k < tiles_K; k++) {
    elem_t *b_ptr = ((elem_t*)B_in) + k * DIM * MATMUL_N;
    uint32_t b_sp  = b_base + (k * tiles_J) * DIM;
    gemmini_extended_mvin2((void*)b_ptr, b_sp, tiles_J * DIM, DIM);
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
