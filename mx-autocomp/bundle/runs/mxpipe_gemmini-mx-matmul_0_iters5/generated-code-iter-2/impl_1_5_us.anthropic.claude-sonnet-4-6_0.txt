#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

void solution(void) {
    // Tile counts
    const int tiles_I = MATMUL_M / DIM;  // 4
    const int tiles_J = MATMUL_N / DIM;  // 4
    const int tiles_K = MATMUL_K / DIM;  // 4

    // Scratchpad layout
    const uint32_t a_base    = 0;
    const uint32_t b_base    = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    const uint32_t SPAD_DEST = 128;

    // -----------------------------------------------------------------------
    // 1. Configure execution (fp8 e4m3 inputs, BF16 output, weight-stationary)
    // -----------------------------------------------------------------------
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        /*C_stride*/1, /*A_stride*/1,
        /*A_transpose*/0, /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

    // -----------------------------------------------------------------------
    // 2. Load scale factors (A: sel=0, B: sel=1)
    // -----------------------------------------------------------------------
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // -----------------------------------------------------------------------
    // 3 & 4. MVIN A and B tiles concurrently using two DMA ports.
    //
    //   Port 0 (mvin,  id=0): A tiles, stride = MATMUL_K * sizeof(elem_t)
    //   Port 1 (mvin2, id=1): B tiles, stride = MATMUL_N * sizeof(elem_t)
    //
    //   tiles_I == tiles_K == 4, so we unify both loops over t=0..3:
    //     - A batch for i=t: DIM rows × (tiles_K*DIM) cols starting at
    //       A_ptr + t*DIM*MATMUL_K, placed at spad a_base + t*tiles_K*DIM
    //     - B batch for k=t: DIM rows × (tiles_J*DIM) cols starting at
    //       B_ptr + t*DIM*MATMUL_N, placed at spad b_base + t*tiles_J*DIM
    //
    //   Issuing port-0 and port-1 instructions back-to-back lets the two DMA
    //   engines overlap their transfers, roughly halving total mvin latency.
    // -----------------------------------------------------------------------

    // Configure port 0 (id=0) for A tiles: row stride = MATMUL_K bytes
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    // Configure port 1 (id=1) for B tiles: row stride = MATMUL_N bytes
    gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

    {
        const elem_t *A_ptr = (const elem_t *)A_in;
        const elem_t *B_ptr = (const elem_t *)B_in;

        // tiles_I == tiles_K == 4; interleave A (i=t) and B (k=t) loads
        for (int t = 0; t < tiles_I; t++) {
            // A: batch all tiles_K k-tiles for row-group i=t on port 0.
            // tiles_K*DIM columns, DIM rows — fills spad rows
            //   a_base + t*tiles_K*DIM  ..  a_base + (t+1)*tiles_K*DIM - 1
            gemmini_extended_mvin(
                (const void *)(A_ptr + (size_t)(t * DIM) * MATMUL_K),
                a_base + t * tiles_K * DIM,
                tiles_K * DIM,   /* cols */
                DIM              /* rows */
            );

            // B: batch all tiles_J j-tiles for row-group k=t on port 1.
            // tiles_J*DIM columns, DIM rows — fills spad rows
            //   b_base + t*tiles_J*DIM  ..  b_base + (t+1)*tiles_J*DIM - 1
            gemmini_extended_mvin2(
                (const void *)(B_ptr + (size_t)(t * DIM) * MATMUL_N),
                b_base + t * tiles_J * DIM,
                tiles_J * DIM,   /* cols */
                DIM              /* rows */
            );
        }
    }

    // -----------------------------------------------------------------------
    // 5. Configure output store and requantizer
    //    i_bound/j_bound/k_bound must exactly match tiles_I/tiles_J/tiles_K
    //    used in the subsequent gemmini_loop_ws_spad call.
    // -----------------------------------------------------------------------
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        tiles_I, tiles_J, tiles_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // -----------------------------------------------------------------------
    // 6. Execute fused weight-stationary tiled matmul
    //    LOOP_WS_CONFIG_SPAD_AB (funct 24) is re-issued inside the macro
    //    (mx_loop_spad_marker is one-shot and must be set before every call).
    //    skip_mask=0x38: skip ldA + ldB + ldD (already in spad).
    // -----------------------------------------------------------------------
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
        a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // -----------------------------------------------------------------------
    // 7. Drain BF16 results from shared memory to DRAM, then fence
    // -----------------------------------------------------------------------
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
