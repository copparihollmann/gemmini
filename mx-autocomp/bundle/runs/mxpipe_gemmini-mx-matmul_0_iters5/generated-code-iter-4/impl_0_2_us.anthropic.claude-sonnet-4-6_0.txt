#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

void solution(void) {
    // Tile counts
    const int tiles_I = MATMUL_M / DIM;  // 4
    const int tiles_J = MATMUL_N / DIM;  // 4
    const int tiles_K = MATMUL_K / DIM;  // 4

    // Scratchpad layout
    const uint32_t a_base    = 0;
    const uint32_t b_base    = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
    const uint32_t SPAD_DEST = 128;

    // -----------------------------------------------------------------------
    // 1. Configure execution (fp8 e4m3 inputs, BF16 output, weight-stationary)
    // -----------------------------------------------------------------------
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
        /*C_stride*/1, /*A_stride*/1,
        /*A_transpose*/0, /*B_transpose*/0,
        /*set_only_strides*/false,
        /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

    // -----------------------------------------------------------------------
    // 2. Load scale factors (A: sel=0, B: sel=1)
    //    Both issued back-to-back; hardware can pipeline them.
    // -----------------------------------------------------------------------
    gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
    gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

    // -----------------------------------------------------------------------
    // 3. Configure mvin port 0 (A tiles) and mvin port 1 (B tiles).
    //    Both A and B have stride = 64 bytes (MATMUL_K == MATMUL_N == 64).
    //    Port 0 (id=0): gemmini_config_ld sets stride for k_MVIN.
    //    Port 1 (id=1): gemmini_extended3_config_ld sets stride for k_MVIN2.
    // -----------------------------------------------------------------------
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
    gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

    // -----------------------------------------------------------------------
    // 4. Interleaved MVIN of A tiles (port 0) and B tiles (port 1).
    //    tiles_I == tiles_K == 4, so we pair them in a single loop.
    //    Each iteration issues:
    //      - One A block mvin  (port 0, k_MVIN)  for row-group i
    //      - One B block mvin2 (port 1, k_MVIN2) for k-group k
    //    This allows both DMA engines to run concurrently, overlapping
    //    the A and B scratchpad loads and reducing total mvin latency.
    //
    //    A layout: a_base + (i * tiles_K) * DIM  (tiles_K contiguous k-tiles)
    //    B layout: b_base + (k * tiles_J) * DIM  (tiles_J contiguous j-tiles)
    //
    //    gemmini_block_mvin  expands to gemmini_extended_mvin  (port 0).
    //    No gemmini_block_mvin2 macro exists, so B is expanded inline using
    //    gemmini_extended_mvin2 with cols = tiles_J * DIM, rows = DIM.
    // -----------------------------------------------------------------------
    {
        const elem_t *A_ptr = (const elem_t *)A_in;
        const elem_t *B_ptr = (const elem_t *)B_in;

        for (int i = 0; i < tiles_I; i++) {
            // A tile group: row-group i, all k tiles contiguous in DRAM
            const void *a_dram = (const void *)(A_ptr + (size_t)(i * DIM) * MATMUL_K);
            uint32_t    a_sp   = a_base + (uint32_t)(i * tiles_K * DIM);
            gemmini_block_mvin(a_dram, a_sp, tiles_K);

            // B tile group: k-group i (reusing loop variable as k), all j tiles contiguous
            // k == i here because tiles_I == tiles_K == 4
            const void *b_dram = (const void *)(B_ptr + (size_t)(i * DIM) * MATMUL_N);
            uint32_t    b_sp   = b_base + (uint32_t)(i * tiles_J * DIM);
            gemmini_extended_mvin2(b_dram, b_sp, (uint32_t)(tiles_J * DIM), DIM);
        }
    }

    // -----------------------------------------------------------------------
    // 5. Configure output store and requantizer.
    //    i_bound/j_bound/k_bound must exactly match tiles_I/J/K used below.
    // -----------------------------------------------------------------------
    gemmini_config_st(OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors,
        tiles_I, tiles_J, tiles_K,
        /*scale_act_sel*/0, /*scale_w_sel*/0,
        /*lut_update_granularity*/1);

    // -----------------------------------------------------------------------
    // 6. Execute fused weight-stationary tiled matmul.
    //    LOOP_WS_CONFIG_SPAD_AB (funct 24) is re-issued inside the macro
    //    (mx_loop_spad_marker is one-shot and must be set before every call).
    //    skip_mask=0x38: skip ldA + ldB + ldD (already loaded into spad).
    // -----------------------------------------------------------------------
    gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
        a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
        false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

    // -----------------------------------------------------------------------
    // 7. Drain BF16 results from shared memory to DRAM, then fence.
    // -----------------------------------------------------------------------
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
    gemmini_fence();
}
