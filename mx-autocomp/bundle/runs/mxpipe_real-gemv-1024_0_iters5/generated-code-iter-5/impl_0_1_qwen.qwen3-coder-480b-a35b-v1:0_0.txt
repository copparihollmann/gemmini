// Optimized MX matmul with Host-Side Scale Factor Pre-Formatting (Strategy 5 + Strategy 8)
void solution(void) {
  // Pre-allocate contiguous formatting buffers for scales to avoid hot-path CPU copies
  static uint8_t A_scales_preformatted[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_preformatted[BLOCKS_N][MATMUL_K / 32][BLK];

  // Pack A scale factors contiguously once during initialization
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_scales_preformatted[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);
    }
  }

  // Pack B scale factors contiguously once during initialization
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_scales_preformatted[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  // Buffer to hold consolidated SMEM outputs contiguously in DRAM
  static out_t temp_block[BLK][BLK / BF16_PER_WORD];

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Partial last M/N block: compute only the real rows/cols, pad the rest.
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // 1. Flush and configure execution format
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

      // 2. Load scale factors directly from pre-formatted contiguous buffers
      gemmini_mx_load_scales((uint64_t)A_scales_preformatted[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_preformatted[j0], B_S_BYTES, 1);

      // 3. MVIN A tiles directly from source DRAM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // 4. MVIN B tiles directly from source DRAM
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (k * tiles + j) * DIM, DIM, DIM);
        }
      }

      // 5. Configure store and requantizer parameters
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);

      // 6. Compute fused weight-stationary loop
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 7. Drain shared memory in a single consolidated read
      // The output block is contiguous in SMEM with width BLK.
      gemmini_mx_read_smem(temp_block, SPAD_DEST * 16, mrows * BLK);
      gemmini_fence();

      // Distribute the contiguous block rows into the strided destination C_hw
      for (int r = 0; r < mrows; r++) {
        memcpy(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
               temp_block[r],
               (ncols / BF16_PER_WORD) * sizeof(out_t));
      }
    }
  }
}
