// Optimized MX matmul with Host-Side Scale Factor Pre-Formatting (Strategy 7)
void solution(void) {
  // Pre-allocate contiguous formatting buffers for scales to avoid hot-path CPU copies
  static uint8_t A_scales_preformatted[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_preformatted[BLOCKS_N][MATMUL_K / 32][BLK];

  // Pack A scale factors contiguously once during initialization
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_scales_preformatted[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);
    }
  }

  // Pack B scale factors contiguously once during initialization
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_scales_preformatted[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  // 1. Flush and configure execution format globally
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Swap loops: Column block (j0) is now outer to allow stationary reuse of B
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    int ncols = REAL_N - j0 * BLK; 
    if (ncols > BLK) ncols = BLK; 
    if (ncols <= 0) continue;

    int tiles_J = BLK / DIM;
    int tiles_K = MATMUL_K / DIM;
    uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM;

    // 2. Load B scale factors once per Column Block (sel=1 -> B)
    gemmini_mx_load_scales((uint64_t)B_scales_preformatted[j0], B_S_BYTES, 1);

    // 3. MVIN B tiles once per Column Block into the upper scratchpad
    gemmini_config_ld(MATMUL_N * sizeof(elem_t));
    for (int j = 0; j < tiles_J; j++) {
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                              b_base + (k * tiles_J + j) * DIM, DIM, DIM);
      }
    }

    // Row block (i0) is now the inner loop
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      int mrows = REAL_M - i0 * BLK; 
      if (mrows > BLK) mrows = BLK; 
      if (mrows <= 0) continue;

      int tiles_I = BLK / DIM;
      uint32_t a_base = 0;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows;
      int pad_j = BLK - ncols;

      // 4. Load A scale factors for the current row block (sel=0 -> A)
      gemmini_mx_load_scales((uint64_t)A_scales_preformatted[i0], A_S_BYTES, 0);

      // 5. MVIN A tiles into the lower scratchpad
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // 6. Configure store and requantizer parameters
      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

      // 7. Compute fused weight-stationary loop (reusing B tiles at b_base)
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // 8. Drain shared memory to DRAM and synchronize
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
    }
  }
}
