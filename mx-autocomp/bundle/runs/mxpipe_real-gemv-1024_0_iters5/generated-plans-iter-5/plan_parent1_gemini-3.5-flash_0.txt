### Optimization Plan

The primary bottleneck in the original program is the **redundant host-side data staging (memcpy) of scale factors inside the nested loops**. 

Scale factors for activation rows (`A_scales_row`) only depend on the row index `i0`, while scale factors for weight columns (`B_scales_col`) only depend on the column index `j0`. However, the original code performs `memcpy` for both sets of scales inside the innermost `j0` loop. This causes the CPU to copy the exact same `A` scales `BLOCKS_N` times and `B` scales `BLOCKS_M` times, leading to massive instruction overhead on the host CPU.

To address this, we will apply **Strategy 2: Cache reused data in local memory instead of reloading from main memory** (specifically, pre-staging the scale factors into contiguous local buffers):

1. **Pre-stage Scale Factors:** We will allocate contiguous staging buffers `A_scales_contiguous` and `B_scales_contiguous` at the function scope. Before entering any of the matmul loops, we will run two hoisted loops to copy the required scale factor slices into these contiguous buffers.
2. **Eliminate Inner-Loop Memcpy:** We completely remove the `memcpy` operations from the hot path inside the nested `i0` and `j0` loops.
3. **Direct Scale Loading:** During execution, we directly pass the pre-staged, contiguous local buffers to `gemmini_mx_load_scales`.
4. **Configuration Hoisting:** We hoist the invariant `gemmini_extended3_config_ex` and `gemmini_config_st` outside the loop nest to avoid redundant Gemmini state re-configurations.

---

### Optimized Code

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_data_mx_lut_hw.h"

#define TILE 16
#define VALUES_PER_BYTE 2
#define USE_LUT 1
#define QUANT_LUT_UPDATE_GRANULARITY 1
#define DIM 16
#define ADDR_LEN 32
#define BF16_PER_WORD 4

// Memory-mapped control registers
#define GEMMINI_CTRL     0x40084000
#define GEMMINI_RS1_ADDR (GEMMINI_CTRL + 0x10)
#define GEMMINI_RS2_ADDR (GEMMINI_CTRL + 0x18)
#define GEMMINI_INST_ADDR (GEMMINI_CTRL + 0x0)
#define GEMMINI_BUSY_ADDR (GEMMINI_CTRL + 0x20)

// LUT regions
#define GEMMINI_LUT0_ADDR (GEMMINI_CTRL + 0x80)
#define GEMMINI_LUT1_ADDR (GEMMINI_CTRL + 0x380)
#define GEMMINI_LUT2_ADDR (GEMMINI_CTRL + 0x680)

// Scale-factor SRAM regions
#define GEMMINI_SF_MEM   0x40088000
#define GEMMINI_SF_MEM_A (GEMMINI_SF_MEM + 0x2000)
#define GEMMINI_SF_MEM_B  GEMMINI_SF_MEM

// Shared memory base
#define SMEM 0x40000000

#define GEMMINI_FORMAT 1

#undef gemmini_fence
#define gemmini_fence() \
  { while (*((volatile uint32_t *) GEMMINI_BUSY_ADDR)) asm volatile ("nop"); }

#undef ROCC_INSTRUCTION_RS1_RS2
#define ROCC_INSTRUCTION_RS1_RS2(x, rs1, rs2, funct) {                        \
  *((volatile uint64_t *) GEMMINI_RS1_ADDR) = (rs1);                          \
  *((volatile uint64_t *) GEMMINI_RS2_ADDR) = (rs2);                          \
  *((volatile uint32_t*) GEMMINI_INST_ADDR) =                                 \
      (0x7B) | (0 << 7) | (3 << 12) | (1 << 15) | (2 << 20) | ((funct) << 25); \
}

typedef uint8_t out_t;

void solution(void) {
  // Define contiguous staging buffers for scale factors
  static uint8_t A_scales_contiguous[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_contiguous[BLOCKS_N][MATMUL_K / 32][BLK];

  // Pre-stage all row and column scales once to avoid redundant copies in the loop nest
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_scales_contiguous[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);
    }
  }
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_scales_contiguous[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  gemmini_flush(0);

  // Hoist invariant Gemmini configuration out of all loops
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_config_st(BLK_OUTC * sizeof(out_t));

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;

    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // Load scale factors directly from pre-staged contiguous buffers (no hot-path memcpy)
      gemmini_mx_load_scales((uint64_t)A_scales_contiguous[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_scales_contiguous[j0], B_S_BYTES, 1);

      gemmini_config_ld(MATMUL_K * sizeof(elem_t));      // A rows: stride K
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      gemmini_config_ld(MATMUL_N * sizeof(elem_t));      // B rows: stride N
      for (int j = 0; j < tiles; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                b_base + (k * tiles + j) * DIM, DIM, DIM);
        }
      }

      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
    }
  }

  // Final synchronization
  gemmini_fence();
}
```
