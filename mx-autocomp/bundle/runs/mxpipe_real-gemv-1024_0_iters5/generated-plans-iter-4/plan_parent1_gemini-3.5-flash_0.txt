An elegant way to optimize this tiled FP8 matrix multiplication is to apply **Strategy 7: Split execution loops into a "core" phase processing perfectly aligned full tiles and a "fringe" phase for partial edge tiles, completely removing branching and dynamic padding logic from the hot path.**

### Analysis of Inefficiencies in the Original Code
In the baseline code, the outer loops for `i0` and `j0` compute the actual row/column counts `mrows` and `ncols` and the padding parameters `pad_i` and `pad_j` dynamically for every single block. This introduces several overheads on the host CPU:
1. **Branching and dynamic calculations:** The compiler cannot optimize loop boundaries, scale-factor loading, or output DMAs because the block dimensions are treated as variable.
2. **Dynamic strides and sizes:** `gemmini_mx_read_smem` has to handle dynamic `ncols` instead of compile-time constants, which prevents the compiler from optimizing address generation and loop unrolling.

### Optimization Plan
1. **Extract Block Processing to an Inlined Helper:** Create a `static inline void run_block(...)` helper function that processes a single block. By declaring it `inline`, passing compile-time constants (like `BLK`) for fully-aligned blocks allows the compiler to perform constant propagation, entirely optimizing away boundary checks, padding math, and dynamic loops for the core phase.
2. **Separate Core and Fringe Phases:**
   - **Core Phase:** Run the main nested loops for `i0` and `j0` over all perfectly aligned full blocks (`core_BLOCKS_M = REAL_M / BLK` and `core_BLOCKS_N = REAL_N / BLK`). For these blocks, `mrows = BLK` and `ncols = BLK`, so `pad_i = 0` and `pad_j = 0`.
   - **Bottom Fringe Phase:** Handle the bottom edge (last row block `i0 = BLOCKS_M - 1` if it is partial) for all core column blocks.
   - **Right Fringe Phase:** Handle the right edge (last column block `j0 = BLOCKS_N - 1` if it is partial) for all core row blocks.
   - **Corner Fringe Phase:** Handle the single bottom-right block if both dimensions have partial fringe blocks.
3. **Move Pool Buffers to File Scope:** Move the static scale staging pools `A_s_pool` and `B_s_pool` to file scope to avoid redeclaring large static arrays inside nested helper functions, ensuring safe and efficient memory layout.

### Optimized Code

```c
#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

#define DIM 16
#define ADDR_LEN 32

// Staging buffers for block scale factors to avoid write-after-read races
static uint8_t A_s_pool[BLOCKS_M][BLOCKS_N][MATMUL_K / 32][BLK];
static uint8_t B_s_pool[BLOCKS_M][BLOCKS_N][MATMUL_K / 32][BLK];

// Inline helper to execute a single block with constant propagation support
static inline void run_block(int i0, int j0, int mrows, int ncols) {
  // Stage per-32-group scales to the designated block buffer
  for (int g = 0; g < MATMUL_K / 32; g++) {
    memcpy(&A_s_pool[i0][j0][g][0], &A_scales_row[g][i0 * BLK], BLK);
    memcpy(&B_s_pool[i0][j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
  }

  int tiles = BLK / DIM;
  int tiles_K = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST = 128;
  int pad_i = BLK - mrows;
  int pad_j = BLK - ncols;

  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)A_s_pool[i0][j0], A_S_BYTES, 0);
  gemmini_mx_load_scales((uint64_t)B_s_pool[i0][j0], B_S_BYTES, 1);

  // Load A tiles
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // Load B tiles
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                            b_base + (k * tiles + j) * DIM, DIM, DIM);
    }
  }

  // Requantizer Configuration and Execution
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Read results from shared memory
  for (int r = 0; r < mrows; r++) {
    gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                         SPAD_DEST * 16 + r * BLK, ncols);
  }
}

// Main orchestrated tiled matmul split into core and fringe phases
void solution(void) {
  int core_BLOCKS_M = REAL_M / BLK;
  int core_BLOCKS_N = REAL_N / BLK;

  // 1. Core Phase: Execute fully aligned blocks with compile-time constant BLK dimensions
  for (int i0 = 0; i0 < core_BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < core_BLOCKS_N; j0++) {
      run_block(i0, j0, BLK, BLK);
    }
  }

  // 2. Bottom Fringe Phase: Execute the last partial row block across core columns
  if (core_BLOCKS_M < BLOCKS_M) {
    int i0 = BLOCKS_M - 1;
    int mrows = REAL_M - i0 * BLK;
    if (mrows > 0) {
      for (int j0 = 0; j0 < core_BLOCKS_N; j0++) {
        run_block(i0, j0, mrows, BLK);
      }
    }
  }

  // 3. Right Fringe Phase: Execute the last partial column block across core rows
  if (core_BLOCKS_N < BLOCKS_N) {
    int j0 = BLOCKS_N - 1;
    int ncols = REAL_N - j0 * BLK;
    if (ncols > 0) {
      for (int i0 = 0; i0 < core_BLOCKS_M; i0++) {
        run_block(i0, j0, BLK, ncols);
      }
    }
  }

  // 4. Corner Fringe Phase: Execute the bottom-right corner block
  if (core_BLOCKS_M < BLOCKS_M && core_BLOCKS_N < BLOCKS_N) {
    int i0 = BLOCKS_M - 1;
    int j0 = BLOCKS_N - 1;
    int mrows = REAL_M - i0 * BLK;
    int ncols = REAL_N - j0 * BLK;
    if (mrows > 0 && ncols > 0) {
      run_block(i0, j0, mrows, ncols);
    }
  }

  // Enforce memory synchronization barrier
  gemmini_fence();
}
```
