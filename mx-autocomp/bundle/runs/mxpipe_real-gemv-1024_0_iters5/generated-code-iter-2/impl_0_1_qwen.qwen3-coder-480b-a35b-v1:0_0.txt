#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

void solution(void) {
  int tiles = BLK / DIM;         // BLK=64, DIM=16 => tiles=4
  int tiles_K = MATMUL_K / DIM;  // MATMUL_K=64 => tiles_K=4
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM; // Place B at top of spad
  int SPAD_DEST = 128;

  int mrows = REAL_M;  // 64
  int ncols = REAL_N;  // 64
  int pad_i = BLK - mrows; // 0
  int pad_j = BLK - ncols; // 0

  // 1. Configuration
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                              /*C_stride=*/1, /*A_stride=*/1,
                              /*A_transpose=*/0, /*B_transpose=*/0,
                              /*set_only_strides=*/false,
                              /*act_mx_fmt=*/0, /*wgt_mx_fmt=*/0, /*out_mx_fmt=*/3, /*uselut=*/0);

  // 2. Load scales directly from global arrays
  gemmini_mx_load_scales((uint64_t)&A_scales_row, A_S_BYTES, /*sel=*/0); // A/activation scales
  gemmini_mx_load_scales((uint64_t)&B_scales_col, B_S_BYTES, /*sel=*/1); // B/weight scales

  // 3. Load A tiles into scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t)); // Stride for A is K
  for (int i = 0; i < tiles; i++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *addr = &A_in[(i * DIM) * MATMUL_K + k * DIM];
      uint32_t spad_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *)addr, spad_addr, DIM, DIM);
    }
  }

  // 4. Load B tiles into scratchpad
  gemmini_config_ld(MATMUL_N * sizeof(elem_t)); // Stride for B is N
  for (int j = 0; j < tiles; j++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *addr = &B_in[(k * DIM) * MATMUL_N + (j * DIM)];
      uint32_t spad_addr = b_base + (k * tiles + j) * DIM;
      gemmini_extended_mvin((void *)addr, spad_addr, DIM, DIM);
    }
  }

  // 5. Configure output (requantization) and execute
  gemmini_config_st(BLK_OUTC * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                               /*i_bound=*/tiles, /*j_bound=*/tiles, /*k_bound=*/tiles_K,
                               /*scale_act_sel=*/0, /*scale_w_sel=*/0,
                               /*lut_update_granularity=*/1);

  gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                       pad_i, pad_j, /*pad_K=*/0,
                       a_base, b_base, /*D=*/0, SPAD_DEST,
                       /*A_transpose=*/false, /*B_transpose=*/false,
                       /*full_C=*/false, /*low_D=*/false, /*ex_accumulate=*/false,
                       NO_ACTIVATION, /*a_spad_id=*/0, /*b_spad_id=*/0,
                       /*is_resadd=*/false, /*skips=*/0x38);

  // 6. Drain results from shared memory
  for (int r = 0; r < mrows; r++) {
    gemmini_mx_read_smem(&C_hw[r][0], SPAD_DEST * 16 + r * BLK, ncols);
  }

  // 7. Synchronize
  gemmini_fence();
}
