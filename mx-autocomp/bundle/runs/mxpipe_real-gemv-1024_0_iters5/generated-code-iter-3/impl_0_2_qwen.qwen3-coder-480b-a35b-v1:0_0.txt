#include <stdint.h>
#include <string.h>
#include "include/gemmini_testutils.h"

void solution(void) {
  // 1. Calculate tile dimensions for monolithic execution
  int tiles_I = MATMUL_M / DIM;
  int tiles_J = MATMUL_N / DIM;
  int tiles_K = MATMUL_K / DIM;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
  int SPAD_DEST = 128;

  // Compute padding if real dimensions differ from padded multiples of 16
  int pad_i = MATMUL_M - REAL_M;
  int pad_j = MATMUL_N - REAL_N;

  // 2. Flush and configure execution format (FP8 WS execution with BF16 outputs)
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // 3. Load scale factors directly from source arrays, eliminating CPU staging copies
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 4. MVIN A tiles directly from DRAM to scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM;
      uint32_t sp_addr = a_base + (i * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *) dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 5. MVIN B tiles directly from DRAM to scratchpad
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      elem_t *dram_ptr = ((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM;
      uint32_t sp_addr = b_base + (j * tiles_K + k) * DIM;
      gemmini_extended_mvin((void *) dram_ptr, sp_addr, DIM, DIM);
    }
  }

  // 6. Configure store and requantizer parameters
  gemmini_config_st((MATMUL_N / BF16_PER_WORD) * sizeof(out_t));
  static uint32_t scale_factors[1024] __attribute__((aligned(32))) = {0};
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  // 7. Compute the monolithic weight-stationary tiled matrix multiplication in hardware
  gemmini_loop_ws_spad(
      tiles_I, tiles_J, tiles_K,
      pad_i, pad_j, 0,
      a_base,
      BANK_NUM * BANK_ROWS,
      0,
      SPAD_DEST,
      false, false,
      false, false, false,
      NO_ACTIVATION,
      0, 0,
      false,
      0x38); // Skip loading A, B, and D from DRAM; perform compute + store to SMEM

  // 8. Drain the shared memory to DRAM using an optimized coalesced block read
  if (pad_j == 0) {
    // Single-pass completely coalesced read
    gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * REAL_N);
  } else {
    // Row-by-row coalesced read to handle any edge padding stride mismatch
    for (int r = 0; r < REAL_M; r++) {
      gemmini_mx_read_smem(&C_hw[r][0], SPAD_DEST * 16 + r * MATMUL_N, REAL_N);
    }
  }
  gemmini_fence();
}
