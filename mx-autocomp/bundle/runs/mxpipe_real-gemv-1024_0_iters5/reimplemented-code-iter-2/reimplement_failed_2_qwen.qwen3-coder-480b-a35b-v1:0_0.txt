void solution(void) {
  // Based on the hardware constraints and typical usage
  int tiles_I = REAL_M / DIM;  // Assuming REAL_M = 64, so tiles_I = 4
  int tiles_J = REAL_N / DIM;  // Assuming REAL_N = 64, so tiles_J = 4  
  int tiles_K = MATMUL_K / DIM; // Assuming MATMUL_K = 1024, so tiles_K = 64
  
  // Calculate memory requirements:
  // A tiles: tiles_I * tiles_K * DIM = 4 * 64 * 16 = 4096 rows
  // B tiles: tiles_J * tiles_K * DIM = 4 * 64 * 16 = 4096 rows
  // Total: 8192 rows - exactly fills scratchpad
  
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_J * tiles_K * DIM; // Place B at top
  
  // Use a safe destination in accumulator
  int SPAD_DEST = 0;
  
  gemmini_flush(0);

  // 1. Configure execution for weight-stationary FP8
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 
                              REAL_N, MATMUL_K, 0, 0, false, 0, 0, 3, 0);

  // 2. Stream all scale factors directly from contiguous DRAM
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // 3. MVIN A tiles to scratchpad
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i * DIM) * MATMUL_K + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // 4. MVIN B tiles to scratchpad 
  gemmini_config_ld(MATMUL_N * sizeof(elem_t));
  for (int j = 0; j < tiles_J; j++) {
    for (int k = 0; k < tiles_K; k++) {
      gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + j * DIM),
                             b_base + (j * tiles_K + k) * DIM, DIM, DIM);
    }
  }

  // 5. Configure the requantizer 
  gemmini_config_st(REAL_N * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 1, 1);

  // 6. Execute fused hardware loop
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, b_base, 0, SPAD_DEST,
                       false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // 7. Vectorized readout of BF16-packed results
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, REAL_M * REAL_N);

  // 8. Synchronize
  gemmini_fence();
}
