// Optimized MX matmul with Host-Side Scale Factor Pre-Formatting (Strategy 5)
void solution(void) {
  // Pre-allocate contiguous formatting buffers for scales to avoid hot-path CPU copies
  static uint8_t A_scales_preformatted[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t B_scales_preformatted[BLOCKS_N][MATMUL_K / 32][BLK];

  // Pack A scale factors contiguously once during initialization
  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&A_scales_preformatted[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);
    }
  }

  // Pack B scale factors contiguously once during initialization
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    for (int g = 0; g < MATMUL_K / 32; g++) {
      memcpy(&B_scales_preformatted[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);
    }
  }

  // 1. Flush and configure execution format ONCE at the start of the computation
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
  int SPAD_DEST = 0; // Start from beginning of shared memory

  // Outer loop over weight columns (N-dimension blocks)
  for (int j0 = 0; j0 < BLOCKS_N; j0++) {
    
    // Inner loop over activation rows (M-dimension blocks)
    for (int i0 = 0; i0 < BLOCKS_M; i0++) {
      // Partial last M/N block: compute only the real rows/cols, pad the rest.
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;
      int pad_i = (BLK - mrows + DIM - 1) / DIM * DIM; // Round up to nearest DIM
      int pad_j = (BLK - ncols + DIM - 1) / DIM * DIM; // Round up to nearest DIM
      int tiles_I = (mrows + DIM - 1) / DIM;
      int tiles_J = (ncols + DIM - 1) / DIM;

      // 2. Load A scale factors directly from pre-formatted contiguous buffers (specific to i0)
      gemmini_mx_load_scales((uint64_t)A_scales_preformatted[i0], A_S_BYTES, 0);
      
      // 3. Load B scale factors directly from pre-formatted contiguous buffers (specific to j0)
      gemmini_mx_load_scales((uint64_t)B_scales_preformatted[j0], B_S_BYTES, 1);

      // 4. MVIN A tiles directly from source DRAM
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // 5. MVIN B tiles directly from source DRAM
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));
      for (int j = 0; j < tiles_J; j++) {
        for (int k = 0; k < tiles_K; k++) {
          gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK + j * DIM)),
                                 b_base + (j * tiles_K + k) * DIM, DIM, DIM);
        }
      }

      // 6. Configure store and requantizer parameters
      gemmini_config_st(MATMUL_N * sizeof(uint16_t)); // BF16 output
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, 
                                   tiles_I, tiles_J, tiles_K, 0, 1, 1); // scale_w_sel should be 1 for B scales

      // 7. Compute fused weight-stationary loop
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 
                           pad_i/DIM, pad_j/DIM, 0,
                           a_base, b_base, 0, SPAD_DEST,
                           false, false, false, false, false, 
                           NO_ACTIVATION, 0, 0, false, 0x38);

      // 8. Drain shared memory to DRAM and synchronize
      int bf16_words = (ncols + 3) / 4; // 4 BF16 values per uint64 word
      for (int r = 0; r < mrows; r++) {
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][j0 * BLK / 4],
                             SPAD_DEST * 16 + r * BLK, ncols);
      }
      gemmini_fence();
      
      // Update SPAD_DEST for next iteration to avoid overwriting
      SPAD_DEST += tiles_I * tiles_J * DIM;
      if (SPAD_DEST >= ACC_ROWS) SPAD_DEST = 0; // Reset if needed
    }
  }
}
