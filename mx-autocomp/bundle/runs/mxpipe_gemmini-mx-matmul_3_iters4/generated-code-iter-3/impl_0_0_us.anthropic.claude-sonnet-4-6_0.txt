void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 0);
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 1);

  // Precompute strides to avoid repeated multiplications in hot loops
  const int spad_stride_i = tiles_K * DIM;  // spad addr step per i tile
  const int spad_stride_j = tiles_K * DIM;  // spad addr step per j tile
  const int dram_stride_i = DIM * MATMUL_K; // dram elem step per i tile (in elem_t units)
  const int dram_stride_j = DIM * MATMUL_K; // dram elem step per j tile (in elem_t units)

  // Running base pointers that advance by DIM each k iteration
  uint32_t k_spad_a = a_base;           // a_base + k*DIM
  uint32_t k_spad_b = b_base;           // b_base + k*DIM
  const elem_t *k_dram_a = (const elem_t *)A_in;  // A_in + k*DIM (in elem_t units)
  const elem_t *k_dram_b = (const elem_t *)B_in;  // B_in + k*DIM (in elem_t units)

  for (int k = 0; k < tiles_K; k++) {
    // A tiles: spad address = a_base + (i*tiles_K + k)*DIM = k_spad_a + i*spad_stride_i
    // DRAM pointer = A_in + i*DIM*MATMUL_K + k*DIM = k_dram_a + i*dram_stride_i
    uint32_t sp_addr_a = k_spad_a;
    const elem_t *dram_a = k_dram_a;
    for (int i = 0; i < tiles_I; i++) {
      gemmini_extended_mvin(dram_a, sp_addr_a, DIM, DIM);
      sp_addr_a += spad_stride_i;
      dram_a    += dram_stride_i;
    }

    // B tiles: spad address = b_base + (j*tiles_K + k)*DIM = k_spad_b + j*spad_stride_j
    // DRAM pointer = B_in + j*DIM*MATMUL_K + k*DIM = k_dram_b + j*dram_stride_j
    uint32_t sp_addr_b = k_spad_b;
    const elem_t *dram_b = k_dram_b;
    for (int j = 0; j < tiles_J; j++) {
      gemmini_extended_mvin2(dram_b, sp_addr_b, DIM, DIM);
      sp_addr_b += spad_stride_j;
      dram_b    += dram_stride_j;
    }

    // Advance k-dimension bases by DIM for next k iteration
    k_spad_a += DIM;
    k_spad_b += DIM;
    k_dram_a += DIM;
    k_dram_b += DIM;
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);

  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
