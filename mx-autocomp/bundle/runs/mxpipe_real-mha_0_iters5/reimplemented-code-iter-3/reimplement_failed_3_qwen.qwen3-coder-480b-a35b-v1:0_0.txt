#include <stdint.h>
#include "include/gemmini_testutils.h"

// Define the missing constants - these would typically come from the test framework
#ifndef ATT_H
#define ATT_H 32
#endif

#ifndef ATT_HKV
#define ATT_HKV 8
#endif

#ifndef ATT_S
#define ATT_S 64
#endif

#ifndef ATT_D
#define ATT_D 64
#endif

#ifndef ATT_CAUSAL
#define ATT_CAUSAL 0
#endif

#define GROUP_SIZE (ATT_H / ATT_HKV)
#define GROUPS_S (ATT_S / 32)

#define DIM 16
#define BANK_NUM 4
#define BANK_ROWS 2048
#define ACC_ROWS 256
#define MAX_BLOCK_LEN 4
#define MAX_BLOCK_LEN_ACC 1

#define ADDR_LEN 32
#define SPAD_DEST 0

#define QSS_BYTES (ATT_S)
#define VSS_BYTES (ATT_S)

#define S_OUT_COLS ATT_S
#define D_OUT_COLS ATT_D

#define out_t uint64_t

// Mock function declarations for the missing external data
static elem_t *QTs[8];
static uint8_t *QTss[8];
static elem_t *Vs[8];
static uint8_t *Vss[8];
static elem_t *Qs[32];
static uint8_t *Qss[32];
static uint8_t scale_factors[ATT_S * ATT_S];

// Mock implementations for missing helper functions
static int64_t mx_ibf16_to_q(uint16_t w) {
    return (int64_t)w;
}

static int64_t mx_iexp(int64_t x) {
    return x > 0 ? x : 0;
}

static int mx_ilog2(uint64_t x) {
    int count = 0;
    while (x > 1) {
        x >>= 1;
        count++;
    }
    return count;
}

static uint8_t mx_ienc_fp8(int64_t p, uint64_t inv_sm, int ls, int e) {
    return (uint8_t)(p >> (ls + e));
}

#define MX_FRAC 8
#define MX_P_TARGET_LOG2 7

#define NO_ACTIVATION 0

// Optimized MHA kernel with Grouped-Query Scale Factor Memory Reuse
void solution(void) {
  // Allocate local buffers to store intermediate states for the group
  uint64_t S1_hw_local[GROUP_SIZE][ATT_S][ATT_S / 4];
  uint8_t P_q_local[GROUP_SIZE][ATT_S][ATT_S];
  uint8_t P_scales_local[GROUP_SIZE][GROUPS_S][ATT_S];

  for (int kv = 0; kv < ATT_HKV; kv++) {
    const elem_t *KT_in = QTs[kv]; 
    const uint8_t *KT_sc = QTss[kv];
    const elem_t *V_in = Vs[kv];  
    const uint8_t *V_sc = Vss[kv];

    // =========================================================================
    // ---- 1) Compute 1 (S1 = Q @ K^T) for all heads in the group ----
    // =========================================================================
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1); // Load shared Key scales once

    for (int gh = 0; gh < GROUP_SIZE; gh++) {
      int h = kv * GROUP_SIZE + gh;
      const elem_t *Q_in = Qs[h];  
      const uint8_t *Q_sc = Qss[h];

      int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

      gemmini_flush(0);
      
      // Load individual Query scales
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);

      // MVIN A (Q_in)
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      }

      // MVIN B (KT_in)
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      }

      // Configure requantizer and execute tiled Gemmini WS loop
      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      
      // Read output into head-specific local buffer
      gemmini_mx_read_smem(&S1_hw_local[gh][0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // =========================================================================
    // ---- 2) FLOAT-FREE integer softmax + fp8 requant for all heads ----
    // =========================================================================
    for (int gh = 0; gh < GROUP_SIZE; gh++) {
      int h = kv * GROUP_SIZE + gh;

      for (int i = 0; i < ATT_S; i++) {
        int64_t q[ATT_S];
        int64_t mq = -((int64_t)1 << 62);
        
        // SWAR BF16 Unpacking
        for (int j = 0; j < ATT_S; j += 4) {
          uint64_t w = S1_hw_local[gh][i][j / 4];
          uint16_t w0 = (uint16_t)(w & 0xFFFF);
          uint16_t w1 = (uint16_t)((w >> 16) & 0xFFFF);
          uint16_t w2 = (uint16_t)((w >> 32) & 0xFFFF);
          uint16_t w3 = (uint16_t)((w >> 48) & 0xFFFF);

#if ATT_CAUSAL
          if (j > i) { q[j] = -((int64_t)1 << 62); } 
          else { int64_t qq = mx_ibf16_to_q(w0); q[j] = qq; if (qq > mq) mq = qq; }

          if (j + 1 > i) { q[j + 1] = -((int64_t)1 << 62); } 
          else { int64_t qq = mx_ibf16_to_q(w1); q[j + 1] = qq; if (qq > mq) mq = qq; }

          if (j + 2 > i) { q[j + 2] = -((int64_t)1 << 62); } 
          else { int64_t qq = mx_ibf16_to_q(w2); q[j + 2] = qq; if (qq > mq) mq = qq; }

          if (j + 3 > i) { q[j + 3] = -((int64_t)1 << 62); } 
          else { int64_t qq = mx_ibf16_to_q(w3); q[j + 3] = qq; if (qq > mq) mq = qq; }
#else
          int64_t qq0 = mx_ibf16_to_q(w0); q[j] = qq0; if (qq0 > mq) mq = qq0;
          int64_t qq1 = mx_ibf16_to_q(w1); q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
          int64_t qq2 = mx_ibf16_to_q(w2); q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
          int64_t qq3 = mx_ibf16_to_q(w3); q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
#endif
        }

        int64_t p[ATT_S];
        int64_t sm = 0;
        int64_t gmax_arr[GROUPS_S];
        for (int g = 0; g < GROUPS_S; g++) {
          int64_t gmax = 0;
          for (int j = g * 32; j < (g + 1) * 32; j++) { 
            int64_t pv = mx_iexp(q[j] - mq); 
            p[j] = pv; 
            sm += pv; 
            if (pv > gmax) gmax = pv; 
          }
          gmax_arr[g] = gmax;
        }
        
        int ls = mx_ilog2((uint64_t)sm);
        uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

        for (int g = 0; g < GROUPS_S; g++) {
          int64_t gmax = gmax_arr[g];
          if (gmax == 0) { 
            P_scales_local[gh][g][i] = 0; 
            for (int j = g * 32; j < (g + 1) * 32; j += 8) {
              *((uint64_t*)&P_q_local[gh][i][j]) = 0;
            }
            continue; 
          }
          
          int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
          P_scales_local[gh][g][i] = (uint8_t)(e + 127);
          
          // SWAR 8-way Parallel Packing of FP8 Elements 
          for (int j = g * 32; j < (g + 1) * 32; j += 8) {
            uint64_t packed_val = 0;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e));
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e)) << 8;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e)) << 16;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e)) << 24;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e)) << 32;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e)) << 40;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e)) << 48;
            packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e)) << 56;
            *((uint64_t*)&P_q_local[gh][i][j]) = packed_val;
          }
        }
      }
    }

    // =========================================================================
    // ---- 3) Compute 3 (O = P @ V) for all heads in the group ----
    // =========================================================================
    gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
    gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1); // Load shared Value scales once

    for (int gh = 0; gh < GROUP_SIZE; gh++) {
      int h = kv * GROUP_SIZE + gh;

      int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

      gemmini_flush(0);
      
      // Load individual Softmax scales
      gemmini_mx_load_scales((uint64_t)&P_scales_local[gh], sizeof(P_scales_local[gh]), 0);

      // MVIN A (P_q_local)
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)P_q_local[gh]) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      }

      // MVIN B (V_in)
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      }

      // Configure requantizer and execute tiled Gemmini WS loop
      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      
      // Read output directly to global output buffer
      gemmini_mx_read_smem(&S1_hw_local[gh][0][0], SPAD_DEST * 16, ATT_S * ATT_D); // Using S1_hw_local as placeholder for O_hw
      gemmini_fence();
    }
  }
}
