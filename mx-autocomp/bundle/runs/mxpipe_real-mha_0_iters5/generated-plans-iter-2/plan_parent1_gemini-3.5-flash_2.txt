An elegant and highly effective way to optimize this kernel is to **reduce data movement and DMA instruction overhead** (Strategy 1). 

### Inefficiency Analysis
In the baseline implementation, the $Q$, $K^T$, $P$, and $V$ matrices (all sized $64 \times 64$) are transferred between DRAM and the scratchpad using inner loops that load only 2 tiles (`bb = 2`, 32 elements) at a time. 
However, the Gemmini hardware supports a `MAX_BLOCK_LEN` of 4, meaning we can transfer up to 4 consecutive `DIM`-row tiles (64 elements) in a single DMA command. Because our matrix dimensions match this limit exactly ($tiles\_K = 4$, $tiles\_J = 4$), we can completely eliminate the inner tile-looping over the columns of $Q$, $K^T$, $P$, and $V$. 

By coalescing these transfers, we:
1. Reduce the number of `gemmini_extended_mvin` instructions by **50%** (from 8 down to 4 calls per matrix load).
2. Drastically lower the CPU branch/loop overhead and the DMA controller's command queue pressure.
3. Maximize DRAM burst transfer sizes to improve bus utilization.

### Optimization Plan
1. **Coalesce $Q$ loading**: Replace the 2D loop over `i` and `k` with a 1D loop over `i`, loading all 4 tiles ($tiles\_K = 4$) along the $K$ dimension in a single `mvin` call with a width of `tiles_K * DIM = 64`.
2. **Coalesce $K^T$ loading**: Replace the 2D loop over `k` and `j` with a 1D loop over `k`, loading all 4 tiles ($tiles\_J = 4$) along the $J$ dimension at once.
3. **Coalesce $P_q$ and $V$ loading**: Apply the same 1D loop flattening to the second matmul ($O = P \times V$).

// Optimized MHA kernel (S=64, D=64, H=4, H_kv=4, causal=0, fp8:e4m3).
void solution(void) {
  for (int h = 0; h < ATT_H; h++) {
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];  const uint8_t *V_sc = Vss[kv];

    // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);
      
      // Coalesced Q loading: Load 4 tiles (64 elements) at once per row
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D),
                              a_base + i * tiles_K * DIM, tiles_K * DIM, DIM);
      }
      
      // Coalesced KT loading: Load 4 tiles (64 elements) at once per row
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)KT_in) + k * DIM * ATT_S),
                              b_base + k * tiles_J * DIM, tiles_J * DIM, DIM);
      }

      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      for (int j = 0; j < ATT_S; j += 4) {                       // unpack 4 bf16/word -> Q(1/1024) int + max
        uint64_t w = S1_hw[i][j / 4];
        for (int l = 0; l < 4; l++) {
#if ATT_CAUSAL
          if (j + l > i) { q[j + l] = -((int64_t)1 << 62); continue; }   // causal: future col -> iexp()=0
#endif
          int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
          q[j + l] = qq; if (qq > mq) mq = qq;
        }
      }
      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];                                // fused group-max: capture per-group max IN the iexp pass
      for (int g = 0; g < GROUPS_S; g++) {                       // (group-nested, register-scalar gmax) -> removes the
        int64_t gmax = 0;                                        // separate 16384-elem max scan. Output bit-identical.
        for (int j = g * 32; j < (g + 1) * 32; j++) { int64_t pv = mx_iexp(q[j] - mq); p[j] = pv; sm += pv; if (pv > gmax) gmax = pv; }
        gmax_arr[g] = gmax;
      }
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;   // one rounded reciprocal/row
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) { P_scales[g][i] = 0; for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0; continue; }
        // group e8m0 scale so the group-max lands at ~2^MX_P_TARGET_LOG2 (moderate; avoids e4 overflow)
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);

        // ---- SWAR OPTIMIZATION: Pack and write 8 FP8 elements at a time ----
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          uint64_t packed = 0;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e);
          packed |= (uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e) << 8;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e) << 16;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e) << 24;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e) << 32;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e) << 40;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e) << 48;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e) << 56;
          
          *((uint64_t *)&P_q[i][j]) = packed;
        }
      }
    }

    // ---- 3) O = P @ V on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
      gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);
      
      // Coalesced P_q loading: Load 4 tiles (64 elements) at once per row
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S),
                              a_base + i * tiles_K * DIM, tiles_K * DIM, DIM);
      }
      
      // Coalesced V loading: Load 4 tiles (64 elements) at once per row
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin((void*)(((elem_t*)V_in) + k * DIM * ATT_D),
                              b_base + k * tiles_J * DIM, tiles_J * DIM, DIM);
      }

      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
  }
}
