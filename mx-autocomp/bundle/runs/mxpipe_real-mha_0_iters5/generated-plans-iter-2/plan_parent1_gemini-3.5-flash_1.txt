An elegant plan to optimize this MHA kernel is to apply **Strategy 2: Hoist redundant operations out of loops**.

### Optimization Plan

1. **Analyze Redundant Configurations:**
   - `gemmini_flush(0)` is called twice inside every head iteration `h`. Since we do not dynamically change configurations during runtime, we can call it only once at the beginning of the program.
   - `gemmini_extended3_config_ex` is called twice inside every head iteration `h` with identical parameters. Since both GEMM operations (Step 1 and Step 3) utilize identical input formats (FP8) and output formats (BF16) in a Weight-Stationary configuration, this call can be hoisted completely outside the `h` loop.
   - `gemmini_config_ld` is called multiple times setting either `ATT_D * sizeof(elem_t)` (64) or `ATT_S * sizeof(elem_t)` (64). Since both `ATT_D` and `ATT_S` are exactly `64` and `elem_t` is `1` byte, the stride is always `64` bytes. We can configure this once at the beginning of `solution()`.
   - `gemmini_config_st` is called with either `S_OUT_COLS * sizeof(out_t)` (128) or `D_OUT_COLS * sizeof(out_t)` (128). Since both store strides are identical, we can configure this once at the beginning of `solution()`.

2. **Refactor Code:**
   - Move the initialization commands (`gemmini_flush`, `gemmini_extended3_config_ex`, `gemmini_config_ld`, and `gemmini_config_st`) to the beginning of the `solution()` function before the outer loop `h` begins.
   - Preserve the strict sequence order within the loop: scale-factor loading -> `mvin` -> requantization configuration (`gemmini_mxquant_config_mvout`) -> loop WS execution -> read shared memory -> fence.

// AUTO-GENERATED optimized MHA kernel (S=64, D=64, H=4, H_kv=4, causal=0, fp8:e4m3).
void solution(void) {
  // Hoist redundant global configurations out of the loops
  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Since ATT_S = 64 and ATT_D = 64, load stride is always 64 bytes
  gemmini_config_ld(64);

  // Since S_OUT_COLS = 16 and D_OUT_COLS = 16, store stride is always 16 * 8 = 128 bytes
  gemmini_config_st(128);

  int tiles_I = ATT_S / DIM;
  int tiles_J = ATT_S / DIM;
  int tiles_K = ATT_D / DIM;

  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  for (int h = 0; h < ATT_H; h++) {
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];  const uint8_t *V_sc = Vss[kv];

    // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);

      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      }

      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      }

      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 2) FLOAT-FREE integer softmax + fp8 requant (I-BERT iexp; host has no FPU) ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      for (int j = 0; j < ATT_S; j += 4) {                       // unpack 4 bf16/word -> Q(1/1024) int + max
        uint64_t w = S1_hw[i][j / 4];
        for (int l = 0; l < 4; l++) {
#if ATT_CAUSAL
          if (j + l > i) { q[j + l] = -((int64_t)1 << 62); continue; }   // causal: future col -> iexp()=0
#endif
          int64_t qq = mx_ibf16_to_q((uint16_t)((w >> (l * 16)) & 0xFFFF));
          q[j + l] = qq; if (qq > mq) mq = qq;
        }
      }
      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];                                // fused group-max: capture per-group max IN the iexp pass
      for (int g = 0; g < GROUPS_S; g++) {                       // (group-nested, register-scalar gmax) -> removes the
        int64_t gmax = 0;                                        // separate 16384-elem max scan. Output bit-identical.
        for (int j = g * 32; j < (g + 1) * 32; j++) { int64_t pv = mx_iexp(q[j] - mq); p[j] = pv; sm += pv; if (pv > gmax) gmax = pv; }
        gmax_arr[g] = gmax;
      }
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;   // one rounded reciprocal/row
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) { P_scales[g][i] = 0; for (int j = g * 32; j < (g + 1) * 32; j++) P_q[i][j] = 0; continue; }
        // group e8m0 scale so the group-max lands at ~2^MX_P_TARGET_LOG2 (moderate; avoids e4 overflow)
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);

        // ---- SWAR OPTIMIZATION: Pack and write 8 FP8 elements at a time ----
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          uint64_t packed = 0;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e);
          packed |= (uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e) << 8;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e) << 16;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e) << 24;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e) << 32;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e) << 40;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e) << 48;
          packed |= (uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e) << 56;
          
          *((uint64_t *)&P_q[i][j]) = packed;
        }
      }
    }

    // ---- 3) O = P @ V on Gemmini (bf16 out) ----
    {
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
      gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);

      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      }

      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      }

      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
  }
}
