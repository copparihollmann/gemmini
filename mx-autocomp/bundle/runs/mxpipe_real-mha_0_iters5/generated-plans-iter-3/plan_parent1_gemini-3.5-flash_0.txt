### Optimization Plan

Our analysis of the original Multi-Head Attention (MHA) kernel shows that the software-based integer softmax step constitutes a significant bottleneck. Specifically, the conversion of BF16 values extracted from the Gemmini shared memory (`S1_hw`) into fixed-point integers (`q[j]`) via `mx_ibf16_to_q` is performed sequentially for every individual element. 

To address this, we will apply **Strategy 6: Vectorized SIMD Conversion of BF16 to Integer**.

#### 1. SWAR-Style Parallel Bitfield Extraction
A 64-bit word `w` contains four packed 16-bit BF16 lanes. Rather than extracting these lanes sequentially and performing bit-manipulation individually, we can extract the exponents, mantissas, and sign bits of all four lanes simultaneously in parallel using 64-bit bitwise masking:
* **Exponents:** Masking the exponent bits of all lanes at once (`0x7F807F807F807F80ULL`) and shifting right by 7. Because the lower bits of each lane's exponent field are zeroed before shifting, we completely eliminate lane-to-lane bit crosstalk.
* **Mantissas:** Masking all mantissa bits simultaneously with `0x007F007F007F007FULL`.
* **Signs:** Extracting sign bits in parallel with `(w & 0x8000800080008000ULL) >> 15`.

#### 2. Inlined Fast Lane Processing
By doing parallel bitfield extraction on the 64-bit register, the sub-fields for all four lanes are computed in parallel. We then unpack these pre-extracted fields and scale them to fixed-point using an inline shift logic. This avoids the overhead of four individual sequential function calls to `mx_ibf16_to_q` and streamlines execution on the host CPU.

// Optimized MHA kernel using SWAR-like SIMD bitfield extraction for BF16 to integer conversion.
void solution(void) {
  for (int h = 0; h < ATT_H; h++) {
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];  const uint8_t *V_sc = Vss[kv];

    // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) {
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);
        }
      }
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) {
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM);
        }
      }
      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 2) FLOAT-FREE integer softmax + fp8 requant with SWAR BF16 unpacking ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      
      // Parallelized BF16 extraction using bitwise SIMD shifts
      for (int j = 0; j < ATT_S; j += 4) {
        uint64_t w = S1_hw[i][j / 4];

        // SWAR extraction of exponent, mantissa, and sign fields
        uint64_t exp_bits = w & 0x7F807F807F807F80ULL;
        uint64_t exps = exp_bits >> 7;
        uint64_t mants = w & 0x007F007F007F007FULL;
        uint64_t signs = (w & 0x8000800080008000ULL) >> 15;

        // Lane 0 processing
#if ATT_CAUSAL
        if (j > i) {
          q[j] = -((int64_t)1 << 62);
        } else {
          uint32_t e = (uint32_t)(exps & 0xFF);
          uint32_t m = (uint32_t)(mants & 0x7F);
          uint32_t s = (uint32_t)(signs & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118; // exp - 127 - 7 + 16
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j] = qq;
          if (qq > mq) mq = qq;
        }
#else
        {
          uint32_t e = (uint32_t)(exps & 0xFF);
          uint32_t m = (uint32_t)(mants & 0x7F);
          uint32_t s = (uint32_t)(signs & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j] = qq;
          if (qq > mq) mq = qq;
        }
#endif

#if ATT_CAUSAL

          uint32_t m = (uint32_t)((mants >> 16) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 16) & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j + 1] = qq;
          if (qq > mq) mq = qq;
        }
#else
        {
          uint32_t e = (uint32_t)((exps >> 16) & 0xFF);
          uint32_t m = (uint32_t)((mants >> 16) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 16) & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j + 1] = qq;
          if (qq > mq) mq = qq;

#endif

        // Lane 2 processing
#if ATT_CAUSAL
        if (j + 2 > i) {
          q[j + 2] = -((int64_t)1 << 62);
        } else {
          uint32_t e = (uint32_t)((exps >> 32) & 0xFF);
          uint32_t m = (uint32_t)((mants >> 32) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 32) & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j + 2] = qq;
          if (qq > mq) mq = qq;
        }
#else
        {
          uint32_t e = (uint32_t)((exps >> 32) & 0xFF);
          uint32_t m = (uint32_t)((mants >> 32) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 32) & 1);

            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j + 2] = qq;
          if (qq > mq) mq = qq;
        }
#endif

        // Lane 3 processing
#if ATT_CAUSAL
        if (j + 3 > i) {
          q[j + 3] = -((int64_t)1 << 62);
        } else {
          uint32_t e = (uint32_t)((exps >> 48) & 0xFF);
          uint32_t m = (uint32_t)((mants >> 48) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 48) & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }

          if (qq > mq) mq = qq;
        }
#else
        {
          uint32_t e = (uint32_t)((exps >> 48) & 0xFF);
          uint32_t m = (uint32_t)((mants >> 48) & 0x7F);
          uint32_t s = (uint32_t)((signs >> 48) & 1);
          int64_t qq = 0;
          if (e != 0 || m != 0) {
            int64_t val = (m | 0x80);
            int shift = (int)e - 118;
            if (shift >= 0) val <<= shift;
            else val >>= (-shift);
            qq = s ? -val : val;
          }
          q[j + 3] = qq;
          if (qq > mq) mq = qq;
        }
#endif
      }

      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) {
          int64_t pv = mx_iexp(q[j] - mq);
          p[j] = pv;
          sm += pv;

      }

      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) { 
          P_scales[g][i] = 0; 
          for (int j = g * 32; j < (g + 1) * 32; j += 8) {
            *((uint64_t*)&P_q[i][j]) = 0;
          }
          continue; 
        }
        
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);
        
        // SWAR 8-way Parallel Packing of FP8 Elements 
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          uint64_t packed_val = 0;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e));
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e)) << 8;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e)) << 16;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e)) << 24;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e)) << 32;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e)) << 40;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e)) << 48;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e)) << 56;
          *((uint64_t*)&P_q[i][j]) = packed_val;
        }
      }
    }

    // ---- 3) O = P @ V on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);

      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++) {
        for (int k = 0; k < tiles_K; k += 2) {
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);

      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++) {
        for (int j = 0; j < tiles_J; j += 2) {
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM);
        }
      }
      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
