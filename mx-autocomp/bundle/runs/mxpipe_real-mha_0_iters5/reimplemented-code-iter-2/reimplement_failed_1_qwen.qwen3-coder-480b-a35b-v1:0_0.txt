#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"

// Double buffers for the pipelined execution
static out_t S1_hw_db[2][ATT_S][S_OUT_COLS];
static elem_t P_q_db[2][ATT_S][ATT_S];
static uint8_t P_scales_db[2][GROUPS_S][ATT_S];

// Helper function to launch Q x K^T on Gemmini asynchronously
void compute_QK(int h, int buf, const uint8_t *Q_sc, const uint8_t *KT_sc, const elem_t *Q_in, const elem_t *KT_in) {
  int tiles_I = ATT_S / DIM;
  int tiles_J = ATT_S / DIM;
  int tiles_K = ATT_D / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
  gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);

  gemmini_config_ld(ATT_D * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += 2) {
      int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
      gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);
    }
  }

  gemmini_config_ld(ATT_S * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += 2) {
      int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
      gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                            b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM);
    }
  }

  gemmini_config_st(S_OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&S1_hw_db[buf][0][0], SPAD_DEST * 16, ATT_S * ATT_S);
}

// Helper function to execute integer softmax and quantization on the CPU
void compute_softmax(int h, int buf_in, int buf_out) {
  for (int i = 0; i < ATT_S; i++) {
    int64_t q[ATT_S];
    int64_t mq = -((int64_t)1 << 62);
    
    // SWAR BF16 Unpacking
    for (int j = 0; j < ATT_S; j += 4) {
      uint64_t w = S1_hw_db[buf_in][i][j / 4];
      uint16_t w0 = (uint16_t)(w & 0xFFFF);
      uint16_t w1 = (uint16_t)((w >> 16) & 0xFFFF);
      uint16_t w2 = (uint16_t)((w >> 32) & 0xFFFF);
      uint16_t w3 = (uint16_t)((w >> 48) & 0xFFFF);

#if ATT_CAUSAL
      if (j > i) { q[j] = -((int64_t)1 << 62); } 
      else { int64_t qq = mx_ibf16_to_q(w0); q[j] = qq; if (qq > mq) mq = qq; }

      if (j + 1 > i) { q[j + 1] = -((int64_t)1 << 62); } 
      else { int64_t qq = mx_ibf16_to_q(w1); q[j + 1] = qq; if (qq > mq) mq = qq; }

      if (j + 2 > i) { q[j + 2] = -((int64_t)1 << 62); } 
      else { int64_t qq = mx_ibf16_to_q(w2); q[j + 2] = qq; if (qq > mq) mq = qq; }

      if (j + 3 > i) { q[j + 3] = -((int64_t)1 << 62); } 
      else { int64_t qq = mx_ibf16_to_q(w3); q[j + 3] = qq; if (qq > mq) mq = qq; }
#else
      int64_t qq0 = mx_ibf16_to_q(w0); q[j] = qq0; if (qq0 > mq) mq = qq0;
      int64_t qq1 = mx_ibf16_to_q(w1); q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
      int64_t qq2 = mx_ibf16_to_q(w2); q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
      int64_t qq3 = mx_ibf16_to_q(w3); q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
#endif
    }

    int64_t p[ATT_S];
    int64_t sm = 0;
    int64_t gmax_arr[GROUPS_S];
    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = 0;
      for (int j = g * 32; j < (g + 1) * 32; j++) { int64_t pv = mx_iexp(q[j] - mq); p[j] = pv; sm += pv; if (pv > gmax) gmax = pv; }
      gmax_arr[g] = gmax;
    }
    
    int ls = mx_ilog2((uint64_t)sm);
    uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

    for (int g = 0; g < GROUPS_S; g++) {
      int64_t gmax = gmax_arr[g];
      if (gmax == 0) { 
        P_scales_db[buf_out][g][i] = 0; 
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          *((uint64_t*)&P_q_db[buf_out][i][j]) = 0;
        }
        continue; 
      }
      
      int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
      P_scales_db[buf_out][g][i] = (uint8_t)(e + 127);
      
      // SWAR 8-way Parallel Packing of FP8 Elements 
      for (int j = g * 32; j < (g + 1) * 32; j += 8) {
        uint64_t packed_val = 0;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e));
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e)) << 8;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e)) << 16;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e)) << 24;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e)) << 32;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e)) << 40;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e)) << 48;
        packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e)) << 56;
        *((uint64_t*)&P_q_db[buf_out][i][j]) = packed_val;
      }
    }
  }
}

// Helper function to launch P x V on Gemmini asynchronously
void compute_PV(int h, int buf, int out_h, const uint8_t *V_sc, const elem_t *V_in) {
  int tiles_I = ATT_S / DIM;
  int tiles_J = ATT_D / DIM;
  int tiles_K = ATT_S / DIM;
  uint32_t a_base = 0;
  uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;

  gemmini_flush(0);
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)P_scales_db[buf], sizeof(P_scales_db[buf]), 0);
  gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);

  gemmini_config_ld(ATT_S * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += 2) {
      int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
      gemmini_extended_mvin((void*)(((elem_t*)P_q_db[buf]) + i * DIM * ATT_S + k * DIM),
                            a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM);
    }
  }

  gemmini_config_ld(ATT_D * sizeof(elem_t));
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += 2) {
      int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
      gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                            b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM);
    }
  }

  gemmini_config_st(D_OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&O_hw[out_h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
}

// Optimized multi-head attention solution using CPU-Gemmini pipelining
void solution(void) {
  // 1. Startup: Launch Q x K^T for Head 0
  {
    int kv = 0 / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[0];  
    const uint8_t *Q_sc = Qss[0];
    const elem_t *KT_in = KTs[kv]; 
    const uint8_t *KT_sc = KTss[kv];
    compute_QK(0, 0, Q_sc, KT_sc, Q_in, KT_in);
    gemmini_fence(); // Ensure S1_hw_db[0] is fully written before starting loop
  }

  // 2. Pipelined Loop over all attention heads
  for (int h = 0; h < ATT_H; h++) {
    int buf = h % 2;

    // A. Overlap Stage: Launch P x V of the previous head (h-1) in the background
    if (h > 0) {
      int prev_kv = (h - 1) / (ATT_H / ATT_HKV);
      const elem_t *V_in = Vs[prev_kv];  
      const uint8_t *V_sc = Vss[prev_kv];
      compute_PV(h - 1, (h - 1) % 2, h - 1, V_sc, V_in);
    }

    // B. Main Compute Stage: CPU performs softmax on head h in parallel
    compute_softmax(h, buf, buf);

    // C. Re-feed Stage: Launch Q x K^T for the next head (h+1)
    if (h + 1 < ATT_H) {
      int next_kv = (h + 1) / (ATT_H / ATT_HKV);
      const elem_t *Q_in = Qs[h + 1];  
      const uint8_t *Q_sc = Qss[h + 1];
      const elem_t *KT_in = KTs[next_kv]; 
      const uint8_t *KT_sc = KTss[next_kv];
      compute_QK(h + 1, (h + 1) % 2, Q_sc, KT_sc, Q_in, KT_in);
    }

    // D. Sync Point: Block the CPU until the background Gemmini operations complete
    gemmini_fence();
  }

  // 3. Wind-down: Launch and complete P x V for the final head (H-1)
  {
    int final_h = ATT_H - 1;
    int kv = final_h / (ATT_H / ATT_HKV);
    const elem_t *V_in = Vs[kv];  
    const uint8_t *V_sc = Vss[kv];
    compute_PV(final_h, final_h % 2, final_h, V_sc, V_in);
    gemmini_fence(); // Ensure the final result is written to DRAM
  }
}
