An elegant and highly effective strategy to optimize this multi-head attention (MHA) kernel is **Asynchronous Gemmini-CPU Pipelining with Maximum-Block DMA Transfers**.

### Plan Explanation

1. **Overlapping Data Movement and Compute (Pipelining):**
   Currently, the CPU and the Gemmini accelerator execute sequentially: Gemmini computes $S1$, the host CPU waits, the host CPU computes Softmax, Gemmini waits, Gemmini computes $O$, and so on. 
   We can overlap these tasks. While the host CPU is busy performing the intense integer softmax computation for head $h$, the Gemmini coprocessor can concurrently calculate the $S1$ projection for head $h+1$ in the background. We achieve this by double-buffering the intermediate $S1$ DRAM buffer (`S1_hw` and a stack-allocated `S1_hw_alt`), preventing data hazard races between Gemmini writing the next head and the CPU reading the current head.

2. **Reducing Data Movement Overhead (Coarser DMA Blocks):**
   In the original code, $Q$, $K^T$, $P$, and $V$ are loaded tile-by-tile via nested loops of width `bb * DIM` (up to 32 bytes). Since `tiles_K = 4` and `tiles_J = 4`, the total width along the $K$ or $J$ dimensions is exactly 64 bytes. 
   We can fully utilize the maximum block length (`MAX_BLOCK_LEN = 4` tiles, i.e., 64 bytes) to load the entire row of tiles in a single `gemmini_extended_mvin` command. This dramatically reduces the number of custom RoCC instructions dispatched to the coprocessor and eliminates the inner loop overhead on the host.

// AUTO-GENERATED optimized MHA kernel (S=64, D=64, H=4, H_kv=4, causal=0, fp8:e4m3).
void solution(void) {
  // Hoist invariant MX configuration to the beginning of the function
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

  // Stack-allocated alternate buffer to double-buffer S1 outputs in DRAM
  alignas(16) uint64_t S1_hw_alt[ATT_S][ATT_S / 4];

  // ---- Startup: Pre-compute S1 for Head 0 ----
  {
    int h = 0;
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];

    gemmini_flush(0);
    gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
    gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);

    // Load Q with maximum DMA width (64 bytes)
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int i = 0; i < 4; i++) {
      gemmini_extended_mvin((void*)(Q_in + i * 16 * ATT_D),
                            0 + i * 4 * 16, 64, 16);
    }

    // Load KT with maximum DMA width (64 bytes)
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int k = 0; k < 4; k++) {
      gemmini_extended_mvin((void*)(KT_in + k * 16 * ATT_S),
                            7936 + k * 4 * 16, 64, 16);
    }

    gemmini_config_st(S_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, 4, 4, 4, 0, 0, 1);
    gemmini_loop_ws_spad(4, 4, 4, 0, 0, 0, 0, 8192, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    gemmini_fence();
  }

  // ---- Main Pipelined Head Loop ----
  for (int h = 0; h < ATT_H; h++) {
    // Alternate between S1_hw and S1_hw_alt
    uint64_t (*current_S1_hw)[ATT_S / 4] = (h % 2 == 0) ? S1_hw : S1_hw_alt;

    // ---- 1) Concurrently run Softmax on the host CPU ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      
      // SWAR BF16 Unpacking
      for (int j = 0; j < ATT_S; j += 4) {
        uint64_t w = current_S1_hw[i][j / 4];
        uint16_t w0 = (uint16_t)(w & 0xFFFF);
        uint16_t w1 = (uint16_t)((w >> 16) & 0xFFFF);
        uint16_t w2 = (uint16_t)((w >> 32) & 0xFFFF);
        uint16_t w3 = (uint16_t)((w >> 48) & 0xFFFF);

        int64_t qq0 = mx_ibf16_to_q(w0); q[j] = qq0; if (qq0 > mq) mq = qq0;
        int64_t qq1 = mx_ibf16_to_q(w1); q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
        int64_t qq2 = mx_ibf16_to_q(w2); q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
        int64_t qq3 = mx_ibf16_to_q(w3); q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
      }

      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) { 
          int64_t pv = mx_iexp(q[j] - mq); 
          p[j] = pv; 
          sm += pv; 
          if (pv > gmax) gmax = pv; 
        }
        gmax_arr[g] = gmax;
      }
      
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];
        if (gmax == 0) { 
          P_scales[g][i] = 0; 
          for (int j = g * 32; j < (g + 1) * 32; j += 8) {
            *((uint64_t*)&P_q[i][j]) = 0;
          }
          continue; 
        }
        
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);
        
        // SWAR 8-way Parallel Packing of FP8 Elements 
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          uint64_t packed_val = 0;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e));
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e)) << 8;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e)) << 16;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e)) << 24;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e)) << 32;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e)) << 40;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e)) << 48;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e)) << 56;
          *((uint64_t*)&P_q[i][j]) = packed_val;
        }
      }
    }

    // ---- 2) Issue O = P @ V to Gemmini queue ----
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *V_in = Vs[kv]; const uint8_t *V_sc = Vss[kv];

    gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
    gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);

    // Load P
    gemmini_config_ld(ATT_S * sizeof(elem_t));
    for (int i = 0; i < 4; i++) {
      gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * 16 * ATT_S),
                            0 + i * 4 * 16, 64, 16);
    }

    // Load V
    gemmini_config_ld(ATT_D * sizeof(elem_t));
    for (int k = 0; k < 4; k++) {
      gemmini_extended_mvin((void*)(V_in + k * 16 * ATT_D),
                            7936 + k * 4 * 16, 64, 16);
    }

    gemmini_config_st(D_OUT_COLS * sizeof(out_t));
    gemmini_mxquant_config_mvout((uint64_t)scale_factors, 4, 4, 4, 0, 0, 1);
    gemmini_loop_ws_spad(4, 4, 4, 0, 0, 0, 0, 8192, 0,
                         SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
    gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);

    // ---- 3) Concurrently queue S1 for the NEXT head ----
    if (h + 1 < ATT_H) {
      int next_h = h + 1;
      int next_kv = next_h / (ATT_H / ATT_HKV);
      const elem_t *next_Q_in = Qs[next_h];  const uint8_t *next_Q_sc = Qss[next_h];
      const elem_t *next_KT_in = KTs[next_kv]; const uint8_t *next_KT_sc = KTss[next_kv];
      uint64_t (*next_S1_hw)[ATT_S / 4] = (next_h % 2 == 0) ? S1_hw : S1_hw_alt;

      gemmini_mx_load_scales((uint64_t)next_Q_sc, QSS_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)next_KT_sc, QSS_BYTES, 1);

      // Load next Q
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < 4; i++) {
        gemmini_extended_mvin((void*)(next_Q_in + i * 16 * ATT_D),
                              0 + i * 4 * 16, 64, 16);
      }

      // Load next KT
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < 4; k++) {
        gemmini_extended_mvin((void*)(next_KT_in + k * 16 * ATT_S),
                              7936 + k * 4 * 16, 64, 16);
      }

      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, 4, 4, 4, 0, 0, 1);
      gemmini_loop_ws_spad(4, 4, 4, 0, 0, 0, 0, 8192, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&next_S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
    }

    // Force synchronization of queued Gemmini commands right before starting next CPU iteration
    gemmini_fence();
  }
}
