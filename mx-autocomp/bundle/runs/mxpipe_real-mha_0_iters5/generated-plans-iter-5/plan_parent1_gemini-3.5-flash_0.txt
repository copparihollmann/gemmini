An elegant and highly effective way to optimize the performance of the host CPU during the softmax and requantization phase is to implement **Strategy 8: Host-side Direct Lookup Table (LUT) for `mx_ibf16_to_q`**.

### How This Strategy Improves Performance

In the baseline kernel, the host CPU performs float-free integer softmax and FP8 requantization. A major bottleneck in this phase is the `mx_ibf16_to_q` function, which is called for every single element in the $S_1$ matrix ($ATT\_S \times ATT\_S$ times per attention head). This function performs complex bitwise manipulation, exponent extraction, shifting, and sign extension on the 16-bit BF16 representation.

Since a BF16 value is exactly 16 bits wide, there are only $2^{16} = 65,536$ possible input values. By precomputing a $65,536$-entry lookup table on the first invocation, we can replace all subsequent expensive bitwise calculations with a single, cache-friendly $O(1)$ array access:
1. **Elimination of Bitwise Overhead:** Completely avoids the shifting, masking, and branching inside `mx_ibf16_to_q`.
2. **Cache Efficiency:** A 128 KB table (using `int16_t` elements) easily fits in the L1/L2 data caches of modern processors, making LUT queries extremely fast.
3. **SWAR Friendly:** It integrates seamlessly into the existing SWAR unpacking loop without interrupting register-level parallelism.

// AUTO-GENERATED optimized MHA kernel with Host-side mx_ibf16_to_q LUT.
void solution(void) {
  // 64 KB (16-bit input, 65536 entries) lookup table to cache mx_ibf16_to_q results
  static int16_t ibf16_to_q_lut[65536];
  static int lut_initialized = 0;

  if (!lut_initialized) {
    for (int i = 0; i < 65536; i++) {
      ibf16_to_q_lut[i] = (int16_t)mx_ibf16_to_q((uint16_t)i);
    }
    lut_initialized = 1;
  }

  for (int h = 0; h < ATT_H; h++) {
    int kv = h / (ATT_H / ATT_HKV);
    const elem_t *Q_in = Qs[h];  const uint8_t *Q_sc = Qss[h];
    const elem_t *KT_in = KTs[kv]; const uint8_t *KT_sc = KTss[kv];
    const elem_t *V_in = Vs[kv];  const uint8_t *V_sc = Vss[kv];
    
    // ---- 1) S1 = Q @ K^T on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_S / DIM, tiles_K = ATT_D / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)Q_sc, QSS_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)KT_sc, QSS_BYTES, 1);
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)Q_in) + i * DIM * ATT_D + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++)
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)KT_in) + (k * DIM) * ATT_S + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      gemmini_config_st(S_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&S1_hw[0][0], SPAD_DEST * 16, ATT_S * ATT_S);
      gemmini_fence();
    }

    // ---- 2) FLOAT-FREE integer softmax + fp8 requant (using fast host-side LUT) ----
    for (int i = 0; i < ATT_S; i++) {
      int64_t q[ATT_S];
      int64_t mq = -((int64_t)1 << 62);
      
      // SWAR BF16 Unpacking utilizing direct LUT lookup
      for (int j = 0; j < ATT_S; j += 4) {
        uint64_t w = S1_hw[i][j / 4];
        uint16_t w0 = (uint16_t)(w & 0xFFFF);
        uint16_t w1 = (uint16_t)((w >> 16) & 0xFFFF);
        uint16_t w2 = (uint16_t)((w >> 32) & 0xFFFF);
        uint16_t w3 = (uint16_t)((w >> 48) & 0xFFFF);

#if ATT_CAUSAL
        if (j > i) { q[j] = -((int64_t)1 << 62); } 
        else { int64_t qq = ibf16_to_q_lut[w0]; q[j] = qq; if (qq > mq) mq = qq; }

        if (j + 1 > i) { q[j + 1] = -((int64_t)1 << 62); } 
        else { int64_t qq = ibf16_to_q_lut[w1]; q[j + 1] = qq; if (qq > mq) mq = qq; }

        if (j + 2 > i) { q[j + 2] = -((int64_t)1 << 62); } 
        else { int64_t qq = ibf16_to_q_lut[w2]; q[j + 2] = qq; if (qq > mq) mq = qq; }

        if (j + 3 > i) { q[j + 3] = -((int64_t)1 << 62); } 
        else { int64_t qq = ibf16_to_q_lut[w3]; q[j + 3] = qq; if (qq > mq) mq = qq; }
#else
        int64_t qq0 = ibf16_to_q_lut[w0]; q[j] = qq0; if (qq0 > mq) mq = qq0;
        int64_t qq1 = ibf16_to_q_lut[w1]; q[j + 1] = qq1; if (qq1 > mq) mq = qq1;
        int64_t qq2 = ibf16_to_q_lut[w2]; q[j + 2] = qq2; if (qq2 > mq) mq = qq2;
        int64_t qq3 = ibf16_to_q_lut[w3]; q[j + 3] = qq3; if (qq3 > mq) mq = qq3;
#endif
      }

      int64_t p[ATT_S];
      int64_t sm = 0;
      int64_t gmax_arr[GROUPS_S];
      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = 0;
        for (int j = g * 32; j < (g + 1) * 32; j++) { 
          int64_t pv = mx_iexp(q[j] - mq); 
          p[j] = pv; 
          sm += pv; 
          if (pv > gmax) gmax = pv; 
        }
        gmax_arr[g] = gmax;
      }
      
      int ls = mx_ilog2((uint64_t)sm);
      uint64_t inv_sm = (((uint64_t)1 << (ls + MX_FRAC)) + ((uint64_t)sm >> 1)) / (uint64_t)sm;

      for (int g = 0; g < GROUPS_S; g++) {
        int64_t gmax = gmax_arr[g];

          P_scales[g][i] = 0;

            *((uint64_t*)&P_q[i][j]) = 0;
          }
          continue; 
        }
        
        int e = mx_ilog2((uint64_t)gmax) - ls - MX_P_TARGET_LOG2;
        P_scales[g][i] = (uint8_t)(e + 127);
        
        // SWAR 8-way Parallel Packing of FP8 Elements 
        for (int j = g * 32; j < (g + 1) * 32; j += 8) {
          uint64_t packed_val = 0;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 0], inv_sm, ls, e));
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 1], inv_sm, ls, e)) << 8;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 2], inv_sm, ls, e)) << 16;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 3], inv_sm, ls, e)) << 24;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 4], inv_sm, ls, e)) << 32;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 5], inv_sm, ls, e)) << 40;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 6], inv_sm, ls, e)) << 48;
          packed_val |= ((uint64_t)mx_ienc_fp8(p[j + 7], inv_sm, ls, e)) << 56;
          *((uint64_t*)&P_q[i][j]) = packed_val;
        }
      }
    }

    // ---- 3) O = P @ V on Gemmini (bf16 out) ----
    {
      int tiles_I = ATT_S / DIM, tiles_J = ATT_D / DIM, tiles_K = ATT_S / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles_K * tiles_J * DIM;
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
      gemmini_mx_load_scales((uint64_t)&P_scales, sizeof(P_scales), 0);
      gemmini_mx_load_scales((uint64_t)V_sc, VSS_BYTES, 1);
      gemmini_config_ld(ATT_S * sizeof(elem_t));
      for (int i = 0; i < tiles_I; i++)
        for (int k = 0; k < tiles_K; k += 2) { 
          int bb = (tiles_K - k < 2) ? (tiles_K - k) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)P_q) + i * DIM * ATT_S + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, bb * DIM, DIM); 
        }
      gemmini_config_ld(ATT_D * sizeof(elem_t));
      for (int k = 0; k < tiles_K; k++)
        for (int j = 0; j < tiles_J; j += 2) { 
          int bb = (tiles_J - j < 2) ? (tiles_J - j) : 2;
          gemmini_extended_mvin((void*)(((elem_t*)V_in) + (k * DIM) * ATT_D + j * DIM),
                                b_base + (k * tiles_J + j) * DIM, bb * DIM, DIM); 
        }
      gemmini_config_st(D_OUT_COLS * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
      gemmini_mx_read_smem(&O_hw[h][0][0], SPAD_DEST * 16, ATT_S * ATT_D);
      gemmini_fence();
    }
  }
}
