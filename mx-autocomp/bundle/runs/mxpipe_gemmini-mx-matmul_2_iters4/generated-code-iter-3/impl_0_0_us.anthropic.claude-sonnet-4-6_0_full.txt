void solution(void) {
  // Compile-time constants for tile counts and scratchpad addresses
  #define TILES_I  (MATMUL_M / DIM)
  #define TILES_J  (MATMUL_N / DIM)
  #define TILES_K  (MATMUL_K / DIM)
  #define A_BASE_ADDR  0
  #define B_BASE_ADDR  (BANK_NUM * BANK_ROWS - TILES_K * TILES_J * DIM)
  #define SPAD_DEST_CONST  128

  // Step 1: Configure execution with MX format fields
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  // Step 2: Load scale factors for A (sel=0) and B (sel=1)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 4: Configure mvin ports and load tiles into scratchpad
  // Port 0 (mvin) for A tiles: DRAM row stride = MATMUL_K bytes (A is M x K, row-major)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);
  // Port 1 (mvin2) for B tiles: DRAM row stride = MATMUL_K bytes (B is K x N, row-major)
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  // Interleave A and B loads across K to overlap DMA transfers
  // All address arithmetic uses compile-time constants to eliminate runtime multiply/add
  for (int k = 0; k < TILES_K; k++) {
    for (int i = 0; i < TILES_I; i++) {
      gemmini_extended_mvin(
          (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
          A_BASE_ADDR + (i * TILES_K + k) * DIM,
          DIM, DIM);
    }
    for (int j = 0; j < TILES_J; j++) {
      gemmini_extended_mvin2(
          (void*)(((elem_t*)B_in) + j * DIM * MATMUL_K + k * DIM),
          B_BASE_ADDR + (j * TILES_K + k) * DIM,
          DIM, DIM);
    }
  }

  // Step 5: Configure output stride and requantizer with compile-time tile counts
  gemmini_config_st(MATMUL_N * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      TILES_I, TILES_J, TILES_K,
      0, 0, 1);

  // Step 6: Execute fused weight-stationary tiled matmul over scratchpad
  // B end address is BANK_NUM*BANK_ROWS (compile-time constant), C dest is SPAD_DEST_CONST
  gemmini_loop_ws_spad(TILES_I, TILES_J, TILES_K, 0, 0, 0,
      A_BASE_ADDR, BANK_NUM * BANK_ROWS, 0, SPAD_DEST_CONST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain BF16 results from mx_smem to DRAM, then fence
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST_CONST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();

  #undef TILES_I
  #undef TILES_J
  #undef TILES_K
  #undef A_BASE_ADDR
  #undef B_BASE_ADDR
  #undef SPAD_DEST_CONST
}
