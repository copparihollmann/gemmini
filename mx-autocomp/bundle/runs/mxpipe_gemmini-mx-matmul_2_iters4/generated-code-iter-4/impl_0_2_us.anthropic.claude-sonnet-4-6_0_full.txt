void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure port 0 (mvin) for A tiles: stride = MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 0);
  // Configure port 1 (mvin2) for B tiles: stride = MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 1);

  // Interleaved A (port 0) and B (port 1) mvin:
  // Outer loop over the larger of tiles_I and tiles_J.
  // For each outer index, we drain all k-batches for row i (A) and
  // column j (B) simultaneously, keeping both DMA ports busy.
  int max_IJ = tiles_I > tiles_J ? tiles_I : tiles_J;

  for (int ij = 0; ij < max_IJ; ij++) {
    // --- A tiles for row ij (if in range) ---
    if (ij < tiles_I) {
      elem_t *a_row_ptr = ((elem_t*)A_in) + ij * DIM * MATMUL_K;
      int k = 0;
      while (k < tiles_K) {
        int batch = tiles_K - k;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        uint32_t sp_addr = a_base + (ij * tiles_K + k) * DIM;
        gemmini_extended_mvin((void*)(a_row_ptr + k * DIM), sp_addr,
                              batch * DIM, DIM);
        k += batch;
      }
    }

    // --- B tiles for column ij (if in range) ---
    if (ij < tiles_J) {
      elem_t *b_col_ptr = ((elem_t*)B_in) + ij * DIM * MATMUL_K;
      int k = 0;
      while (k < tiles_K) {
        int batch = tiles_K - k;
        if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;
        uint32_t sp_addr = b_base + (ij * tiles_K + k) * DIM;
        gemmini_extended_mvin2((void*)(b_col_ptr + k * DIM), sp_addr,
                               batch * DIM, DIM);
        k += batch;
      }
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      0, 0, tiles_I * tiles_J);

  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);

  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
