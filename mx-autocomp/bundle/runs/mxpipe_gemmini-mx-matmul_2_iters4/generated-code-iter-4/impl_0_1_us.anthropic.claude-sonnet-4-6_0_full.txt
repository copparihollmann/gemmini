void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Configure port 0 (mvin) for A tiles: stride = MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 0);
  // Configure port 1 (mvin2) for B tiles: stride = MATMUL_K bytes per row
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), SCALE_IDENTITY, false, 1);

  // Interleaved A and B tile loads: issue mvin (port 0) and mvin2 (port 1) back-to-back
  // each iteration so both DMA engines run concurrently.
  // Since tiles_K <= MAX_BLOCK_LEN=4, each row of A (all k-tiles) and each column of B
  // (all k-tiles) fits in a single mvin/mvin2 call — no inner while-loop needed.
  int total_A_groups = tiles_I;
  int total_B_groups = tiles_J;
  int total_groups = (total_A_groups > total_B_groups) ? total_A_groups : total_B_groups;

  for (int idx = 0; idx < total_groups; idx++) {
    if (idx < total_A_groups) {
      elem_t *dram_ptr_a = ((elem_t*)A_in) + idx * DIM * MATMUL_K;
      uint32_t sp_addr_a = a_base + idx * tiles_K * DIM;
      gemmini_extended_mvin((void*)dram_ptr_a, sp_addr_a, tiles_K * DIM, DIM);
    }
    if (idx < total_B_groups) {
      elem_t *dram_ptr_b = ((elem_t*)B_in) + idx * DIM * MATMUL_K;
      uint32_t sp_addr_b = b_base + idx * tiles_K * DIM;
      gemmini_extended_mvin2((void*)dram_ptr_b, sp_addr_b, tiles_K * DIM, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, tiles_I * tiles_J);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
