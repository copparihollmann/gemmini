void solution(void) {
  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      for (int g = 0; g < MATMUL_K / 32; g++) {
        memcpy(&A_s[g][0], &A_scales_row[g][i0 * BLK], BLK);
        memcpy(&B_s[g][0], &B_scales_col[g][j0 * BLK], BLK);
      }

      int tiles   = BLK / DIM;
      int tiles_K = MATMUL_K / DIM;
      int pad_i   = BLK - mrows;
      int pad_j   = BLK - ncols;

      // A tiles: a_base=0, layout a_base + (i*tiles_K + k)*DIM
      uint32_t a_base  = 0;
      // B tiles: b_base at top of scratchpad, layout b_base + (j*tiles_K + k)*DIM
      uint32_t b_base  = BANK_NUM * BANK_ROWS - (uint32_t)(tiles * tiles_K * DIM);
      int      SPAD_DEST = 128;

      gemmini_flush(0);

      // Step 1: configure execution (fp8 formats, BF16 output)
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
                                  /*C_stride*/1, /*A_stride*/1,
                                  /*A_transpose*/0, /*B_transpose*/0,
                                  /*set_only_strides*/false,
                                  /*act_mx_fmt*/0, /*wgt_mx_fmt*/0, /*out_mx_fmt*/3, /*uselut*/0);

      // Step 2: load scales (sel=0 for A activations, sel=1 for B weights)
      gemmini_mx_load_scales((uint64_t)A_s, A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)B_s, B_S_BYTES, 1);

      // Step 3: mvin A tiles into scratchpad using mvin (port 0)
      // A layout: a_base + (i*tiles_K + k)*DIM
      // Stride = MATMUL_K * sizeof(elem_t) bytes (row stride of A in DRAM)
      gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

      for (int i = 0; i < tiles; i++) {
        int k = 0;
        while (k < tiles_K) {
          // Batch up to MAX_BLOCK_LEN=4 consecutive K-tiles for the same i
          int batch = tiles_K - k;
          if (batch > MAX_BLOCK_LEN) batch = MAX_BLOCK_LEN;

          const elem_t *a_ptr = ((const elem_t *)A_in)
                                + (size_t)(i0 * BLK + i * DIM) * MATMUL_K
                                + (size_t)(k * DIM);
          uint32_t spad_addr = a_base + (uint32_t)((i * tiles_K + k) * DIM);

          gemmini_extended_mvin((void *)a_ptr, spad_addr, batch * DIM, DIM);
          k += batch;
        }
      }

      // Step 4: mvin B tiles into scratchpad using mvin2 (port 1)
      // B layout expected by gemmini_loop_ws_spad: b_base + (j*tiles_K + k)*DIM
      // B in DRAM is stored row-major: B[k_row][n_col], stride = MATMUL_N cols
      gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

      for (int j = 0; j < tiles; j++) {
        int k = 0;
        while (k < tiles_K) {
          // Batch consecutive K-tiles for same j (they are NOT contiguous in DRAM
          // since B is k-major in DRAM: row = k*DIM, col = j0*BLK+j*DIM)
          // Each k step advances by DIM rows in DRAM = DIM*MATMUL_N bytes
          // consecutive k tiles for same j ARE contiguous in DRAM column sense:
          // B_in[k*DIM .. (k+batch)*DIM - 1][j0*BLK+j*DIM .. +DIM)
          // They are NOT row-contiguous (each successive k row is stride MATMUL_N apart)
          // So we can only batch if the spad destination is also contiguous.
          // spad: b_base + (j*tiles_K + k)*DIM .. + batch*DIM  --> contiguous in spad
          // dram: ((k*DIM)*MATMUL_N + j0*BLK+j*DIM) -- rows separated by MATMUL_N*DIM bytes
          // With block_mvin_stride we could batch, but safest is batch=1 here
          // unless MATMUL_N*DIM == DIM (i.e. MATMUL_N==1, not our case).
          // Use batch=1 for correctness; each B tile is a DIM×DIM block.
          const elem_t *b_ptr = ((const elem_t *)B_in)
                                + (size_t)(k * DIM) * MATMUL_N
                                + (size_t)(j0 * BLK + j * DIM);
          uint32_t spad_addr = b_base + (uint32_t)((j * tiles_K + k) * DIM);

          gemmini_extended_mvin2((void *)b_ptr, spad_addr, DIM, DIM);
          k++;
        }
      }

      // Step 5: configure output stride and requantizer
      gemmini_extended_config_st(BLK_OUTC * sizeof(out_t), 0, ACC_SCALE_IDENTITY);
      gemmini_mxquant_config_mvout((uint64_t)scale_factors,
                                   tiles, tiles, tiles_K,
                                   /*scale_act_sel*/0, /*scale_w_sel*/0,
                                   /*lut_update_granularity*/1);

      // Step 6: fused weight-stationary matmul in hardware
      // B end address = BANK_NUM*BANK_ROWS (top of scratchpad, per convention)
      gemmini_loop_ws_spad(tiles, tiles, tiles_K,
                           pad_i, pad_j, 0,
                           a_base, BANK_NUM * BANK_ROWS,
                           /*D*/0, /*C=SPAD_DEST*/SPAD_DEST,
                           /*A_transpose*/false, /*B_transpose*/false,
                           /*full_C*/false, /*low_D*/false,
                           /*ex_accumulate*/false,
                           NO_ACTIVATION,
                           /*a_spad_id*/0, /*b_spad_id*/0,
                           /*is_resadd*/false,
                           /*skips*/0x38);

      // Step 7: drain BF16 results from mx_smem to DRAM, row by row
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK,
                             ncols);

      // Step 8: fence before host reads output
      gemmini_fence();
    }
}
