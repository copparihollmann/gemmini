#include <stdint.h>
#include <string.h>
#include "gemmini.h"
#include "gemmini_params.h"

// Matrix dimensions - must be multiples of DIM=16
#define MATMUL_M 64
#define MATMUL_N 64
#define MATMUL_K 64

#define DIM_SIZE 16
#define TILES_I (MATMUL_M / DIM_SIZE)
#define TILES_J (MATMUL_N / DIM_SIZE)
#define TILES_K (MATMUL_K / DIM_SIZE)

// Number of MX blocks along K dimension (one scale per DIM elements along K)
#define GK TILES_K

// Scale memory: A has GK*TILES_I scales (one per row-block, per K-block)
// B has GK*TILES_J scales
static uint8_t A_scales[GK * TILES_I];
static uint8_t B_scales[GK * TILES_J];

// Input matrices (FP8 E4M3 packed as uint8)
static uint8_t A_mat[MATMUL_M][MATMUL_K];
static uint8_t B_mat[MATMUL_N][MATMUL_K]; // weight-stationary: B is N x K

// Output matrix (BF16 packed, 2 bytes per element)
static uint16_t C_hw[MATMUL_M][MATMUL_N];

// Reference output for verification
static float C_ref[MATMUL_M][MATMUL_N];

// Scratchpad layout:
// b_base at top of scratchpad, growing downward
// a_base at bottom, growing upward
#define SPAD_ROWS (BANK_NUM * BANK_ROWS)  // 4 * 2048 = 8192
#define b_base_spad (SPAD_ROWS - TILES_K * TILES_J * DIM_SIZE)
#define a_base_spad 0

// Output destination in mx_smem (word offset)
#define SPAD_DEST 0

void matmul_mx_fp8(
    uint8_t A[MATMUL_M][MATMUL_K],
    uint8_t B[MATMUL_N][MATMUL_K],
    uint8_t a_scales[GK * TILES_I],
    uint8_t b_scales[GK * TILES_J],
    uint16_t C[MATMUL_M][MATMUL_N])
{
    // Verify scratchpad fit
    // A needs TILES_I * TILES_K * DIM rows
    // B needs TILES_J * TILES_K * DIM rows
    // Total = (TILES_I + TILES_J) * TILES_K * DIM <= SPAD_ROWS
    const int a_rows_needed = TILES_I * TILES_K * DIM_SIZE;
    const int b_rows_needed = TILES_J * TILES_K * DIM_SIZE;
    // a_base=0 + a_rows_needed must not exceed b_base_spad
    // This is guaranteed by our layout choice

    // =========================================================
    // Step 1: Configure execution (FP8 E4M3 input, BF16 output)
    // act_mx_fmt=0 (FP8 E4M3), wgt_mx_fmt=0 (FP8 E4M3), out_mx_fmt=3 (BF16)
    // uselut=0 (FP8 does not need LUT)
    // =========================================================
    gemmini_extended3_config_ex(
        WEIGHT_STATIONARY,  // dataflow
        NO_ACTIVATION,      // sys_act
        0,                  // sys_shift
        ACC_SCALE_IDENTITY, // sys_acc_scale
        1,                  // C_stride
        MATMUL_K,           // A_stride (row stride of A in elements)
        0,                  // A_transpose
        0,                  // B_transpose
        false,              // set_only_strides
        0,                  // act_mx_fmt: FP8 E4M3
        0,                  // wgt_mx_fmt: FP8 E4M3
        3,                  // out_mx_fmt: BF16
        0                   // uselut: not needed for FP8
    );

    // =========================================================
    // Step 2: Load scale factors
    // sel=0 -> A (activation) scales into mx_scale_a_mem
    // sel=1 -> B (weight) scales into mx_scale_b_mem
    // =========================================================
    gemmini_mx_load_scales((uint64_t)a_scales, (uint64_t)(GK * TILES_I), 0);
    gemmini_mx_load_scales((uint64_t)b_scales, (uint64_t)(GK * TILES_J), 1);

    // =========================================================
    // Step 3: No LUT needed for FP8 E4M3 (uselut=0)
    // =========================================================

    // =========================================================
    // Step 4: MVIN A and B tiles into scratchpad
    // A tile layout: a_base + (i*TILES_K + k)*DIM rows
    // B tile layout: b_base + (j*TILES_K + k)*DIM rows
    //
    // A is M x K, row stride = MATMUL_K bytes (elem_t = uint8)
    // B is N x K, row stride = MATMUL_K bytes
    // Each tile is DIM x DIM elements
    // =========================================================

    // Configure load for A (id=0, stride = MATMUL_K * sizeof(elem_t) = MATMUL_K bytes)
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

    // Load all A tiles: iterate over i (row tiles) and k (K tiles)
    for (int i = 0; i < TILES_I; i++) {
        for (int k = 0; k < TILES_K; k++) {
            uint64_t dram_addr = (uint64_t)&A[i * DIM_SIZE][k * DIM_SIZE];
            uint32_t spad_addr = a_base_spad + (i * TILES_K + k) * DIM_SIZE;
            // Use block mvin, up to MAX_BLOCK_LEN=4 tiles at once
            // Each tile is DIM cols x DIM rows
            gemmini_extended_mvin(dram_addr, spad_addr, DIM_SIZE, DIM_SIZE);
        }
    }

    // Configure load for B (id=1, stride = MATMUL_K * sizeof(elem_t))
    gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

    // Load all B tiles: iterate over j (col tiles) and k (K tiles)
    for (int j = 0; j < TILES_J; j++) {
        for (int k = 0; k < TILES_K; k++) {
            uint64_t dram_addr = (uint64_t)&B[j * DIM_SIZE][k * DIM_SIZE];
            uint32_t spad_addr = b_base_spad + (j * TILES_K + k) * DIM_SIZE;
            gemmini_extended_mvin2(dram_addr, spad_addr, DIM_SIZE, DIM_SIZE);
        }
    }

    // =========================================================
    // Step 5: Configure requantized mvout
    // scale_dram: pointer to scale memory (we use a_scales as reference)
    // i_bound=TILES_I, j_bound=TILES_J, k_bound=TILES_K
    // scale_act_sel=0: use mx_scale_a_mem buffer
    // scale_w_sel=0: use mx_scale_b_mem buffer
    // lut_update_granularity=1
    // =========================================================
    gemmini_extended_config_st(MATMUL_N * sizeof(uint16_t), NO_ACTIVATION, ACC_SCALE_IDENTITY);

    // scale_dram: The requantizer reads scales from DRAM at:
    //   scale_dram + m * N_blocks + bi
    // We provide the base address of A_scales (used as output scale reference)
    // For BF16 output, the requantizer uses A and B scales from the loaded buffers
    gemmini_mxquant_config_mvout(
        (uint64_t)a_scales,  // scale_dram base (used for output requantization indexing)
        TILES_I,             // i_bound
        TILES_J,             // j_bound
        TILES_K,             // k_bound
        0,                   // scale_act_sel: use mx_scale_a_mem
        0,                   // scale_w_sel: use mx_scale_b_mem
        1                    // lut_update_granularity
    );

    // =========================================================
    // Step 6: Execute fused weight-stationary matmul
    // B spad end address = BANK_NUM * BANK_ROWS (top of scratchpad)
    // SPAD_DEST = output word offset in mx_smem
    // skips=0x38: suppress D preload (no bias) and other unused ops
    // =========================================================
    gemmini_loop_ws_spad(
        TILES_I,          // I tile count
        TILES_J,          // J tile count
        TILES_K,          // K tile count
        0, 0, 0,          // pad_I, pad_J, pad_K (no padding)
        a_base_spad,      // A spad base address
        SPAD_ROWS,        // B spad end address (top of scratchpad)
        0,                // D (no bias, use 0)
        SPAD_DEST,        // C output destination in mx_smem
        false,            // A_transpose
        false,            // B_transpose
        false,            // full_C
        false,            // low_D
        false,            // ex_accumulate
        NO_ACTIVATION,    // activation
        0,                // a_spad_id (no reuse)
        0,                // b_spad_id (no reuse)
        false,            // is_resadd
        0x38              // skips: suppress D preload and redundant ops
    );

    // =========================================================
    // Step 7: Drain BF16 results from mx_smem to DRAM
    // Results are packed 4 BF16 per uint64 word
    // smem_off_bf16 = SPAD_DEST * 16 (word offset in 16-bit units)
    // num_bf16 = total number of BF16 elements = M * N
    // =========================================================
    gemmini_mx_read_smem(
        (uint64_t)C,           // DRAM destination
        SPAD_DEST * 16,        // smem word offset (16-bit units)
        MATMUL_M * MATMUL_N    // number of BF16 values to read
    );
    gemmini_fence();
}

// fp8_e4m3 to float conversion helper
static float fp8_e4m3_to_float(uint8_t bits) {
    // E4M3: sign=bit7, exp=bits[6:3], mantissa=bits[2:0]
    int sign = (bits >> 7) & 1;
    int exp = (bits >> 3) & 0xF;
    int mant = bits & 0x7;
    float val;
    if (exp == 0) {
        // subnormal
        val = (float)mant / 8.0f * (1.0f / (1 << 6));
    } else if (exp == 15 && mant == 7) {
        // NaN
        val = 0.0f; // treat as 0 for reference
    } else {
        val = (1.0f + (float)mant / 8.0f) * (1.0f * (1 << (exp - 7)));
    }
    return sign ? -val : val;
}

// e8m0 to float: pure power of two, 2^(exp - 127)
static float e8m0_to_float(uint8_t exp) {
    if (exp == 0) return 0.0f; // zero scale
    // e8m0: value = 2^(exp - 127)
    int e = (int)exp - 127;
    if (e >= 0) {
        return (float)(1 << e);
    } else {
        return 1.0f / (float)(1 << (-e));
    }
}

// bf16 bits to float
static float bf16_to_float(uint16_t bf16) {
    uint32_t f32 = (uint32_t)bf16 << 16;
    float result;
    memcpy(&result, &f32, sizeof(float));
    return result;
}

// Initialize test data and run verification
void run_matmul_test(void) {
    // Initialize A and B with simple FP8 E4M3 values
    // Use value encoding for 1.0 in FP8 E4M3: exp=7, mant=0 -> bits = 0x38
    // (sign=0, exp=7, mant=0): value = (1 + 0/8) * 2^(7-7) = 1.0
    for (int i = 0; i < MATMUL_M; i++) {
        for (int k = 0; k < MATMUL_K; k++) {
            A_mat[i][k] = 0x38; // FP8 E4M3 value 1.0
        }
    }
    for (int j = 0; j < MATMUL_N; j++) {
        for (int k = 0; k < MATMUL_K; k++) {
            B_mat[j][k] = 0x38; // FP8 E4M3 value 1.0
        }
    }

    // Initialize scales: e8m0 value for scale=1.0 is exp=127 -> bits=0x7F
    for (int i = 0; i < GK * TILES_I; i++) {
        A_scales[i] = 0x7F; // e8m0: 2^(127-127) = 1.0
    }
    for (int j = 0; j < GK * TILES_J; j++) {
        B_scales[j] = 0x7F; // e8m0: 1.0
    }

    // Compute reference: C[i][j] = sum_k A[i][k] * B[j][k] * a_scale * b_scale
    // With all values = 1.0 and all scales = 1.0:
    // C[i][j] = MATMUL_K * 1.0 * 1.0 = 64.0
    for (int i = 0; i < MATMUL_M; i++) {
        for (int j = 0; j < MATMUL_N; j++) {
            float sum = 0.0f;
            for (int k_blk = 0; k_blk < GK; k_blk++) {
                float a_scale = e8m0_to_float(A_scales[k_blk * TILES_I + i / DIM_SIZE]);
                float b_scale = e8m0_to_float(B_scales[k_blk * TILES_J + j / DIM_SIZE]);
                float block_sum = 0.0f;
                for (int k = k_blk * DIM_SIZE; k < (k_blk + 1) * DIM_SIZE; k++) {
                    block_sum += fp8_e4m3_to_float(A_mat[i][k]) * fp8_e4m3_to_float(B_mat[j][k]);
                }
                sum += block_sum * a_scale * b_scale;
            }
            C_ref[i][j] = sum;
        }
    }

    // Run hardware matmul
    matmul_mx_fp8(A_mat, B_mat, A_scales, B_scales, C_hw);

    // Verify results
    int errors = 0;
    for (int i = 0; i < MATMUL_M; i++) {
        for (int j = 0; j < MATMUL_N; j++) {
            float hw_val = bf16_to_float(C_hw[i][j]);
            float ref_val = C_ref[i][j];
            float diff = hw_val - ref_val;
            if (diff < 0) diff = -diff;
            float abs_ref = ref_val < 0 ? -ref_val : ref_val;
            float rel_err = (abs_ref > 1e-6f) ? diff / abs_ref : diff;
            if (rel_err > 0.02f) { // 2% relative tolerance for BF16
                errors++;
            }
        }
    }

    if (errors == 0) {
        // Test passed
    }
}

int main(void) {
    gemmini_flush(0);
    run_matmul_test();
    return 0;
}
