An elegant and highly effective optimization strategy to apply to this code is **Vectorized Multi-Tile DRAM DMA Transactions (Block MVIN) with Dedicated Port Mapping**.

### How This Strategy Works

In the baseline implementation, the host CPU issues a separate `gemmini_extended_mvin` RoCC instruction for every individual $16 \times 16$ tile of matrices A and B. For a block size of $64 \times 64$ and a $K$-dimension of $512$, this fine-grained loading pattern requires:
* **Matrix A**: $4 \text{ (I tiles)} \times 32 \text{ (K tiles)} = 128$ load instructions.
* **Matrix B**: $4 \text{ (J tiles)} \times 32 \text{ (K tiles)} = 128$ load instructions.

This results in **256 load instructions per iteration**, creating a severe CPU-to-coprocessor instruction dispatch bottleneck that leaves the systolic array starved for data. 

Since the data is contiguous in DRAM along the row dimension, we can consolidate multiple individual tile loads into single wide DMA transactions up to the hardware-supported limit of `MAX_BLOCK_LEN = 4` tiles (i.e., $4 \times 16 = 64$ bytes):
1. **For Matrix A**: We load 4 consecutive $K$-tiles in a single wider transaction. This reduces the number of load commands for A from $128$ to just $32$ per iteration.
2. **For Matrix B**: Since the column tile count ($J$) is $4$, we can load all 4 column tiles for a given row in one single block transfer of length 4. This reduces B's load commands from $128$ to $32$ per iteration.
3. **Dedicated Port Mapping**: Mapping Matrix B to the dedicated `gemmini_extended_mvin2` instruction instead of sharing the primary `mvin` port with A enables the hardware to utilize dual-ported SRAM interfaces and pipeline the transfers in parallel.

This reduces the total number of load instructions by **4x** (from 256 to 64 per iteration), dramatically lowering host CPU overhead and maximizing systolic array duty cycles.

### Optimized Code

```c
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include "include/gemmini_testutils.h"
#include "include/matmul_fp8_64x64.h"

#define GEMMINI_SF_MEM 0x40088000
#define GEMMINI_SF_MEM_A (GEMMINI_SF_MEM + 0x2000)
#define GEMMINI_SF_MEM_B GEMMINI_SF_MEM
#define SMEM 0x40000000

#define DIM 16
#define ADDR_LEN 32

// BF16 values packed 4 per uint64_t output word
#define BF16_PER_WORD 4
#define OUT_COLS (MATMUL_M / BF16_PER_WORD)

typedef uint8_t  elem_t;   // fp8:e4m3 input elements
typedef uint8_t  welem_t;  // fp8:e4m3 weight elements
typedef uint64_t out_t;    // 4x bf16 packed per word

void solution(void) {
  // Pre-stage scale buffers once for all blocks, amortizing the restriding
  // cost across all BLOCKS_M * BLOCKS_N iterations.
  const int GK = MATMUL_K / 32;

  // Static pre-staged scale arrays: [block_idx][scale_group][elem_in_block]
  // Sized for the maximum number of blocks along each dimension.
  static uint8_t pre_A_s[BLOCKS_M][MATMUL_K / 32][BLK];
  static uint8_t pre_B_s[BLOCKS_N][MATMUL_K / 32][BLK];

  // One-time restride: copy from strided A_scales_row[g][i0*BLK] into
  // contiguous pre_A_s[i0][g][0..BLK-1], done once before the main loop.
  for (int i0 = 0; i0 < BLOCKS_M; i0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_A_s[i0][g][0], &A_scales_row[g][i0 * BLK], BLK);

  for (int j0 = 0; j0 < BLOCKS_N; j0++)
    for (int g = 0; g < GK; g++)
      memcpy(&pre_B_s[j0][g][0], &B_scales_col[g][j0 * BLK], BLK);

  for (int i0 = 0; i0 < BLOCKS_M; i0++) {
    for (int j0 = 0; j0 < BLOCKS_N; j0++) {
      // Partial last M/N block: compute only the real rows/cols, pad the rest.
      int mrows = REAL_M - i0 * BLK; if (mrows > BLK) mrows = BLK; if (mrows <= 0) continue;
      int ncols = REAL_N - j0 * BLK; if (ncols > BLK) ncols = BLK; if (ncols <= 0) continue;

      int tiles = BLK / DIM, tiles_K = MATMUL_K / DIM;
      uint32_t a_base = 0;
      uint32_t b_base = BANK_NUM * BANK_ROWS - tiles * tiles_K * DIM;
      int SPAD_DEST = 128;
      int pad_i = BLK - mrows, pad_j = BLK - ncols;

      // PROF accel
      gemmini_flush(0);
      gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);

      // No per-iteration memcpy: use pre-staged contiguous scale pointers directly.
      gemmini_mx_load_scales((uint64_t)pre_A_s[i0], A_S_BYTES, 0);
      gemmini_mx_load_scales((uint64_t)pre_B_s[j0], B_S_BYTES, 1);

      // A rows: stride K (Vectorized block load: 4 K-tiles at a time)
      gemmini_config_ld(MATMUL_K * sizeof(elem_t));      
      for (int i = 0; i < tiles; i++) {
        for (int k = 0; k < tiles_K; k += 4) {
          gemmini_extended_mvin((void*)(((elem_t*)A_in) + (i0 * BLK + i * DIM) * MATMUL_K + k * DIM),
                                a_base + (i * tiles_K + k) * DIM, 
                                4 * DIM, DIM);
        }
      }

      // B rows: stride N (Vectorized block load: all 'tiles' of J at once using mvin2)
      gemmini_config_ld(MATMUL_N * sizeof(elem_t));      
      for (int k = 0; k < tiles_K; k++) {
        gemmini_extended_mvin2((void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j0 * BLK)),
                              b_base + (k * tiles) * DIM, 
                              tiles * DIM, DIM);
      }

      gemmini_config_st(BLK_OUTC * sizeof(out_t));
      gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles, tiles, tiles_K, 0, 0, 1);
      gemmini_loop_ws_spad(tiles, tiles, tiles_K, pad_i, pad_j, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                           SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

      // Accelerator-centric readout: per-row hardware DMA from smem straight into the
      // bf16-packed row-major output (block-internal smem row stride is BLK).
      for (int r = 0; r < mrows; r++)
        gemmini_mx_read_smem(&C_hw[i0 * BLK + r][(j0 * BLK) / BF16_PER_WORD],
                             SPAD_DEST * 16 + r * BLK, ncols);
      gemmini_fence();
    }
  }
}
```
