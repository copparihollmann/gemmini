void solution(void) {
  // Step 1: Configure execution (FP8 E4M3, BF16 output)
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY,
      1, 1, 0, 0, false, 0, 0, 3, 0);

  // Step 2: Load scale factors (A=sel0, B=sel1)
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Step 3 (skipped): uselut=0 for FP8

  // Step 4: MVIN tiles - use mvin for A, mvin2 for B to overlap loads
  // tiles_I=8, tiles_J=8, tiles_K=16
  // a_base=0, b_base=8192-16*8*16=6144
  // A layout: a_base + (i*tiles_K + k)*DIM
  // B layout: b_base + (k*tiles_J + j)*DIM

  // Load A tiles: batch up to MAX_BLOCK_LEN=4 consecutive k tiles per i row
  gemmini_config_ld(MATMUL_K * sizeof(elem_t));
  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {
      int batch = (k + MAX_BLOCK_LEN <= tiles_K) ? MAX_BLOCK_LEN : (tiles_K - k);
      for (int b = 0; b < batch; b++) {
        gemmini_extended_mvin(
            (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + (k + b) * DIM),
            a_base + (i * tiles_K + k + b) * DIM,
            DIM, DIM);
      }
    }
  }

  // Load B tiles via mvin2 port: batch consecutive j tiles for same k
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), SCALE_IDENTITY, false, 1);
  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {
      int batch = (j + MAX_BLOCK_LEN <= tiles_J) ? MAX_BLOCK_LEN : (tiles_J - j);
      for (int b = 0; b < batch; b++) {
        gemmini_extended_mvin2(
            (void*)(((elem_t*)B_in) + (k * DIM) * MATMUL_N + (j + b) * DIM),
            b_base + (k * tiles_J + j + b) * DIM,
            DIM, DIM);
      }
    }
  }

  // Step 5: Configure requantized mvout
  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors,
      tiles_I, tiles_J, tiles_K,
      0, 0, tiles_K);

  // Step 6: Compute tiled matmul (one-shot hardware loop)
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
      a_base, BANK_NUM * BANK_ROWS, 0, SPAD_DEST,
      false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);

  // Step 7: Drain results from mx_smem to DRAM
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
