Looking at the original code, I need to understand what it does:
- 128×128×256 FP8 E4M3 matmul
- tiles_I = 128/16 = 8, tiles_J = 128/16 = 8, tiles_K = 256/16 = 16
- Total scratchpad needed: A tiles = 8*16*16 = 2048 rows, B tiles = 8*16*16 = 2048 rows
- b_base = BANK_NUM*BANK_ROWS - tiles_K*tiles_J*DIM = 8192 - 16*8*16 = 8192 - 2048 = 6144

The main optimization opportunities:
1. Use `gemmini_extended_mvin2` for B tiles to overlap A and B loads
2. Batch mvin calls using MAX_BLOCK_LEN=4 (load 4 tiles at once where possible)
3. Increase `lut_update_granularity` since uselut=0, this doesn't apply but tiles_J=8 so G=8 is safe
4. The mvin loops can be restructured to issue larger block transfers

For batching: when loading A tiles, consecutive k tiles for same i are contiguous in scratchpad (stride DIM=16 rows). We can load up to 4 at once using cols=4*DIM.

For B tiles: consecutive j tiles for same k are contiguous. We can batch up to 4.

Let me restructure the mvin to use block loads and overlap A/B loading via mvin vs mvin2:

```c
void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  // Load A tiles: for each i, load k tiles in blocks of MAX_BLOCK_LEN=4
  // A[i][k] in DRAM: row i*DIM, col k*DIM, stride MATMUL_K
  // Spad: a_base + (i*tiles_K + k)*DIM
  gemmini_extended3_config_ld(MATMUL_K * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 0);

  // Load B tiles via mvin2: B[k][j] in DRAM, stride MATMUL_N
  // Spad: b_base + (k*tiles_J + j)*DIM
  gemmini_extended3_config_ld(MATMUL_N * sizeof(elem_t), MVIN_SCALE_IDENTITY, false, 1);

  for (int i = 0; i < tiles_I; i++) {
    for (int k = 0; k < tiles_K; k += MAX_BLOCK_LEN) {
      int batch = (k + MAX_BLOCK_LEN <= tiles_K) ? MAX_BLOCK_LEN : (tiles_K - k);
      gemmini_extended_mvin(
        (void*)(((elem_t*)A_in) + i * DIM * MATMUL_K + k * DIM),
        a_base + (i * tiles_K + k) * DIM,
        batch * DIM, DIM);
    }
  }

  for (int k = 0; k < tiles_K; k++) {
    for (int j = 0; j < tiles_J; j += MAX_BLOCK_LEN) {
      int batch = (j + MAX_BLOCK_LEN <= tiles_J) ? MAX_BLOCK_LEN : (tiles_J - j);
      gemmini_extended_mvin2(
        (void*)(((elem_t*)B_in) + k * DIM * MATMUL_N + j * DIM),
        b_base + (k * tiles_J + j) * DIM,
        batch * DIM, DIM);
    }
  }

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, tiles_J);

  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0,
                       a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false,
                       NO_ACTIVATION, 0, 0, false, 0x38);

  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * 16, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
```
