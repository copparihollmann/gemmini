void solution(void) {
  gemmini_extended3_config_ex(WEIGHT_STATIONARY, 0, 0, ACC_SCALE_IDENTITY, 1, 1, 0, 0, false, 0, 0, 3, 0);
  gemmini_mx_load_scales((uint64_t)&A_scales_row, sizeof(A_scales_row), 0);
  gemmini_mx_load_scales((uint64_t)&B_scales_col, sizeof(B_scales_col), 1);

  gemmini_config_ld(MATMUL_K * sizeof(elem_t));

  // Interleave A tiles (via mvin, port 0) and B tiles (via mvin2, port 1)
  // so that the two DMA engines can work in parallel.

  // A tile (i=0, k=0) -> a_base + 0*DIM  via port 0
  // B tile (j=0, k=0) -> b_base + 0*DIM  via port 1
  gemmini_extended_mvin( (void*)(((elem_t*)A_in) + 0 * DIM * MATMUL_K + 0 * DIM),
                         a_base + 0 * DIM, DIM, DIM);
  gemmini_extended_mvin2((void*)(((elem_t*)B_in) + 0 * DIM * MATMUL_K + 0 * DIM),
                         b_base + 0 * DIM, DIM, DIM);

  // A tile (i=0, k=1) -> a_base + 1*DIM  via port 0
  // B tile (j=0, k=1) -> b_base + 1*DIM  via port 1
  gemmini_extended_mvin( (void*)(((elem_t*)A_in) + 0 * DIM * MATMUL_K + 1 * DIM),
                         a_base + 1 * DIM, DIM, DIM);
  gemmini_extended_mvin2((void*)(((elem_t*)B_in) + 0 * DIM * MATMUL_K + 1 * DIM),
                         b_base + 1 * DIM, DIM, DIM);

  // A tile (i=1, k=0) -> a_base + 2*DIM  via port 0
  // B tile (j=1, k=0) -> b_base + 2*DIM  via port 1
  gemmini_extended_mvin( (void*)(((elem_t*)A_in) + 1 * DIM * MATMUL_K + 0 * DIM),
                         a_base + 2 * DIM, DIM, DIM);
  gemmini_extended_mvin2((void*)(((elem_t*)B_in) + 1 * DIM * MATMUL_K + 0 * DIM),
                         b_base + 2 * DIM, DIM, DIM);

  // A tile (i=1, k=1) -> a_base + 3*DIM  via port 0
  // B tile (j=1, k=1) -> b_base + 3*DIM  via port 1
  gemmini_extended_mvin( (void*)(((elem_t*)A_in) + 1 * DIM * MATMUL_K + 1 * DIM),
                         a_base + 3 * DIM, DIM, DIM);
  gemmini_extended_mvin2((void*)(((elem_t*)B_in) + 1 * DIM * MATMUL_K + 1 * DIM),
                         b_base + 3 * DIM, DIM, DIM);

  gemmini_config_st(OUT_COLS * sizeof(out_t));
  gemmini_mxquant_config_mvout((uint64_t)scale_factors, tiles_I, tiles_J, tiles_K, 0, 0, 1);
  gemmini_loop_ws_spad(tiles_I, tiles_J, tiles_K, 0, 0, 0, a_base, BANK_NUM * BANK_ROWS, 0,
                       SPAD_DEST, false, false, false, false, false, NO_ACTIVATION, 0, 0, false, 0x38);
  gemmini_mx_read_smem(&C_hw[0][0], SPAD_DEST * DIM, MATMUL_M * MATMUL_N);
  gemmini_fence();
}
